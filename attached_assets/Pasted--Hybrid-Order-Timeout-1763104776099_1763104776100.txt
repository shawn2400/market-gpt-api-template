```python
class SmartTimeoutManager:
    def __init__(self):
        self.base_timeout = 60  # 60 שניות בסיסי
        self.adaptive_multipliers = {
            'VOLATILITY_BASED': True,
            'MARKET_CONDITIONS': True, 
            'ORDER_TYPE': True,
            'TIME_OF_DAY': True
        }
    
    async def calculate_dynamic_timeout(self, symbol, order_type, current_conditions):
        """חישוב טיימאאוט דינמי לפי תנאים"""
        
        # 1. זמן בסיסי
        timeout = self.base_timeout
        
        # 2. התאמה לפי תנודתיות
        volatility = await self.get_symbol_volatility(symbol)
        if volatility > 0.04:  # תנודתיות גבוהה
            timeout *= 0.6     # 60% מהזמן - יותר אגרסיבי
        elif volatility < 0.015:  # תנודתיות נמוכה
            timeout *= 1.4     # 140% מהזמן - יותר סבלנות
        
        # 3. התאמה לפי סוג Order
        order_multipliers = {
            'ENTRY_LIMIT': 1.2,   # כניסה - יותר סבלנות
            'TP_LIMIT': 0.8,      # Take Profit - פחות סבלנות
            'GRID_ORDER': 1.5,    # GRID - הכי הרבה סבלנות
            'STOP_LIMIT': 0.7     # Stop Limit - אגרסיבי
        }
        timeout *= order_multipliers.get(order_type, 1.0)
        
        # 4. התאמה לפי שעה ביום
        current_hour = datetime.now().hour
        if 14 <= current_hour <= 22:  # שעות פעילות
            timeout *= 0.8  # יותר מהיר
        else:  # שעות שקטות
            timeout *= 1.3  # יותר סבלנות
        
        return max(30, min(timeout, 180))  # הגבל ל-30-180 שניות
```

### **🎯 Smart Timeout Monitor משודרג:**

```python
class SmartOrderTimeoutMonitor:
    def __init__(self):
        self.order_timers = {}  # {order_id: {'created_at': timestamp, 'timeout': seconds}}
        self.check_interval = 3  # בודק כל 3 שניות - יותר מדויק!
    
    async def start_smart_monitoring(self):
        """ניטור orders עם טיימאאוט דינמי"""
        while True:
            try:
                current_time = time.time()
                
                for order_id, order_data in list(self.order_timers.items()):
                    order_age = current_time - order_data['created_at']
                    
                    # בדוק אם הגיע זמן טיימאאוט
                    if order_age >= order_data['timeout']:
                        await self.handle_order_timeout(order_id, order_data)
                    
                    # 📊 ניטור pro-active - לפני הטיימאאוט!
                    elif order_age >= order_data['timeout'] * 0.8:  # 80% מזמן הטיימאאוט
                        await self.pre_timeout_check(order_id, order_data)
                
                await asyncio.sleep(self.check_interval)
                
            except Exception as e:
                logger.error(f"❌ Smart timeout monitor error: {e}")
                await asyncio.sleep(10)
    
    async def pre_timeout_check(self, order_id, order_data):
        """בדיקה pro-active לפני הטיימאאוט"""
        symbol = order_data['symbol']
        current_conditions = await self.get_current_market_conditions(symbol)
        
        # אם השוק השתנה - אולי צריך להתאים את הטיימאאוט
        if await self.should_extend_timeout(symbol, current_conditions):
            new_timeout = order_data['timeout'] * 1.3  # הארך ב-30%
            self.order_timers[order_id]['timeout'] = new_timeout
            logger.info(f"⏱️ Extended timeout for {symbol} to {new_timeout}s")
```

### **🚀 Intelligent Order Escalation:**

```python
class IntelligentOrderEscalator:
    async def handle_order_timeout(self, order_id, order_data):
        """טיפול חכם ב-order שהגיע טיימאאוט"""
        
        symbol = order_data['symbol']
        order_type = order_data['order_type']
        
        # 1. בדוק את המצב הנוכחי לפני שמבטלים
        current_market = await self.analyze_current_market(symbol)
        
        # 2. החלטה חכמה - לא אוטומטית!
        if current_market['is_near_fill']:
            # אם קרוב להתמלא - תן עוד זמן
            if await self.is_near_fill_price(symbol, order_data['price']):
                logger.info(f"🎯 Order {order_id} near fill - extending timeout")
                return await self.extend_timeout(order_id, 30)  # +30 שניות
        
        # 3. החלטת escalation חכמה
        escalation_decision = await self.make_escalation_decision(
            order_data, current_market
        )
        
        if escalation_decision['action'] == 'CONVERT_TO_MARKET':
            await self.convert_to_market_order(order_id, order_data)
        elif escalation_decision['action'] == 'ADJUST_AND_RETRY':
            await self.adjust_and_retry_order(order_id, order_data)
        elif escalation_decision['action'] == 'CANCEL_AND_WAIT':
            await self.cancel_and_wait(order_id, order_data)
    
    async def make_escalation_decision(self, order_data, market_conditions):
        """החלטה חכמה איך לטפל ב-order שהגיע טיימאאוט"""
        
        symbol = order_data['symbol']
        
        # קריטריונים להחלטה:
        factors = {
            'urgency': await self.calculate_urgency(symbol, order_data['order_type']),
            'market_volatility': market_conditions['volatility'],
            'distance_from_price': await self.get_price_distance(symbol, order_data['price']),
            'order_importance': self.get_order_importance(order_data['order_type']),
            'time_sensitivity': await self.get_time_sensitivity(symbol)
        }
        
        # החלטה מבוססת scoring
        score = sum(factors.values())
        
        if score >= 4.0:
            return {'action': 'CONVERT_TO_MARKET', 'reason': 'High urgency and good conditions'}
        elif score >= 2.5:
            return {'action': 'ADJUST_AND_RETRY', 'reason': 'Medium urgency - worth retry'}
        else:
            return {'action': 'CANCEL_AND_WAIT', 'reason': 'Low urgency - better to wait'}
```

```python
class MarketConditionsWatcher:
    async def get_current_market_conditions(self, symbol):
        """ניתוח תנאי שוק בזמן אמת"""
        return {
            'volatility': await self.get_current_volatility(symbol),
            'volume_profile': await self.get_volume_profile(symbol),
            'order_book_imbalance': await self.get_order_book_imbalance(symbol),
            'recent_price_action': await self.get_recent_price_action(symbol),
            'is_near_fill': await self.is_near_fill_price(symbol, target_price),
            'market_momentum': await self.get_market_momentum(symbol)
        }
    
    async def is_near_fill_price(self, symbol, target_price):
        """בדיקה אם המחיר קרוב ל-target price"""
        current_price = await self.get_current_price(symbol)
        price_diff = abs(current_price - target_price) / target_price
        
        return price_diff <= 0.002  # within 0.2%
```

```python
class HybridOrderManager:
    def __init__(self):
        self.timeout_manager = SmartTimeoutManager()
        self.monitor = SmartOrderTimeoutMonitor() 
        self.escalator = IntelligentOrderEscalator()
        self.market_watcher = MarketConditionsWatcher()
    
    async def place_order_with_hybrid_timeout(self, symbol, order_type, price, quantity):
        """הגשת order עם טיימאאוט היברידי"""
        
        # 1. חשב טיימאאוט דינמי
        conditions = await self.market_watcher.get_current_market_conditions(symbol)
        dynamic_timeout = await self.timeout_manager.calculate_dynamic_timeout(
            symbol, order_type, conditions
        )
        
        # 2. הגש את ה-order
        order_result = await self.binance.place_limit_order(
            symbol=symbol,
            price=price,
            quantity=quantity
        )
        
        # 3. התחל ניטור עם הטיימאאוט הדינמי
        await self.monitor.start_monitoring_order(
            order_id=order_result['orderId'],
            symbol=symbol,
            order_type=order_type,
            timeout=dynamic_timeout,
            original_price=price
        )
        
        logger.info(f"⏰ Order {order_result['orderId']} - Hybrid timeout: {dynamic_timeout}s")
        
        return order_result
```
