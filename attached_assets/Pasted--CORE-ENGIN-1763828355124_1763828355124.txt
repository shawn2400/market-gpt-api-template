```python
class EfficientRegimeDetector:
    def __init__(self):
        self.required_data = {
            'price': deque(maxlen=50),      # רק 50 נקודות
            'volume': deque(maxlen=20),     # רק 20 נקודות
            'volatility': deque(maxlen=30)  # רק 30 נקודות
        }
    
    def detect_regime(self, current_data):
        """זיהוי regime עם 3 אינדיקטורים בלבד"""
        # 1. Trend Detection (EMA פשוט)
        trend_strength = self._calculate_trend_strength()
        
        # 2. Volatility Detection (ATR מקוצר)
        volatility_regime = self._calculate_volatility()
        
        # 3. Momentum (RSI בסיסי)
        momentum = self._calculate_momentum()
        
        # Simple decision tree
        if volatility_regime > 0.8:  # High volatility
            return 'HIGH_VOL', -0.07, 0.85
        elif trend_strength > 0.6:   # Strong trend
            return 'TRENDING', +0.05, 0.75
        elif trend_strength < 0.3:   # Weak trend
            return 'RANGING', +0.03, 0.65
        else:                        # Neutral
            return 'NEUTRAL', 0.0, 0.5
    
    def _calculate_trend_strength(self):
        """חישוב trend עם EMA 20 ו-50 בלבד"""
        if len(self.required_data['price']) < 50:
            return 0.5
        
        prices = list(self.required_data['price'])
        ema_20 = self._ema(prices, 20)
        ema_50 = self._ema(prices, 50)
        
        if ema_50 == 0:
            return 0.5
            
        trend_ratio = (ema_20 - ema_50) / ema_50
        return abs(trend_ratio)  # רק העוצמה, לא הכיוון
```

```python
class RedisOptimizer:
    def __init__(self, redis_client):
        self.redis = redis_client
        self.compression_enabled = True
        
    # Redis keys עם TTL אוטומטי
    REDIS_KEYS = {
        'performance:current': {'ttl': 3600},           # 1 hour
        'trades:recent': {'ttl': 86400},               # 24 hours
        'market:regime': {'ttl': 1800},                # 30 minutes
        'adaptive:params': {'ttl': 7200},              # 2 hours
        'learning:patterns': {'ttl': 3600}             # 1 hour
    }
    
    def store_trade(self, trade_data):
        """אחסון trade עם compression"""
        key = f"trade:{trade_data['id']}"
        
        # compressed data
        compressed_data = {
            's': trade_data['symbol'],           # symbol
            'p': trade_data['pnl'],              # pnl
            'd': trade_data['duration'],         # duration
            't': trade_data['timestamp']         # timestamp
        }
        
        # store with TTL
        self.redis.hset(key, mapping=compressed_data)
        self.redis.expire(key, 86400)  # 24 hours
        
        # update recent trades list (keep only 50)
        self.redis.lpush('trades:recent', trade_data['id'])
        self.redis.ltrim('trades:recent', 0, 49)
    
    def get_performance_metrics(self):
        """איסוף metrics ללא overload"""
        try:
            recent_trades = self.redis.lrange('trades:recent', 0, 29)  # רק 30 אחרונים
            
            if not recent_trades:
                return {'win_rate': 0.5, 'avg_pnl': 0.0}
            
            wins = 0
            total_pnl = 0.0
            
            for trade_id in recent_trades:
                trade_data = self.redis.hgetall(f"trade:{trade_id}")
                if trade_data:
                    pnl = float(trade_data.get('p', 0))
                    total_pnl += pnl
                    if pnl > 0:
                        wins += 1
            
            win_rate = wins / len(recent_trades) if recent_trades else 0.5
            avg_pnl = total_pnl / len(recent_trades) if recent_trades else 0.0
            
            return {
                'win_rate': win_rate,
                'avg_pnl': avg_pnl,
                'sample_size': len(recent_trades)
            }
            
        except Exception as e:
            # fallback to default
            return {'win_rate': 0.5, 'avg_pnl': 0.0, 'error': str(e)}
```

```python
class MinimalPerformanceTracker:
    def __init__(self, redis_optimizer):
        self.redis = redis_optimizer
        self.metrics = {
            'daily_trades': 0,
            'daily_pnl': 0.0,
            'peak_memory': 0
        }
    
    def update_and_analyze(self, trade_result):
        """עדכון ביצועים עם ניתוח מינימלי"""
        # store trade
        self.redis.store_trade(trade_result)
        
        # update daily counters
        self.metrics['daily_trades'] += 1
        self.metrics['daily_pnl'] += trade_result['pnl']
        
        # adaptive learning - רק אם יש מספיק data
        if self.metrics['daily_trades'] % 10 == 0:  # כל 10 טריידים
            self._adaptive_learning_cycle()
        
        # memory cleanup - כל 20 טריידים
        if self.metrics['daily_trades'] % 20 == 0:
            self._cleanup_old_data()
    
    def _adaptive_learning_cycle(self):
        """מחזור למידה חסכוני"""
        try:
            metrics = self.redis.get_performance_metrics()
            current_win_rate = metrics['win_rate']
            
            # simple adaptive rules
            if current_win_rate > 0.65:
                # increase confidence - winning streak
                self._adjust_parameters('INCREASE_CONFIDENCE', 0.1)
            elif current_win_rate < 0.45:
                # decrease confidence - losing streak  
                self._adjust_parameters('DECREASE_CONFIDENCE', -0.15)
                
        except Exception as e:
            # fail silently - don't crash on learning errors
            print(f"Learning cycle error: {e}")
```

```python
class ResourceAwareExecutor:
    def __init__(self, adaptive_manager):
        self.adaptive_manager = adaptive_manager
        self.concurrency_limiter = asyncio.Semaphore(2)  # רק 2 concurrent
        self.last_execution = 0
        self.min_interval = 1.0  # minimum 1 second between trades
    
    async def execute_adaptive_trade(self, signal):
        """ביצוע trade עם התאמה דינמית"""
        async with self.concurrency_limiter:
            # rate limiting
            await self._rate_limit()
            
            # get market data snapshot (lightweight)
            market_snapshot = await self._get_market_snapshot(signal.symbol)
            
            # calculate dynamic parameters
            dynamic_params = self.adaptive_manager.calculate_dynamic_parameters(market_snapshot)
            
            # prepare trade with adaptive sizing
            trade_order = {
                'symbol': signal.symbol,
                'side': signal.direction,
                'quantity': self._calculate_quantity(signal, dynamic_params),
                'stop_loss': self._calculate_stop_loss(signal, dynamic_params),
                'take_profit': self._calculate_take_profit(signal, dynamic_params),
                'metadata': {
                    'confidence': dynamic_params['confidence'],
                    'regime': self.adaptive_manager.regime_state['current'],
                    'adaptive_size': dynamic_params['size']
                }
            }
            
            # execute trade
            return await self._execute_order(trade_order)
    
    def _calculate_quantity(self, signal, dynamic_params):
        """חישוב כמות עם התאמה דינמית"""
        base_size = self.adaptive_manager.adaptive_params['position_size']
        adaptive_size = base_size * (1 + dynamic_params['confidence'])
        
        # clamp between 1% and 5%
        final_size = max(0.01, min(0.05, adaptive_size))
        
        # convert to actual quantity based on account size
        account_size = self._get_account_balance()
        return (account_size * final_size) / signal.current_price
```

```python
class EfficientContinuousLearner:

    def analyze_trade_patterns(self, recent_trades):
        """ניתוח patterns עם מינימום חישובים"""
        if len(recent_trades) < 15:  # need minimum data
            return
            
        # analyze winning patterns
        winning_trades = [t for t in recent_trades if t['pnl'] > 0]
        losing_trades = [t for t in recent_trades if t['pnl'] < 0]
        
        if not winning_trades or not losing_trades:
            return
        
        # simple pattern extraction - only key features
        win_patterns = self._extract_common_features(winning_trades[:10])  # only 10
        loss_patterns = self._extract_common_features(losing_trades[:10])  # only 10
        
        # store patterns in Redis with compression
        self.redis.store_patterns({
            'win_patterns': win_patterns,
            'loss_patterns': loss_patterns,
            'timestamp': time.time(),
            'sample_size': len(recent_trades)
        })
    
    def _extract_common_features(self, trades):

            'entry_rsi': []
```

```python
class MainAdaptiveSystem:
    def __init__(self):
        # initialize components
        self.redis_optimizer = RedisOptimizer(redis_client)
        self.regime_detector = EfficientRegimeDetector()
        self.adaptive_manager = UltraLightAdaptiveManager()
        self.performance_tracker = MinimalPerformanceTracker(self.redis_optimizer)
        self.execution_engine = ResourceAwareExecutor(self.adaptive_manager)
        self.continuous_learner = EfficientContinuousLearner(self.redis_optimizer)
        
        # system state
        self.is_running = False
        self.trade_count = 0
        
    async def run_adaptive_cycle(self):
        """מחזור מסחר אדפטיבי ראשי"""
        while self.is_running:
            try:
                # 1. detect market regime
                market_data = await self._get_market_data_snapshot()
                regime, adjustment, confidence = self.regime_detector.detect_regime(market_data)
                
                # 2. update adaptive manager
                self.adaptive_manager.regime_state = {
                    'current': regime,
                    'confidence': confidence,
                    'adjustment': adjustment
                }
                
                # 3. check for trade signals
                signals = await self._get_trade_signals()
                
                # 4. execute trades with adaptive parameters
                for signal in signals[:3]:  # max 3 signals per cycle
                    if self._should_accept_signal(signal, confidence):
                        trade_result = await self.execution_engine.execute_adaptive_trade(signal)
                        self.performance_tracker.update_and_analyze(trade_result)
                        self.trade_count += 1
                
                # 5. learning cycle (every 10 trades)
                if self.trade_count % 10 == 0:
                    recent_trades = self.redis_optimizer.get_recent_trades(20)
                    self.continuous_learner.analyze_trade_patterns(recent_trades)
                
                # 6. resource management
                await self._resource_management()
                
                # 7. sleep between cycles
                await asyncio.sleep(5)  # 5 seconds between cycles
                
            except Exception as e:
                print(f"Adaptive cycle error: {e}")
                await asyncio.sleep(10)  # longer sleep on error
```
