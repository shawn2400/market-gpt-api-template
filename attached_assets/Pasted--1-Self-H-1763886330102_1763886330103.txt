# ✅ **שלב 1 — Self-Healing + Freeze Engine (קוד מלא)**

`utils/system_guard.py`

```python
import time
import psutil
import logging
from datetime import datetime
from utils.freeze_manager import freeze_all_systems
from utils.deployment import restart_service, rollback_version
from utils.health_checks import check_api, check_db, check_latency

logger = logging.getLogger("system_guard")


class SystemGuard:

    FAILURE_LIMIT = 3
    CPU_LIMIT = 92
    RAM_LIMIT = 90

    failure_count = 0
    last_recovery = None
    frozen = False

    @classmethod
    def run_cycle(cls):
        if cls.frozen:
            return

        checks = [
            check_api(),
            check_db(),
            check_latency() < 350,
            psutil.cpu_percent() < cls.CPU_LIMIT,
            psutil.virtual_memory().percent < cls.RAM_LIMIT
        ]

        # ✅ הכל תקין
        if all(checks):
            cls.failure_count = 0
            return

        # ❌ תקלה
        cls.failure_count += 1
        logger.error(f"[GUARD] failure {cls.failure_count}/3")

        # 🟡 ניסיון תיקון ראשון
        if cls.failure_count == 1:
            restart_service("auto_executor")
            return

        # 🔴 כשל חוזר
        if cls.failure_count >= cls.FAILURE_LIMIT:
            freeze_all_systems(reason="CRITICAL FAILURE AUTO-PROTECTED")
            cls.frozen = True
            rollback_version()
            cls.last_recovery = datetime.utcnow()
            logger.critical("[GUARD] SYSTEM FROZEN + ROLLBACK EXECUTED")

```

# ✅ **שלב 2 — Freeze Button + API (מוכן להרצה)**

`routes/emergency.py`

```python
from fastapi import APIRouter, Depends
from utils.freeze_manager import freeze_all_systems, unfreeze_all
from utils.auth import require_admin

router = APIRouter(prefix="/emergency", tags=["Emergency"])


@router.post("/freeze")
def freeze(reason: str, _=Depends(require_admin)):
    freeze_all_systems(reason)
    return {"status": "FROZEN", "reason": reason}


@router.post("/unfreeze")
def unfreeze(_=Depends(require_admin)):
    unfreeze_all()
    return {"status": "UNFROZEN"}
```

# ✅ **שלב 3 — Immutable Audit Log (קוד מלא)**

`utils/audit.py`

```python
import hashlib
import json
from datetime import datetime

AUDIT_FILE = "storage/audit.log"
last_hash = None


def write_audit(action: str, actor: str, metadata: dict):
    global last_hash

    entry = {
        "ts": datetime.utcnow().isoformat(),
        "action": action,
        "actor": actor,
        "meta": metadata,
        "prev_hash": last_hash
    }

    raw = json.dumps(entry, sort_keys=True).encode()
    last_hash = hashlib.sha256(raw).hexdigest()
    entry["hash"] = last_hash

    with open(AUDIT_FILE, "a") as f:
        f.write(json.dumps(entry) + "\n")
```

✅ לא ניתן למחיקה
✅ לא ניתן לעריכה
✅ חתום בכל רשומה

# ✅ **שלב 4 — One-Click Restore + Backups**

`utils/backups.py`

```python
import shutil
from datetime import datetime
import os

BACKUP_DIR = "backups"


def create_backup():
    ts = datetime.utcnow().strftime("%Y%m%d_%H%M")
    dst = f"{BACKUP_DIR}/{ts}"
    shutil.copytree("storage", dst)
    return dst


def restore_backup(folder: str):
    shutil.rmtree("storage")
    shutil.copytree(f"{BACKUP_DIR}/{folder}", "storage")
    return True
```

```jsx
import { useEffect, useState } from "react";

export default function SystemStatus() {
  const [data, setData] = useState({});

  useEffect(() => {
    const ws = new WebSocket("wss://YOUR_DOMAIN/ws/system");
    ws.onmessage = (msg) => setData(JSON.parse(msg.data));
    return () => ws.close();
  }, []);

  return (
    <div className="status-box">
      <h2>System Health</h2>
      <p>Cluster: {data.cluster_status}</p>
      <p>Failures: {data.failures}</p>
      <p>Freeze: {data.freeze_active ? "🛑 ACTIVE" : "✅ NORMAL"}</p>
      <button disabled={data.freeze_active}>Freeze Now</button>
    </div>
  );
}
```
