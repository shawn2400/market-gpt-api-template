```python
class MultiLevelProtector:
    def check_and_activate_protections(self):
        win_rate = performance_tracker.get_win_rate()
        consecutive_losses = performance_tracker.get_consecutive_losses()
```

### **4.1 Symbol Tiering Engine**
```python
# מודול חדש: workers/symbol_tiering_engine.py
class SymbolTieringEngine:
    def calculate_tier(self, symbol_stats):
        score = (symbol_stats.win_rate * 0.4 + 
                symbol_stats.volume_score * 0.3 +
                symbol_stats.volatility_score * 0.3)
        
        if score >= 7.5: return "A"
        elif score >= 5.0: return "B" 
        else: return "C"
```

### **4.2 Dynamic Blacklist Manager**
```python
# מודול חדש: workers/blacklist_manager.py
class BlacklistManager:
    def auto_blacklist_symbols(self):
        for symbol, stats in symbol_aggregator.get_all_stats():
            if stats.win_rate < 25 and stats.total_trades >= 5:
                self.blacklist_24h(symbol, f"Win rate too low: {stats.win_rate}%")
```
