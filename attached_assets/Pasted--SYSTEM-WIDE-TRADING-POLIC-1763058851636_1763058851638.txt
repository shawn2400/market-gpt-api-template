### **פקודת אכיפה מרכזית:**
```python
def enforce_trading_policy(symbol, order_type, leverage):
    """אכיפת כל מדיניות המערכת לפני ביצוע trade"""
    
    # 1. בדיקת סמל
    if not is_symbol_approved(symbol):
        return "BLOCKED: Symbol not in approved list"
    
    # 2. בדיקת מינוף לפי Tier
    tier = get_symbol_tier(symbol)
    max_leverage = TIER_LEVERAGE_LIMITS[tier]
    if leverage > max_leverage:
        return f"BLOCKED: Max leverage for {tier} is {max_leverage}x"
    
    # 3. בדיקת ביצועים היסטוריים
    if get_symbol_efficiency(symbol) < MINIMUM_FILL_RATE_REQUIREMENT:
        return "BLOCKED: Low fill rate history"
    
    # 4. בדיקת תנאי שוק
    if is_high_volatility_period():
        return "REDUCED_LEVERAGE: High volatility period"
    
    return "APPROVED"
```
