# -*- coding: utf-8 -*-
from __future__ import annotations
import math

def floor_to_step(value: float, step: float) -> float:
    if step <= 0: return value
    return math.floor(value / step) * step

def round_tick(price: float, tick_size: float) -> float:
    if tick_size <= 0: return price
    return floor_to_step(price, tick_size)

def round_qty(qty: float, step_size: float) -> float:
    if step_size <= 0: return qty
    return floor_to_step(qty, step_size)

# -*- coding: utf-8 -*-
from __future__ import annotations
import time, uuid
from tenacity import retry, stop_after_attempt, wait_exponential_jitter

def _cid(prefix: str) -> str:
    return f"{prefix}_{int(time.time()*1000)}_{uuid.uuid4().hex[:8]}"

@retry(stop=stop_after_attempt(3), wait=wait_exponential_jitter(0.1, 1.0))
def safe_replace_sl(binance, symbol: str, position_side: str, qty: float, new_sl_price: float) -> dict:
    """
    1) Place new SL (ReduceOnly) with unique clientOrderId (idempotent-ish)
    2) Verify it is Working
    3) Cancel old STOP orders
    Avoids window without SL and common -2021 errors.
    """
    side = "SELL" if position_side.upper() == "LONG" else "BUY"
    new_params = dict(
        symbol=symbol,
        side=side,
        type="STOP_MARKET",
        stopPrice=str(new_sl_price),
        reduceOnly=True,
        quantity=str(qty),
        newClientOrderId=_cid("SLNEW"),
        workingType="MARK_PRICE",
    )
    placed = binance.create_futures_order(**new_params)

    time.sleep(0.2)
    open_orders = binance.futures_get_open_orders(symbol=symbol)
    new_id = placed.get("orderId")
    if not any(o.get("orderId") == new_id for o in open_orders):
        raise RuntimeError("New SL not found/working")

    for o in open_orders:
        if o.get("orderId") != new_id and str(o.get("type","")).upper().startswith("STOP"):
            try: binance.futures_cancel_order(symbol=symbol, orderId=o["orderId"])
            except Exception: pass

    return placed
PY
