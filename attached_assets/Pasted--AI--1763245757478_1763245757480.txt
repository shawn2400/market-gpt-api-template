```python
# utils/data_collector.py
import requests
import pandas as pd
from datetime import datetime, timedelta

class BinanceDataCollector:
    def __init__(self):
        self.base_url = "https://api.binance.com/api/v3"
    
    def get_historical_data(self, symbol, interval='5m', limit=1000):
        """מביא נתונים היסטוריים מ-Binance"""
        url = f"{self.base_url}/klines"
        params = {
            'symbol': symbol,
            'interval': interval,
            'limit': limit
        }
        
        try:
            response = requests.get(url, params=params)
            data = response.json()
            
            # המרה ל-DataFrame
            df = pd.DataFrame(data, columns=[
                'open_time', 'open', 'high', 'low', 'close', 'volume',
                'close_time', 'quote_asset_volume', 'number_of_trades',
                'taker_buy_base_asset_volume', 'taker_buy_quote_asset_volume', 'ignore'
            ])
            
            # המרת טיפוסים
            numeric_columns = ['open', 'high', 'low', 'close', 'volume']
            for col in numeric_columns:
                df[col] = pd.to_numeric(df[col])
            
            df['timestamp'] = pd.to_datetime(df['open_time'], unit='ms')
            
            return df
            
        except Exception as e:
            print(f"❌ שגיאה בקבלת נתונים: {e}")
            return None
```

import yfinance as yf

def get_yahoo_data(symbol, period="6mo"):
    """מביא נתונים מ-Yahoo Finance"""
    try:
        ticker = yf.Ticker(f"{symbol}-USD")
        data = ticker.history(period=period, interval="1h")
        return data
    except:
        return None
```

# core/ai_data_manager.py
import asyncio
from utils.data_collector import BinanceDataCollector

class AIDataManager:
    def __init__(self):
        self.collector = BinanceDataCollector()
        self.training_symbols = ['BTCUSDT', 'ETHUSDT', 'ADAUSDT', 'XRPUSDT', 'SOLUSDT']
    
    async def collect_training_data(self):
        """אוסף נתונים מכל הסמלים"""
        all_data = {}
        
        for symbol in self.training_symbols:
            print(f"📥 אוסף נתונים עבור {symbol}...")
            data = self.collector.get_historical_data(symbol, interval='5m', limit=2000)
            
            if data is not None:
                all_data[symbol] = data
                print(f"✅ נאספו {len(data)} רשומות עבור {symbol}")
            else:
                print(f"❌ לא הצלחנו לאסוף נתונים עבור {symbol}")
            
            # המתן בין בקשות כדי לא לעבור rate limits
            await asyncio.sleep(0.5)
        
        return all_data
    
    def prepare_training_dataset(self, raw_data):
        """מכין את הנתונים לאימון"""
        features_list = []
        labels_list = []
        
        for symbol, data in raw_data.items():
            if data is None or len(data) < 100:
                continue
            
            # יצירת features בסיסיים
            data['returns'] = data['close'].pct_change()
            data['volume_sma'] = data['volume'].rolling(20).mean()
            data['price_sma'] = data['close'].rolling(20).mean()
            data['rsi'] = self.calculate_rsi(data['close'])
            
            # יצירת labels (רווח/הפסד בעתיד)
            data['future_return'] = data['close'].shift(-5) / data['close'] - 1
            
            # סינון נתונים תקינים
            data = data.dropna()
            
            # הכנת features
            for i in range(len(data) - 1):
                features = {
                    'current_return': data['returns'].iloc[i],
                    'volume_ratio': data['volume'].iloc[i] / data['volume_sma'].iloc[i],
                    'price_vs_sma': data['close'].iloc[i] / data['price_sma'].iloc[i],
                    'rsi_value': data['rsi'].iloc[i],
                    'symbol': hash(symbol) % 100  # encoding סמל
                }
                
                label = 1 if data['future_return'].iloc[i] > 0.002 else 0  # 0.2%
                
                features_list.append(features)
                labels_list.append(label)
        
        return pd.DataFrame(features_list), pd.Series(labels_list)
```

### **שלב 2: AI בסיסי שיכול לעבוד בלי נתונים רבים**
```python
# core/basic_ai.py
import pandas as pd
import numpy as np
from sklearn.ensemble import RandomForestClassifier
from sklearn.preprocessing import StandardScaler

class BasicTradingAI:
    """
    AI בסיסי שיכול לעבוד גם עם נתונים מוגבלים
    """
    
    def __init__(self):
        self.model = RandomForestClassifier(
            n_estimators=20,  # מאוד קל
            max_depth=5,      # מאוד קל  
            random_state=42
        )
        self.scaler = StandardScaler()
        self.is_trained = False
        self.default_confidence = 0.6
    
    def train_with_minimal_data(self, market_data_list):
        """
        מאמן עם נתונים מינימליים - יכול לעבוד גם עם 100-200 דגימות
        """
        try:
            if len(market_data_list) < 50:
                print("⚠️ נתונים מוגבלים - AI ישתמש בחיזוי בסיסי")
                return False
            
            # יצירת features בסיסיים
            features = []
            labels = []
            
            for data in market_data_list:
                # features בסיסיים מהנתונים הנוכחיים
                feature_vector = [
                    data.get('rsi', 50) / 100,  # normalize
                    data.get('volume_ratio', 1),
                    data.get('trend_strength', 0),
                    1 if data.get('above_ema', False) else 0
                ]
                
                # label - האם זה היה trade מוצלח (מבוסס על היסטוריה)
                # אם אין נתונים - נשתמש בחיזוי בסיסי
                successful_trade = data.get('successful', True)  # ברירת מחדל
                
                features.append(feature_vector)
                labels.append(1 if successful_trade else 0)
            
            # אימון
            X = np.array(features)
            y = np.array(labels)
            
            X_scaled = self.scaler.fit_transform(X)
            self.model.fit(X_scaled, y)
            self.is_trained = True
            
            print(f"✅ AI בסיסי אומן עם {len(X)} דגימות")
            return True
            
        except Exception as e:
            print(f"❌ שגיאה באימון AI בסיסי: {e}")
            return False
    
    def predict_simple_signal(self, market_data):
        """חיזוי בסיסי - עובד גם בלי אימון"""
        if not self.is_trained:
            return self._get_fallback_signal(market_data)
        
        try:
            # features בסיסיים
            features = np.array([[
                market_data.get('rsi', 50) / 100,
                market_data.get('volume_ratio', 1),
                market_data.get('trend_strength', 0),
                1 if market_data.get('above_ema', False) else 0
            ]])
            
            features_scaled = self.scaler.transform(features)
            prediction = self.model.predict_proba(features_scaled)[0]
            confidence = float(prediction[1])
            
            # החלטה פשוטה
            if confidence > 0.6:
                direction = 'long'
            elif confidence < 0.4:
                direction = 'short'
            else:
                direction = 'hold'
            
            return {
                'direction': direction,
                'confidence': confidence,
                'ai_used': True,
                'type': 'basic_ai'
            }
            
        except Exception as e:
            print(f"❌ שגיאה בחיזוי AI: {e}")
            return self._get_fallback_signal(market_data)
    
    def _get_fallback_signal(self, market_data):
        """חיזוי נפילה - כאשר AI לא מאומן"""
        rsi = market_data.get('rsi', 50)
        
        if rsi < 35:
            direction = 'long'
            confidence = 0.7
        elif rsi > 65:
            direction = 'short' 
            confidence = 0.7
        else:
            direction = 'hold'
            confidence = 0.5
        
        return {
            'direction': direction,
            'confidence': confidence,
            'ai_used': False,
            'type': 'fallback_rsi'
        }
```
