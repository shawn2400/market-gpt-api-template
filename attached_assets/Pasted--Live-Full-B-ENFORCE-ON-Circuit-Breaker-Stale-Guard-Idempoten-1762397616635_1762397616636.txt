### 1.1 `utils/regime_detector_v2.py`

```python
# -*- coding: utf-8 -*-
from __future__ import annotations
from dataclasses import dataclass
from typing import Dict

@dataclass
class RegimeResult:
    regime: str          # "TRENDING"|"CHOPPY"|"VOLATILE"|"SIDEWAYS"
    confidence: float    # 0..1

def detect_market_regime_v2(features: Dict[str, float]) -> RegimeResult:
    atr_pct = float(features.get("atr_pct", 0.0))
    adx     = float(features.get("adx", 0.0))
    macd_s  = float(features.get("macd_slope", 0.0))
    rsi     = float(features.get("rsi", 50.0))

    trending = min(max((adx/48.0)*0.55 + (abs(macd_s)/5.0)*0.35 + (0.1 if (rsi<35 or rsi>65) else 0.0), 0), 1)
    volatile = min(max(atr_pct/90.0, 0), 1)
    choppy   = max(0.0, 0.9 - trending - 0.3*volatile)
    sideways = max(0.0, 1.0 - max(trending, volatile, choppy))
    scores = {"TRENDING": trending, "VOLATILE": volatile, "CHOPPY": choppy, "SIDEWAYS": sideways}
    regime = max(scores, key=scores.get)
    conf   = float(scores[regime])
    ordered = sorted(scores.values(), reverse=True)
    if len(ordered) > 1 and (ordered[0]-ordered[1]) < 0.12:
        conf *= 0.75
    return RegimeResult(regime=regime, confidence=conf)
```

### 1.2 `utils/adaptive_mixer.py`

```python
# -*- coding: utf-8 -*-
from __future__ import annotations
from typing import Dict

def _clamp(v, lo, hi): return max(lo, min(hi, v))

def adaptive_mix(regime: str, confidence: float, atr_pct: float, pnl_state: str, time_in_pos_min: float) -> Dict[str, float]:
    # בסיס לפי Regime
    base = {
        "TRENDING": {"sl_atr": 1.8, "tp_rr": 2.3, "trail": True,  "tp_ladder": [1.8, 3.2, 4.8]},
        "CHOPPY":   {"sl_atr": 1.1, "tp_rr": 1.5, "trail": False, "tp_ladder": [1.6, 2.6]},
        "VOLATILE": {"sl_atr": 2.2, "tp_rr": 2.1, "trail": True,  "tp_ladder": [2.0, 3.6]},
        "SIDEWAYS": {"sl_atr": 1.3, "tp_rr": 1.4, "trail": False, "tp_ladder": [1.4, 2.2]},
    }.get(regime, {"sl_atr": 1.5, "tp_rr": 1.8, "trail": False, "tp_ladder": [1.6, 2.6]})

    k = _clamp(confidence, 0.2, 1.0)
    v = _clamp(atr_pct/100.0, 0.3, 1.2)

    # התאמות לפי ביטחון/תנודתיות
    base["sl_atr"] *= (1.0 + (1.0-k)*0.28) * (1.0 + (v-0.6)*0.35)
    base["tp_rr"]  *= (0.95 + 0.15*k)
    base["sl_atr"] = _clamp(base["sl_atr"], 0.9, 3.0)
    base["tp_rr"]  = _clamp(base["tp_rr"], 1.3, 3.2)

    # התאמות לפי מצב PnL
    if pnl_state in ("drawdown", "cooldown"):
        base["tp_rr"] *= 0.9
        base["sl_atr"] *= 0.95

    # התאמות לפי זמן בפוזיציה
    if time_in_pos_min > 120:
        base["trail"] = True

    return base
```

### 1.3 `utils/precision.py` (אם חסר)

```python
# -*- coding: utf-8 -*-
import math
def quantize_price(p: float, tick: float) -> float:
    return p if tick<=0 else math.floor(p/tick)*tick
def quantize_qty(q: float, step: float) -> float:
    return q if step<=0 else math.floor(q/step)*step
```

### 1.4 `utils/idempotency.py`

```python
# -*- coding: utf-8 -*-
import time, hashlib, hmac, os
WINDOW_SEC = int(os.getenv("IDEMP_WINDOW_SEC", "90"))
SECRET = os.getenv("IDEMP_SECRET", "demo_secret_change")
_cache = {}

def make_key(route: str, payload: str) -> str:
    msg = f"{route}|{payload}|{int(time.time()/WINDOW_SEC)}".encode()
    return hmac.new(SECRET.encode(), msg, hashlib.sha256).hexdigest()

def seen(key: str) -> bool:
    t = int(time.time()/WINDOW_SEC)
    _cache.setdefault(t, set())
    # נקה דלי ישן
    for old in list(_cache.keys()):
        if old < t-2: _cache.pop(old, None)
    if key in _cache[t]: return True
    _cache[t].add(key); return False
```

### 1.5 `utils/circuit_breaker.py`

```python
# -*- coding: utf-8 -*-
import time, os
MAX_FAILS   = int(os.getenv("CB_MAX_FAILS", "5"))
COOLDOWN_S  = int(os.getenv("CB_COOLDOWN_SEC", "900"))
_state = {"fails": 0, "until": 0}

def track(ok: bool) -> None:
    global _state
    now = time.time()
    if now < _state["until"]: return
    if ok: _state["fails"] = 0
    else:
        _state["fails"] += 1
        if _state["fails"] >= MAX_FAILS:
            _state["until"] = now + COOLDOWN_S

def allow() -> bool:
    return time.time() >= _state["until"]
```

**בתוך `manage_open_trades()` — לפני נתיב legacy, הוסף נתיב דינמי (ENFORCE חי):**

```python
now = time.time()
if MANAGER_DYN_PATH:
    try:
        # context חייב להכיל שדות אלה (אם חסרים – נעשה skip):
        required = ["symbol","side","entry_price","atr","atr_pct","adx","macd_slope","rsi","tick_size","step_size","position_qty","last_market_update_ts"]
        if any(k not in context for k in required):
            dyn_skips.labels(reason="missing_context").inc()
        elif (now - float(context["last_market_update_ts"])) > DYN_SAFE_STALE_SEC:
            age_guard_hit.inc(); dyn_skips.labels(reason="stale_data").inc()
        elif BTC_GATE_ENABLE and not bool(context.get("btc_gate_ok", True)):
            dyn_skips.labels(reason="btc_gate").inc()
        elif not cb_allow():
            cb_blocks.inc(); dyn_skips.labels(reason="circuit_block").inc()
        else:
            feats = {
                "atr_pct": float(context["atr_pct"]),
                "adx": float(context["adx"]),
                "macd_slope": float(context["macd_slope"]),
                "rsi": float(context["rsi"]),
            }
            r = detect_market_regime_v2(feats)
            if r.confidence < DYN_MIN_CONF:
                conf_low_hit.inc(); dyn_skips.labels(reason="low_conf").inc()
            else:
                entry = float(context["entry_price"])
                atr   = float(context["atr"])
                side  = context["side"]
                regime = r.regime
                # time in position (דקות) — אם יש, אחרת 0
                tip_min = float(context.get("time_in_position_min", 0.0))
                pnl_state = str(context.get("pnl_state", "normal"))
                mix = adaptive_mix(regime, r.confidence, feats["atr_pct"], pnl_state, tip_min)

                sl_dist = mix["sl_atr"] * atr
                sl_p = (entry - sl_dist) if side=="LONG" else (entry + sl_dist)
                rr   = mix["tp_rr"]
                tp_p = (entry + rr*sl_dist) if side=="LONG" else (entry - rr*sl_dist)

                qtz = quantize_qty(abs(float(context["position_qty"])), float(context["step_size"]))
                sl_p = quantize_price(sl_p, float(context["tick_size"]))
                tp_p = quantize_price(tp_p, float(context["tick_size"]))

                # Idempotency – מונע כפילות
                payload = json.dumps({"sym":context["symbol"],"sl":sl_p,"tp":tp_p,"qty":qtz,"side":side,"reg":regime,"conf":r.confidence}) 
                idem_key = make_key("manage_dyn", payload)
                if seen(idem_key):
                    dyn_skips.labels(reason="idem_dup").inc()
                else:
                    if DYN_SHADOW and not DYN_ENFORCE:
                        dyn_decisions.labels(symbol=context["symbol"], regime=regime).inc()
                        print(json.dumps({"evt":"dyn_shadow","sym":context["symbol"],"regime":regime,"conf":round(r.confidence,3),"sl_atr":round(mix["sl_atr"],3),"tp_rr":round(mix["tp_rr"],3),"side":side,"sl_p":sl_p,"tp_p":tp_p,"qty":qtz}, ensure_ascii=False))
                        cb_track(ok=True)
                    else:
                        # === LIVE ENFORCE ===
                        ok1 = _sl_manager.safe_replace_sl(symbol=context["symbol"], new_stop_price=sl_p, qty=qtz, side=side)
                        if ok1: sl_changes.labels(symbol=context["symbol"]).inc()
                        ok2 = _tp_ladder.set_tp_ladder(context["symbol"], entry, qtz, side, mix.get("tp_ladder", []), reduce_only=True, post_only=True)
                        if ok2: tp_sets.labels(symbol=context["symbol"]).inc()
                        dyn_decisions.labels(symbol=context["symbol"], regime=regime).inc()
                        cb_track(ok=(ok1 and ok2))
    except Exception as e:
        dyn_errors.labels(stage="manage_dyn").inc()
        cb_track(ok=False)
        print(json.dumps({"evt":"dyn_error","err":str(e),"sym":context.get("symbol")}, ensure_ascii=False))
# legacy path ממשיך אחר כך ללא שינוי
```
