# 🚀 **שלב 1 — קובץ יחיד להפעלה: `full-bootstrap.sh`**

העתק-הדבק — זה *הכול*:

```bash
#!/bin/bash
set -e

echo "✅ Starting FULL AUTO INSTALL (AlgoGPT + OLLaMA + Dashboard)"

### --- 1) SYSTEM PREP ---
apt-get update -y
apt-get install -y curl wget git ufw

### --- 2) ENABLE FIREWALL ---
ufw allow ssh
ufw allow 10000   # AlgoGPT API
ufw allow 3000    # Grafana
ufw allow 11434   # OLLaMA
ufw --force enable

### --- 3) INSTALL DOCKER ---
if ! command -v docker >/dev/null; then
  curl -fsSL https://get.docker.com | sh
  systemctl enable docker
fi

### --- 4) CREATE DEPLOY DIR ---
mkdir -p /opt/algogpt
cd /opt/algogpt

### --- 5) FETCH COMPOSE FILE ---
cat > docker-compose.yml << 'EOF'
version: "3.9"

services:
  algogpt:
    image: algogpt/core:latest
    restart: always
    environment:
      AUTO_RUN: "1"
      EXECUTE_TRADES: "0"
      MIN_QUALITY_SCORE: "8.5"
      TZ: Asia/Jerusalem

      AUTO_SCALE: "on"
      AUTO_RECOVERY: "on"
      AUTO_EXPAND: "on"
      SELF_HEALING: "on"

      HA_CLUSTER: "standby"
      MULTI_ACCOUNT: "standby"

    ports:
      - "10000:10000"
    volumes:
      - ./data:/app/data
      - ./backups:/app/backups

  ollama:
    image: ollama/ollama:latest
    restart: always
    ports:
      - "11434:11434"
    volumes:
      - ./ollama:/root/.ollama

  dashboard:
    image: algogpt/dashboard:latest
    restart: always
    environment:
      API_URL: "http://algogpt:10000"
      OLLAMA_URL: "http://ollama:11434"
    ports:
      - "8080:8080"

volumes:
  data:
EOF

### --- 6) START SYSTEM ---
docker compose pull
docker compose up -d

### --- 7) LOAD BASE OLLAMA MODEL ---
curl -X POST http://localhost:11434/api/pull -d '{"name":"llama3"}'

echo ""
echo "✅ INSTALL COMPLETE"
echo "🌍 AlgoGPT:      http://<SERVER-IP>:10000/readyz"
echo "🤖 Dashboard:    http://<SERVER-IP>:8080"
echo "🧠 OLLaMA API:   http://<SERVER-IP>:11434"
echo "⚡ Auto-Scale + Auto-Recovery READY"
```
