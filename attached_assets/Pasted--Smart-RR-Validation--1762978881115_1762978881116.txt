```python
# 🎯 SMART RR VALIDATION - REPLACEMENT CODE
def validate_risk_reward(rr_ratio, ai_consensus=None, symbol=""):
    """
    Smart RR Validation עם hard floor + AI consensus
    מחליף את ה-RR check הקיים
    """
    
    # HARD FLOOR - אף trade מתחת ל-0.8
    if rr_ratio < 0.8:
        print(f"❌ HARD FLOOR REJECT: {symbol} RR={rr_ratio} < 0.8")
        return False, "RR below hard floor"
    
    # AUTO APPROVE - מעל 1.15
    if rr_ratio >= 1.15:
        print(f"✅ AUTO APPROVE: {symbol} RR={rr_ratio} ≥ 1.15")
        return True, "RR auto approved"
    
    # CONSENSUS ZONE - 0.8 עד 1.15
    if 0.8 <= rr_ratio < 1.15:
        if ai_consensus and check_ai_consensus(ai_consensus):
            print(f"🎯 CONSENSUS APPROVE: {symbol} RR={rr_ratio} - 3/5 AI agents approved")
            return True, "RR approved by AI consensus"
        else:
            print(f"❌ CONSENSUS REJECT: {symbol} RR={rr_ratio} - insufficient AI approval")
            return False, "RR needs AI consensus"
    
    # Fallback
    return False, "RR validation failed"

def check_ai_consensus(ai_consensus):
    """
    בודק אם יש consensus של 3/5 סוכני AI
    """
    if not ai_consensus or len(ai_consensus) < 5:
        return False
    
    approve_count = sum(1 for agent in ai_consensus if agent.get('approve', False))
    return approve_count >= 3
```
