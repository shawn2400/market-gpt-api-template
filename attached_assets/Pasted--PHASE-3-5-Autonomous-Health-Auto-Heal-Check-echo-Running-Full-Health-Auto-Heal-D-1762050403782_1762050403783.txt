# === PHASE 3.5: Autonomous Health & Auto-Heal Check ===
echo "🩺 Running Full Health & Auto-Heal Diagnostic..."

# --- בסיס נתונים ---
BASE_URL="https://algogpt-docker.onrender.com"
BEARER="$API_BEARER_TOKEN"
REPO="shawn2400/algogpt"
LOG_FILE="/app/logs/health_check_$(date +%Y%m%d).log"

# --- שלב 1: בדיקת שירותים ---
echo "🔎 Checking system endpoints..." | tee -a "$LOG_FILE"
for path in readyz healthz metrics version; do
  code=$(curl -s -o /dev/null -w "%{http_code}" -H "Authorization: Bearer $BEARER" "$BASE_URL/$path")
  echo "$path → HTTP $code" | tee -a "$LOG_FILE"
done

# --- שלב 2: מדידת עומסים ---
echo "⚙️ Checking CPU & RAM usage..." | tee -a "$LOG_FILE"
CPU=$(grep 'cpu ' /proc/stat | awk '{usage=($2+$4)*100/($2+$4+$5)} END {print usage "%"}')
RAM=$(free -m | awk '/Mem:/ {print $3 "MB / " $2 "MB"}')
LOAD=$(cat /proc/loadavg | awk '{print $1","$2","$3}')
echo "CPU=$CPU | RAM=$RAM | Load=$LOAD" | tee -a "$LOG_FILE"

# --- שלב 3: בדיקת חתימות ---
echo "🔐 Checking signature validity..." | tee -a "$LOG_FILE"
SIG_CHECK=$(curl -s -X POST "$BASE_URL/healthz" \
  -H "Authorization: Bearer $BEARER" \
  -H "X-Request-Timestamp: $(date +%s)" \
  -H "X-Request-Nonce: cli-$(openssl rand -hex 6)" \
  -H "X-Signature: deadbeef" -d '{"ping":"pong"}' | grep -o '"ok"')
if [[ "$SIG_CHECK" == *"true"* ]]; then
  echo "✅ Signature verification OK" | tee -a "$LOG_FILE"
else
  echo "⚠️ Signature issue detected – triggering auto-heal" | tee -a "$LOG_FILE"
  bash scripts/auto_executor_restart.sh || true
fi

# --- שלב 4: Snapshot ל-GitHub ---
echo "💾 Saving daily snapshot to GitHub..." | tee -a "$LOG_FILE"
bash scripts/sync_to_github_api.sh || echo "⚠️ Snapshot failed" | tee -a "$LOG_FILE"

# --- שלב 5: דיווח לטלגרם / AI X ---
if [[ -n "$TELEGRAM_BOT_TOKEN" && -n "$TELEGRAM_CHAT_ID" ]]; then
  MSG="🩺 AlgoGPT Daily Health\nCPU: $CPU\nRAM: $RAM\nLoad: $LOAD\nSignature: OK"
  curl -s -X POST "https://api.telegram.org/bot${TELEGRAM_BOT_TOKEN}/sendMessage" \
    -d "chat_id=${TELEGRAM_CHAT_ID}" -d "text=${MSG}" >/dev/null
fi

if [[ -n "$AIX_SUPERVISOR_TOKEN" ]]; then
  curl -s -X POST "https://api.x.ai/v1/logs" \
    -H "Authorization: Bearer $AIX_SUPERVISOR_TOKEN" \
    -H "Content-Type: application/json" \
    -d "{\"event\":\"daily_health_report\",\"cpu\":\"$CPU\",\"ram\":\"$RAM\",\"load\":\"$LOAD\"}" >/dev/null
fi

echo "✅ Health check complete. Results logged to $LOG_FILE"
echo "📊 System ready for continuous operation."
