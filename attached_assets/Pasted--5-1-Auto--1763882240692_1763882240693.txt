# ✅ 1) AUTO-HEDGE SCALING לפי גודל חשיפה

### 📌 עדכון לקובץ: `utils/auto_hedger.py`

```python
class AutoHedger:
    def __init__(self):
        self.active = False
        self.min_atr_spike = 2.5
        self.base_hedge = 0.25   # מינימום 25%
        self.max_hedge = 0.75    # מקסימום 75%

    def calc_dynamic_hedge(self, exposure_usd):
        if exposure_usd < 100:
            return self.base_hedge
        if exposure_usd > 2000:
            return self.max_hedge
        return self.base_hedge + (exposure_usd / 2000) * (self.max_hedge - self.base_hedge)

    def check_and_hedge(self, symbol, position, price, atr):
        exposure = position.size * price
        hedge_ratio = self.calc_dynamic_hedge(exposure)

        if self.active:
            return

        if atr_spike(atr, threshold=self.min_atr_spike):
            self.open_hedge(symbol, position, hedge_ratio)
            self.active = True

    def open_hedge(self, symbol, position, hedge_ratio):
        side = "SELL" if position.side == "LONG" else "BUY"
        qty = position.size * hedge_ratio

        futures_market_order(
            symbol=symbol,
            side=side,
            quantity=qty,
            reduce_only=False
        )
```

### ✅ מה זה נותן בפועל

✔ חשיפה קטנה → הגנה קטנה
✔ חשיפה גדולה → הגנה עולה אוטומטית
✔ אין סכנת over-hedge
✔ 0 התערבות ידנית

# ✅ 2) CROSS-EXCHANGE BALANCING (BINANCE ↔ BYBIT)

### 📌 מודול חדש: `utils/cross_balancer.py`

```python
from utils.account_router import router
from utils.binance_client import get_positions as binance_pos
from utils.bybit_client import get_positions as bybit_pos

THRESHOLD = 0.22  # 22% סטייה

def rebalance():
    b = binance_pos()
    y = bybit_pos()

    for sym in b:
        if sym not in y:
            continue

        diff = abs(b[sym].size - y[sym].size) / max(b[sym].size, 1)

        if diff > THRESHOLD:
            adjust(sym, b[sym], y[sym])

def adjust(symbol, binance_p, bybit_p):
    if binance_p.size > bybit_p.size:
        router.broadcast(lambda c: c.reduce(symbol, qty=binance_p.size - bybit_p.size))
    else:
        router.broadcast(lambda c: c.reduce(symbol, qty=bybit_p.size - binance_p.size))
```

### ✅ מה זה נותן

✔ סיכון לא מרוכז בבורסה אחת
✔ הגנות מפני קריסה צדית (FTX-style)
✔ יישור חשיפה אוטומטי

# ✅ 4) INSURANCE-MODE (HEDGE + REDUCE EXPOSURE)

### 📌 קובץ חדש: `utils/insurance_mode.py`

```python
from utils.auto_hedger import AutoHedger
from utils.binance_client import futures_reduce_only

class InsuranceMode:
    def __init__(self):
        self.triggered = False
        self.hedger = AutoHedger()

    def activate(self, position, price, atr):
        if self.triggered:
            return

        self.hedger.check_and_hedge(position.symbol, position, price, atr)

        futures_reduce_only(
            symbol=position.symbol,
            quantity=position.size * 0.25
        )

        self.triggered = True

    def release(self, position):
        if not self.triggered:
            return

        futures_reduce_only(
            symbol=position.symbol,
            quantity=position.size * -0.25
        )

        self.triggered = False
```

### ✅ מה זה נותן

✔ גם hedge וגם הקטנת פוזיציה
✔ בולע תנודות חדות בלי לברוח מהשוק
✔ חוזר לגודל רגיל אוטומטית

# ✅ 5) REAL-TIME FRAUD / ANOMALY DETECTOR

### 📌 קובץ חדש: `utils/anomaly_detector.py`

```python
import numpy as np
from utils.metrics import get_latency, get_order_reject_rate

class AnomalyDetector:
    def __init__(self):
        self.latency_window = []
        self.reject_window = []

    def track(self):
        self.latency_window.append(get_latency())
        self.reject_window.append(get_order_reject_rate())

        if len(self.latency_window) > 20:
            self.latency_window.pop(0)
            self.reject_window.pop(0)

        return self.detect()

    def detect(self):
        if np.mean(self.latency_window[-5:]) > np.mean(self.latency_window) * 2:
            return "LATENCY_ATTACK"

        if np.mean(self.reject_window[-5:]) > 0.35:
            return "ORDER_REJECTION_SPIKE"

        return None
```
