### **שלב 1: הוסף את הקוד ל-`workers/gpt_auto_suggest.py`**

```python
# הוסף בחלק העליון של הקובץ
import requests
import os
from datetime import datetime

class TelegramNotifier:
    def __init__(self):
        self.bot_token = os.getenv('TELEGRAM_BOT_TOKEN')
        self.chat_id = os.getenv('TELEGRAM_CHAT_ID')
        self.base_url = f"https://api.telegram.org/bot{self.bot_token}"
    
    def send_trade_alert(self, trade_data):
        """שלח התראת trade לטלגרם"""
        message = f"""
🎯 **בוצע Trade חדש**
────────────────
• סימבול: {trade_data['symbol']}
• פעולה: {trade_data['action']}
• מחיר: {trade_data['price']}
• כמות: {trade_data['amount']}
• אסטרטגיה: {trade_data['strategy']}
• זמן: {trade_data['timestamp']}
────────────────
        """
        
        try:
            response = requests.post(
                f"{self.base_url}/sendMessage",
                json={
                    "chat_id": self.chat_id,
                    "text": message,
                    "parse_mode": "Markdown"
                },
                timeout=10
            )
            return response.status_code == 200
        except Exception as e:
            print(f"❌ שליחת התראה נכשלה: {e}")
            return False
    
    def send_error_alert(self, error_msg):
        """שלח התראת error"""
        message = f"❌ **שגיאה במערכת:** {error_msg}"
        self._send_message(message)
    
    def send_signal_alert(self, signal_data):
        """שלח התראת signal חדש"""
        message = f"""
🔔 **התגלה Signal חדש**
────────────────
• סימבול: {signal_data['symbol']}
• סוג: {signal_data['type']}
• ביטחון: {signal_data['confidence']}%
• זמן: {signal_data['timestamp']}
────────────────
        """
        self._send_message(message)
    
    def _send_message(self, text):
        """פונקציית עזר לשליחת הודעות"""
        try:
            requests.post(
                f"{self.base_url}/sendMessage",
                json={
                    "chat_id": self.chat_id,
                    "text": text,
                    "parse_mode": "Markdown"
                }
            )
        except Exception as e:
            print(f"❌ שליחת הודעה נכשלה: {e}")

# צור instance גלובלי
telegram_notifier = TelegramNotifier()
```

### **שלב 2: עדכן את פונקציית ה-trade execution**

```python
# חפש את הפונקציה שמבצעת trades והוסף:

def execute_trade_with_notification(trade_signal, exchange_api):
    """
    בצע trade ושלוח התראה
    """
    try:
        # בצע את ה-trade כרגיל
        trade_result = exchange_api.execute_order(
            symbol=trade_signal.symbol,
            side=trade_signal.action,
            quantity=trade_signal.amount,
            price=trade_signal.price
        )
        
        # הכין נתונים להתראה
        trade_data = {
            'symbol': trade_signal.symbol,
            'action': trade_signal.action,
            'price': trade_result.get('fill_price', trade_signal.price),
            'amount': trade_result.get('executed_qty', trade_signal.amount),
            'strategy': getattr(trade_signal, 'strategy_name', 'Auto Scanner'),
            'timestamp': datetime.now().strftime("%Y-%m-%d %H:%M:%S")
        }
        
        # שלח התראה לטלגרם
        telegram_notifier.send_trade_alert(trade_data)
        
        return trade_result
        
    except Exception as e:
        # שלח התראת error
        error_msg = f"Trade failed for {trade_signal.symbol}: {str(e)}"
        telegram_notifier.send_error_alert(error_msg)
        raise e
```

## 🛠️ **איך להשיג את ה-Token ו-Chat ID:**

### **1. צור Bot חדב:**
- חפש `@BotFather` בטלגרם
- שלח `/newbot`
- תן שם והקבל את ה-`BOT_TOKEN`

### **2. קבל את ה-Chat ID:**
- שלח הודעה ל-Bot שיצרת
- גלוש ל: `https://api.telegram.org/bot<YOUR_BOT_TOKEN>/getUpdates`
- חפש את ה-`chat.id` בתגובה
