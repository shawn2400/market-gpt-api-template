class SentimentIntelligence:
    def __init__(self):
        self.sources = [
            'twitter', 'reddit', 'telegram', 'discord',
            'news_sites', 'youtube', 'tradingview', 'coindesk'
        ]
        self.sentiment_engine = SentimentAnalysisEngine()
    
    def analyze_market_sentiment(self, symbol: str) -> dict:
        """ניתוח סנטימנט משוקלל מרובה מקורות"""
        sentiment_scores = {}
        
        for source in self.sources:
            try:
                data = self.fetch_sentiment_data(source, symbol)
                score = self.sentiment_engine.analyze(data)
                sentiment_scores[source] = score
            except Exception as e:
                self.logger.warning(f"⚠️ שגיאה בניתוח סנטימנט מ-{source}: {e}")
        
        return {
            'composite_score': np.mean(list(sentiment_scores.values())),
            'source_breakdown': sentiment_scores,
            'momentum_indicator': self.calculate_sentiment_momentum(sentiment_scores),
            'extremeness_alert': self.detect_sentiment_extremes(sentiment_scores)
        }

class PortfolioOptimizer:

### 7. **ניתוח Correlation Matrix דינאמי**

```python
class DynamicCorrelationEngine:
    def __init__(self):
        self.correlation_matrix = {}
        self.rolling_correlations = {}
    
    def calculate_dynamic_correlations(self, portfolio: list) -> dict:
        """חישוב קורלציות דינאמיות עם regime detection"""
        correlations = {}
        
        for i, asset1 in enumerate(portfolio):
            for j, asset2 in enumerate(portfolio):
                if i < j:
                    corr_data = self.calculate_rolling_correlation(asset1, asset2)
                    correlations[f"{asset1}_{asset2}"] = {
                        'current_correlation': corr_data['current'],
                        'rolling_correlation': corr_data['rolling'],
                        'correlation_regime': self.detect_correlation_regime(corr_data),
                        'regime_stability': self.assess_regime_stability(corr_data),
                        'breakout_alert': self.detect_correlation_breakout(corr_data)
                    }
        
        return {
            'correlation_matrix': correlations,
            'portfolio_beta': self.calculate_portfolio_beta(correlations),
            'diversification_score': self.calculate_diversification_score(correlations),
            'concentration_risk': self.assess_concentration_risk(correlations)
        }
```

class AdvancedBacktester:
    def __init__(self):
        self.walk_forward = WalkForwardAnalyzer()
        self.monte_carlo = MonteCarloSimulator()
        self.strategy_robustness = StrategyRobustnessTester()
    
    def comprehensive_backtest(self, strategy: dict, data: dict) -> dict:
        """באקסטיסטינג מקיף עם ולידציה מתקדמת"""
        results = {}
        
        # Walk-Forward Analysis
        results['walk_forward'] = self.walk_forward.analyze(strategy, data)
        
        # Monte Carlo Simulations
        results['monte_carlo'] = self.monte_carlo.simulate(strategy, data, iterations=1000)
        
        # Strategy Robustness Testing
        results['robustness'] = self.strategy_robustness.test(strategy, data)
        
        # Regime-Based Performance
        results['regime_performance'] = self.analyze_regime_performance(strategy, data)
        
        return {
            'performance_metrics': results,
            'strategy_quality_score': self.calculate_strategy_quality(results),
            'optimization_recommendations': self.generate_optimization_recommendations(results),
            'live_trading_probability': self.calculate_success_probability(results)
        }
