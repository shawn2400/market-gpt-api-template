### **1. 🧠 Breakeven State Management - הכי חשוב!**
```python
class BreakevenStateManager:
    def __init__(self):
        self.breakeven_states = {}  # {symbol: {'last_notification': timestamp, 'status': 'SENT'}}
        self.notification_cooldown = 300  # 5 דקות בין התראות
    
    async def should_send_breakeven_notification(self, symbol):
        """מניעת הצפת הודעות - זיכרון מצבים"""
        current_time = time.time()
        
        if symbol in self.breakeven_states:
            last_notification = self.breakeven_states[symbol]['last_notification']
            if current_time - last_notification < self.notification_cooldown:
                return False  # עדיין בקולאון
        
        self.breakeven_states[symbol] = {
            'last_notification': current_time,
            'status': 'SENT'
        }
        return True
```

### **3. 🧹 Automatic Cleanup Worker - חדש!**
```python
class AutoCleanupWorker:
    def __init__(self):
        self.cleanup_interval = 300  # 5 דקות
    
    async def start_auto_cleanup(self):
        """ניקוי אוטומטי של orphaned orders"""
        while True:
            try:
                # 1. נקה orphaned orders
                orphaned_count = await self.cleanup_orphaned_orders()
                
                # 2. נקה orders ישנים מדי
                old_orders_count = await self.cleanup_old_orders()
                
                # 3. נקה state cache ישן
                await self.cleanup_old_states()
                
                if orphaned_count > 0 or old_orders_count > 0:
                    logger.info(f"🧹 Auto-cleanup: {orphaned_count} orphaned, {old_orders_count} old orders")
                
                await asyncio.sleep(self.cleanup_interval)
                
            except Exception as e:
                logger.error(f"❌ Auto-cleanup failed: {e}")
                await asyncio.sleep(60)
```

```python
class SmartOrderRetry:
    async def smart_retry_order(self, original_order, reason):
        """ניסיון חוזר חכם ל-order שנכשל"""
        
        retry_strategies = {
            'PRICE_TOO_HIGH': {
                'action': 'ADJUST_PRICE',
                'adjustment': -0.002  # הורד 0.2% במחיר
            },
            'INSUFFICIENT_BALANCE': {
                'action': 'REDUCE_SIZE', 
                'adjustment': 0.5  # חצי גודל
            },
            'TIMEOUT': {
                'action': 'SWITCH_TO_MARKET',
                'immediate': True
            }
        }
        
        strategy = retry_strategies.get(reason, {'action': 'CANCEL'})
        
        if strategy['action'] == 'ADJUST_PRICE':
            new_price = original_order['price'] * (1 + strategy['adjustment'])
            return await self.place_limit_order(
                symbol=original_order['symbol'],
                price=new_price,
                quantity=original_order['quantity']
            )
```

### **6. 🔄 Order Lifecycle Manager - חדש!**
```python
class OrderLifecycleManager:
    def __init__(self):
        self.order_states = {}  # מעקב אחרי כל order
    
    async def track_order_lifecycle(self, order_id, symbol, order_type):
        """מעקב מלא אחרי מחזור חיי order"""
        self.order_states[order_id] = {
            'symbol': symbol,
            'type': order_type,
            'created_at': time.time(),
            'status': 'PENDING',
            'events': []
        }
    
    async def log_order_event(self, order_id, event, data=None):
        """רישום אירוע ב-order"""
        if order_id in self.order_states:
            self.order_states[order_id]['events'].append({
                'timestamp': time.time(),
                'event': event,
                'data': data
            })
```

### **7. 🚨 Circuit Breaker for Orders - חדש!**

```python
class OrderCircuitBreaker:
    def __init__(self):
        self.failed_orders_count = 0
        self.circuit_open = False
        self.max_failures = 5
        self.reset_timeout = 300  # 5 דקות
    
    async def should_allow_orders(self, symbol):
        """בדיקה אם מותר לשלוח orders"""
        if self.circuit_open:
            return False
        
        if self.failed_orders_count >= self.max_failures:
            self.circuit_open = True
            logger.error(f"🚨 ORDER CIRCUIT BREAKER: Too many failures for {symbol}")
            
            # Schedule reset
            asyncio.create_task(self.reset_circuit())
            return False
        
        return True
    
    async def record_order_failure(self):
        """רישום כשלון order"""
        self.failed_orders_count += 1
    
    async def reset_circuit(self):
        """איפוס circuit breaker"""
        await asyncio.sleep(self.reset_timeout)
        self.circuit_open = False
        self.failed_orders_count = 0
        logger.info("✅ ORDER CIRCUIT BREAKER: Reset after cooldown")
```
