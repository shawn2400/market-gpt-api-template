## ✅ **3) frontend/src/api.js**

```js
import axios from "axios";

const api = axios.create({
  baseURL: "/api",
});

export const getFiles = () => api.get("/files/tree");
export const openFile = (path) => api.get(`/files/open?path=${path}`);
export const saveFile = (path, content) =>
  api.post("/files/save", { path, content });

export const runCmd = (cmd) =>
  api.post("/exec/run", { cmd }).then((r) => r.data.output);

export const listProjects = () => api.get("/project/list");
export const switchProject = (name) =>
  api.post("/project/switch", { name });

export const aiMessage = (msg) =>
  api.post("/ai/chat", { prompt: msg }).then((r) => r.data);
```

## ✅ **5) File Tree Component**

`frontend/src/components/FileTree.jsx`

```jsx
import { useEffect, useState } from "react";
import { getFiles } from "../api";

export default function FileTree({ onSelect }) {
  const [files, setFiles] = useState([]);

  useEffect(() => {
    getFiles().then((res) => setFiles(res.data));
  }, []);

  return (
    <div style={{ width: 250, background: "#111", color: "#fff", padding: 10 }}>
      <h4>Files</h4>
      {files.map((f) => (
        <div
          key={f}
          onClick={() => onSelect(f)}
          style={{ cursor: "pointer", marginBottom: 4 }}
        >
          {f}
        </div>
      ))}
    </div>
  );
}
```

## ✅ **6) Editor Component**

`frontend/src/components/Editor.jsx`

```jsx
import { useEffect, useState } from "react";
import MonacoEditor from "react-monaco-editor";
import { openFile, saveFile } from "../api";

export default function Editor({ file }) {
  const [content, setContent] = useState("");

  useEffect(() => {
    if (file) openFile(file).then((res) => setContent(res.data.content));
  }, [file]);

  if (!file) return <div style={{ flex: 1 }}>Select a file</div>;

  return (
    <div style={{ flex: 1 }}>
      <MonacoEditor
        value={content}
        onChange={setContent}
        language="python"
      />
      <button onClick={() => saveFile(file, content)}>Save</button>
    </div>
  );
}
```

## ✅ **7) Terminal Component**

`frontend/src/components/Terminal.jsx`

```jsx
import { useState } from "react";
import { runCmd } from "../api";

export default function Terminal() {
  const [cmd, setCmd] = useState("");
  const [output, setOutput] = useState("");

  async function execute() {
    const out = await runCmd(cmd);
    setOutput(out);
  }

  return (
    <div style={{ background: "#000", color: "#0f0", padding: 10, height: 200 }}>
      <input
        style={{ width: "100%" }}
        value={cmd}
        onChange={(e) => setCmd(e.target.value)}
        placeholder="run command..."
      />
      <pre>{output}</pre>
      <button onClick={execute}>Run</button>
    </div>
  );
}
```

import { useEffect, useState } from "react";
import { listProjects, switchProject } from "../api";

## ✅ **9) AI Chat with Ollama**

`frontend/src/components/AIChat.jsx`

```jsx
import { useState } from "react";
import { aiMessage } from "../api";

export default function AIChat() {
  const [msg, setMsg] = useState("");
  const [history, setHistory] = useState([]);

  async function send() {
    const res = await aiMessage(msg);
    setHistory([...history, { you: msg, ai: res.reply }]);
    setMsg("");
  }

  return (
    <div style={{ width: 300, background: "#1a1a1a", color: "#fff", padding: 10 }}>
      <h4>AI Assistant</h4>
      <div style={{ height: "80%", overflow: "auto" }}>
        {history.map((h, i) => (
          <div key={i}>
            <b>You:</b> {h.you}
            <br />
            <b>AI:</b> {h.ai}
            <hr />
          </div>
        ))}
      </div>
      <input
        value={msg}
        onChange={(e) => setMsg(e.target.value)}
        placeholder="Ask..."
      />
      <button onClick={send}>Send</button>
    </div>
  );
}
```
