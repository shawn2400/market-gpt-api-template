### 🔧 שלב 4: עדכון פונקציית המסחר הראשית

**עדכן את פונקציית המסחר הראשית שלך:**

```python
def run_trading_cycle(self):
    """מחזור מסחר מעודכן עם ניהול סיכונים"""
    
    # עדכן פרמטרי סיכון based on ביצועים
    self.risk_manager.update_risk_parameters()
    self.risk_manager.analyze_market_conditions()
    
    for symbol in self.trading_pairs:
        try:
            # 📈 קבל אות מהמערכת הקיימת שלך
            signal = self.generate_your_existing_signal(symbol)
            
            if signal and signal.get('confidence', 0) > 0.65:
                # 🛡️ השתמש במערכת ניהול הסיכונים החדשה
                result = self.risk_manager.execute_safe_trade(
                    symbol=symbol,
                    action=signal['action'],
                    entry_price=signal['entry_price'],
                    signal_confidence=signal['confidence']
                )
                
                if result:
                    self.logger.info(f"✅ מסחר בטוח בוצע: {symbol}")
            
        except Exception as e:
            self.logger.error(f"❌ שגיאה במסחר: {e}")
```

## 🛠️ דוגמת קוד מלא משולב

**הנה דוגמה איך הקוד הסופי שלך אמור להיראות:**

```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
מערכת המסחר המלאה שלך - עם ניהול סיכונים משולב
YOUR_SYSTEM_WITH_RISK_MANAGEMENT_V2
"""

import pandas as pd
import numpy as np
from binance.client import Client
import time
import logging
from datetime import datetime

# 🛡️ יבוא מערכת ניהול הסיכונים
from dynamic_risk_system import DynamicRiskManagementSystem

class YourCompleteTradingSystem:
    def __init__(self, api_key, api_secret):
        self.client = Client(api_key, api_secret)
        self.equity = self.get_account_balance()
        
        # 🎯 המערכת הקיימת שלך כאן
        self.trading_pairs = ['BTCUSDT', 'ETHUSDT', 'BNBUSDT', 'XRPUSDT', 'ADAUSDT']
        self.active_signals = []
        self.open_positions = []
        
        # 🛡️ מערכת ניהול סיכונים חדשה
        self.risk_manager = DynamicRiskManagementSystem(api_key, api_secret)
        
        self.setup_logging()
    
    def setup_logging(self):
        """הגדר לוגינג"""
        logging.basicConfig(level=logging.INFO)
        self.logger = logging.getLogger(__name__)
    
    def get_account_balance(self):
        """קבל יתרת חשבון"""
        try:
            account = self.client.get_account()
            usdt_balance = next(
                (float(x['free']) for x in account['balances'] if x['asset'] == 'USDT'), 
                1000.0
            )
            return usdt_balance
        except:
            return 1000.0
    
    def generate_your_existing_signal(self, symbol):
        """
        🎯 כאן נמצא האלגוריתם הקיים שלך ליצירת אותות
        החלף בפונקציה האמיתית שלך
        """
        try:
            # 📊 זה המקום שבו אתה מכניס את הלוגיקה הקיימת שלך
            # לדוגמה - ניתוח טכני, אינדיקטורים, וכו'
            
            klines = self.client.get_klines(symbol=symbol, interval='15m', limit=50)
            df = pd.DataFrame(klines, columns=[
                'timestamp', 'open', 'high', 'low', 'close', 'volume',
                'close_time', 'quote_volume', 'trades', 'taker_buy', 'taker_quote', 'ignore'
            ])
            
            for col in ['open', 'high', 'low', 'close', 'volume']:
                df[col] = pd.to_numeric(df[col])
            
            # 🎯 החלק הזה יהיה האלגוריתם הספציפי שלך
            current_price = df['close'].iloc[-1]
            
            # דוגמה פשוטה - החלף בלוגיקה האמיתית שלך
            signal_confidence = 0.75
            action = "BUY"
            
            return {
                'symbol': symbol,
                'action': action,
                'entry_price': current_price,
                'confidence': signal_confidence
            }
            
        except Exception as e:
            self.logger.error(f"❌ שגיאה ביצירת אות: {e}")
            return None
    
    def run_enhanced_trading_cycle(self):
        """מחזור מסחר מעודכן עם ניהול סיכונים"""
        self.logger.info("🚀 מתחיל מחזור מסחר עם הגנות סיכון...")
        
        # 🔄 עדכן פרמטרי סיכון
        self.risk_manager.update_risk_parameters()
        self.risk_manager.analyze_market_conditions()
        
        for symbol in self.trading_pairs:
            try:
                # 📈 קבל אות מהמערכת הקיימת שלך
                signal = self.generate_your_existing_signal(symbol)
                
                if signal and signal.get('confidence', 0) > 0.65:
                    # 🛡️ השתמש במערכת ניהול הסיכונים לביצוע בטוח
                    result = self.risk_manager.execute_safe_trade(
                        symbol=symbol,
                        action=signal['action'],
                        entry_price=signal['entry_price'],
                        signal_confidence=signal['confidence']
                    )
                    
                    if result:
                        self.logger.info(f"✅ מסחר בטוח בוצע: {symbol}")
                        self.active_signals.append(signal)
                
                # ⏳ המתנה בין סריקות
                time.sleep(1)
                
            except Exception as e:
                self.logger.error(f"❌ שגיאה במסחר עבור {symbol}: {e}")
                continue
        
        # 📊 הצג דוח ביצועים
        self.display_enhanced_dashboard()
    
    def display_enhanced_dashboard(self):
        """הצג דשבורד מעודכן"""
        risk_report = self.risk_manager.get_performance_report()
        
        print(f"""
🎯 מערכת מסחר מתקדמת - דשבורד בזמן אמת
══════════════════════════════════════
{risk_report}
📈 אותות פעילים: {len(self.active_signals)}
💼 פוזיציות פתוחות: {len(self.open_positions)}
💰 יתרה: {self.equity:.2f} USDT
🛡️ הגנות סיכון: פעילות
══════════════════════════════════════
        """)
    
    def main_trading_loop(self):
        """לולאת מסחר ראשית"""
        while True:
            try:
                self.run_enhanced_trading_cycle()
                
                # ⏰ המתנה בין מחזורים
                time.sleep(60)  # המתנה 1 דקה
                
            except KeyboardInterrupt:
                self.logger.info("⏹️ המערכת הופסקה ידנית")
                break
            except Exception as e:
                self.logger.error(f"❌ שגיאה כללית: {e}")
                time.sleep(30)

# 🚀 הרצת המערכת
if __name__ == "__main__":
    API_KEY = "your_api_key_here"
    API_SECRET = "your_api_secret_here"
    
    # אתחול המערכת המשולבת
    trading_system = YourCompleteTradingSystem(API_KEY, API_SECRET)
    
    # הפעל את לולאת המסחר
    trading_system.main_trading_loop()
```
