from pathlib import Path
p=Path("utils/mesh_bus.py")
if not p.exists():
    p.parent.mkdir(parents=True, exist_ok=True)
    p.write_text(r'''# -*- coding: utf-8 -*-
from __future__ import annotations
import os, json, time, hmac, hashlib, asyncio, aiohttp
from typing import Dict, Any

AI_MESH_PATH = os.getenv("AI_MESH_PATH","ai_mesh.json")
AI_MESH_SECRET = os.getenv("AI_MESH_SECRET","")
NODE_ID = os.getenv("NODE_ID","core-api")

_STATE: Dict[str, Any] = {
  "node": {"id": NODE_ID, "ts": int(time.time())},
  "peers": {},    # url -> meta
  "ok": True,
  "last_ping_ts": 0
}

def _sig(msg: str) -> str:
    return hmac.new(AI_MESH_SECRET.encode(), msg.encode(), hashlib.sha256).hexdigest() if AI_MESH_SECRET else ""

def snapshot() -> Dict[str,Any]:
    return {"node": _STATE["node"], "peers": dict(_STATE["peers"]), "ok": _STATE["ok"], "last_ping_ts": _STATE["last_ping_ts"]}

def _load_mesh() -> Dict[str,Any]:
    try:
        with open(AI_MESH_PATH,"r",encoding="utf-8") as f: return json.load(f)
    except Exception: return {}

def ensure_peer(peer: Dict[str,Any]) -> None:
    url = (peer.get("url") or "").rstrip("/")
    if not url: return
    rec = _STATE["peers"].get(url, {})
    rec.update({
      "id": peer.get("id"), "role": peer.get("role"),
      "scopes": peer.get("scopes") or [], "weight": int(peer.get("weight",1)),
      "enabled": bool(peer.get("enabled", True))
    })
    _STATE["peers"][url]=rec

async def _ping_one(sess: aiohttp.ClientSession, url: str) -> None:
    t0=time.perf_counter()
    try:
        async with sess.get(url, timeout=8) as r: j = await r.json()
        lat=int((time.perf_counter()-t0)*1000)
        rec=_STATE["peers"].setdefault(url, {})
        rec.update({"ok": bool(j.get("ok",True)), "last_ts": int(time.time()),
                    "latency_ms": lat, "version": (j.get("app") or {}).get("version"),
                    "id": (j.get("node") or {}).get("id")})
    except Exception as e:
        rec=_STATE["peers"].setdefault(url, {})
        rec.update({"ok": False, "last_ts": int(time.time()), "last_error": str(e)})

async def ping_loop() -> None:
    while True:
        try:
            _STATE["last_ping_ts"]=int(time.time())
            urls=list(_STATE["peers"].keys())
            if urls:
                async with aiohttp.ClientSession() as s:
                    await asyncio.gather(*[_ping_one(s,u) for u in urls])
            oks=[m.get("ok") for m in _STATE["peers"].values()]
            _STATE["ok"] = (not oks) or (sum(1 for x in oks if x) >= max(1, len(oks)//2))
        except Exception: pass
        await asyncio.sleep(15)
''', encoding="utf-8")
    print("✅ created utils/mesh_bus.py")
else:
    print("ℹ️ utils/mesh_bus.py exists")
PY

# =============== 2) routes/mesh_public.py ===============
python - <<'PY'
from pathlib import Path
p=Path("routes/mesh_public.py")
if not p.exists():
    p.parent.mkdir(parents=True, exist_ok=True)
    p.write_text(r'''from __future__ import annotations
import os, time, hmac, hashlib
from typing import Any, Dict
from fastapi import APIRouter, HTTPException, Request
from utils.mesh_bus import snapshot, ensure_peer, _sig

router = APIRouter(tags=["Mesh"])

@router.get("/mesh/public")
def mesh_public() -> Dict[str,Any]:
    s = snapshot()
    app = {"version": os.getenv("APP_VERSION","")}
    # מסננים רק שדות ציבוריים
    pub_peers = {}
    for url, meta in (s.get("peers") or {}).items():
        pub_peers[url] = {k: meta.get(k) for k in ("id","role","scopes","weight","enabled","ok","latency_ms","last_ts","version")}
    return {"ok": True, "node": s.get("node"), "peers": pub_peers, "app": app, "ts": int(time.time())}

@router.post("/mesh/register")
async def mesh_register(req: Request) -> Dict[str,Any]:
    body = await req.json()
    for k in ("id","url","role","scopes","ts","sig"):
        if k not in body: raise HTTPException(400, "bad_payload")
    msg = f"{body['id']}:{body['ts']}:{body['url']}"
    secret = os.getenv("AI_MESH_SECRET","")
    if not secret: raise HTTPException(500, "mesh_secret_missing")
    sig = hmac.new(secret.encode(), msg.encode(), hashlib.sha256).hexdigest()
    if sig != body["sig"]: raise HTTPException(403, "bad_sig")
    peer = {"id": body["id"], "url": body["url"].rstrip("/"), "role": body["role"],
            "scopes": body.get("scopes", []), "weight": int(body.get("weight",1)), "enabled": bool(body.get("enabled",True))}
    ensure_peer(peer)
    return {"ok": True, "registered": peer["id"]}

@router.get("/mesh/peers")
def mesh_peers() -> Dict[str,Any]:
    return {"ok": True, "peers": list((snapshot().get("peers") or {}).keys())}
''', encoding="utf-8")
    print("✅ created routes/mesh_public.py")
else:
    print("ℹ️ routes/mesh_public.py exists")
PY

import os, time
from fastapi import APIRouter

router = APIRouter(tags=["Public"])
def _now(): return int(time.time())

@router.get("/status/public")
def status_public():
    app = {"name": os.getenv("APP_TITLE","AlgoGPT Service"),
           "version": os.getenv("APP_VERSION",""),
           "ts": _now()}
    sup = {"ok": False, "interval_sec": None}
    with suppress(Exception):
        from workers import ai_supervisor as w  # type: ignore

    mesh_ok=True
    with suppress(Exception):
        from utils import mesh_bus as mb  # type: ignore
        mesh_ok = bool(mb.snapshot().get("ok", True))
    return {"ok": True, "app": app, "supervisors": sup, "mesh_ok": mesh_ok, "ts": _now()}
