### **🔴 1. CORE FILTERING ENGINE**
```python
class SymbolFilterEngine:
    def __init__(self):
        self.approved_symbols = set()
        self.symbol_metrics = {}
        self.blacklist = {}
        
    def pre_trade_validation(self, symbol, order_type, amount):
        """Validation gate לפני כל trade"""
        
        # 1. BASIC SYMBOL CHECKS
        checks = [
            self._check_market_cap(symbol),
            self._check_24h_volume(symbol),
            self._check_exchange_listing(symbol),
            self._check_liquidity_depth(symbol),
            self._check_blacklist_status(symbol)
        ]
        
        # 2. PERFORMANCE CHECKS
        performance_checks = [
            self._check_fill_rate(symbol),
            self._check_slippage_history(symbol),
            self._check_win_rate(symbol),
            self._check_recent_performance(symbol)
        ]
        
        # 3. POSITION LIMITS
        limit_checks = [
            self._check_max_positions_per_symbol(symbol),
            self._check_total_open_orders(),
            self._check_correlation_exposure(symbol)
        ]
        
        # אם כל הבדיקות עברו
        if all(checks + performance_checks + limit_checks):
            return {"status": "APPROVED", "message": "Trade validation passed"}
        else:
            return {"status": "BLOCKED", "message": "Failed validation checks"}
```

### **🟡 2. MARKET DATA FILTERS**
```python
class MarketDataFilters:
    # MINIMUM REQUIREMENTS
    MIN_MARKET_CAP = 500_000_000  # $500M
    MIN_24H_VOLUME = 10_000_000   # $10M
    MIN_LIQUIDITY_DEPTH = 50_000  # $50k order book depth
    
    def _check_market_cap(self, symbol):
        market_cap = self.get_symbol_market_cap(symbol)
        return market_cap >= self.MIN_MARKET_CAP
        
    def _check_24h_volume(self, symbol):
        volume_24h = self.get_24h_volume(symbol)
        return volume_24h >= self.MIN_24H_VOLUME
        
    def _check_liquidity_depth(self, symbol):
        """בדיקת עומק order book"""
        bid_depth, ask_depth = self.get_order_book_depth(symbol)
        return min(bid_depth, ask_depth) >= self.MIN_LIQUIDITY_DEPTH
        
    def _check_exchange_listing(self, symbol):
        """וידוא שהסמל ברשימת TOP exchange"""
        top_exchanges = ["BINANCE", "COINBASE", "KRAKEN", "BYBIT"]
        return any(exchange in symbol for exchange in top_exchanges)
```

### **🟢 3. PERFORMANCE FILTERS**
```python
class PerformanceFilters:
    # PERFORMANCE THRESHOLDS
    MIN_FILL_RATE = 0.65    # 65%
    MAX_AVG_SLIPPAGE = 0.02 # 2%
    MIN_WIN_RATE = 0.40     # 40%
    MAX_CONSECUTIVE_LOSSES = 3
    
    def _check_fill_rate(self, symbol):
        fill_rate = self.calculate_fill_rate(symbol)
        return fill_rate >= self.MIN_FILL_RATE
        
    def _check_slippage_history(self, symbol):
        avg_slippage = self.get_avg_slippage(symbol)
        return avg_slippage <= self.MAX_AVG_SLIPPAGE
        
    def _check_win_rate(self, symbol):
        win_rate = self.get_symbol_win_rate(symbol)
        return win_rate >= self.MIN_WIN_RATE
        
    def _check_recent_performance(self, symbol):
        recent_trades = self.get_recent_trades(symbol, count=10)
        losses = sum(1 for trade in recent_trades if trade['pnl'] < 0)
        return losses <= self.MAX_CONSECUTIVE_LOSSES
```

### **🔵 4. POSITION LIMIT FILTERS**
```python
class PositionLimitFilters:
    MAX_POSITIONS_PER_SYMBOL = 2
    MAX_TOTAL_OPEN_ORDERS = 25
    MAX_CORRELATED_EXPOSURE = 0.3  # 30% portfolio
    
    def _check_max_positions_per_symbol(self, symbol):
        current_positions = self.get_open_positions(symbol)
        return len(current_positions) < self.MAX_POSITIONS_PER_SYMBOL
        
    def _check_total_open_orders(self):
        total_orders = self.get_total_open_orders()
        return total_orders < self.MAX_TOTAL_OPEN_ORDERS
        
    def _check_correlation_exposure(self, symbol):
        """בדיקת exposure ל-correlated assets"""
        correlated_symbols = self.get_correlated_symbols(symbol)
        total_exposure = self.get_total_exposure_to_group(correlated_symbols)
        portfolio_value = self.get_portfolio_value()
        
        return total_exposure <= (portfolio_value * self.MAX_CORRELATED_EXPOSURE)
```

### **🟣 5. INTEGRATION WITH DYNAMIC LEVERAGE**

```python
class TradingGatekeeper:
    def process_trade_request(self, trade_request):
        """Full trading pipeline integration"""
        
        # 1. SYMBOL FILTERING
        filter_result = self.symbol_filter.pre_trade_validation(
            trade_request.symbol, 
            trade_request.order_type,
            trade_request.amount
        )
        
        if filter_result["status"] != "APPROVED":
            return filter_result  # BLOCKED at filter stage
            
        # 2. DYNAMIC LEVERAGE CALCULATION
        leverage_result = self.leverage_engine.calculate_leverage(
            trade_request.symbol,
            trade_request.confidence_score,
            trade_request.market_regime
        )
        
        # 3. FINAL EXECUTION APPROVAL
        if leverage_result["approved"]:
            return {
                "status": "APPROVED",
                "leverage": leverage_result["leverage"],
                "max_position_size": leverage_result["position_size"],
                "filters_passed": True
            }
        else:
            return {
                "status": "BLOCKED", 
                "message": "Failed leverage validation",
                "reason": leverage_result["reason"]
            }
```

```python
class RealTimeSymbolMonitor:
    def __init__(self):

    def update_symbol_metrics(self):
        """עדכון מטריקות בזמן אמת"""
        for symbol in self.tracked_symbols:
            self.symbol_metrics[symbol] = {
                'current_volume': self.get_current_volume(symbol),
                'liquidity_depth': self.get_liquidity_depth(symbol),
                'fill_rate': self.calculate_recent_fill_rate(symbol),
                'slippage_trend': self.get_slippage_trend(symbol),
                'performance_score': self.calculate_performance_score(symbol)
            }
            
    def auto_blacklist_symbol(self, symbol, reason):
        """הוספה אוטומטית ל-blacklist"""
        self.blacklist[symbol] = {
            'timestamp': datetime.now(),
            'reason': reason,
            'duration_hours': 24  # 24 hours auto-removal
        }
        
    def check_and_update_approved_list(self):
        """עדכון רשימת הסמלים המאושרים"""
        for symbol in self.all_symbols:
            metrics = self.symbol_metrics.get(symbol, {})
            
            if self._passes_all_filters(symbol, metrics):
                self.approved_symbols.add(symbol)
            else:
                self.approved_symbols.discard(symbol)
```
