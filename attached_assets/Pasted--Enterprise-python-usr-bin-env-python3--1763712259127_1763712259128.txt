                risk_score=portfolio_impact['risk_score']

    def analyze_market_microstructure(self, symbol: str) -> Dict:
        """ניתוח מיקרו-מבנה שוק מתקדם"""
        try:
            # קבל נתונים ברזולוציה גבוהה
            klines = self.client.get_klines(symbol=symbol, interval='1m', limit=100)
            df = pd.DataFrame(klines, columns=[
                'timestamp', 'open', 'high', 'low', 'close', 'volume',
                'close_time', 'quote_volume', 'trades', 'taker_buy', 'taker_quote', 'ignore'
            ])
            
            for col in ['open', 'high', 'low', 'close', 'volume']:
                df[col] = pd.to_numeric(df[col])
            
            current_price = df['close'].iloc[-1]
            
            # מדדים מיקרו-מבניים
            microstructure = {
                'tradable': True,
                'current_price': current_price,
                'volatility_regime': self.calculate_volatility_regime(df),
                'liquidity_score': self.calculate_liquidity_score(df),
                'order_imbalance': self.calculate_order_imbalance(df),
                'market_impact_cost': self.estimate_market_impact(df, 1000),  # עבור 1000 USDT
                'spread_percentage': self.calculate_spread(symbol),
                'momentum_regime': self.analyze_momentum_regime(df)
            }
            
            # תנאי סף למסחר
            if (microstructure['liquidity_score'] < 0.3 or 
                microstructure['market_impact_cost'] > 0.002 or
                microstructure['spread_percentage'] > 0.001):
                microstructure['tradable'] = False
            
            return microstructure
            
        except Exception as e:
            self.logger.error(f"❌ שגיאה בניתוח מיקרו-מבנה: {e}")
            return {'tradable': False}
    
    def calculate_portfolio_impact(self, signal: Dict) -> Dict:
        """חישוב השפעה על הפורטפוליו עם קורלציות"""
        try:
            # חשיפה נוכחית
            current_exposure = sum(
                pos.quantity * pos.entry_price 
                for pos in self.active_positions.values()
            )
            
            # חשיפה חדשה
            new_position_value = signal['position_size'] * signal['entry_price']
            total_exposure = current_exposure + new_position_value
            
            # קורלציה עם פוזיציות קיימות
            correlation_scores = []
            for pos in self.active_positions.values():
                corr = self.calculate_pair_correlation(signal['symbol'], pos.symbol)
                correlation_scores.append(abs(corr))
            
            avg_correlation = np.mean(correlation_scores) if correlation_scores else 0
            concentration_risk = len(self.active_positions) / 20  # נורמליזציה ל-20 פוזיציות מקסימום
            
            risk_score = min(1.0, (
                (total_exposure / self.get_account_balance()) * 0.4 +
                avg_correlation * 0.3 +
                concentration_risk * 0.3
            ))
            
            return {
                'exposure': total_exposure / self.get_account_balance(),
                'correlation_score': avg_correlation,
                'risk_score': risk_score,
                'concentration_risk': concentration_risk
            }
            
        except Exception as e:
            self.logger.error(f"❌ שגיאה בחישוב השפעת פורטפוליו: {e}")
            return {'risk_score': 1.0, 'exposure': 1.0, 'correlation_score': 1.0}
    
    def optimize_entry_execution(self, signal: Dict, market_conditions: Dict) -> Optional[Dict]:
        """אופטימיזציית ביצוע כניסה מתקדמת"""
        try:
            base_quantity = signal['position_size']
            entry_price = market_conditions['current_price']
            
            # התאם based on תנאי שוק
            if market_conditions['volatility_regime'] == "HIGH_VOL":
                # בוולאטיליות גבוהה - פיצול כניסה
                quantity = base_quantity * 0.7
                stop_loss_multiplier = 1.2
            elif market_conditions['liquidity_score'] < 0.5:
                # נזילות נמוכה - כמות קטנה יותר
                quantity = base_quantity * 0.8
                stop_loss_multiplier = 1.1
            else:
                quantity = base_quantity
                stop_loss_multiplier = 1.0
            
            # חישוב SL/TP אופטימלי
            atr = self.calculate_atr(signal['symbol'], period=14)
            stop_loss_pct = min(0.05, atr * 1.5 * stop_loss_multiplier)  # מקסימום 5%
            take_profit_pct = stop_loss_pct * 2.0  # R:R של 2:1
            
            if signal['action'] == "BUY":
                stop_loss = entry_price * (1 - stop_loss_pct)
                take_profit = entry_price * (1 + take_profit_pct)
            else:
                stop_loss = entry_price * (1 + stop_loss_pct)
                take_profit = entry_price * (1 - take_profit_pct)
            
            return {
                'entry_price': entry_price,
                'quantity': quantity,
                'stop_loss': stop_loss,
                'take_profit': take_profit,
                'execution_type': 'OPTIMIZED_MARKET'
            }
            
        except Exception as e:
            self.logger.error(f"❌ שגיאה באופטימיזציית כניסה: {e}")
            return None
    
    def execute_quantum_entry(self, position: QuantumPosition, execution_params: Dict) -> bool:
        """ביצוע כניסה קוונטי עם ניהול errors"""
        max_retries = 3
        for attempt in range(max_retries):
            try:
                if execution_params['execution_type'] == 'OPTIMIZED_MARKET':
                    if position.direction == "BUY":
                        order = self.client.order_market_buy(
                            symbol=position.symbol,
                            quantity=position.quantity
                        )
                    else:
                        order = self.client.order_market_sell(
                            symbol=position.symbol,
                            quantity=position.quantity
                        )
                
                # רישום SL/TP מורכב
                self.place_advanced_stop_loss_take_profit(position)
                
                # עדכון מטא-דאטה
                position.entry_price = float(order['fills'][0]['price']) if order.get('fills') else position.entry_price
                
                return True
                
            except Exception as e:
                self.logger.warning(f"⚠️ נסיון {attempt + 1} נכשל: {e}")
                if attempt == max_retries - 1:
                    return False
                time.sleep(1)
        
        return False
    
    def place_advanced_stop_loss_take_profit(self, position: QuantumPosition):
        """רישום SL/TP מתקדם עם תנאים מורכבים"""
        try:
            # SL דינאמי based on וולאטיליות
            if position.direction == "BUY":
                # Stop Loss
                self.client.create_order(
                    symbol=position.symbol,
                    side=Client.SIDE_SELL,
                    type=Client.ORDER_TYPE_STOP_LOSS_LIMIT,
                    timeInForce='GTC',
                    quantity=position.quantity,
                    stopPrice=round(position.current_stop_loss, 6),
                    price=round(position.current_stop_loss, 6)
                )
                
                # Take Profit עם OCO
                self.client.create_order(
                    symbol=position.symbol,
                    side=Client.SIDE_SELL,
                    type=Client.ORDER_TYPE_TAKE_PROFIT_LIMIT,
                    timeInForce='GTC',
                    quantity=position.quantity,
                    stopPrice=round(position.current_take_profit, 6),
                    price=round(position.current_take_profit, 6)
                )
            
            self.logger.info(f"🛡️ הגנות מתקד�ות רשומות עבור {position.position_id}")
            
        except Exception as e:
            self.logger.error(f"❌ שגיאה ברישום הגנות: {e}")

    def realtime_exposure_monitor(self):
        """מוניטור חשיפה ריאל-טיים"""
        while True:
            try:
                total_exposure = 0.0
                symbol_exposure = {}
                
                for position in self.active_positions.values():
                    current_price = self.get_current_price(position.symbol)
                    position_value = position.quantity * current_price
                    total_exposure += position_value
                    
                    if position.symbol not in symbol_exposure:
                        symbol_exposure[position.symbol] = 0.0
                    symbol_exposure[position.symbol] += position_value
                
                # עדכן מטריצת חשיפה
                self.realtime_metrics['total_exposure'] = total_exposure
                self.realtime_metrics['symbol_exposure'] = symbol_exposure
                
                # בדוק גבולות חשיפה
                account_balance = self.get_account_balance()
                if total_exposure / account_balance > 0.8:  # 80% חשיפה מקסימלית
                    self.logger.warning(f"🚨 חשיפה כוללת גבוהה: {total_exposure/account_balance:.2%}")
                
                time.sleep(5)  # עדכון כל 5 שניות
                
            except Exception as e:
                self.logger.error(f"❌ שגיאה במוניטור חשיפה: {e}")
                time.sleep(10)
    
    def position_health_monitor(self):
        """מוניטור בריאות פוזיציות"""
        while True:
            try:
                for position_id, position in list(self.active_positions.items()):
                    self.assess_position_health(position)
                    self.manage_position_dynamics(position)
                
                time.sleep(10)  # בדיקה כל 10 שניות
                
            except Exception as e:
                self.logger.error(f"❌ שגיאה במוניטור בריאות: {e}")
                time.sleep(30)
    
    def assess_position_health(self, position: QuantumPosition):
        """הערכת בריאות הפוזיציה"""
        current_price = self.get_current_price(position.symbol)
        
        # חשב PNL לא ממומש
        if position.direction == "BUY":
            unrealized_pnl = (current_price - position.entry_price) * position.quantity
        else:
            unrealized_pnl = (position.entry_price - current_price) * position.quantity
        
        position.unrealized_pnl = unrealized_pnl
        
        # עדכן תנועות מקסימליות
        pnl_pct = unrealized_pnl / (position.quantity * position.entry_price)
        if pnl_pct > position.max_favorable_move:
            position.max_favorable_move = pnl_pct
        if pnl_pct < position.max_adverse_move:
            position.max_adverse_move = pnl_pct
        
        # זמן בפוזיציה
        position.time_in_trade = (datetime.now() - position.timestamp).total_seconds() / 3600  # בשעות
        
        # הערכת בריאות
        health_score = self.calculate_position_health_score(position)
        
        if health_score < 0.3:
            self.logger.warning(f"⚠️ פוזיציה {position.position_id} בבריאות ירודה: {health_score:.2f}")
    
    def calculate_position_health_score(self, position: QuantumPosition) -> float:
        """חישוב ציון בריאות פוזיציה"""
        scores = []
        
        # 1. PNL score
        pnl_pct = position.unrealized_pnl / (position.quantity * position.entry_price)
        if pnl_pct >= 0:
            pnl_score = min(1.0, pnl_pct / 0.05)  # נורמליזציה ל-5%
        else:
            pnl_score = max(0.0, 1 + (pnl_pct / 0.02))  # הפסד של 2% = 0
        
        scores.append(pnl_score * 0.3)
        
        # 2. Time score
        time_score = max(0.0, 1 - (position.time_in_trade / 24))  # 24 שעות מקסימום
        scores.append(time_score * 0.2)
        
        # 3. Volatility score
        if position.volatility_regime == "HIGH_VOL":
            vol_score = 0.6
        elif position.volatility_regime == "LOW_VOL":
            vol_score = 0.9
        else:
            vol_score = 0.8
        scores.append(vol_score * 0.2)
        
        # 4. Correlation score
        corr_score = 1 - position.correlation_score
        scores.append(corr_score * 0.3)
        
        return sum(scores)
    
    def manage_position_dynamics(self, position: QuantumPosition):
        """ניהול דינאמי של הפוזיציה"""
        current_price = self.get_current_price(position.symbol)
        
        # Trail Stop Logic
        if position.direction == "BUY" and current_price > position.entry_price * 1.02:
            # הרווחת 2% - התחל trailing
            new_stop = current_price * 0.995  # 0.5% מתחת למחיר נוכחי
            if new_stop > position.current_stop_loss:
                self.update_stop_loss(position, new_stop)
                position.state = TradeState.TRAILING
        
        # Partial Close Logic
        if (position.direction == "BUY" and 
            current_price >= position.current_take_profit * 0.8 and
            position.state != TradeState.PARTIAL_CLOSE):
            # סגור 30% מהפוזיציה ב-80% מה-TP
            self.execute_partial_close(position, 0.3)
            position.state = TradeState.PARTIAL_CLOSE
    
    def update_stop_loss(self, position: QuantumPosition, new_stop: float):
        """עדכון Stop Loss דינאמי"""
        try:
            # בטל את ה-SL הישן
            open_orders = self.client.get_open_orders(symbol=position.symbol)
            for order in open_orders:
                if order['type'] in ['STOP_LOSS_LIMIT', 'STOP_LOSS']:
                    self.client.cancel_order(symbol=position.symbol, orderId=order['orderId'])
            
            # צור SL חדש
            if position.direction == "BUY":
                self.client.create_order(
                    symbol=position.symbol,
                    side=Client.SIDE_SELL,
                    type=Client.ORDER_TYPE_STOP_LOSS_LIMIT,
                    timeInForce='GTC',
                    quantity=position.quantity,
                    stopPrice=round(new_stop, 6),
                    price=round(new_stop, 6)
                )
            else:
                self.client.create_order(
                    symbol=position.symbol,
                    side=Client.SIDE_BUY,
                    type=Client.ORDER_TYPE_STOP_LOSS_LIMIT,
                    timeInForce='GTC',
                    quantity=position.quantity,
                    stopPrice=round(new_stop, 6),
                    price=round(new_stop, 6)
                )
            
            position.current_stop_loss = new_stop
            self.logger.info(f"🔄 Stop Loss עודכן: {position.position_id} -> {new_stop:.6f}")
            
        except Exception as e:
            self.logger.error(f"❌ שגיאה בעדכון Stop Loss: {e}")
    
    def execute_partial_close(self, position: QuantumPosition, close_percentage: float):
        """ביצוע סגירה חלקית"""
        try:
            close_quantity = position.quantity * close_percentage
            
            if position.direction == "BUY":
                order = self.client.order_market_sell(
                    symbol=position.symbol,
                    quantity=close_quantity
                )
            else:
                order = self.client.order_market_buy(
                    symbol=position.symbol,
                    quantity=close_quantity
                )
            
            # עדכן את הכמות הנותרת
            position.quantity *= (1 - close_percentage)
            
            self.logger.info(f"📊 סגירה חלקית: {position.position_id} - {close_percentage:.1%}")
            
            return True
            
        except Exception as e:
            self.logger.error(f"❌ שגיאה בסגירה חלקית: {e}")
            return False

    def calculate_volatility_regime(self, df: pd.DataFrame) -> str:
        """חישוב רגים וולאטיליות"""
        returns = df['close'].pct_change().dropna()
        volatility = returns.std() * np.sqrt(365)  # וולאטיליות שנתית
        
        if volatility > 0.8:  # 80% וולאטיליות שנתית
            return "HIGH_VOL"
        elif volatility < 0.3:  # 30% וולאטיליות שנתית
            return "LOW_VOL"
        else:
            return "NORMAL"
    
    def calculate_liquidity_score(self, df: pd.DataFrame) -> float:
        """חישוב ציון נזילות"""
        volume_score = df['volume'].iloc[-1] / df['volume'].rolling(20).mean().iloc[-1]
        spread_score = 1.0  # ניתן לחשב based on order book
        
        liquidity_score = (volume_score * 0.7 + spread_score * 0.3)
        return min(1.0, liquidity_score)
    
    def calculate_pair_correlation(self, symbol1: str, symbol2: str, period: int = 30) -> float:
        """חישוב קורלציה בין זוגות"""
        try:
            # מימוש פשוט - בפרודקשן יש להשתמש בנתונים היסטוריים
            if symbol1 == symbol2:
                return 1.0
            
            # דוגמה - ניתן להרחיב עם נתונים היסטוריים אמיתיים
            common_pairs = [('BTCUSDT', 'ETHUSDT'), ('BTCUSDT', 'BNBUSDT')]
            if (symbol1, symbol2) in common_pairs or (symbol2, symbol1) in common_pairs:
                return 0.7
            else:
                return 0.3
                
        except:
            return 0.5  # ברירת מחדל
    
    def get_current_price(self, symbol: str) -> float:
        """קבל מחיר נוכחי"""
        try:
            ticker = self.client.get_symbol_ticker(symbol=symbol)
            return float(ticker['price'])
        except:
            return 0.0

        try:

            usdt_balance = next(

            return usdt_balance
        except:
