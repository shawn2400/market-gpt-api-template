# -*- coding: utf-8 -*-
from __future__ import annotations
import os, time, hmac, hashlib, requests

BINANCE_FAPI = os.getenv("BINANCE_FAPI", "https://fapi.binance.com")
API_KEY = os.getenv("BINANCE_API_KEY", "")
API_SECRET = os.getenv("BINANCE_API_SECRET", "")

def _sign(qs: str) -> str:
    return hmac.new(API_SECRET.encode(), qs.encode(), hashlib.sha256).hexdigest()

def get_dual_position_mode() -> bool:
    """True אם Hedge (dualSidePosition=true), אחרת False.
       אם אין מפתחות/כשל — נניח False ונפעל ב-Single Mode סלחני."""
    if not API_KEY or not API_SECRET:
        return False
    try:
        ts = int(time.time()*1000)
        qs = f"timestamp={ts}"
        sig = _sign(qs)
        r = requests.get(f"{BINANCE_FAPI}/fapi/v1/positionSide/dual?{qs}&signature={sig}",
                         headers={"X-MBX-APIKEY": API_KEY}, timeout=5)
        r.raise_for_status()
        data = r.json()
        return bool(data.get("dualSidePosition", False))
    except Exception:
        return False
PY

# -*- coding: utf-8 -*-
from __future__ import annotations
import os
from decimal import Decimal, ROUND_DOWN
from utils.position_mode import get_dual_position_mode

TICK_FALLBACK = Decimal("0.0001")

def quantize_price(price: float, tick_size: str | None) -> str:
    if not tick_size:
        return str(Decimal(price).quantize(TICK_FALLBACK, rounding=ROUND_DOWN))
    q = Decimal(str(price)).quantize(Decimal(str(tick_size)), rounding=ROUND_DOWN)
    return format(q, "f")

def build_order_params(
    *,
    symbol: str,
    side: str,             # BUY/SELL
    type_: str,            # LIMIT/MARKET/STOP/TAKE_PROFIT/STOP_MARKET/TAKE_PROFIT_MARKET
    qty: float | None,
    price: float | None,
    reduce_only: bool,
    close_order: bool,     # האם זו פקודת סגירה (TP/SL/Close Only)
    hedge_hint: str | None,    # LONG/SHORT אם ידוע (אופציונלי)
    tick_size: str | None,
    time_in_force: str | None = None,
) -> dict:
    """
    כללים:
    - Hedge OFF: לא שולחים positionSide בכלל (מונע -4061).
    - reduceOnly: נשלח רק אם close_order=True ובסוגי פקודה נתמכים (LIMIT/STOP/STOP_MARKET/TAKE_PROFIT/TAKE_PROFIT_MARKET).
      לא נשלח ב-MARKET פתיחה (מונע -1106).
    """
    params: dict = {"symbol": symbol, "side": side, "type": type_}

    # positionSide רק אם החשבון ב-Hedge ולנו יש רמז
    if get_dual_position_mode() and hedge_hint:
        params["positionSide"] = hedge_hint  # "LONG" / "SHORT"

    # כמות/מחיר
    if qty is not None:
        params["quantity"] = qty
    if price is not None and type_ in {"LIMIT","STOP","TAKE_PROFIT"}:
        params["price"] = quantize_price(price, tick_size)

    # TIF
    if time_in_force and type_ == "LIMIT":
        params["timeInForce"] = time_in_force

    # reduceOnly רק כשסוג פקודה מתאים וסגירה
    supported_reduce = {"LIMIT","STOP","STOP_MARKET","TAKE_PROFIT","TAKE_PROFIT_MARKET"}
    if reduce_only and close_order and type_ in supported_reduce:
        params["reduceOnly"] = True

    return params
PY

# -*- coding: utf-8 -*-
from __future__ import annotations
from typing import List
from decimal import Decimal, ROUND_DOWN

def _q(val: float, step: float) -> float:
    d = Decimal(str(val)).quantize(Decimal(str(step)), rounding=ROUND_DOWN)
    return float(d)

def tp_levels_from_entry(
    *,
    side: str,                 # "LONG" / "SHORT"
    entry: float,
    atr: float | None,
    percents: list[float],     # [1.6, 3.0, 4.8] וכו'
    tick_size: float = 0.0001
) -> List[float]:
    entry = float(entry)
    tick = float(tick_size)
    out: List[float] = []
    for pct in percents:
        k = float(pct) / 100.0
        if side.upper() == "LONG":
            price = entry * (1.0 + k)
        else:
            price = entry * (1.0 - k)
        # Guard: לא שלילי ולא אפס
        price = max(price, tick)
        out.append(_q(price, tick))
    return out
PY

# -*- coding: utf-8 -*-
from __future__ import annotations
import os, time, threading
from collections import deque
from typing import Deque, Tuple, Optional, List

DIGEST_ENABLE = os.getenv("TELEGRAM_DIGEST_ENABLE", "1") == "1"
DIGEST_WINDOW_SEC = int(os.getenv("TELEGRAM_DIGEST_WINDOW_SEC", "900"))

class _DigestBus:
    def __init__(self) -> None:
        self.q: Deque[Tuple[str, str]] = deque()
        self.last_flush = 0.0
        self.lock = threading.Lock()

    def push(self, level: str, msg: str) -> None:
        if not DIGEST_ENABLE:
            return
        with self.lock:
            self.q.append((level, msg))

    def pop_all(self) -> List[Tuple[str,str]]:
        with self.lock:
            items = list(self.q)
            self.q.clear()
            return items

BUS = _DigestBus()

def enqueue(level: str, msg: str) -> None:
    BUS.push(level, msg)

def should_flush() -> bool:
    if not DIGEST_ENABLE:
        return False
    now = time.time()
    if now - BUS.last_flush >= DIGEST_WINDOW_SEC:
        BUS.last_flush = now
        return True
    return False

def format_digest(items: List[Tuple[str,str]]) -> str:
    lines = ["📬 *Digest Window*"]
    bucket = {}
    for lvl, m in items:
        bucket.setdefault(lvl, []).append(m)
    for lvl, arr in bucket.items():
        lines.append(f"\n*{lvl}* ({len(arr)}):")
        for s in arr[:15]:
            lines.append(f"• {s}")
        if len(arr) > 15:
            lines.append(f"… ועוד {len(arr)-15}")
    return "\n".join(lines)

### מה זה פותר (בקצר):

1. **Neon Disabled** → `scripts/neon_resume.sh` (דרך API).
2. **Binance -4061/-1106** → בניית פרמטרים בטוחה (`utils/order_params.py`) + זיהוי hedge אמיתי (`utils/position_mode.py`) → לא שולחים `positionSide` כשאין hedge; `reduceOnly` רק כשבאמת צריך.
3. **TP שלילי** → מחשבון בטוח `utils/tp_calc.py` + התאמה ל־`tp_ladder` (בלי מחירים שליליים).
4. **Telegram Flood** → `utils/telegram_digest.py` לאיגוד התראות לחלונות זמן (ברירת מחדל 15 דק׳).
5. **API Keys חסרים** → הרחבת `.env.example` לכל המפתחות/דגלים הרלוונטיים.

אם תרצה—אוסיף גם **רול-אאוט פרוגרסיבי (Shadow→Symbol אחד→Full)** בקובץ ENV/דגלים, אבל קודם נסגור את ה־5 הקריטיות.
