# Deploys full AlgoGPT infra:

# with daily reports, fault recovery, and no overload.

echo "🚀 Deploying AlgoGPT Phase-2 infrastructure..."

# === ENVIRONMENT SETUP ===
export BASE_URL="https://algogpt-docker.onrender.com"
export APP_MODE="production"
export HEARTBEAT_INTERVAL=600
export DAILY_REPORT_HOUR="23:00"
export NETWORK_MODE="interlinked"
export TELEGRAM_BOT_TOKEN="$TELEGRAM_BOT_TOKEN"
export TELEGRAM_CHAT_ID="$TELEGRAM_CHAT_ID"
export API_BEARER_TOKEN="$API_BEARER_TOKEN"
export GITHUB_TOKEN="$GITHUB_TOKEN"
export OPENAI_API_KEY="$OPENAI_API_KEY"
export DEEPSEEK_API_KEY="$DEEPSEEK_API_KEY"
export AIX_SUPERVISOR_TOKEN="$AIX_SUPERVISOR_TOKEN"

sudo apt-get update -qq && sudo apt-get install -y -qq docker-compose procps curl >/dev/null

# === PHASE-2 docker-compose ===
cat > docker-compose.yml <<'EOF'
version: "3.9"
services:
  core:
    image: algogpt/core:latest
    container_name: algogpt-core
    restart: always
    depends_on: [db, redis]
    environment:
      - APP_MODE=production
      - DATABASE_URL=postgresql://algogpt:algogpt@db:5432/algogpt
      - REDIS_URL=redis://redis:6379/0
      - OPENAI_API_KEY=${OPENAI_API_KEY}
      - DEEPSEEK_API_KEY=${DEEPSEEK_API_KEY}
      - TELEGRAM_BOT_TOKEN=${TELEGRAM_BOT_TOKEN}
      - TELEGRAM_CHAT_ID=${TELEGRAM_CHAT_ID}
      - API_BEARER_TOKEN=${API_BEARER_TOKEN}
    ports:
      - "8000:8000"

  db:
    image: postgres:15-alpine
    container_name: algogpt-db
    restart: always
    environment:
      POSTGRES_USER: algogpt
      POSTGRES_PASSWORD: algogpt
      POSTGRES_DB: algogpt
    volumes: ["db_data:/var/lib/postgresql/data"]

  redis:
    image: redis:7-alpine
    container_name: algogpt-redis
    restart: always
    volumes: ["redis_data:/data"]

  dashboard:
    image: algogpt/dashboard:latest
    container_name: algogpt-dashboard
    restart: always
    depends_on: [core]
    ports: ["8500:8500"]
    environment:
      - CORE_API=http://core:8000
      - REDIS_URL=redis://redis:6379/0
      - REFRESH_INTERVAL=15

  heartbeat:
    image: alpine:latest
    container_name: algogpt-heartbeat
    restart: always
    environment:
      - BASE_URL=${BASE_URL}
      - TELEGRAM_BOT_TOKEN=${TELEGRAM_BOT_TOKEN}
      - TELEGRAM_CHAT_ID=${TELEGRAM_CHAT_ID}
      - HEARTBEAT_INTERVAL=${HEARTBEAT_INTERVAL}
      - DAILY_REPORT_HOUR=${DAILY_REPORT_HOUR}
    command: /bin/sh -c "
      while true; do
        NOW=\$(date +%H:%M);
        CPU=\$(grep 'cpu ' /proc/stat | awk '{u=(\$2+\$4)*100/(\$2+\$4+\$5)} END {print u\"%\"}');
        MEM=\$(free -m | awk '/Mem:/ {printf(\"%.0f%%\", \$3/\$2*100)}');
        STATUS=\$(curl -fsS \$BASE_URL/readyz >/dev/null && echo '✅ Alive' || echo '❌ Down');
        curl -s -o /dev/null -X POST https://api.telegram.org/bot\$TELEGRAM_BOT_TOKEN/sendMessage -d chat_id=\$TELEGRAM_CHAT_ID -d text=\"🫀 Heartbeat \$STATUS | CPU:\$CPU MEM:\$MEM\";
        if [ \"\$NOW\" = \"\$DAILY_REPORT_HOUR\" ]; then
          UPTIME=\$(uptime -p);
          MSG=\"📊 <b>Daily Health Report</b>%0A💾 CPU:\$CPU%0A🧮 MEM:\$MEM%0A⏱️ \$UPTIME%0A📅 \$(date '+%d/%m/%Y %H:%M')\";
          curl -s -o /dev/null -X POST https://api.telegram.org/bot\$TELEGRAM_BOT_TOKEN/sendMessage -d chat_id=\$TELEGRAM_CHAT_ID -d parse_mode=HTML -d text=\"\$MSG\";
          sleep 3600;
        fi;
        sleep \$HEARTBEAT_INTERVAL;
      done
    "

volumes:
  db_data:
  redis_data:
EOF

sudo docker-compose up -d
echo "✅ AlgoGPT Phase-2 containers are running."

# === PHASE-2.2: Autonomous Self-Fix + Supervisor Sync ===
cat > /usr/local/bin/algo-supervisor.sh <<'SCRIPT'
#!/bin/bash
LOGFILE="/var/log/algo-supervisor.log"
echo "👁️ Supervisor started at $(date)" >> $LOGFILE

while true; do
  # Verify containers
  for svc in core redis db dashboard; do
    if ! docker ps | grep -q "$svc"; then
      echo "⚠️ $svc down, restarting..." >> $LOGFILE
      docker restart $svc >/dev/null 2>&1
    fi
  done
  # Git sync
  if [ -n "$GITHUB_TOKEN" ]; then
    git -C /app pull origin main >> $LOGFILE 2>&1 || echo "⚠️ Git pull failed" >> $LOGFILE
  fi
  # AI-X status ping
  curl -s -X POST https://api.x.ai/v1/chat/completions \
    -H "Authorization: Bearer $AIX_SUPERVISOR_TOKEN" \
    -H "Content-Type: application/json" \
    -d '{"model":"grok-4-fast-non-reasoning","messages":[{"role":"user","content":"Ping from AlgoGPT Supervisor"}]}' >/dev/null 2>&1
  sleep 900
done
SCRIPT
chmod +x /usr/local/bin/algo-supervisor.sh
nohup /usr/local/bin/algo-supervisor.sh >/dev/null 2>&1 &

# === GITHUB INTEGRATION ===
if [ -n "$GITHUB_TOKEN" ]; then
  git config --global user.name "AlgoGPT-System"
  git config --global user.email "algogpt@system.local"
  git remote set-url origin "https://${GITHUB_TOKEN}@github.com/shawn2400/algogpt.git"
  echo "🔗 GitHub Linked (auto-sync enabled)"
fi

curl -fsS "$BASE_URL/readyz" && echo "🟢 Core Ready"
curl -fsS "$BASE_URL/version" && echo "🧠 Version OK"
echo "✅ Phase-2 + 2.2 Fully Active (Autonomous + Secure + Resilient)"
