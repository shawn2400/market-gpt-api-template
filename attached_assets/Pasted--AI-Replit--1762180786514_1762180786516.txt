הנה הודעה אחת, מוכנה להדבקה לסוכן ה-AI שלך (Replit). היא מקימה “**מנהל אסטרטגיה אוטומטי**” שמרכז מטרות (OKR), מפעיל *סיבוב מוחות* פנימי+חיצוני לפני כל החלטה, עושה סימולציה/בק־טסט מהיר, מצביע בקונצנזוס משוקלל, שומר ראיות (Audit Chain), ומבצע רק אחרי אישור כללי סיכון. 4GB-safe, מאובטח, פרודקשן.

---

**TASK NAME:** AlgoGPT — AI Strategy Manager (Brain-Round → Sim → Vote → Commit)

**מטרה:**
להוסיף מודול ניהול אסטרטגי מרכזי שאחראי על יעדי המערכת (OKR), על הפעלת *סיבוב מוחות* (פנימי: GPT-5/DeepSeek/Local + חיצוני: Cohere/Mistral/Perplexity/OpenRouter/TV CSV/Lean JSON), הרצת סימולציה/בדיקות מהירות, הצבעה משוקללת לפי אמינות/רקורד, ואישור החלטות רק לאחר עמידה בשומרי-סף (Risk & Cost). כל ריצה יוצרת דו״ח חתום, שקוף ונבדק.

**אל תשנה:** אימות/HMAC/Bearer, Anti-Replay, מנועי טרייד/ניהול קיימים, CORS, Docker/Render, נתיבי PROD אחרים.

## 2) סכמת DB (מיגרציה חדשה: `migrations/xxxx_strategy_manager.sql`)

```sql
CREATE TABLE IF NOT EXISTS okr_goals (
  id SERIAL PRIMARY KEY,
  code VARCHAR(48) UNIQUE NOT NULL,   -- e.g., "PNL_GROWTH_Q4"
  title TEXT NOT NULL,
  kpis JSONB NOT NULL,                -- [{"name":"PF","target":1.4}, ...]
  active BOOLEAN NOT NULL DEFAULT TRUE,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS brain_rounds (
  id BIGSERIAL PRIMARY KEY,
  ts TIMESTAMPTZ DEFAULT NOW(),
  topic TEXT NOT NULL,                -- תיאור ההחלטה/שאלה
  context JSONB NOT NULL,             -- מצב שוק, פוזיציות, משאבים
  sources JSONB NOT NULL,             -- מי נקרא לדיון הפעם
  status VARCHAR(16) NOT NULL DEFAULT 'OPEN' -- OPEN/DECIDED/ABORTED
);

CREATE TABLE IF NOT EXISTS brain_opinions (
  id BIGSERIAL PRIMARY KEY,
  round_id BIGINT REFERENCES brain_rounds(id) ON DELETE CASCADE,
  reviewer VARCHAR(32) NOT NULL,      -- gpt5/deepseek/cohere/...
  answer JSONB NOT NULL,              -- נימוק+הצעה+ציון ביטחון
  confidence NUMERIC(5,4) NOT NULL,   -- 0..1
  cost_usd NUMERIC(8,4) DEFAULT 0.0,
  latency_ms INT,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS sim_results (
  id BIGSERIAL PRIMARY KEY,
  round_id BIGINT REFERENCES brain_rounds(id) ON DELETE CASCADE,
  setup JSONB NOT NULL,               -- פרמטרי סימולציה
  metrics JSONB NOT NULL,             -- PF, Sharpe, Win%, DD, etc.
  verdict VARCHAR(16) NOT NULL,       -- PASS/FAIL
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS strategy_decisions (
  id BIGSERIAL PRIMARY KEY,
  round_id BIGINT REFERENCES brain_rounds(id) ON DELETE CASCADE,
  decision JSONB NOT NULL,            -- ההחלטה הסופית (מה לשנות/להפעיל)
  vote JSONB NOT NULL,                -- פירוט משקולות/ציונים/נומינציות
  passed BOOLEAN NOT NULL,            -- עבר סף קונצנזוס?
  risk_gate JSONB NOT NULL,           -- בדיקות סיכון/עלות
  audit_prev_hash TEXT,               -- hash השרשור הקודם
  audit_hash TEXT NOT NULL,           -- hash נוכחי (immutability)
  created_by VARCHAR(32) DEFAULT 'STRAT_MGR',
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE INDEX IF NOT EXISTS idx_brain_rounds_status ON brain_rounds(status);
```

## 3) לוגיקה — `utils/strategy_manager.py`

```python
# -*- coding: utf-8 -*-
from __future__ import annotations
import os, time, json, hashlib, datetime as dt
from typing import Dict, Any, List, Tuple
from decimal import Decimal

# שים לב: השתמשו ב-DB helpers הקיימים אצלכם (psycopg/async)
from utils.db import q, q1  # ספקו פונקציות עזר פשוטות (או החלף ב-psycopg2 כאן)
from utils.metrics_tracker import inc_counter as metr  # אם קיים
from utils.telegram_notifier import send_message

RSS_CAP = int(os.getenv("STRAT_RSS_CAP_MB","3100"))
CPU_CAP = int(os.getenv("STRAT_CPU_CAP_PCT","65"))
COST_CAP = Decimal(os.getenv("STRAT_COST_CAP_USD","4.00"))
MIN_CONF = float(os.getenv("STRAT_MIN_CONFIDENCE","0.62"))
VOTE_WEIGHTS = os.getenv("STRAT_VOTE_WEIGHTS","internal:0.45,external:0.35,sim:0.20")

def _parse_weights(s:str)->Dict[str,float]:
    out={}
    for kv in s.split(","):
        k,v=kv.split(":"); out[k.strip()]=float(v)
    return out
W = _parse_weights(VOTE_WEIGHTS)

def _sys_ok()->Tuple[bool,str]:
    # התחבר למדידות קיימות אם יש; כאן נניח OK
    # אפשר לקרוא /metrics או פונקציה קיימת אצלכם
    return True,"ok"

def open_round(topic:str, context:Dict[str,Any], sources:List[str])->int:
    r = q1("""INSERT INTO brain_rounds (topic,context,sources) VALUES (%s,%s,%s) RETURNING id""",
           (topic, json.dumps(context), json.dumps(sources)))
    return r["id"]

def record_opinion(round_id:int, reviewer:str, answer:Dict[str,Any], conf:float, cost:Decimal, latency_ms:int):
    q("""INSERT INTO brain_opinions (round_id,reviewer,answer,confidence,cost_usd,latency_ms)
         VALUES (%s,%s,%s,%s,%s,%s)""",
      (round_id, reviewer, json.dumps(answer), conf, str(cost), latency_ms))

def record_sim(round_id:int, setup:Dict[str,Any], metrics:Dict[str,Any], verdict:str):
    q("""INSERT INTO sim_results (round_id,setup,metrics,verdict) VALUES (%s,%s,%s,%s)""",
      (round_id, json.dumps(setup), json.dumps(metrics), verdict))

def _hash(prev_hash:str|None, payload:Dict[str,Any])->str:
    s=(prev_hash or "")+json.dumps(payload,sort_keys=True)
    return hashlib.sha256(s.encode("utf-8")).hexdigest()

def _vote(round_id:int)->Tuple[bool,Dict[str,Any]]:
    # שליפת דעות
    ops = q("""SELECT reviewer,confidence,answer FROM brain_opinions WHERE round_id=%s""",(round_id,))
    sims = q("""SELECT verdict,metrics FROM sim_results WHERE round_id=%s""",(round_id,))
    score_internal=0.0; score_external=0.0; score_sim=0.0
    n_int=0; n_ext=0
    # נאפיין reviewers פנימיים/חיצוניים בשם
    for o in ops:
        name=o["reviewer"]
        conf=float(o["confidence"])
        if name in ("gpt5","deepseek","local"):
            score_internal+=conf; n_int+=1
        else:
            score_external+=conf; n_ext+=1
    if n_int: score_internal/=n_int
    if n_ext: score_external/=n_ext
    # סימולציה: PASS=1 FAIL=0; או מיפוי לפי PF/Sharpe
    if sims:
        passed=sum(1 for s in sims if s["verdict"]=="PASS")
        score_sim = passed/len(sims)
    final = W.get("internal",0)*score_internal + W.get("external",0)*score_external + W.get("sim",0)*score_sim
    return (final>=MIN_CONF, {
        "score_internal": round(score_internal,3),
        "score_external": round(score_external,3),
        "score_sim": round(score_sim,3),
        "weights": W,
        "final": round(final,3)
    })

def _risk_gates(topic:str, cost_used:Decimal)->Tuple[bool,Dict[str,Any]]:
    ok, reason = _sys_ok()
    if not ok: 
        return False, {"system":"overload", "reason":reason}
    if cost_used > COST_CAP:
        return False, {"cost":"cap_exceeded", "spent": str(cost_used), "cap": str(COST_CAP)}
    return True, {"ok":True}

def commit_decision(round_id:int, decision:Dict[str,Any], vote:Dict[str,Any], risk:Dict[str,Any], prev_hash:str|None)->Dict[str,Any]:
    payload={"round_id":round_id,"decision":decision,"vote":vote,"risk":risk,"ts":dt.datetime.utcnow().isoformat()}
    h=_hash(prev_hash, payload)
    q("""INSERT INTO strategy_decisions (round_id,decision,vote,passed,risk_gate,audit_prev_hash,audit_hash)
         VALUES (%s,%s,%s,%s,%s,%s,%s)""",
      (round_id, json.dumps(decision), json.dumps(vote), True, json.dumps(risk), prev_hash, h))
    return {"audit_hash":h}

def abort_round(round_id:int, reason:str):
    q("""UPDATE brain_rounds SET status='ABORTED' WHERE id=%s""",(round_id,))
    # אפשר להוסיף audit/log

def close_round(round_id:int):
    q("""UPDATE brain_rounds SET status='DECIDED' WHERE id=%s""",(round_id,))
```

## 4) חיבור “מבצעי” — Orchestrator + Reviewers + סימולציה

**קבצים:**

* `utils/reviewers_external.py` — מחברים בקלות ל-Cohere/Mistral/Perplexity/OpenRouter (קריאה בלבד; עם timeouts ותקציב).
* `utils/strategy_orchestrator.py` — הפייפליין המלא: *Round → Gather → Sim → Vote → Commit → Notify*.
* `utils/strategy_sim.py` — סימולציית בזק (≤ STRAT_SIM_MAX_MIN) על סט מדדים/סימבולים מצומצם (4GB-safe).

**דוגמת Orchestrator (תקציר):**

```python
# -*- coding: utf-8 -*-
from __future__ import annotations
import os, time, json, datetime as dt
from decimal import Decimal
from utils.strategy_manager import (open_round, record_opinion, record_sim, _vote, _risk_gates, commit_decision, close_round, abort_round)
from utils.reviewers_external import call_reviewers   # תממשו עם timeouts ועלות
from utils.strategy_sim import quick_simulate         # מפיק PF/Sharpe/Win%/DD + PASS/FAIL
from utils.telegram_notifier import send_message

def run_brain_round(topic:str, context:dict)->dict:
    # 1) פתיחת סבב
    sources = ["gpt5","deepseek","cohere","mistral","perplexity","openrouter"]
    rid = open_round(topic, context, sources)
    cost_used = Decimal("0.0")

    # 2) איסוף דעות (פנימיים+חיצוניים)
    outs = call_reviewers(topic=topic, context=context, reviewers=sources, budget_usd=float(os.getenv("STRAT_COST_CAP_USD","4")))
    for o in outs["opinions"]:
        record_opinion(rid, o["reviewer"], o["answer"], o["confidence"], Decimal(str(o.get("cost_usd",0))), o.get("latency_ms",0))
        cost_used += Decimal(str(o.get("cost_usd",0)))

    # 3) סימולציית בזק (≤ N דקות)
    sim = quick_simulate(topic, context, max_minutes=int(os.getenv("STRAT_SIM_MAX_MIN","6")))
    for s in sim["runs"]:
        record_sim(rid, s["setup"], s["metrics"], s["verdict"])

    # 4) הצבעה משוקללת
    passed, vote = _vote(rid)

    # 5) שומרי סף (משאבים/עלות)
    ok, risk = _risk_gates(topic, cost_used)
    if not (passed and ok):
        abort_round(rid, "vote_fail_or_risk")
        send_message(f"🤖 BrainRound ABORTED\nTopic: {topic}\nVote: {json.dumps(vote)}\nRisk: {json.dumps(risk)}")
        return {"ok": False, "round_id": rid, "vote": vote, "risk": risk}

    # 6) החלטה (דוגמה: שינוי פרמטרים/הפעלת אסטרטגיה)
    decision = {
        "action": "update_strategy_params",
        "params": sim.get("suggested_params", {"ATR_MULT":1.5, "ADX_MIN":18}),
        "reason": "consensus_and_sim_pass"
    }
    audit = commit_decision(rid, decision, vote, risk, prev_hash=None)
    close_round(rid)

    # 7) נוטיפיקציה
    send_message(f"✅ BrainRound PASSED\nTopic: {topic}\nVote: {json.dumps(vote)}\nDecision: {json.dumps(decision)}\nAudit: {audit['audit_hash']}")
    return {"ok": True, "round_id": rid, "vote": vote, "decision": decision, "audit": audit}
```

## 5) נתיבי API — `routes/strategy_manager.py` (Bearer/HMAC)

```python
# -*- coding: utf-8 -*-
from __future__ import annotations
from fastapi import APIRouter, Depends, HTTPException
from pydantic import BaseModel
from typing import Any, Dict
from utils.strategy_orchestrator import run_brain_round
from utils.strategy_manager import q, q1

def bearer_ok(): return True  # חברו לבודק אצלכם

router = APIRouter(prefix="/ai/strategy", tags=["AI-Strategy"], dependencies=[Depends(bearer_ok)])

class RoundIn(BaseModel):
    topic: str
    context: Dict[str, Any]

@router.post("/round")
def open_and_run_round(body: RoundIn):
    return run_brain_round(body.topic, body.context)

@router.get("/okr")
def okr_list():
    return {"okrs": q("SELECT * FROM okr_goals WHERE active=TRUE ORDER BY id")}

@router.get("/decision/{round_id}")
def decision(round_id: int):
    r = q1("SELECT * FROM strategy_decisions WHERE round_id=%s",(round_id,))
    if not r: return {"found": False}
    return {"found": True, "decision": r}
```

## 7) סימולציית בזק — `utils/strategy_sim.py` (תקציר 4GB-safe)

* מריץ Backtest קצר על 6–12 סמלים/טיימפריימים עם data מה-cache (Redis/Batch).
* מדדים: PF, Sharpe, Win%, MaxDD, Time-to-TP.
* פסק דין: PASS אם PF≥1.2 ו-DD≤12% ו-Sharpe≥0.7 (דוגמה).
* מכין `suggested_params` ע״פ תוצאות.

## 9) Guardrails & מדיניות החלטה

* **חובה**: Brain-Round ≥ 1 פנימי + 1 חיצוני + סימולציית בזק.
* **סף קונצנזוס**: ציון סופי ≥ `STRAT_MIN_CONFIDENCE`.
* **סף סיכון**: RSS/CPU/Cost מתחת לתקרות ENV.
* **אישור ידני (טלגרם)**: אם `final∈[0.58..0.62]` → דורש קליק `/approve`.
* **איסור**: אין שינוי אסטרטגיות בלייב אם יש >0 פוזיציות פעילות באותו נכס.
