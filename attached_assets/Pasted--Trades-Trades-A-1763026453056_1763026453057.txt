```python
def calculate_position_size(symbol_volatility, account_balance):
    base_size = 0.02 * account_balance  # 2% מהחשבון
    
    if symbol_volatility > 0.08:
        return base_size * 0.5   # חצי position לתנודתי
    elif symbol_volatility < 0.02:
        return base_size * 1.5   # position מלא ליציב
    
    return base_size
```
