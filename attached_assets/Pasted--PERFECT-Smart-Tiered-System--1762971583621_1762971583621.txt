# 🎯 SMART TIERED QUALITY SYSTEM
# ==============================

class SmartConfluenceHelper:
    def __init__(self):
        self.market_strength_thresholds = {
            'strong': 7.0,
            'normal': 4.0,
            'weak': 0.0
        }
        
        self.quality_tiers = {
            'strong_market': {
                'min_quality': 4.4,
                'filters': ['basic_validation'],
                'description': 'Standard trading - good market conditions'
            },
            'smart_filter': {
                'min_quality': 4.5,
                'filters': [
                    'volume_1.2x_average',
                    'trend_4h_alignment', 
                    'risk_reward_1.8+',
                    'momentum_confirmation'
                ],
                'description': 'Enhanced filters - moderate market conditions'
            },
            'protection_mode': {
                'min_quality': 6.0,
                'filters': ['maximum_protection'],
                'description': 'Only high conviction - weak market conditions'
            }
        }
    
    def calculate_market_strength(self, volatility, volume_trend, btc_dominance):
        """Calculate overall market strength score"""
        strength = 0.0
        
        # Volatility component (30%)
        if volatility < 2.0:
            strength += 2.0  # Low volatility = good
        elif volatility < 4.0:
            strength += 1.0  # Moderate volatility
        else:
            strength += 0.0  # High volatility = bad
            
        # Volume trend component (40%)
        if volume_trend > 1.3:
            strength += 2.8  # Strong volume
        elif volume_trend > 1.0:
            strength += 1.4  # Normal volume
        else:
            strength += 0.0  # Weak volume
            
        # BTC dominance component (30%)
        if 40 <= btc_dominance <= 60:
            strength += 2.1  # Healthy dominance
        else:
            strength += 0.9  # Extreme dominance = risk
            
        return min(10.0, strength)
    
    def get_trading_tier(self, market_strength):
        """Determine which trading tier to use"""
        if market_strength >= self.market_strength_thresholds['strong']:
            return self.quality_tiers['strong_market']
        elif market_strength >= self.market_strength_thresholds['normal']:
            return self.quality_tiers['smart_filter']
        else:
            return self.quality_tiers['protection_mode']
    
    def validate_trade(self, symbol, quality_score, market_data, technicals):
        """Validate trade using smart tiered system"""
        
        # Calculate market strength
        market_strength = self.calculate_market_strength(
            market_data['volatility'],
            market_data['volume_trend'], 
            market_data['btc_dominance']
        )
        
        # Get appropriate tier
        tier = self.get_trading_tier(market_strength)
        
        print(f"🎯 Market Strength: {market_strength:.1f}/10.0")
        print(f"📊 Trading Tier: {tier['description']}")
        print(f"🎯 Quality Required: {tier['min_quality']}")
        
        # Base quality check
        if quality_score < tier['min_quality']:
            return False, f"Quality {quality_score} < {tier['min_quality']}"
        
        # Tier-specific validation
        if tier['min_quality'] == 4.5:  # Smart filter tier
            if not self._validate_smart_filters(technicals):
                return False, "Failed smart filters"
        
        return True, f"Trade validated - {tier['description']}"
    
    def _validate_smart_filters(self, technicals):
        """Validate smart filters for tier 2"""
        checks = {
            'volume': technicals.get('volume_ratio', 0) >= 1.2,
            'trend': technicals.get('trend_alignment_4h', False),
            'risk_reward': technicals.get('rr_ratio', 0) >= 1.8,
            'momentum': technicals.get('momentum_score', 0) >= 5.0
        }
        
        print("🔍 Smart Filters Check:")
        for check, passed in checks.items():
            print(f"   {'✅' if passed else '❌'} {check}")
            
        return all(checks.values())

# Test the system
print("🧪 Testing Smart Tiered System")

helper = SmartConfluenceHelper()

    },
    {

        'market_data': {'volatility': 5.0, 'volume_trend': 0.8, 'btc_dominance': 65},
        'quality': 5.8,

for test in test_cases:
