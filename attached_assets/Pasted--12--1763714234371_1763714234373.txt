```python
class MultiTimeframeMomentumEngine:
    def __init__(self):
        self.timeframes = ['1m', '5m', '15m', '1h', '4h', '1d']
        self.momentum_weights = {
            '1m': 0.05, '5m': 0.15, '15m': 0.25, 
            '1h': 0.30, '4h': 0.20, '1d': 0.05
        }
    
    def calculate_fused_momentum(self, symbol: str) -> dict:
        """מיזוג מומנטום מכל ה-timeframes"""
        momentum_data = {}
        
        for tf in self.timeframes:
            tf_data = self.get_timeframe_data(symbol, tf)
            momentum_data[tf] = self.calculate_tf_momentum(tf_data)
        
        # מיזוג משוקלל
        fused_momentum = self.fuse_momentum_signals(momentum_data)
        
        return {
            'fused_momentum_score': fused_momentum['composite_score'],
            'momentum_alignment': fused_momentum['alignment_score'],
            'timeframe_confluence': self.calculate_confluence(momentum_data),
            'momentum_trend': self.calculate_momentum_trend(momentum_data),
            'key_timeframe_signals': self.extract_key_signals(momentum_data)
        }
    
    def fuse_momentum_signals(self, momentum_data: dict) -> dict:
        """מיזוג אותות מומנטום"""
        weighted_scores = []
        directions = []
        
        for tf, data in momentum_data.items():
            weight = self.momentum_weights.get(tf, 0.1)
            weighted_scores.append(data['score'] * weight)
            directions.append(data['direction'])
        
        composite_score = sum(weighted_scores)
        
        # בדיקת alignement בין timeframes
        alignment_score = len(set(directions)) / len(directions)
        
        return {
            'composite_score': composite_score,
            'alignment_score': alignment_score,
            'direction_consensus': max(set(directions), key=directions.count),
            'momentum_strength': np.std(weighted_scores)  # וולאטיליות במומנטום
        }
```

class DynamicCorrelationMatrix:

        """עדכן מטריצת קורלציה דינאמית"""
        correlation_data = {}
        
        for i, asset1 in enumerate(portfolio):
            for j, asset2 in enumerate(portfolio):
                if i < j:
                    corr_data = self.calculate_pair_correlation(asset1, asset2)
                    correlation_data[f"{asset1}_{asset2}"] = corr_data

        clusters = self.cluster_analyzer.detect_clusters(correlation_data)
        
        return {
            'correlation_matrix': correlation_data,
            'cluster_analysis': clusters,

            'concentration_risk': self.assess_concentration_risk(correlation_data, clusters),
            'hedging_opportunities': self.identify_hedging_opportunities(correlation_data)

    def calculate_pair_correlation(self, asset1: str, asset2: str) -> dict:

        returns1 = self.get_asset_returns(asset1)

class AdaptiveLearningLoop:

        performance_analysis = self.performance_tracker.analyze(trading_data)

            strategy_improvements, 
            parameter_updates
        )

            'performance_insights': performance_analysis,
            'strategy_updates': strategy_improvements,
            'parameter_updates': parameter_updates,
            'validation_results': validation_results,
            'learning_effectiveness': self.measure_learning_effectiveness(performance_analysis)
        }

        """מדידת effectiveness של הלמידה"""

            'sharpe_improvement': performance['current_sharpe'] - performance['baseline_sharpe'],
            'drawdown_improvement': performance['baseline_drawdown'] - performance['current_drawdown'],

        """אימות וסינון אנומליות"""
        validated = {}
        
        for model_name, model_anomalies in anomalies.items():
            for anomaly in model_anomalies:
                # בדיקת cross-validation עם מודלים אחרים
                cross_validation_score = self.cross_validate_anomaly(anomaly, anomalies)
                
                if cross_validation_score > 0.7:  # threshold
                    validated[anomaly['id']] = {
                        **anomaly,
                        'validation_score': cross_validation_score,
                        'confirmed_by_models': self.count_confirming_models(anomaly, anomalies)
                    }
        
        return validated
