### 1) Neon Resume – קריאות API אוטומטיות + בריאות

**צור קובץ חדש:** `utils/neon_resume.py`

```python
# -*- coding: utf-8 -*-
import os, json, time
import urllib.request

NEON_API = "https://console.neon.tech/api/v2"

def _req(method: str, path: str, data: dict|None=None) -> dict:
    api_key = os.getenv("NEON_API_KEY", "")
    if not api_key:
        return {"ok": False, "err": "NEON_API_KEY missing"}
    url = f"{NEON_API}{path}"
    body = None
    headers = {"Authorization": f"Bearer {api_key}", "Content-Type": "application/json"}
    if data is not None:
        body = json.dumps(data).encode("utf-8")
    req = urllib.request.Request(url, data=body, method=method, headers=headers)
    with urllib.request.urlopen(req, timeout=20) as r:
        return {"ok": True, "data": json.loads(r.read().decode("utf-8"))}

def resume_endpoint(project_id: str, endpoint_id: str) -> dict:
    # Neon: POST /projects/{project_id}/endpoints/{endpoint_id}/resume
    return _req("POST", f"/projects/{project_id}/endpoints/{endpoint_id}/resume")

def ensure_neon_running() -> dict:
    pid = os.getenv("NEON_PROJECT_ID", "")
    eid = os.getenv("NEON_ENDPOINT_ID", "")
    if not pid or not eid:
        return {"ok": False, "err": "NEON_PROJECT_ID / NEON_ENDPOINT_ID missing"}
    res = resume_endpoint(pid, eid)
    # המתן רגע שה־compute יעלה
    time.sleep(3)
    return res
```

### 3) TP Ladder – מניעת מחירים שליליים + דיוק Tick/Precision

**קובץ חדש:** `utils/price_math.py`

```python
def clamp_tick(price: float, tick_size: float) -> float:
    if tick_size <= 0: return max(price, 0.0)
    steps = max(int(round(price / tick_size)), 0)
    return max(steps * tick_size, 0.0)

def compute_tp_prices(entry: float, side: str, atr: float, levels=(1.8, 3.2), tick=0.0001) -> list[float]:
    """
    side: 'LONG' or 'SHORT'
    levels: ATR multipliers
    """
    entry = float(entry); atr = abs(float(atr))
    if entry <= 0: return []
    tps = []
    for m in levels:
        delta = atr * float(m)
        if side.upper() == "LONG":
            p = entry + delta
        else:
            p = entry - delta
        p = clamp_tick(p, tick)
        # אין מצב שלילי/אפס ב-TP – אם יצא 0, קדם מינימום Tick
        if p <= 0 and tick > 0:
            p = tick
        tps.append(p)
    return tps
```
