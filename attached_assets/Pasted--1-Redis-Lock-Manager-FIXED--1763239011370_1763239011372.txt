```python
# 📁 utils/advanced_risk_manager.py - התחלה

class AdvancedRiskManager:
    def __init__(self, account_balance=1000, max_risk_per_trade=0.02):
        self.account_balance = account_balance
        self.max_risk_per_trade = max_risk_per_trade
        self.active_trades = 0
        self.max_concurrent_trades = 3
    
    def calculate_dynamic_sl_tp(self, atr_value, current_price, direction):
        """מחשב Stop Loss ו-Take Profit דינמי based on ATR"""
        atr_multiplier = 1.5
        sl_distance = atr_value * atr_multiplier
        
        if direction == "long":
            stop_loss = current_price - sl_distance
            take_profit = current_price + (sl_distance * 2)  # 1:2 risk/reward
        else:
            stop_loss = current_price + sl_distance  
            take_profit = current_price - (sl_distance * 2)
        
        return {
            "stop_loss": round(stop_loss, 4),
            "take_profit": round(take_profit, 4),
            "sl_distance_percent": round((sl_distance / current_price) * 100, 2)
        }
    
    def get_position_size(self, stop_loss_distance, confidence_score):
        """מחזיר גודל פוזיציה מותאם ל-confidence"""
        base_risk = self.max_risk_per_trade * confidence_score
        position_size = (self.account_balance * base_risk) / (stop_loss_distance / 100)
        
        return min(position_size, self.account_balance * 0.1)  # max 10% מהון
```
