# main_system_integrator.py
import asyncio
import time
import logging
from typing import Dict, List
from central_orchestrator import CentralOrderOrchestrator
from hybrid_timeout_manager import HybridTimeoutManager
from smart_cleanup_manager import SmartCleanupManager
from resilient_error_handler import ResilientErrorHandler
from unified_state_tracker import UnifiedStateTracker

logger = logging.getLogger(__name__)

class RootCauseSolution:
    """
    🚀 הפתרון השורשי המלא - כל המערכת במקום אחד
    """
    
    def __init__(self, binance_client, redis_client=None):
        self.binance = binance_client
        self.redis_client = redis_client
        self.orchestrator = CentralOrderOrchestrator(binance_client)
        self.is_running = False
        self.start_time = time.time()
        
        # Inject dependencies
        self.orchestrator.timeout_manager = HybridTimeoutManager(self.orchestrator)
        self.orchestrator.cleanup_manager = SmartCleanupManager(self.orchestrator)
        self.orchestrator.error_handler = ResilientErrorHandler(self.orchestrator)
        self.orchestrator.state_tracker = UnifiedStateTracker(self.orchestrator)
        
        # Set Redis client if available
        if redis_client:
            self.orchestrator.state_tracker.redis_client = redis_client
        
    async def start_complete_system(self):
        """הפעלת המערכת השורשית המלאה"""
        if self.is_running:
            logger.warning("⚠️ System already running")
            return
        
        logger.info("🎯 STARTING ROOT CAUSE SOLUTION - COMPLETE SYSTEM")
        
        try:
            # Set start time for uptime tracking
            self.orchestrator.start_time = self.start_time
            
            # Initialize all components
            await self.initialize_components()
            
            # Start all engines
            await self.start_all_engines()
            
            # Verify system health
            await self.verify_system_health()
            
            self.is_running = True
            logger.info("✅ ROOT CAUSE SOLUTION ACTIVATED - System is now stable!")
            
            # Continuous health monitoring
            asyncio.create_task(self.continuous_health_monitoring())
            
        except Exception as e:
            logger.error(f"❌ Failed to start root cause solution: {e}")
            await self.emergency_fallback()
            raise
    
    async def initialize_components(self):
        """אתחול כל הקומפוננטות"""
        logger.info("🔧 Initializing system components...")
        
        components = [
            self.orchestrator.timeout_manager,
            self.orchestrator.cleanup_manager, 
            self.orchestrator.error_handler,
            self.orchestrator.state_tracker
        ]
        
        for component in components:
            component_name = component.__class__.__name__
            try:
                if hasattr(component, 'initialize'):
                    await component.initialize()
                    logger.info(f"✅ {component_name} initialized")
                else:
                    logger.info(f"✅ {component_name} ready (no initialize method)")
            except Exception as e:
                logger.error(f"❌ Failed to initialize {component_name}: {e}")
                raise
        
        logger.info("🎉 All components initialized successfully")
    
    async def start_all_engines(self):
        """הפעלת כל המנועים"""
        logger.info("🚀 Starting all system engines...")
        
        # Start orchestrator (starts all child processes)
        await self.orchestrator.start_orchestration()
        
        # Start cleanup engine
        asyncio.create_task(self.orchestrator.cleanup_manager.start_cleanup_engine())
        
        # Additional system processes
        additional_tasks = [
            self.system_health_monitor(),
            self.performance_reporter(),
            self.emergency_recovery_monitor(),
            self.state_tracker.track_system_state()
        ]
        
        for task in additional_tasks:
            asyncio.create_task(task)
        
        logger.info("✅ All engines started")
    
    async def system_health_monitor(self):
        """ניטור בריאות המערכת"""
        logger.info("🏥 Starting system health monitor...")
        
        while self.is_running:
            try:
                health_status = await self.check_system_health()
                
                if health_status['status'] != 'HEALTHY':
                    logger.warning(f"⚠️ System health issue: {health_status['issues']}")
                    
                    # Add to recovery queue
                    await self.orchestrator.recovery_queue.put({
                        'type': 'SYSTEM_HEALTH',
                        'issues': health_status['issues'],
                        'severity': health_status['severity']
                    })
                
                await asyncio.sleep(60)  # Check every minute
                
            except Exception as e:
                logger.error(f"❌ Health monitor failed: {e}")
                await asyncio.sleep(30)
    
    async def check_system_health(self) -> Dict:
        """בדיקת בריאות מקיפה של המערכת"""
        health_issues = []
        
        try:
            # 1. Check order states consistency
            active_orders_count = len([
                state for state in self.orchestrator.order_states.values() 
                if state['status'] == 'ACTIVE'
            ])
            
            if active_orders_count > 50:  # Arbitrary limit
                health_issues.append("High number of active orders")
            
            # 2. Check error rates
            total_errors = sum(self.orchestrator.error_handler.error_counts.values())
            if total_errors > 20:
                health_issues.append(f"High error count: {total_errors}")
            
            # 3. Check recovery queue size
            queue_size = self.orchestrator.recovery_queue.qsize()
            if queue_size > 10:
                health_issues.append(f"Large recovery queue: {queue_size}")
            
            # 4. Check system responsiveness
            current_time = time.time()
            stale_orders = 0
            for order_state in self.orchestrator.order_states.values():
                if current_time - order_state['last_checked'] > 300:  # 5 minutes
                    stale_orders += 1
            
            if stale_orders > 5:
                health_issues.append(f"Stale orders detected: {stale_orders}")
            
            # 5. Check memory usage (simplified)
            if len(self.orchestrator.order_states) > 1000:
                health_issues.append("Large state memory usage")
            
            # Determine overall status
            if not health_issues:
                status = 'HEALTHY'
                severity = 'LOW'
            elif len(health_issues) <= 2:
                status = 'DEGRADED'
                severity = 'MEDIUM'
            else:
                status = 'UNHEALTHY'
                severity = 'HIGH'
            
            return {
                'status': status,
                'issues': health_issues,
                'severity': severity,
                'timestamp': current_time,
                'active_orders': active_orders_count,
                'total_errors': total_errors,
                'recovery_queue_size': queue_size
            }
            
        except Exception as e:
            logger.error(f"❌ Health check failed: {e}")
            return {
                'status': 'UNKNOWN',
                'issues': [f'Health check error: {e}'],
                'severity': 'HIGH',
                'timestamp': time.time()
            }
    
    async def performance_reporter(self):
        """דיווח ביצועים"""
        logger.info("📊 Starting performance reporter...")
        
        report_count = 0
        
        while self.is_running:
            try:
                report_count += 1
                
                # Generate performance report every 5 minutes
                if report_count % 30 == 0:  # 30 * 10 seconds = 5 minutes
                    await self.generate_performance_report()
                
                # Basic stats logging every 10 minutes
                if report_count % 60 == 0:
                    await self.log_system_stats()
                
                await asyncio.sleep(10)  # Every 10 seconds
                
            except Exception as e:
                logger.error(f"❌ Performance reporter failed: {e}")
                await asyncio.sleep(30)
    
    async def generate_performance_report(self):
        """יצירת דוח ביצועים"""
        try:
            system_summary = await self.orchestrator.state_tracker.get_system_summary()
            
            report = {
                'timestamp': time.time(),
                'uptime': time.time() - self.start_time,
                'performance_metrics': self.orchestrator.performance_metrics,
                'system_summary': system_summary,
                'error_stats': self.orchestrator.error_handler.error_counts,
                'circuit_breakers': len(self.orchestrator.error_handler.circuit_breakers)
            }
            
            logger.info(f"📈 Performance Report:")
            logger.info(f"   Uptime: {report['uptime']:.0f}s")
            logger.info(f"   Active Orders: {system_summary['orders_summary']['active']}")
            logger.info(f"   Success Rate: {system_summary['performance'].get('success_rate', 0):.1%}")
            logger.info(f"   Health: {system_summary['health'].get('status', 'UNKNOWN')}")
            logger.info(f"   Circuit Breakers: {report['circuit_breakers']}")
            
            # Store report for historical analysis
            if hasattr(self, 'performance_reports'):
                self.performance_reports.append(report)
                # Keep only last 100 reports
                if len(self.performance_reports) > 100:
                    self.performance_reports.pop(0)
            
        except Exception as e:
            logger.error(f"❌ Performance report generation failed: {e}")
    
    async def log_system_stats(self):
        """רישום סטטיסטיקות מערכת"""
        try:
            stats = {
                'timestamp': time.time(),
                'order_states_count': len(self.orchestrator.order_states),
                'active_orders': len([
                    s for s in self.orchestrator.order_states.values() 
                    if s['status'] == 'ACTIVE'
                ]),
                'recovery_queue_size': self.orchestrator.recovery_queue.qsize(),
                'error_counts': dict(self.orchestrator.error_handler.error_counts),
                'system_uptime': time.time() - self.start_time
            }
            
            logger.info(f"📊 System Stats: {stats}")
            
        except Exception as e:
            logger.error(f"❌ System stats logging failed: {e}")
    
    async def emergency_recovery_monitor(self):
        """ניטור התאוששות חירום"""
        logger.info("🛡️ Starting emergency recovery monitor...")
        
        while self.is_running:
            try:
                # Check for critical conditions
                critical_conditions = await self.check_critical_conditions()
                
                if critical_conditions:
                    logger.critical(f"🚨 Critical conditions detected: {critical_conditions}")
                    await self.initiate_emergency_recovery(critical_conditions)
                
                await asyncio.sleep(30)  # Check every 30 seconds
                
            except Exception as e:
                logger.error(f"❌ Emergency recovery monitor failed: {e}")
                await asyncio.sleep(60)
    
    async def check_critical_conditions(self) -> List[str]:
        """בדיקת תנאים קריטיים"""
        critical_conditions = []
        current_time = time.time()
        
        try:
            # 1. Check for too many errors
            total_errors = sum(self.orchestrator.error_handler.error_counts.values())
            if total_errors > 50:
                critical_conditions.append(f"Very high error count: {total_errors}")
            
            # 2. Check for system unresponsiveness
            recent_activity = 0
            for order_state in self.orchestrator.order_states.values():
                if current_time - order_state['last_checked'] < 60:  # Last minute
                    recent_activity += 1
            
            if recent_activity == 0 and len(self.orchestrator.order_states) > 0:
                critical_conditions.append("System completely unresponsive")
            
            # 3. Check for memory explosion
            if len(self.orchestrator.order_states) > 5000:
                critical_conditions.append("Extremely large state memory")
            
            # 4. Check for too many circuit breakers
            if len(self.orchestrator.error_handler.circuit_breakers) > 5:
                critical_conditions.append("Multiple circuit breakers active")
            
            return critical_conditions
            
        except Exception as e:
            logger.error(f"❌ Critical conditions check failed: {e}")
            return ["Critical check failure"]
    
    async def initiate_emergency_recovery(self, conditions: List[str]):
        """הפעלת התאוששות חירום"""
        try:
            logger.critical("🔄 INITIATING EMERGENCY RECOVERY")
            
            # 1. Stop all active monitoring temporarily
            self.is_running = False
            await asyncio.sleep(2)  # Brief pause
            
            # 2. Clear recovery queue
            while not self.orchestrator.recovery_queue.empty():
                try:
                    self.orchestrator.recovery_queue.get_nowait()
                except:
                    break
            
            # 3. Reset error counts
            self.orchestrator.error_handler.error_counts.clear()
            
            # 4. Clear circuit breakers
            self.orchestrator.error_handler.circuit_breakers.clear()
            
            # 5. Clean up order states (keep only recent active ones)
            current_time = time.time()
            orders_to_keep = {}
            
            for order_id, order_state in self.orchestrator.order_states.items():
                if (order_state['status'] == 'ACTIVE' and 
                    current_time - order_state['created_at'] < 3600):  # Keep active orders from last hour
                    orders_to_keep[order_id] = order_state
            
            self.orchestrator.order_states = orders_to_keep
            
            # 6. Restart the system
            self.is_running = True
            logger.critical("✅ EMERGENCY RECOVERY COMPLETED - System restarted")
            
        except Exception as e:
            logger.critical(f"💥 EMERGENCY RECOVERY FAILED: {e}")
            # Ultimate fallback - complete shutdown might be needed
            await self.complete_shutdown()
    
    async def verify_system_health(self):
        """ולידציה של בריאות המערכת לאחר האתחול"""
        logger.info("🔍 Verifying system health after startup...")
        
        checks = [
            self.verify_exchange_connection(),
            self.verify_order_tracking(),
            self.verify_error_handler(),
            self.verify_cleanup_manager()
        ]
        
        results = await asyncio.gather(*checks, return_exceptions=True)
        
        successful_checks = 0
        for i, result in enumerate(results):
            check_name = ["Exchange Connection", "Order Tracking", "Error Handler", "Cleanup Manager"][i]
            
            if isinstance(result, Exception):
                logger.error(f"❌ {check_name} check failed: {result}")
            else:
                logger.info(f"✅ {check_name} check passed")
                successful_checks += 1
        
        if successful_checks == len(checks):
            logger.info("🎉 All system health checks passed!")
        else:
            logger.warning(f"⚠️ {successful_checks}/{len(checks)} health checks passed")
    
    async def verify_exchange_connection(self) -> bool:
        """ולידציה של חיבור Exchange"""
        try:
            await self.binance.ping()
            account = await self.binance.get_account()
            return account is not None
        except Exception as e:
            logger.error(f"❌ Exchange connection verification failed: {e}")
            return False
    
    async def verify_order_tracking(self) -> bool:
        """ולידציה של מעקב הזמנות"""

            test_order = {

            return False
    
    async def verify_error_handler(self) -> bool:
        """ולידציה של מטפל השגיאות"""
        try:
            # Test error handling with a mock error
            test_context = {'message': 'Verification test'}
            await self.orchestrator.error_handler.handle_error('NETWORK_ERROR', test_context)
            return True
        except Exception as e:
            logger.error(f"❌ Error handler verification failed: {e}")
            return False
    
    async def verify_cleanup_manager(self) -> bool:
        """ולידציה של מנהל הניקוי"""
        try:
            # Test cleanup initialization
            await self.orchestrator.cleanup_manager.cleanup_old_order_states()
            return True
        except Exception as e:
            logger.error(f"❌ Cleanup manager verification failed: {e}")
            return False
    
    async def continuous_health_monitoring(self):
        """ניטור בריאות רציף"""
        while self.is_running:
            try:
                # Basic heartbeat
                if self.orchestrator.performance_metrics:
                    logger.debug("💓 System heartbeat - running normally")
                
                await asyncio.sleep(300)  # Every 5 minutes
                
            except Exception as e:
                logger.error(f"❌ Continuous health monitoring failed: {e}")
                await asyncio.sleep(60)
    
    async def emergency_fallback(self):
        """נפילת חירום למערכת בסיסית"""
        logger.critical("🆘 ACTIVATING EMERGENCY FALLBACK MODE")
        
        try:
            # Implement basic order management without advanced features
            # This ensures the system can still function minimally
            logger.info("🔧 Emergency fallback activated - basic functionality only")
            
        except Exception as e:
            logger.critical(f"💥 EMERGENCY FALLBACK ALSO FAILED: {e}")
    
    async def complete_shutdown(self):
        """כיבוי מלא של המערכת"""
        logger.critical("🛑 INITIATING COMPLETE SYSTEM SHUTDOWN")
        
        self.is_running = False
        
        try:
            # Cancel all active orders
            active_orders = [
                (order_id, state) for order_id, state in self.orchestrator.order_states.items()
                if state['status'] == 'ACTIVE'
            ]
            
            for order_id, state in active_orders:
                try:
                    await self.binance.cancel_order(state['symbol'], order_id)
                    logger.info(f"✅ Shutdown: Cancelled order {order_id}")
                except Exception as e:
                    logger.error(f"❌ Shutdown: Failed to cancel {order_id}: {e}")
            
            # Clear all state
            self.orchestrator.order_states.clear()
            self.orchestrator.error_handler.error_counts.clear()
            self.orchestrator.error_handler.circuit_breakers.clear()
            
            logger.critical("🛑 SYSTEM SHUTDOWN COMPLETED")
            
        except Exception as e:
            logger.critical(f"💥 SHUTDOWN FAILED: {e}")
    
    async def get_system_status(self) -> Dict:
        """קבלת סטטוס מערכת נוכחי"""
        return {
            'is_running': self.is_running,
            'uptime': time.time() - self.start_time,
            'orchestrator': {
                'order_states_count': len(self.orchestrator.order_states),
                'recovery_queue_size': self.orchestrator.recovery_queue.qsize(),
                'performance_metrics': self.orchestrator.performance_metrics
            },
            'error_handler': {
                'error_counts': dict(self.orchestrator.error_handler.error_counts),
                'circuit_breakers': len(self.orchestrator.error_handler.circuit_breakers)
            },
            'health': await self.check_system_health()
        }


# Utility function for easy creation
def create_root_solution(binance_client, redis_client=None):
