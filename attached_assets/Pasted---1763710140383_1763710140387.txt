    def trend_fusion_combination(self, df):
        """שילוב טרנד רב-שכבתי - הליבה של המערכת"""
        close = df['close']
        
        # 📊 שילוב EMA רב-שכבתי עם משקלים דינאמיים
        ema_8 = ta.trend.EMAIndicator(close, window=8).ema_indicator()
        ema_21 = ta.trend.EMAIndicator(close, window=21).ema_indicator()
        ema_34 = ta.trend.EMAIndicator(close, window=34).ema_indicator()
        ema_55 = ta.trend.EMAIndicator(close, window=55).ema_indicator()
        ema_89 = ta.trend.EMAIndicator(close, window=89).ema_indicator()
        ema_144 = ta.trend.EMAIndicator(close, window=144).ema_indicator()
        
        # חישוב זוויות EMA
        df['ema_angle_8'] = self.calculate_ema_angle(ema_8)
        df['ema_angle_21'] = self.calculate_ema_angle(ema_21)
        
        # ציון alignement טרנד
        trend_alignment = (
            (ema_8 > ema_21).astype(int) +
            (ema_21 > ema_34).astype(int) +
            (ema_34 > ema_55).astype(int) +
            (ema_55 > ema_89).astype(int) +
            (ema_89 > ema_144).astype(int)
        ) / 5.0
        
        df['trend_alignment_score'] = trend_alignment
        
        # 📈 Ichimoku Cloud מתקדם
        ichimoku = ta.trend.IchimokuIndicator(high=df['high'], low=df['low'])
        df['ichimoku_a'] = ichimoku.ichimoku_a()
        df['ichimoku_b'] = ichimoku.ichimoku_b()
        df['ichimoku_base'] = ichimoku.ichimoku_base_line()
        df['ichimoku_conversion'] = ichimoku.ichimoku_conversion_line()
        
        # ציון Ichimoku
        df['ichimoku_score'] = self.calculate_ichimoku_score(df)
        
        # 🎯 SuperTrend עם אופטימיזציה דינאמית
        for multiplier in [2, 3, 4]:
            super_trend = ta.trend.SuperTrendIndicator(

                close=df['close'],
                period=10,
                multiplier=multiplier
            )
            df[f'supertrend_{multiplier}'] = super_trend.super_trend()
            df[f'supertrend_dir_{multiplier}'] = super_trend.super_trend_direction()

    def momentum_fusion_combination(self, df):
        """שילוב מומנטום מתקדם עם קורלציה בין אינדיקטורים"""
        close = df['close']
        high, low = df['high'], df['low']
        
        # 📊 RSI משוקלל מרובה timeframes (ווירטואלי)
        rsi_6 = ta.momentum.RSIIndicator(close, window=6).rsi()
        rsi_14 = ta.momentum.RSIIndicator(close, window=14).rsi()
        rsi_21 = ta.momentum.RSIIndicator(close, window=21).rsi()
        
        # RSI משוקלל עם הגבלות
        df['rsi_weighted'] = (rsi_6 * 0.4 + rsi_14 * 0.4 + rsi_21 * 0.2)
        df['rsi_momentum'] = rsi_6 - rsi_21  # מומנטום קצר vs ארוך
        
        # 📈 MACD מתקדם עם multiple signal lines
        macd = ta.trend.MACD(close)
        df['macd'] = macd.macd()
        df['macd_signal'] = macd.macd_signal()
        df['macd_histogram'] = macd.macd_diff()
        
        # MACD נוסף עם פרמטרים שונים
        macd_fast = ta.trend.MACD(close, window_fast=8, window_slow=21, window_sign=5)
        df['macd_fast'] = macd_fast.macd()
        
        # 🎯 Stochastic משולב עם RSI
        stoch = ta.momentum.StochasticOscillator(high=high, low=low, close=close)
        df['stoch_k'] = stoch.stoch()
        df['stoch_d'] = stoch.stoch_signal()
        
        # שילוב מומנטום - RSI + Stochastic + MACD
        df['momentum_composite'] = (
            (df['rsi_weighted'] - 50) * 0.4 +
            (df['stoch_k'] - 50) * 0.3 +
            (df['macd_histogram'] * 100) * 0.3
        )
        
        # 🔄 Williams %R עם התאמה
        williams = ta.momentum.WilliamsRIndicator(high=high, low=low, close=close)
        df['williams_r'] = williams.williams_r()
        
        # 📊 Awesome Oscillator
        ao = ta.momentum.AwesomeOscillatorIndicator(high=high, low=low)
        df['awesome_oscillator'] = ao.awesome_oscillator()
        
        # ציון מומנטום משולב
        df['momentum_score'] = self.calculate_momentum_score(df)

    def volume_intelligence_combination(self, df):
        """ניתוח ווליום חכם עם קורלציה למחיר"""
        volume, close = df['volume'], df['close']
        
        # 📊 OBV מתקדם
        obv = ta.volume.OnBalanceVolumeIndicator(close=close, volume=volume)
        df['obv'] = obv.on_balance_volume()
        
        # נורמליזציית OBV
        df['obv_normalized'] = (df['obv'] - df['obv'].rolling(50).mean()) / df['obv'].rolling(50).std()
        
        # 📈 Volume Weighted MACD
        vwap = ta.volume.VolumeWeightedAveragePrice(
            high=df['high'], low=df['low'], close=df['close'], volume=df['volume']
        )
        df['vwap'] = vwap.volume_weighted_average_price()
        
        # 🎯 Chaikin Money Flow
        cmf = ta.volume.ChaikinMoneyFlowIndicator(
            high=df['high'], low=df['low'], close=df['close'], volume=df['volume']
        )
        df['cmf'] = cmf.chaikin_money_flow()
        
        # 📊 Accumulation/Distribution Line
        adl = ta.volume.AccDistIndexIndicator(
            high=df['high'], low=df['low'], close=df['close'], volume=df['volume']
        )
        df['adl'] = adl.acc_dist_index()
        
        # 🔄 Volume ROC עם מחיר ROC
        df['volume_roc'] = volume.pct_change(periods=10)
        df['price_roc'] = close.pct_change(periods=10)
        df['volume_price_correlation'] = df['volume_roc'].rolling(20).corr(df['price_roc'])
        
        # 🧠 Volume Zone (אזורי ווליום קריטיים)
        df['volume_ma_20'] = volume.rolling(20).mean()
        df['volume_ma_50'] = volume.rolling(50).mean()
        df['volume_zone'] = np.where(
            volume > df['volume_ma_20'] * 1.5, 2,  # HIGH
            np.where(volume > df['volume_ma_20'], 1, 0)  # NORMAL / LOW
        )
        
        # ציון ווליום משולב
        df['volume_score'] = self.calculate_volume_score(df)
        
        return df
    
    def volatility_clustering_combination(self, df):
        """אשכולות וולאטיליות עם התאמה דינאמית"""
        high, low, close = df['high'], df['low'], df['close']
        
        # 📊 Bollinger Bands מתקדם עם multiple deviations
        for std in [1, 2, 2.5]:
            bb = ta.volatility.BollingerBands(close, window=20, window_dev=std)
            df[f'bb_upper_{std}'] = bb.bollinger_hband()
            df[f'bb_lower_{std}'] = bb.bollinger_lband()
            df[f'bb_middle_{std}'] = bb.bollinger_mavg()
            df[f'bb_width_{std}'] = (df[f'bb_upper_{std}'] - df[f'bb_lower_{std}']) / df[f'bb_middle_{std}']
        
        # 📈 Keltner Channel
        keltner = ta.volatility.KeltnerChannel(
            high=high, low=low, close=close, window=20
        )
        df['keltner_upper'] = keltner.keltner_channel_hband()
        df['keltner_lower'] = keltner.keltner_channel_lband()
        df['keltner_middle'] = keltner.keltner_channel_mband()
        
        # 🎯 Donchian Channel
        donchian = ta.volatility.DonchianChannel(
            high=high, low=low, close=close, window=20
        )
        df['donchian_upper'] = donchian.donchian_channel_hband()
        df['donchian_lower'] = donchian.donchian_channel_lband()
        df['donchian_middle'] = donchian.donchian_channel_mband()
        
        # 🔄 ATR מרובה תקופות
        for period in [7, 14, 21]:
            atr = ta.volatility.AverageTrueRange(
                high=high, low=low, close=close, window=period
            )
            df[f'atr_{period}'] = atr.average_true_range()
            df[f'atr_pct_{period}'] = df[f'atr_{period}'] / close
        
        # 📊 וולאטיליות היסטורית
        df['volatility_20'] = close.pct_change().rolling(20).std()
        df['volatility_50'] = close.pct_change().rolling(50).std()
        
        # 🧠 Volatility Regime Detection
        df['volatility_ratio'] = df['volatility_20'] / df['volatility_50']
        df['volatility_regime'] = np.where(
            df['volatility_ratio'] > 1.5, 'HIGH',
            np.where(df['volatility_ratio'] < 0.7, 'LOW', 'NORMAL')
        )
        
        # ציון וולאטיליות
        df['volatility_score'] = self.calculate_volatility_score(df)
        
        return df
    
    def mean_reversion_quantum(self, df):
        """שילוב רוורסיון מתקדם עם זיהוי נקודות מפנה"""
        close = df['close']
        
        # 📊 CCI - Commodity Channel Index
        cci = ta.trend.CCIIndicator(
            high=df['high'], low=df['low'], close=close, window=20
        )
        df['cci'] = cci.cci()
        
        # 📈 Detrended Price Oscillator
        dpo = ta.trend.DPOIndicator(close=close, window=20)
        df['dpo'] = dpo.dpo()
        
        # 🎯 Price Distance from Mean
        df['price_zscore_20'] = (close - close.rolling(20).mean()) / close.rolling(20).std()
        df['price_zscore_50'] = (close - close.rolling(50).mean()) / close.rolling(50).std()
        
        # 🔄 RSI2 for fast mean reversion
        rsi_2 = ta.momentum.RSIIndicator(close, window=2).rsi()
        df['rsi_2'] = rsi_2
        
        # 📊 Fibonacci Retracement Levels (ווירטואלי)
        recent_high = df['high'].rolling(50).max()
        recent_low = df['low'].rolling(50).min()
        range_total = recent_high - recent_low
        
        for level in [0.236, 0.382, 0.5, 0.618, 0.786]:
            df[f'fib_{level}'] = recent_high - (range_total * level)
        
        # 🧠 Mean Reversion Composite Score
        df['mean_reversion_score'] = (
            (abs(df['cci']) / 100) * 0.3 +
            (abs(df['price_zscore_20']) / 2) * 0.3 +
            (abs(df['rsi_2'] - 50) / 50) * 0.4
        )
        
        return df
    
    def breakout_confirmation_combination(self, df):
        """אישורי פריצה עם multiple confirmation"""
        high, low, close = df['high'], df['low'], df['close']
        
        # 📊 Pivot Points
        pivot = ta.pivot.PivotPointIndicator(high=high, low=low, close=close)
        df['pivot'] = pivot.pivot_point()
        df['resistance1'] = pivot.resistance1()
        df['support1'] = pivot.support1()
        
        # 📈 Parabolic SAR
        psar = ta.trend.PSARIndicator(high=high, low=low, close=close)
        df['psar'] = psar.psar()
        df['psar_up'] = psar.psar_up()
        df['psar_down'] = psar.psar_down()
        
        # 🎯 ADX for trend strength
        adx = ta.trend.ADXIndicator(high=high, low=low, close=close)
        df['adx'] = adx.adx()
        df['di_plus'] = adx.adx_pos()
        df['di_minus'] = adx.adx_neg()
        
        # 🔄 Price Action Patterns (פשוט)
        df['inside_bar'] = ((high < high.shift(1)) & (low > low.shift(1))).astype(int)
        df['outside_bar'] = ((high > high.shift(1)) & (low < low.shift(1))).astype(int)
        df['engulfing_bullish'] = ((close > open) & (close.shift(1) < open.shift(1)) & 
                                 (close > open.shift(1)) & (open < close.shift(1))).astype(int)
        
        # 📊 Breakout Strength Indicator
        df['breakout_strength'] = self.calculate_breakout_strength(df)
        
        return df
    
    def ml_enhancement_combination(self, df):
        """שיפורי ML וסטטיסטיקה מתקדמת"""
        close = df['close']
        
        # 📊 Hurst Exponent for market regime
        df['hurst_exponent'] = self.calculate_hurst_exponent(close)
        
        # 📈 Z-Score for anomaly detection
        df['price_zscore'] = (close - close.rolling(50).mean()) / close.rolling(50).std()
        
        # 🎯 Rolling Skewness and Kurtosis
        df['skewness_20'] = close.rolling(20).apply(lambda x: stats.skew(x))
        df['kurtosis_20'] = close.rolling(20).apply(lambda x: stats.kurtosis(x))
        
        # 🔄 Quantile-based levels
        df['price_quantile_20'] = close.rolling(50).apply(
            lambda x: pd.qcut(x, 4, labels=False).iloc[-1] if len(x) == 50 else np.nan
        )
        
        # 📊 Correlation with BTC (אם יש multiple pairs)
        # df['btc_correlation'] = close.rolling(20).corr(btc_close)
        
        return df
    
    def calculate_ema_angle(self, ema_series):
        """חישוב זווית EMA - מדד לכוח הטרנד"""
        angle = np.degrees(np.arctan(
            (ema_series - ema_series.shift(5)) / (ema_series.shift(5))
        ))
        return angle
    
    def calculate_ichimoku_score(self, df):
        """ציון Ichimoku משולב"""
        score = 0
        
        # מחיר מעל הענן
        if df['close'] > max(df['ichimoku_a'], df['ichimoku_b']):
            score += 1
        # ענן עולה
        if df['ichimoku_a'] > df['ichimoku_b']:
            score += 1
        # Tenkan-sen above Kijun-sen
        if df['ichimoku_conversion'] > df['ichimoku_base']:
            score += 1
        # מחיר above Kijun-sen
        if df['close'] > df['ichimoku_base']:
            score += 1
            
        return score / 4.0
    
    def calculate_momentum_score(self, df):
        """ציון מומנטום משולב"""
        score = 0
        
        # RSI באזור אופטימלי
        if 45 <= df['rsi_weighted'] <= 65:
            score += 1
        
        # MACD חיובי
        if df['macd'] > df['macd_signal']:
            score += 1
            
        # Stochastic באזור ניטרלי-חיובי
        if 20 <= df['stoch_k'] <= 80:
            score += 0.5
        if df['stoch_k'] > df['stoch_d']:
            score += 0.5
            
        # Awesome Oscillator חיובי
        if df['awesome_oscillator'] > 0:
            score += 1
            
        return score / 4.0
    
    def calculate_volume_score(self, df):
        """ציון ווליום משולב"""
        score = 0
        
        # OBV בעלייה
        if df['obv'] > df['obv'].shift(1):
            score += 1
            
        # CMF חיובי
        if df['cmf'] > 0:
            score += 1
            
        # קורלציה חיובית בין ווליום למחיר
        if df['volume_price_correlation'] > 0:
            score += 1
            
        # ווליום גבוה מממוצע
        if df['volume_zone'] >= 1:
            score += 1
            
        return score / 4.0
    
    def calculate_volatility_score(self, df):
        """ציון וולאטיליות עבור ניהול סיכונים"""
        # וולאטיליות נמוכה-בינונית עדיפה למסחר
        atr_pct = df['atr_pct_14']
        
        if atr_pct < 0.01:
            return 0.8  # וולאטיליות נמוכה - טוב
        elif atr_pct < 0.025:
            return 0.9  # וולאטיליות אופטימלית
        elif atr_pct < 0.04:
            return 0.6  # וולאטיליות גבוהה - זהיר
        else:
            return 0.3  # וולאטיליות מאוד גבוהה - מסוכן
    
    def calculate_breakout_strength(self, df):
        """חישוב חוזק פריצה"""
        strength = 0
        
        # פריצת resistance
        if df['close'] > df['resistance1']:
            strength += 1
            
        # ADX מעל 25 (טרנד חזק)
        if df['adx'] > 25:
            strength += 1
            
        # DI+ above DI-
        if df['di_plus'] > df['di_minus']:
            strength += 1
            
        # ווליום גבוה
        if df['volume_zone'] >= 1:
            strength += 1
            
        return strength / 4.0
    
    def calculate_hurst_exponent(self, prices, max_lag=20):
        """חישוב Hurst Exponent - זיהוי סדרות זמן"""
        lags = range(2, max_lag)
        tau = [np.std(np.subtract(prices[lag:], prices[:-lag])) for lag in lags]
        
        if len(tau) > 1:
            hurst = np.polyfit(np.log(lags), np.log(tau), 1)[0]
            return hurst
        return 0.5
    
    def generate_trading_signals(self, df):
        """הפקת אותות מסחר based on שילוב אינדיקטורים"""
        signals = []
        
        for i in range(2, len(df)):
            current = df.iloc[i]
            prev = df.iloc[i-1]
            
            # 🎯 LONG SIGNAL - שילוב תנאים מורכב
            long_conditions = (
                # טרנד חזק
                current['trend_alignment_score'] > 0.7 and
                current['ichimoku_score'] > 0.75 and
                
                # מומנטום חיובי
                current['momentum_score'] > 0.7 and
                current['macd'] > current['macd_signal'] and
                
                # ווליום תומך
                current['volume_score'] > 0.6 and
                current['volume_price_correlation'] > 0 and
                
                # וולאטיליות מתאימה
                current['volatility_score'] > 0.6 and
                
                # אישורי נוספים
                current['adx'] > 20 and
                current['di_plus'] > current['di_minus'] and
                
                # הגנה מפני overbought
                current['rsi_weighted'] < 70
            )
            
            # 🎯 SHORT SIGNAL - שילוב תנאים מורכב
            short_conditions = (
                # טרנד שלילי
                current['trend_alignment_score'] < 0.3 and
                current['ichimoku_score'] < 0.25 and
                
                # מומנטום שלילי
                current['momentum_score'] < 0.3 and
                current['macd'] < current['macd_signal'] and
                
                # ווליום תומך
                current['volume_score'] > 0.6 and
                current['volume_price_correlation'] < 0 and
                
                # וולאטיליות מתאימה
                current['volatility_score'] > 0.6 and
                
                # אישורי נוספים
                current['adx'] > 20 and
                current['di_plus'] < current['di_minus'] and
                
                # הגנה מפני oversold
                current['rsi_weighted'] > 30
            )
            
            if long_conditions:
                signals.append(1)
            elif short_conditions:
                signals.append(-1)
            else:
                signals.append(0)
        
        return signals

            current['ichimoku_score'] * 0.15
        )
        
        # 🎯 קביעת כיוון based on רוב האינדיקטורים
        bullish_indicators = 0
        bearish_indicators = 0
        
        # ספירת אינדיקטורים חיוביים/שליליים
        if current['trend_alignment_score'] > 0.6: bullish_indicators += 1
        if current['momentum_score'] > 0.6: bullish_indicators += 1
        if current['macd'] > current['macd_signal']: bullish_indicators += 1
        if current['volume_price_correlation'] > 0: bullish_indicators += 1
        if current['di_plus'] > current['di_minus']: bullish_indicators += 1
        
        if current['trend_alignment_score'] < 0.4: bearish_indicators += 1
        if current['momentum_score'] < 0.4: bearish_indicators += 1
        if current['macd'] < current['macd_signal']: bearish_indicators += 1
        if current['volume_price_correlation'] < 0: bearish_indicators += 1
        if current['di_plus'] < current['di_minus']: bearish_indicators += 1
        
        if bullish_indicators >= 3 and confidence_score > 0.7:
            return {'action': 'BUY', 'confidence': confidence_score}
        elif bearish_indicators >= 3 and confidence_score > 0.7:
            return {'action': 'SELL', 'confidence': confidence_score}
        else:
            return {'action': 'HOLD', 'confidence': confidence_score}
