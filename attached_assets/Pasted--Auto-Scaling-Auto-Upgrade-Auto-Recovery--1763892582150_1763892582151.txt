```python
from fastapi import WebSocket

@app.websocket("/ws/system")
async def system_ws(ws: WebSocket):
    await ws.accept()
    while True:
        await ws.send_json(get_current_auto_state())
        await asyncio.sleep(5)
```

## ✅ 3) UI כפתורי שליטה מהדשבורד

להוסיף API:

```python
@app.post("/system/control")
async def system_control(cmd: str):
    if cmd == "disable_autoscale":
        state.autoscale_active = False
    if cmd == "enable_autoscale":
        state.autoscale_active = True
    if cmd == "force_scale_up":
        trigger_scale_up()
    if cmd == "force_upgrade":
        trigger_upgrade_now()
    if cmd == "freeze_all":
        activate_kill_switch()
    return {"ok": True, "cmd": cmd}
```

✔ אתה שולט מהדשבורד
✔ בלי SSH
✔ בלי להיכנס לשרת

## ✅ 4) דשבורד מוכן (React component)

הסוכן ייצור קובץ:

`dashboard/components/SystemAutoPanel.jsx`

```jsx
export default function SystemAutoPanel() {
  const [data, setData] = useState({});

  useEffect(() => {
    const ws = new WebSocket("wss://YOUR_DOMAIN/ws/system");
    ws.onmessage = (msg) => setData(JSON.parse(msg.data));
  }, []);

  return (
    <Card>
      <h2>AUTO-SYSTEM STATUS</h2>
      <Badge>Status: {data.health}</Badge>
      <p>CPU Load: {data.cpu_load}</p>
      <p>Free RAM: {data.free_ram} MB</p>
      <p>Last Scale: {data.last_scale}</p>
      <p>Last Upgrade: {data.last_upgrade}</p>

      <Button onClick={() => sendCmd("disable_autoscale")}>Disable Auto-Scale</Button>
      <Button onClick={() => sendCmd("enable_autoscale")}>Enable Auto-Scale</Button>
      <Button onClick={() => sendCmd("force_upgrade")}>Force Upgrade</Button>
      <Button onClick={() => sendCmd("freeze_all")}>KILL SWITCH</Button>
    </Card>
  );
}
```

✔ נראה כמו Replit
✔ שליטה מלאה
✔ חי 100%
