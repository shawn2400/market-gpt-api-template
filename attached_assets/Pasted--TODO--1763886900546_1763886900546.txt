```python
import time
import json
import random
from datetime import datetime

class AutoPilotV5:
    def __init__(self):
        self.system_state = {
            "market_mode": "TREND_UP",
            "auto_pilot_enabled": True,
            "kill_switch": False,
            "news_pause": False,
            "funding_regime": "NORMAL",
            "hedge_active": False,
            "profit_share_pending": False,
            "open_positions": {},
            "last_recovery": None,
        }

    # =====================================================
    # ✅ 1) זיהוי מצב שוק דינמי — 5 מצבים
    # =====================================================
    def detect_market_mode(self, data):
        atr = data.get("atr", 0)
        adx = data.get("adx", 0)
        rsi = data.get("rsi", 50)

        if adx > 28 and rsi > 55:
            mode = "TREND_UP"
        elif adx > 28 and rsi < 45:
            mode = "TREND_DOWN"
        elif atr > 3.5:
            mode = "HIGH_VOL"
        elif abs(rsi - 50) < 3:
            mode = "CHOPPY"
        else:
            mode = "REVERSAL_SETUP"

        self.system_state["market_mode"] = mode
        return mode

    # =====================================================
    # ✅ 2) ניהול SL/TP דינמי + Profit-Lock חובה
    # =====================================================
    def update_sl_tp(self, position, price, atr):
        entry = position["entry"]
        sl = position["sl"]
        tp1 = position["tp1"]

        # MOVE TO BE+
        if not position.get("be_done") and price >= tp1:
            position["sl"] = entry * 1.0006
            position["be_done"] = True
            position["last_high"] = price
            return position

        # TRAILING אחרי BE
        if position.get("be_done") and price > position.get("last_high", price):
            new_sl = price - (atr * 1.2)
            if new_sl > sl:
                position["sl"] = new_sl
                position["last_high"] = price

        return position

    # =====================================================
    # ✅ 3) Auto-Hedge בזמן אירוע קיצוני
    # =====================================================
    def hedge_if_needed(self, volatility, news_impact):
        if volatility > 4.5 or news_impact == "HIGH":
            self.system_state["hedge_active"] = True
        else:
            self.system_state["hedge_active"] = False

        return self.system_state["hedge_active"]

    # =====================================================
    # ✅ 4) Auto-Pause לפני CPI / FOMC / חדשות
    # =====================================================
    def apply_news_pause(self, event):
        if event in ["CPI", "FOMC", "NFP", "FED_SPEECH"]:
            self.system_state["news_pause"] = True
        else:
            self.system_state["news_pause"] = False

        return self.system_state["news_pause"]

    # =====================================================
    # ✅ 5) Auto-Freeze + Kill-Switch חכם
    # =====================================================
    def kill_switch_trigger(self, reason):
        self.system_state["kill_switch"] = True
        print(f"⚠️ KILL-SWITCH ACTIVATED — Reason: {reason}")

    def release_kill_switch(self):
        self.system_state["kill_switch"] = False

    # =====================================================
    # ✅ 6) Failover + Self-Healing אוטומטי
    # =====================================================
    def failover_recovery(self, component):
        print(f"🔁 FAILOVER engaged for: {component}")
        time.sleep(1)
        self.system_state["last_recovery"] = datetime.utcnow()
        print("✅ System restored")

    # =====================================================
    # ✅ 7) Profit-Share Weekly (18%)
    # =====================================================
    def calculate_profit_share(self, pnl_usd):
        fee = round(pnl_usd * 0.18, 2)
        self.system_state["profit_share_pending"] = True
        return {
            "amount_due": fee,
            "wallet": "USDT-TRC20-PAY-HERE",
            "deadline_hours": 24
        }

    # =====================================================
    # ✅ 8) הפעלה מלאה
    # =====================================================
    def run_cycle(self, market):
        if self.system_state["kill_switch"]:
            return "STOPPED — KILL SWITCH ACTIVE"

        mode = self.detect_market_mode(market)
        hedge = self.hedge_if_needed(market["atr"], market["news"])
        return {
            "mode": mode,
            "hedge": hedge,
            "timestamp": datetime.utcnow().isoformat()
        }


# =====================================================
# ✅ הפעלה — מוכן להרצה / לסוכן
# =====================================================

if __name__ == "__main__":
    bot = AutoPilotV5()

    sample_market = {
        "atr": 3.2,
        "adx": 31,
        "rsi": 61,
        "news": "LOW"
    }

    print(bot.run_cycle(sample_market))
```
