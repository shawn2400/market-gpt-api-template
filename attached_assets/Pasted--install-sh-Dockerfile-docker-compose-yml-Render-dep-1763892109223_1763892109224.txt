# ✅ **1) שמור כ־install.sh**

```bash
#!/usr/bin/env bash
set -e

echo "🚀 Starting FULL INSTALL for AlgoGPT Control-Agent…"

### ===== 1) SYSTEM PREP =====
apt update -y
apt install -y curl git ufw docker.io docker-compose

systemctl enable docker
systemctl start docker

### ===== 2) CREATE APP FOLDER =====
mkdir -p /opt/algogpt-agent
cd /opt/algogpt-agent

### ===== 3) CLONE MAIN REPO (placeholder) =====
# ✅ הסוכן מושך את קוד המערכת
git clone https://github.com/YOUR_ORG/algogpt-control-agent.git app || true
cd app

### ===== 4) CREATE .ENV (placeholder values) =====
cat > .env <<EOF
APP_ENV=prod
PORT=8080

# ✅ Binance / Telegram / GitHub / OpenAI / Ollama
BINANCE_API_KEY=
BINANCE_API_SECRET=
TELEGRAM_BOT_TOKEN=
GITHUB_TOKEN=
OPENAI_API_KEY=
OLLAMA_HOST=http://localhost:11434

# ✅ Admin control
ADMIN_MASTER_KEY=CHANGE_ME_NOW
FREEZE_ALLOWED=true
AUTO_RECOVERY=true
EOF

### ===== 5) START OLLAMA =====
curl -fsSL https://ollama.com/install.sh | sh
systemctl restart ollama || true

# ✅ preload models
ollama pull llama3 || true
ollama pull codellama || true

### ===== 6) BUILD & START FULL SYSTEM =====
docker-compose down || true
docker-compose up -d --build

echo "✅ Install completed!"
echo "Dashboard available at: http://SERVER_IP:8080"
```

✅ הפעלה:

```bash
chmod +x install.sh
./install.sh
```

# ✅ **2) שמור כ־docker-compose.yml**

```yaml
version: '3.9'

services:
  control-agent:
    build: .
    container_name: algogpt-control-agent
    restart: always
    env_file: .env
    ports:
      - "8080:8080"
    volumes:
      - ./data:/app/data
      - /var/run/docker.sock:/var/run/docker.sock

  ollama:
    image: ollama/ollama
    restart: always
    ports:
      - "11434:11434"
    volumes:
      - ollama-data:/root/.ollama

volumes:
  ollama-data:
```

# ✅ **3) שמור כ־Dockerfile**

```dockerfile
FROM python:3.11-slim

WORKDIR /app

COPY requirements.txt .
RUN pip install --no-cache-dir -r requirements.txt

COPY . .

EXPOSE 8080

CMD ["python", "main.py"]
```
