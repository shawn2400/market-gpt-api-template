# ✅ 2) Auto-Scaling לסמלים לפי ביצועים

### 📌 קובץ חדש: `utils/auto_scaler.py`

```python
from utils.pnl_tracker import get_symbol_pnl
from utils.metrics_tracker import risk_level

class AutoScaler:
    def __init__(self):
        self.min_risk = 0.25
        self.max_risk = 1.5

    def calc_multiplier(self, symbol):
        pnl = get_symbol_pnl(symbol, window_days=7)

        if pnl <= -5:
            return self.min_risk   # מוריד מינוף

        if pnl >= 20:
            return self.max_risk   # מגדיל חשיפה

        scale = 1 + (pnl / 20) * (self.max_risk - 1)
        return max(self.min_risk, min(scale, self.max_risk))

    def apply(self, symbol, base_size):
        multiplier = self.calc_multiplier(symbol)
        risk_level.set(multiplier)
        return base_size * multiplier
```

### ✅ מה זה נותן

✔ סמלים מרוויחים → גדלים אוטומטית
✔ סמלים מפסידים → קטנים לבד
✔ ללא מגע יד אדם

# ✅ 3) AI Hedge-Ratio Decision Engine

### 📌 קובץ חדש: `utils/ai_hedge_decider.py`

```python
from openai import OpenAI
from utils.market_state import get_state_snapshot

client = OpenAI()

class AIHedgeDecider:
    def decide(self, symbol, exposure, funding, sentiment):
        snapshot = get_state_snapshot(symbol)

        prompt = f"""
        Symbol: {symbol}
        Exposure: {exposure}
        Funding: {funding}
        Sentiment: {sentiment}
        Volatility: {snapshot['volatility']}
        Trend: {snapshot['trend']}

        Decide ideal hedge ratio between 0.15 and 0.85.
        Only return a number.
        """

        res = client.chat.completions.create(
            model="gpt-5",
            messages=[{"role": "user", "content": prompt}]
        )

        return float(res.choices[0].message.content.strip())
```

### ✅ שימוש ב־Auto-Hedger


# ✅ 4) FAILOVER מלא אם בורסה נופלת

### 📌 קובץ חדש: `utils/exchange_failover.py`

```python
from utils.binance_client import ping as b_ping
from utils.bybit_client import ping as y_ping
from utils.metrics_tracker import exchange_failovers

ACTIVE_EXCHANGE = "binance"

def health_check():
    global ACTIVE_EXCHANGE

    try:
        b_ping()
        return "binance"
    except:
        try:
            y_ping()
            if ACTIVE_EXCHANGE != "bybit":
                exchange_failovers.inc()
            ACTIVE_EXCHANGE = "bybit"
            return "bybit"
        except:
            raise Exception("🚨 ALL EXCHANGES DOWN")

def route_order(order):
    exchange = health_check()

    if exchange == "binance":
        return binance_execute(order)
    else:
        return bybit_execute(order)
```
