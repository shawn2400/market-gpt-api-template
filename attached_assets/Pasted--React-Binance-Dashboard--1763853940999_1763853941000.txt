# 🧠 Zustand Store — **src/store/bots.js**

```javascript
import { create } from "zustand";
import { api } from "../api/api";

export const useBotsStore = create((set) => ({
  bots: [],
  loadBots: async () => {
    const { data } = await api.get("/external/status");
    set({ bots: data.bots });
  },
  setMode: async (name, mode) => {
    await api.post(`/external/control/${name}/${mode}`);
    const { data } = await api.get("/external/status");
    set({ bots: data.bots });
  }
}));
```

export default function BotCard({ bot, setMode }) {
  return (
    <div className="bg-card border border-border p-4 rounded-xl m-2 w-72">
      <h2 className="text-xl font-bold text-white mb-2">{bot.name}</h2>
      <p className="text-text">Type: {bot.type}</p>
      <p className="text-text">Score: {bot.score}/10</p>
      <p className="text-text">Signals Today: {bot.signals}</p>

      <div className="flex mt-3 gap-2">
        <button className="bg-accent px-3 py-1 rounded" onClick={() => setMode(bot.name, "on")}>ON</button>
        <button className="bg-red-600 px-3 py-1 rounded" onClick={() => setMode(bot.name, "off")}>OFF</button>
        <button className="bg-blue-600 px-3 py-1 rounded" onClick={() => setMode(bot.name, "auto")}>AUTO</button>
      </div>
    </div>
  );
}

# 🟦 **Dashboard Page — src/pages/Bots.jsx**

```jsx
import { useEffect } from "react";
import BotCard from "../components/BotCard";
import { useBotsStore } from "../store/bots";

export default function Bots() {
  const { bots, loadBots, setMode } = useBotsStore();

  useEffect(() => {
    loadBots();
    const interval = setInterval(loadBots, 5000);
    return () => clearInterval(interval);
  }, []);

  return (
    <div className="p-6 flex flex-wrap gap-4">
      {bots.map((bot) => (
        <BotCard key={bot.name} bot={bot} setMode={setMode} />
      ))}
    </div>
  );
}
```
