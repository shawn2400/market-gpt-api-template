# ✅ 1) `mode_engine.py` — 5 מצבי שוק (ULTRA)

```python
# utils/mode_engine.py

class ModeEngine:
    """
    ULTRA market mode classification (5 modes)
    Input: price action metrics
    Output: mode string
    """

    def classify(self, metrics: dict) -> str:
        atr = metrics["atr"]
        vol = metrics["volatility"]
        trend = metrics["trend_strength"]
        structure = metrics["structure_breaks"]

        # MODE 1 — STRONG TREND
        if trend > 0.75 and vol < 0.65:
            return "TREND"

        # MODE 2 — BREAKOUT MOMENTUM
        if structure == "fresh_break" and vol >= 0.65:
            return "MOMENTUM"

        # MODE 3 — CHOP / RANGE
        if vol < 0.35 and trend < 0.45:
            return "CHOP"

        # MODE 4 — REVERSAL
        if metrics["divergence"] and structure == "failed_break":
            return "REVERSAL"

        # MODE 5 — HIGH-SHOCK
        if vol > 0.85 or metrics["liquidations"] > 3:
            return "SHOCK"

        return "NEUTRAL"
```

# ✅ 2) `news_engine.py` — AI NEWS IMPACT

```python
# utils/news_engine.py

import time

class NewsEngine:
    """
    Lightweight AI-news impact engine
    Returns:
      score: -1 to +1
      type: normal | high | critical
    """

    def __init__(self):
        self.cache = {}
        self.last_ts = 0

    def evaluate(self, headlines: list) -> dict:
        now = time.time()
        if now - self.last_ts < 900:   # 15m cache
            return self.cache

        score = 0
        impact = "normal"

        for h in headlines:
            text = h.lower()

            if any(x in text for x in ["hack", "security breach", "shutdown"]):
                score -= 0.7
                impact = "critical"

            elif any(x in text for x in ["etf approval", "partnership", "listing"]):
                score += 0.6
                impact = "high"

        score = max(-1, min(1, score))

        result = {
            "score": score,
            "impact": impact
        }

        self.cache = result
        self.last_ts = now
        return result
```

# ✅ 3) `funding_filter.py` — FUNDING REGIME

```python
# utils/funding_filter.py

class FundingFilter:
    """
    Funding regime classifier:
      A — extreme positive (long crowded)
      B — extreme negative (short crowded)
      C — neutral
      D — transition
    """

    def classify(self, rate: float, delta_1h: float) -> str:

        # REGIME A — extreme positive
        if rate > 0.025 or delta_1h > 0.015:
            return "A"

        # REGIME B — extreme negative
        if rate < -0.025 or delta_1h < -0.015:
            return "B"

        # REGIME D — transition
        if abs(delta_1h) > 0.008:
            return "D"

        return "C"
```

# ✅ 4) `trade_manager.py` — SL/TP דינמי + סנכרון AUTO-FLIP

```python
# utils/trade_manager.py

class TradeManager:
    """
    ULTRA-PLUS dynamic SL engine:
    - BE → Trailing ATR
    - Auto-flip safety rules
    - News + Funding sync
    """

    def update(self, pos: dict, price: float, atr: float,
               mode: str, news: dict, regime: str) -> dict:

        # 1) CRITICAL NEWS — FULL PROTECTION
        if news["impact"] == "critical":
            pos["sl"] = max(pos["sl"], price * 0.992)
            pos["auto_flip_block"] = "critical_news"
            return pos

        # 2) FUNDING EXTREMES
        if regime == "A" and pos["side"] == "long":
            pos["sl"] = max(pos["sl"], price - atr * 0.6)
            pos["auto_flip_block"] = "extreme_long_funding"

        if regime == "B" and pos["side"] == "short":
            pos["sl"] = min(pos["sl"], price + atr * 0.6)
            pos["auto_flip_block"] = "extreme_short_funding"

        # 3) BE MOVE
        if not pos.get("be_done") and price >= pos["entry"] * 1.012:
            pos["sl"] = pos["entry"] * 1.001
            pos["be_done"] = True
            pos["last_high"] = price
            return pos

        # 4) TRAILING AFTER BE
        if pos.get("be_done") and price > pos.get("last_high", 0):
            new_sl = price - atr * 1.2
            if new_sl > pos["sl"]:
                pos["sl"] = new_sl
                pos["last_high"] = price

        # 5) AUTO-FLIP ALLOWED?
        pos["auto_flip_allowed"] = (
            mode in ["REVERSAL"] and
            regime in ["B", "C"] and
            news["impact"] != "critical"
        )

        return pos
```

# ✅ 5) שילוב במערכת (הדבקה ב-`auto_executor.py`)

```python
from utils.mode_engine import ModeEngine
from utils.news_engine import NewsEngine
from utils.funding_filter import FundingFilter
from utils.trade_manager import TradeManager

mode_engine = ModeEngine()
news_engine = NewsEngine()
funding = FundingFilter()
manager = TradeManager()

def manage(position, market_data, headlines, funding_rate, funding_delta):
    mode = mode_engine.classify(market_data)
    news = news_engine.evaluate(headlines)
    regime = funding.classify(funding_rate, funding_delta)

    updated = manager.update(
        position,
        price=market_data["price"],
        atr=market_data["atr"],
        mode=mode,
        news=news,
        regime=regime
    )

    return updated
```
