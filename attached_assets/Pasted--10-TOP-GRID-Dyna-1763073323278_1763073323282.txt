```python
class DynamicGridApprover:
    def __init__(self):
        self.min_grid_volume = 75_000_000  # $75M - נפח מינימלי ל-GRID
        self.max_grid_atr = 0.035          # 3.5% ATR מקסימלי
        self.min_liquidity_score = 8.5     # ציון liquidity מינימלי
        
    async def calculate_dynamic_grid_list(self):
        """חישוב דינמי של הסמלים המורשים ל-GRID"""
        
        # 1. קח את ה-TOP 50 הנוכחי
        top_50 = await self.get_current_top_50()
        
        # 2. חשב ציון GRID לכל סמל
        grid_candidates = []
        for symbol in top_50:
            grid_score = await self.calculate_grid_score(symbol)
            
            if grid_score >= self.min_grid_score:
                grid_candidates.append({
                    'symbol': symbol,
                    'grid_score': grid_score,
                    'volume': await self.get_24h_volume(symbol),
                    'liquidity': await self.get_liquidity_depth(symbol),
                    'volatility': await self.get_atr(symbol)
                })
        
        # 3. מיין לפי ציון GRID וקח את הטובים ביותר
        grid_candidates.sort(key=lambda x: x['grid_score'], reverse=True)
        
        # 4. מספר דינמי - בין 10 ל-30 לפי תנאי שוק
        optimal_count = await self.calculate_optimal_grid_count()
        approved_grid_symbols = [c['symbol'] for c in grid_candidates[:optimal_count]]
        
        return approved_grid_symbols
    
    async def calculate_grid_score(self, symbol):
        """חישוב ציון GRID 0-100"""
        scores = {
            'volume_score': await self.calculate_volume_score(symbol),
            'liquidity_score': await self.calculate_liquidity_score(symbol),
            'volatility_score': await self.calculate_volatility_score(symbol),
            'spread_score': await self.calculate_spread_score(symbol),
            'stability_score': await self.calculate_stability_score(symbol)
        }
        
        # ממוצע משוקלל - liquidity ו-volume יותר חשובים ל-GRID
        weights = [0.25, 0.30, 0.20, 0.15, 0.10]
        total_score = sum(score * weight for score, weight in zip(scores.values(), weights))
        
        return total_score
    
    async def calculate_optimal_grid_count(self):
        """חישוב מספר אופטימלי של סמלי GRID לפי תנאי שוק"""
        market_volatility = await self.get_market_volatility()
        total_capital = await self.get_available_capital()
        
        if market_volatility > 0.05:    # שוק מאוד תנודתי
            return 10  # GRID מצומצם - רק הבטוחים ביותר
        elif market_volatility > 0.03:  # שוק תנודתי
            return 15  # GRID בינוני
        elif total_capital > 10000:     # הון גבוה + שוק רגוע
            return 25  # GRID מורחב
        else:                           # תנאים רגילים
            return 20  # GRID סטנדרטי
```

class GridCriteriaCalculator:
    async def calculate_volume_score(self, symbol):
        """ציון נפח 0-25 points"""
        volume = await self.get_24h_volume(symbol)
        
        if volume > 200_000_000: return 25    # $200M+
        elif volume > 100_000_000: return 22  # $100M-200M
        elif volume > 50_000_000: return 18   # $50M-100M
        elif volume > 25_000_000: return 12   # $25M-50M
        else: return 0                        # מתחת לסף
    
    async def calculate_liquidity_score(self, symbol):
        """ציון liquidity 0-30 points"""
        liquidity = await self.get_liquidity_depth(symbol)
        spread = await self.get_spread(symbol)
        
        if liquidity > 1_000_000 and spread < 0.005: return 30   # liquidity מעולה
        elif liquidity > 500_000 and spread < 0.01: return 25    # liquidity טוב
        elif liquidity > 250_000 and spread < 0.02: return 18    # liquidity סביר
        elif liquidity > 100_000 and spread < 0.03: return 10    # liquidity בסיסי
        else: return 0                                           # לא מספיק
    
    async def calculate_volatility_score(self, symbol):
        """ציון תנודתיות 0-20 points (יותר נמוך = יותר טוב ל-GRID)"""
        atr = await self.get_atr(symbol)
        
        if atr < 0.02: return 20    # תנודתיות נמוכה - אידיאלי ל-GRID
        elif atr < 0.03: return 16  # תנודתיות בינונית-נמוכה
        elif atr < 0.04: return 10  # תנודתיות בינונית
        elif atr < 0.05: return 5   # תנודתיות גבוהה
        else: return 0              # תנודתיות גבוהה מדי
```

class TieredGridSystem:
    async def assign_grid_tiers(self, approved_symbols):
        """הקצאת רמות GRID לפי ביצועים"""
        
        tiered_symbols = {
            'PLATINUM_GRID': [],  # GRID מלא - 50 orders
            'GOLD_GRID': [],      # GRID מורחב - 35 orders  
            'SILVER_GRID': [],    # GRID בסיסי - 20 orders
            'BRONZE_GRID': []     # GRID מוגבל - 10 orders
        }
        
        for symbol in approved_symbols:
            tier = await self.calculate_grid_tier(symbol)
            tiered_symbols[tier].append(symbol)
        
        return tiered_symbols
    
    async def calculate_grid_tier(self, symbol):
        """חישוב רמת GRID לסמל"""
        grid_score = await self.calculate_grid_score(symbol)
        
        if grid_score >= 90: return 'PLATINUM_GRID'
        elif grid_score >= 80: return 'GOLD_GRID'
        elif grid_score >= 70: return 'SILVER_GRID'
        else: return 'BRONZE_GRID'
    
    async def get_grid_parameters(self, tier):
        """פרמטרי GRID לפי רמה"""
        grid_configs = {
            'PLATINUM_GRID': {
                'max_orders': 50,

class GridListManager:
    def __init__(self):
        self.current_grid_list = []
        self.grid_list_expiry = 0
        
    async def get_current_grid_list(self):
        """קבלת רשימת GRID נוכחית - עם caching"""
        current_time = time.time()
        
        # אם הרשימה בת יותר מ-15 דקות - עדכן
        if current_time > self.grid_list_expiry or not self.current_grid_list:
            await self.update_grid_list()
            
        return self.current_grid_list
    
    async def update_grid_list(self):
        """עדכון רשימת GRID"""
        grid_approver = DynamicGridApprover()
        new_grid_list = await grid_approver.calculate_dynamic_grid_list()
        
        # השוואה לרשימה הקודמת
        changes = await self.analyze_grid_list_changes(new_grid_list)
        
        if changes['has_changes']:
            logger.info(f"🔄 Grid List Updated: {changes['added']} added, {changes['removed']} removed")
        
        self.current_grid_list = new_grid_list
        self.grid_list_expiry = time.time() + 900  # תוקף 15 דקות
        
        # שמור ב-Redis
        await self.save_to_redis(new_grid_list)
```

class GridPerformanceTracker:
    async def monitor_grid_performance(self):
        """ניטור ביצועי GRID וקבלת החלטות דינמיות"""
        while True:
            try:
                # בדוק ביצועי GRID לכל סמל
                for symbol in self.get_active_grid_symbols():
                    performance = await self.analyze_grid_performance(symbol)
                    
                    # אם ביצועים גרועים - הסר מרשימת GRID
                    if performance['score'] < 60:
                        await self.demote_from_grid(symbol, performance['reason'])
                    
                    # אם ביצועים מעולים - שדרג רמת GRID
                    elif performance['score'] > 85:
                        await self.promote_grid_tier(symbol)
                
                await asyncio.sleep(300)  # סריקה כל 5 דקות
                
            except Exception as e:
                logger.error(f"❌ Grid performance monitoring failed: {e}")
