```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
מערכת לוגינג בלבד - NO TRADE MODE
LOGGING_ONLY_SYSTEM - הגנה מפני מסחר
"""

import pandas as pd
import numpy as np
import ta
from binance.client import Client
import time
import logging
from datetime import datetime, timedelta
import redis
import json
import threading
from concurrent.futures import ThreadPoolExecutor
import psutil
import gc

class LoggingOnlySystem:
    def __init__(self, api_key: str, api_secret: str, redis_host: str = 'localhost'):
        self.client = Client(api_key, api_secret)
        self.redis = redis.Redis(host=redis_host, port=6379, decode_responses=True)
        
        # 🚨 NO TRADE MODE - הגנה
        self.TRADING_DISABLED = True
        self.LOGGING_ONLY = True
        
        # הגדרות מערכת
        self.trading_pairs = ['BTCUSDT', 'ETHUSDT', 'BNBUSDT', 'ADAUSDT', 'DOTUSDT']
        self.signals_history = []
        
        self.setup_safe_logging()
        self.initialize_safe_system()
    
    def setup_safe_logging(self):
        """הגדרת לוגינג בטוח"""
        logging.basicConfig(
            level=logging.INFO,
            format='%(asctime)s | 🛡️ %(levelname)s | %(message)s',
            handlers=[
                logging.FileHandler('safe_logging_only.log', encoding='utf-8'),
                logging.StreamHandler()
            ]
        )
        self.logger = logging.getLogger(__name__)
        
        # הודעה ברורה על מצב NO TRADE
        self.logger.info("🛡️" * 50)
        self.logger.info("🛡️ NO TRADE MODE - LOGGING ONLY")
        self.logger.info("🛡️ ALL TRADING DISABLED FOR SAFETY")
        self.logger.info("🛡️" * 50)
    
    def initialize_safe_system(self):
        """אתחול מערכת בטוחה"""
        self.logger.info("🚀 מאתחל מערכת לוגינג בלבד...")
        
        # בדיקת חיבורים
        self.test_connections_safe()
        
        self.logger.info("✅ מערכת לוגינג בלבד מוכנה - NO TRADES!")
    
    def test_connections_safe(self):
        """בדיקת חיבורים בטוחה"""
        try:
            # בדיקת Binance (read-only)
            server_time = self.client.get_server_time()
            self.logger.info("✅ Binance API מחובר (Read Only)")
            
            # בדיקת Redis
            self.redis.ping()
            self.logger.info("✅ Redis מחובר")
            
        except Exception as e:
            self.logger.error(f"❌ שגיאה בחיבורים: {e}")
    
    def run_safe_monitoring(self):
        """ניטור בטוח - לוגינג בלבד"""
        self.logger.info("🔍 מתחיל ניטור בטוח - לוגינג בלבד...")
        
        cycle_count = 0
        while True:
            try:
                cycle_count += 1
                self.logger.info(f"📊 מחזור ניטור #{cycle_count} - NO TRADES")
                
                # 1. ניתוח שוק (לוגינג בלבד)
                self.analyze_market_safe()
                
                # 2. סריקת אותות (לוגינג בלבד)
                self.scan_signals_safe()
                
                # 3. דוח ביצועים
                if cycle_count % 10 == 0:
                    self.generate_safe_report()
                
                # המתנה בין סריקות
                time.sleep(60)  # סריקה כל דקה
                
            except KeyboardInterrupt:
                self.logger.info("⏹️ ניטור הופסק ידנית")
                break
            except Exception as e:
                self.logger.error(f"❌ שגיאה בניטור: {e}")
                time.sleep(30)
    
    def analyze_market_safe(self):
        """ניתוח שוק בטוח - לוגינג בלבד"""
        try:
            for symbol in self.trading_pairs:
                # קבל נתונים (read-only)
                data = self.get_market_data_safe(symbol)
                
                if data:
                    # חשב אינדיקטורים (לוגינג בלבד)
                    analysis = self.analyze_data_safe(data, symbol)
                    
                    # רשם לוג
                    self.log_market_analysis(symbol, analysis)
                
                time.sleep(1)  # הפסקה בין סריקות
                
        except Exception as e:
            self.logger.error(f"❌ שגיאה בניתוח שוק: {e}")
    
    def get_market_data_safe(self, symbol):
        """קבלת נתוני שוק בטוחה"""
        try:
            klines = self.client.get_klines(
                symbol=symbol,
                interval='15m',
                limit=20
            )
            
            df = pd.DataFrame(klines, columns=[
                'timestamp', 'open', 'high', 'low', 'close', 'volume',
                'close_time', 'quote_volume', 'trades', 'taker_buy', 'taker_quote', 'ignore'
            ])
            
            for col in ['open', 'high', 'low', 'close', 'volume']:
                df[col] = pd.to_numeric(df[col])
            
            return df
            
        except Exception as e:
            self.logger.error(f"❌ שגיאה בקבלת נתונים ל-{symbol}: {e}")
            return None
    
    def analyze_data_safe(self, data, symbol):
        """ניתוח נתונים בטוח - לוגינג בלבד"""
        try:
            close = data['close']
            current_price = close.iloc[-1]
            
            # אינדיקטורים בסיסיים
            rsi = ta.momentum.RSIIndicator(close, window=14).rsi().iloc[-1]
            macd = ta.trend.MACD(close).macd().iloc[-1]
            macd_signal = ta.trend.MACD(close).macd_signal().iloc[-1]
            
            # ניתוח מגמות
            sma_20 = ta.trend.SMAIndicator(close, window=20).sma_indicator().iloc[-1]
            trend_direction = "עולה" if current_price > sma_20 else "יורד"
            
            # ניתוח ווליום
            volume = data['volume']
            volume_ma = ta.trend.SMAIndicator(volume, window=20).sma_indicator().iloc[-1]
            volume_ratio = volume.iloc[-1] / volume_ma
            
            return {
                'symbol': symbol,
                'current_price': current_price,
                'rsi': rsi,
                'macd_signal': "חיובי" if macd > macd_signal else "שלילי",
                'trend': trend_direction,
                'volume_boost': volume_ratio,
                'signal_strength': self.calculate_signal_strength(rsi, macd, macd_signal),
                'timestamp': datetime.now()
            }
            
        except Exception as e:
            self.logger.error(f"❌ שגיאה בניתוח {symbol}: {e}")
            return {}
    
    def calculate_signal_strength(self, rsi, macd, macd_signal):
        """חישוב חוזק אות (לוגינג בלבד)"""
        strength = 0
        
        # RSI analysis
        if 30 < rsi < 70:
            if rsi > 50:
                strength += 1
            else:
                strength -= 1
        
        # MACD analysis
        if macd > macd_signal:
            strength += 1
        else:
            strength -= 1
        
        if strength > 0:
            return "קניה"
        elif strength < 0:
            return "מכירה"
        else:
            return "ניטרלי"
    
    def log_market_analysis(self, symbol, analysis):
        """רישום ניתוח שוק"""
        if not analysis:
            return
        
        self.logger.info(f"""
📈 ניתוח שוק - {symbol}:
   💰 מחיר: {analysis['current_price']:.6f}
   📊 RSI: {analysis['rsi']:.1f}
   📈 MACD: {analysis['macd_signal']}
   🎯 מגמה: {analysis['trend']}
   🔊 ווליום: {analysis['volume_boost']:.2f}x
   🎯 אות: {analysis['signal_strength']}
        """)
    
    def scan_signals_safe(self):
        """סריקת אותות בטוחה - לוגינג בלבד"""
        try:
            signals_found = 0
            
            for symbol in self.trading_pairs:
                # ניתוח מתקדם יותר
                signal = self.generate_safe_signal(symbol)
                
                if signal and signal['confidence'] > 0.6:
                    signals_found += 1
                    self.log_potential_trade(signal)
                
                time.sleep(1)
            
            if signals_found > 0:
                self.logger.info(f"🎯 נמצאו {signals_found} אותות מסחר פוטנציאליים (NO TRADE)")
            else:
                self.logger.info("🔍 לא נמצאו אותות מסחר משמעותיים")
                
        except Exception as e:
            self.logger.error(f"❌ שגיאה בסריקת אותות: {e}")
    
    def generate_safe_signal(self, symbol):
        """יצירת אות בטוח - לוגינג בלבד"""
        try:
            data = self.get_market_data_safe(symbol)
            if data is None:
                return None
            
            analysis = self.analyze_data_safe(data, symbol)
            
            # חישוב ביטחון
            confidence = self.calculate_confidence(analysis)
            
            if confidence > 0.6:
                return {
                    'symbol': symbol,
                    'action': "BUY" if analysis['signal_strength'] == "קניה" else "SELL",
                    'confidence': confidence,
                    'price': analysis['current_price'],
                    'analysis': analysis,
                    'timestamp': datetime.now()
                }
            
            return None
            
        except Exception as e:
            self.logger.error(f"❌ שגיאה ביצירת אות: {e}")
            return None
    
    def calculate_confidence(self, analysis):
        """חישוב ביטחון אות (לוגינג בלבד)"""
        confidence = 0.5  # בסיס
        
        # RSI confidence
        rsi = analysis['rsi']
        if 40 < rsi < 60:
            confidence += 0.2
        elif 30 < rsi < 70:
            confidence += 0.1
        
        # Volume confidence
        volume_boost = analysis['volume_boost']
        if volume_boost > 1.5:
            confidence += 0.2
        elif volume_boost > 1.2:
            confidence += 0.1
        
        # Trend confidence
        if analysis['trend'] == "עולה" and analysis['signal_strength'] == "קניה":
            confidence += 0.1
        elif analysis['trend'] == "יורד" and analysis['signal_strength'] == "מכירה":
            confidence += 0.1
        
        return min(confidence, 1.0)
    
    def log_potential_trade(self, signal):
        """רישום מסחר פוטנציאלי (לוגינג בלבד)"""
        self.logger.info(f"""
🎯 אות מסחר פוטנציאלי - NO TRADE:
   📈 {signal['symbol']} | {signal['action']}
   💰 מחיר: {signal['price']:.6f}
   📊 ביטחון: {signal['confidence']:.2f}
   ⚠️  מצב: NO TRADE MODE - לא בוצע מסחר!
   🕒 זמן: {signal['timestamp'].strftime('%H:%M:%S')}
        """)
        
        # שמור בהיסטוריה
        self.signals_history.append(signal)
    
    def generate_safe_report(self):
        """יצירת דוח בטוח"""
        try:
            total_signals = len(self.signals_history)
            recent_signals = [s for s in self.signals_history 
                            if datetime.now() - s['timestamp'] < timedelta(hours=1)]
            
            buy_signals = len([s for s in recent_signals if s['action'] == "BUY"])
            sell_signals = len([s for s in recent_signals if s['action'] == "SELL"])
            
            self.logger.info(f"""
📊 דוח ניטור בטוח - NO TRADE:
   🔍 אותות אחרונה: {len(recent_signals)} (שעה אחרונה)
   📈 קניות: {buy_signals}
   📉 מכירות: {sell_signals}
   📈 סה"כ אותות: {total_signals}
   🛡️  מצב: NO TRADE - אבטחה מקסימלית
   💾 זיכרון: {psutil.virtual_memory().percent}%
            """)
            
        except Exception as e:
            self.logger.error(f"❌ שגיאה ביצירת דוח: {e}")
    
    def simulate_trading_disabled(self):
        """סימולציה שמראה מה היה קורה אילו"""
        self.logger.info("🔮 סימולציה: מה היה קורה עם מסחר פעיל...")
        
        # ניתוח האותות האחרונים
        if self.signals_history:
            recent_signals = self.signals_history[-5:]  # 5 האותות האחרונים
            
            for signal in recent_signals:
                current_price = self.get_current_price_safe(signal['symbol'])
                if current_price:
                    price_change = (current_price - signal['price']) / signal['price'] * 100
                    
                    self.logger.info(f"""
💡 סימולציה עבור {signal['symbol']}:
   🎯 אות: {signal['action']} ב-{signal['price']:.6f}
   💰 מחיר נוכחי: {current_price:.6f}
   📈 שינוי: {price_change:+.2f}%
   📊 ביטחון: {signal['confidence']:.2f}
   🔮 תוצאה: {"רווח" if (signal['action'] == "BUY" and price_change > 0) or (signal['action'] == "SELL" and price_change < 0) else "הפסד"}
                    """)
    
    def get_current_price_safe(self, symbol):
        """קבלת מחיר נוכחי בטוחה"""
        try:
            ticker = self.client.get_symbol_ticker(symbol=symbol)
            return float(ticker['price'])
        except:
            return None

# 🚀 הרצת מערכת הלוגינג הבטוחה
```
