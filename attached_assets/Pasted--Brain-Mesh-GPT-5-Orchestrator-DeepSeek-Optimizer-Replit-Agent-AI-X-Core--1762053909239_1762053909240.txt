# ========== 2) utils/mesh_bus.py — State + Ping + Registry in-memory ==========
python - <<'PY'
from pathlib import Path
Path("utils").mkdir(exist_ok=True)
Path("utils/mesh_bus.py").write_text(r"""# -*- coding: utf-8 -*-
from __future__ import annotations
import os, json, time, hmac, hashlib, asyncio, aiohttp
from typing import Dict, Any, List

AI_MESH_PATH = os.getenv("AI_MESH_PATH","ai_mesh.json")
AI_MESH_SECRET = os.getenv("AI_MESH_SECRET","")
NODE_ID = os.getenv("NODE_ID","core-api")

_STATE: Dict[str, Any] = {
  "node": {"id": NODE_ID, "ts": int(time.time())},
  "peers": {},    # url -> {id, role, scopes, weight, ok, latency_ms, last_ts, version, enabled}
  "registry": {}, # id  -> url
  "ok": True,
  "last_ping_ts": 0
}

def _sig(msg: str) -> str:
    return hmac.new(AI_MESH_SECRET.encode(), msg.encode(), hashlib.sha256).hexdigest() if AI_MESH_SECRET else ""

def _load_mesh() -> Dict[str, Any]:
    try:
        with open(AI_MESH_PATH,"r",encoding="utf-8") as f:
            return json.load(f)
    except Exception:
        return {}

def snapshot() -> Dict[str,Any]:
    s=dict(_STATE)
    s["peers"]=dict(_STATE["peers"])
    s["registry"]=dict(_STATE["registry"])
    return s

def ensure_peer(peer: Dict[str,Any]) -> None:
    """idempotent add/update from file/registry"""
    url = peer.get("url","").rstrip("/")
    if not url: return
    rec = _STATE["peers"].get(url, {})
    rec.update({
      "id": peer.get("id"),
      "role": peer.get("role"),
      "scopes": peer.get("scopes", []),
      "weight": peer.get("weight", 1),
      "enabled": bool(peer.get("enabled", True))
    })
    _STATE["peers"][url] = rec
    if rec.get("id"):
        _STATE["registry"][rec["id"]] = url

def refresh_from_file() -> None:
    j = _load_mesh()
    for p in j.get("peers", []):
        ensure_peer(p)

async def _ping_one(sess: aiohttp.ClientSession, url: str) -> None:
    t0=time.perf_counter()
    try:
        async with sess.get(url, timeout=8) as r:
            j = await r.json()
        lat=int((time.perf_counter()-t0)*1000)
        rec=_STATE["peers"].setdefault(url, {})
        rec["ok"]=bool(j.get("ok", True))
        rec["last_ts"]=int(time.time())
        rec["latency_ms"]=lat
        rec["version"]=j.get("app",{}).get("version")
        # שמור node_id מהצד השני אם קיים
        nid = j.get("node",{}).get("id")
        if nid: rec["id"]=nid; _STATE["registry"][nid]=url
    except Exception as e:
        rec=_STATE["peers"].setdefault(url, {})
        rec.update({"ok": False, "last_ts": int(time.time()), "last_error": str(e)})

async def ping_loop() -> None:
    refresh_from_file()
    while True:
        try:
            urls=[u for u,meta in _STATE["peers"].items() if meta.get("enabled",True)]
            _STATE["last_ping_ts"]=int(time.time())
            if urls:
                async with aiohttp.ClientSession() as sess:
                    await asyncio.gather(*[_ping_one(sess,u) for u in urls])
            oks=[meta.get("ok") for meta in _STATE["peers"].values() if meta.get("enabled",True)]
            _STATE["ok"] = (not oks) or (sum(1 for x in oks if x) >= max(1, len(oks)//2))
        except Exception:
            pass
        await asyncio.sleep(15)
""", encoding="utf-8")
print("✅ utils/mesh_bus.py ready")
PY

# ========== 3) utils/mesh_consensus.py — קונסנזוס per-scope ==========
python - <<'PY'
from pathlib import Path
Path("utils/mesh_consensus.py").write_text(r"""# -*- coding: utf-8 -*-
from __future__ import annotations
from typing import Dict, Any, List

def required_roles_for_scope(scope: str) -> List[str]:
    # מיפוי אילו תפקידים חייבים לאשר עבור יכולת מסוימת
    return {
        "route:decide": ["orchestrator","co_supervisor"],
        "signal:confirm": ["co_supervisor"],
        "route:improve": ["optimizer"],
        "ops:apply": ["executor_assist"],
        "risk:gate": ["orchestrator","co_supervisor"],
    }.get(scope, [])

def has_scope(peer: Dict[str,Any], scope: str) -> bool:
    return scope in (peer.get("scopes") or [])

def quorum(peers: List[Dict[str,Any]], scope: str, min_score: float=8.5) -> bool:
    roles = set(required_roles_for_scope(scope))
    if not roles: return True
    ok_roles=set()
    for p in peers:
        if not p.get("ok"): continue
        if p.get("role") in roles and has_scope(p, scope):
            ok_roles.add(p.get("role"))
    return ok_roles == roles
""", encoding="utf-8")
print("✅ utils/mesh_consensus.py ready")
PY

Path("routes/mesh_public.py").write_text(r"""from __future__ import annotations
import os, time, hmac, hashlib
from typing import Any, Dict, List
from fastapi import APIRouter, HTTPException, Request
from utils.mesh_bus import snapshot, ensure_peer, _load_mesh, _sig

router = APIRouter(tags=["Mesh"])

@router.get("/mesh/public")
def mesh_public() -> Dict[str,Any]:
    s = snapshot()
    app = {"version": os.getenv("APP_VERSION","")}
    # סינון לשדות ציבוריים בלבד
    pub_peers = {}
    for url, meta in (s.get("peers") or {}).items():
        pub_peers[url] = {
            "id": meta.get("id"),
            "role": meta.get("role"),
            "scopes": meta.get("scopes"),
            "weight": meta.get("weight"),
            "ok": meta.get("ok"),
            "latency_ms": meta.get("latency_ms"),
            "last_ts": meta.get("last_ts"),
            "enabled": meta.get("enabled", True),
            "version": meta.get("version")
        }
    return {"ok": True, "node": s.get("node"), "peers": pub_peers, "app": app, "ts": int(time.time())}

@router.get("/mesh/peers")
def mesh_peers() -> Dict[str,Any]:
    j=_load_mesh()
    return {"ok": True, "peers": j.get("peers", [])}

@router.post("/mesh/register")
async def mesh_register(req: Request) -> Dict[str,Any]:
    body = await req.json()
    # נדרש: id, url, role, scopes, ts, sig
    need = ["id","url","role","scopes","ts","sig"]
    if any(k not in body for k in need):
        raise HTTPException(400, "bad_payload")
    msg = f"{body['id']}:{body['ts']}:{body['url']}"
    secret = os.getenv("AI_MESH_SECRET","")
    if not secret: raise HTTPException(500, "mesh_secret_missing")
    sig = hmac.new(secret.encode(), msg.encode(), hashlib.sha256).hexdigest()
    if sig != body["sig"]:
        raise HTTPException(403, "bad_sig")
    # שמירה בזיכרון; הקובץ יעדכן מחוץ למסלול (PR/DevOps) אם רוצים
    peer = {
        "id": body["id"],
        "url": body["url"].rstrip("/"),
        "role": body["role"],
        "scopes": body.get("scopes", []),
        "weight": int(body.get("weight", 1)),
        "enabled": bool(body.get("enabled", True))
    }
    ensure_peer(peer)
    return {"ok": True, "registered": peer}
""", encoding="utf-8")
print("✅ routes/mesh_public.py ready")
