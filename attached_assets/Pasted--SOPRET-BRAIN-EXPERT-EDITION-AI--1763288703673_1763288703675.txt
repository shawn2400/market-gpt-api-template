# utils/sopret_expert_engine.py
import time
import numpy as np
import pandas as pd
from collections import defaultdict, deque
from typing import Dict, List, Tuple, Optional, Any
import logging
import redis
import json
from sklearn.ensemble import RandomForestRegressor
from sklearn.preprocessing import StandardScaler
import joblib
import asyncio

class SopretExpertEngine:
    """
    🧠 SOPRET Brain - EXPERT Edition
    עם Redis Long-Term Memory + Machine Learning + Predictive Analytics
    """
    
    def __init__(self, redis_url: str = "redis://localhost:6379", memory_size: int = 10000):
        # Redis Connection for Long-Term Memory
        try:
            self.redis_client = redis.from_url(redis_url, decode_responses=True)
            self.redis_client.ping()
            logging.info("✅ Redis connected for long-term memory")
        except Exception as e:
            logging.warning(f"❌ Redis unavailable: {e}")
            self.redis_client = None

        # ML Model for Weight Optimization
        self.ml_model = None
        self.scaler = StandardScaler()
        self.model_trained = False
        self.last_training_time = 0
        
        # Enhanced Memory Architecture
        self.ensemble_weights = {
            "cryptohopper": 0.25,
            "technical": 0.20, 
            "regime": 0.20,
            "risk": 0.20,
            "alpha": 0.15
        }
        
        # Advanced Pattern Detection
        self.pattern_performance = defaultdict(lambda: deque(maxlen=memory_size))
        self.source_performance = defaultdict(lambda: {'success': 0, 'total': 0, 'profit': 0.0})
        self.banned_patterns = set()
        self.toxic_sources = set()
        
        # Multi-Exchange Toxic Pattern Detection
        self.exchange_patterns = defaultdict(lambda: defaultdict(lambda: deque(maxlen=1000)))
        
        # Predictive Health Analytics
        self.health_history = deque(maxlen=500)
        self.failure_predictions = deque(maxlen=100)
        self.system_health_score = 1.0
        self.health_threshold = 0.7
        
        # Advanced Learning
        self.adaptive_learning_rate = 0.01
        self.performance_history = deque(maxlen=2000)
        self.market_regime_history = deque(maxlen=1000)
        
        # Backtesting Engine
        self.backtest_results = {}
        
        logging.info("🧠 SOPRET EXPERT Brain initialized with ML + Redis")

    def save_to_redis(self, key: str, data: Any):
        """שומר נתונים ב-Redis לזיכרון ארוך טווח"""
        if self.redis_client:
            try:
                self.redis_client.setex(
                    f"sopret:{key}", 
                    86400 * 30,  # 30 days expiry
                    json.dumps(data)
                )
            except Exception as e:
                logging.error(f"Redis save error: {e}")

    def load_from_redis(self, key: str) -> Any:
        """טוען נתונים מ-Redis"""
        if self.redis_client:
            try:
                data = self.redis_client.get(f"sopret:{key}")
                return json.loads(data) if data else None
            except Exception as e:
                logging.error(f"Redis load error: {e}")
        return None

    def record_trade_result(self, source: str, success: bool, profit: float = 0.0, 
                          pattern_used: Optional[str] = None, exchange: str = "binance"):
        """רושם תוצאות מסחר מתקדמות"""
        
        # Update source performance with profit tracking
        self.source_performance[source]['total'] += 1
        self.source_performance[source]['profit'] += profit
        if success:
            self.source_performance[source]['success'] += 1

        # Multi-exchange pattern tracking
        if pattern_used:
            self.pattern_performance[pattern_used].append(1.0 if success else 0.0)
            self.exchange_patterns[exchange][pattern_used].append(1.0 if success else 0.0)

        # Save to long-term memory
        trade_data = {
            'timestamp': time.time(),
            'source': source,
            'success': success,
            'profit': profit,
            'pattern': pattern_used,
            'exchange': exchange
        }
        self.save_to_redis(f"trade:{int(time.time())}", trade_data)

    def train_ml_model(self):
        """מאמן מודל ML לחיזוי ביצועי מקורות"""
        try:
            # איסוף נתונים מאימון
            training_data = self._prepare_training_data()
            if len(training_data) < 50:  # צריך מספיק נתונים
                return

            X, y = training_data
            self.scaler.fit(X)
            X_scaled = self.scaler.transform(X)

            # אימון מודל Random Forest
            self.ml_model = RandomForestRegressor(
                n_estimators=100,
                max_depth=10,
                random_state=42
            )
            self.ml_model.fit(X_scaled, y)
            self.model_trained = True
            self.last_training_time = time.time()
            
            # שמירת המודל
            if self.redis_client:
                model_bytes = joblib.dumps(self.ml_model)
                self.redis_client.setex("sopret:ml_model", 86400, model_bytes)
            
            logging.info("✅ ML model trained and saved")
            
        except Exception as e:
            logging.error(f"ML training error: {e}")

    def _prepare_training_data(self):
        """מכין נתוני אימון עבור המודל"""
        features = []
        targets = []
        
        # איסוף נתונים היסטוריים
        for source, perf in self.source_performance.items():
            if perf['total'] > 10:
                # מאפיינים: success_rate, total_trades, recent_performance
                success_rate = perf['success'] / perf['total']
                recent_trades = list(self.performance_history)[-50:]
                recent_success = np.mean(recent_trades) if recent_trades else 0.5
                
                feature_vector = [
                    success_rate,
                    np.log(perf['total'] + 1),
                    recent_success,
                    self.system_health_score,
                    len(self.banned_patterns) / 100.0  # normalized
                ]
                
                features.append(feature_vector)
                targets.append(success_rate)  # חיזוי success rate עתידי
                
        return np.array(features), np.array(targets)

    def predict_source_performance(self, source: str) -> float:
        """מנבא ביצועים עתידיים של מקור"""
        if not self.model_trained:
            return self.source_performance[source]['success'] / max(1, self.source_performance[source]['total'])
        
        try:
            perf = self.source_performance[source]
            success_rate = perf['success'] / max(1, perf['total'])
            recent_trades = list(self.performance_history)[-50:]
            recent_success = np.mean(recent_trades) if recent_trades else 0.5
            
            feature_vector = np.array([[
                success_rate,
                np.log(perf['total'] + 1),
                recent_success,
                self.system_health_score,
                len(self.banned_patterns) / 100.0
            ]])
            
            X_scaled = self.scaler.transform(feature_vector)
            prediction = self.ml_model.predict(X_scaled)[0]
            return max(0.0, min(1.0, prediction))
            
        except Exception as e:
            logging.error(f"Prediction error: {e}")
            return success_rate

    def detect_cross_exchange_toxic_patterns(self) -> List[Tuple[str, str]]:
        """מזהה דפוסים רעילים across multiple exchanges"""
        toxic_patterns = []
        
        for pattern in set().union(*[set(patterns) for patterns in self.exchange_patterns.values()]):
            exchange_performances = []
            
            for exchange, patterns in self.exchange_patterns.items():
                if pattern in patterns and len(patterns[pattern]) >= 5:
                    performance = np.mean(patterns[pattern])
                    exchange_performances.append((exchange, performance))
            
            # אם הדפוס רעיל ב-2 exchanges או יותר
            toxic_exchanges = [ep for ep in exchange_performances if ep[1] < 0.4]
            if len(toxic_exchanges) >= 2:
                toxic_patterns.extend([(pattern, exchange) for exchange, _ in toxic_exchanges])
                logging.warning(f"🚨 Cross-exchange toxic pattern: {pattern} on {toxic_exchanges}")
                
        return toxic_patterns

    def predictive_health_analytics(self) -> Dict[str, float]:
        """ניתוח בריאות חיזויי - מנבא התדרדרות לפני שהיא קורה"""
        if len(self.health_history) < 20:
            return {"current_health": self.system_health_score, "predicted_health": self.system_health_score}
        
        recent_health = list(self.health_history)[-20:]
        times = np.arange(len(recent_health)).reshape(-1, 1)
        health_values = np.array(recent_health)
        
        # רגרסיה לינארית פשוטה לחיזוי
        try:
            slope = np.polyfit(times.flatten(), health_values, 1)[0]
            predicted_health = self.system_health_score + slope * 5  # חיזוי 5 steps קדימה
            
            risk_score = max(0.0, min(1.0, (0.5 - predicted_health) * 2))
            
            return {
                "current_health": self.system_health_score,
                "predicted_health": max(0.1, predicted_health),
                "trend_slope": slope,
                "failure_risk": risk_score,
                "action_required": risk_score > 0.7
            }
        except:
            return {"current_health": self.system_health_score, "predicted_health": self.system_health_score}

    def advanced_weight_optimization(self):
        """אופטימיזציית משקלים מתקדמת עם ML"""
        if not self.model_trained and len(self.performance_history) > 100:
            self.train_ml_model()
        
        new_weights = {}
        total_predicted = 0
        
        for source in self.ensemble_weights.keys():
            predicted_perf = self.predict_source_performance(source)
            new_weights[source] = predicted_perf
            total_predicted += predicted_perf
        
        if total_predicted > 0:
            # Normalize and apply smooth update
            for source in new_weights:
                target_weight = new_weights[source] / total_predicted
                current_weight = self.ensemble_weights[source]
                
                # Update with adaptive learning rate
                lr = self.adaptive_learning_rate * (1.0 + abs(target_weight - current_weight))
                self.ensemble_weights[source] = (1 - lr) * current_weight + lr * target_weight

        logging.info(f"🔧 Advanced weights optimized: {self.ensemble_weights}")

    def run_expert_cycle(self) -> Dict:
        """מריץ מחזור מומחה מלא"""
        
        # חיזוי בריאות
        health_analytics = self.predictive_health_analytics()
        
        # זיהוי דפוסים רעילים cross-exchange
        cross_toxic = self.detect_cross_exchange_toxic_patterns()
        for pattern, exchange in cross_toxic:
            self.banned_patterns.add(f"{pattern}::{exchange}")

        # אופטימיזציית משקלים עם ML
        self.advanced_weight_optimization()
        
        # עדכון בריאות מערכת
        self.system_health_score = health_analytics["current_health"]
        self.health_history.append(self.system_health_score)
        
        # Self-healing אם יש סיכון גבוה
        if health_analytics.get("failure_risk", 0) > 0.8:
            self.emergency_self_heal()
        
        # שמירת סטטוס ל-Redis
        status = {
            "health_score": round(self.system_health_score, 3),
            "predicted_health": round(health_analytics.get("predicted_health", self.system_health_score), 3),
            "failure_risk": round(health_analytics.get("failure_risk", 0), 3),
            "banned_patterns_count": len(self.banned_patterns),
            "toxic_sources_count": len(self.toxic_sources),
            "cross_exchange_toxic": len(cross_toxic),
            "ml_model_trained": self.model_trained,
            "weights": {k: round(v, 3) for k, v in self.ensemble_weights.items()}
        }
        
        self.save_to_redis("expert_status", status)
        return status

    def emergency_self_heal(self):
        """ריפוי חירום כאשר המערכת בצניחה חריפה"""
        logging.warning("🚑 EMERGENCY SELF-HEALING ACTIVATED")
        
        # איפוס אגרסיבי
        self.banned_patterns.clear()
        self.toxic_sources.clear()
        
        # איפוס משקלים לבסיסיים
        base_weights = {k: 0.2 for k in self.ensemble_weights}
        self.ensemble_weights = base_weights
        
        # רענון מודל ML
        if self.model_trained:
            self.train_ml_model()
        
        logging.info("✅ Emergency self-healing completed")

    def get_expert_insights(self) -> Dict:
        """מחזיר תובנות מומחה למסחר"""
        insights = {
            "recommended_sources": [],
            "risk_warnings": [],
            "market_adaptation": {},
            "performance_forecast": {}
        }
        
        # המלצות מקורות based on predictions
        for source in self.ensemble_weights:
            predicted_perf = self.predict_source_performance(source)
            if predicted_perf > 0.6:
                insights["recommended_sources"].append({
                    "source": source,
                    "predicted_performance": round(predicted_perf, 3),
                    "current_weight": round(self.ensemble_weights[source], 3)
                })
        
        # אזהרות סיכון
        health_analytics = self.predictive_health_analytics()
        if health_analytics.get("failure_risk", 0) > 0.6:
            insights["risk_warnings"].append(
                f"High system failure risk: {health_analytics['failure_risk']:.1%}"
            )
        
        return insights

# utils/expert_decision_engine.py
from utils.sopret_expert_engine import SopretExpertEngine
import logging
import numpy as np

class ExpertDecisionEngine:
    """
    🧠 Expert Decision Engine - עם יכולות חיזוי וניתוח מתקדמות
    """
    
    def __init__(self, redis_url: str = "redis://localhost:6379"):
        self.sopret = SopretExpertEngine(redis_url)
        self.market_adaptation_factors = {
            "high_volatility": 1.2,
            "low_volatility": 0.8,
            "trending_up": 1.1,
            "trending_down": 0.9,
            "ranging": 1.0
        }

    async def analyze_market_regime(self, volatility: float, trend_strength: float) -> str:
        """מנתח מצב שוק ומתאים החלטות"""
        if volatility > 0.05:
            return "high_volatility"
        elif volatility < 0.01:
            return "low_volatility"
        elif trend_strength > 0.7:
            return "trending_up"
        elif trend_strength < -0.7:
            return "trending_down"
        else:
            return "ranging"

    def apply_market_adaptation(self, score: float, market_regime: str) -> float:
        """מתאים את הציון לפי מצב השוק"""
        adaptation_factor = self.market_adaptation_factors.get(market_regime, 1.0)
        adapted_score = score * adaptation_factor
        return max(0.0, min(1.0, adapted_score))

    async def make_expert_decision(self, 
                                 hopper: float, 
                                 technical: float, 
                                 regime: float, 
                                 risk: float, 
                                 alpha: float,
                                 market_volatility: float = 0.02,
                                 trend_strength: float = 0.0) -> Dict:
        """מקבל החלטה מומחה עם ניתוח שוק מלא"""
        
        # ניתוח מצב שוק
        market_regime = await self.analyze_market_regime(market_volatility, trend_strength)
        
        # ציון בסיסי
        base_score = self.sopret.aggregate_sources(hopper, technical, regime, risk, alpha)
        
        # התאמה למצב שוק
        adapted_score = self.apply_market_adaptation(base_score, market_regime)
        
        # התאמה סופית לפי בריאות המערכת
        health_adjusted_score = adapted_score * self.sopret.system_health_score
        
        # החלטה סופית עם thresholds מותאמים
        if health_adjusted_score >= 0.7:
            decision = "STRONG_BUY"
            confidence = "HIGH"
        elif health_adjusted_score >= 0.6:
            decision = "BUY"
            confidence = "MEDIUM"
        elif health_adjusted_score <= 0.3:
            decision = "STRONG_SELL" 
            confidence = "HIGH"
        elif health_adjusted_score <= 0.4:
            decision = "SELL"
            confidence = "MEDIUM"
        else:
            decision = "HOLD"
            confidence = "LOW"

        # תובנות מומחה
        expert_insights = self.sopret.get_expert_insights()
        
        # הרצת מחזור מומחה
        system_status = self.sopret.run_expert_cycle()

        return {
            "decision": decision,
            "confidence": confidence,
            "final_score": round(health_adjusted_score, 3),
            "market_regime": market_regime,
            "base_score": round(base_score, 3),
            "adapted_score": round(adapted_score, 3),
            "system_health": round(self.sopret.system_health_score, 3),
            "expert_insights": expert_insights,
            "system_status": system_status,
            "timestamp": time.time()
        }

    def get_comprehensive_analytics(self) -> Dict:
        """מחזיר אנליטיקות מקיפות למערכת"""
        health_analytics = self.sopret.predictive_health_analytics()
        expert_insights = self.sopret.get_expert_insights()
        system_status = self.sopret.run_expert_cycle()
        
        return {
            "predictive_analytics": health_analytics,
            "expert_insights": expert_insights,
            "system_status": system_status,
            "performance_metrics": {
                "total_trades": sum([p['total'] for p in self.sopret.source_performance.values()]),
                "success_rate": np.mean(list(self.sopret.performance_history)) if self.sopret.performance_history else 0.5,
                "avg_profit": np.mean([p['profit'] for p in self.sopret.source_performance.values()])
            },
            "ml_status": {
                "model_trained": self.sopret.model_trained,
                "last_training": self.sopret.last_training_time,
                "training_samples": len(self.sopret.performance_history)
            }
        }

# routes/expert_sopret.py
from fastapi import APIRouter, HTTPException, BackgroundTasks
from utils.expert_decision_engine import ExpertDecisionEngine
from pydantic import BaseModel
import time
import asyncio

router = APIRouter(prefix="/expert", tags=["SOPRET Expert"])
expert_engine = ExpertDecisionEngine()

class ExpertTradeResult(BaseModel):
    source: str
    success: bool 
    profit: float = 0.0
    pattern_used: str = None
    exchange: str = "binance"

class ExpertDecisionRequest(BaseModel):
    hopper: float
    technical: float
    regime: float
    risk: float
    alpha: float
    market_volatility: float = 0.02
    trend_strength: float = 0.0

@router.post("/decision")
async def expert_decision(request: ExpertDecisionRequest):
    """מקבל החלטת מומחה עם ניתוח שוק מלא"""
    try:
        decision = await expert_engine.make_expert_decision(
            hopper=request.hopper,
            technical=request.technical,
            regime=request.regime,
            risk=request.risk,
            alpha=request.alpha,
            market_volatility=request.market_volatility,
            trend_strength=request.trend_strength
        )
        return decision
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))

@router.post("/record-trade")
def record_expert_trade(result: ExpertTradeResult):
    """רושם תוצאות מסחר מתקדמות"""
    try:
        expert_engine.sopret.record_trade_result(
            source=result.source,
            success=result.success,
            profit=result.profit,
            pattern_used=result.pattern_used,
            exchange=result.exchange
        )
        return {"status": "success", "message": "Trade recorded with expert analytics"}
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))

@router.get("/analytics")
def get_comprehensive_analytics():
    """מחזיר אנליטיקות מקיפות של המערכת"""
    try:
        analytics = expert_engine.get_comprehensive_analytics()
        return analytics
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))

@router.post("/train-ml")
def train_ml_model(background_tasks: BackgroundTasks):
    """מאמן את מודל ה-ML ברקע"""
    try:
        background_tasks.add_task(expert_engine.sopret.train_ml_model)
        return {"status": "training_started", "message": "ML model training initiated"}
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))

@router.get("/redis-status")
def redis_status():
    """בודק סטטוס חיבור Redis"""
    try:
        if expert_engine.sopret.redis_client:
            info = expert_engine.sopret.redis_client.info()
            return {
                "status": "connected",
                "used_memory": info.get('used_memory_human', 'N/A'),
                "connected_clients": info.get('connected_clients', 'N/A')
            }
        else:
            return {"status": "disconnected"}
    except Exception as e:
        return {"status": "error", "detail": str(e)}

@router.get("/cross-exchange-toxic")
def get_cross_exchange_toxic():
    """מחזיר דפוסים רעילים across exchanges"""
    try:
        toxic_patterns = expert_engine.sopret.detect_cross_exchange_toxic_patterns()
        return {
            "toxic_patterns": toxic_patterns,
            "count": len(toxic_patterns)
        }
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))

@app.on_event("startup")
async def startup_event():
    logging.info("🧠 SOPRET EXPERT Brain - Advanced AI Trading System Ready")
    # טעינת מודל ML אם קיים
    if expert_engine.sopret.redis_client:
        try:
            model_bytes = expert_engine.sopret.redis_client.get("sopret:ml_model")
            if model_bytes:
                expert_engine.sopret.ml_model = joblib.loads(model_bytes)
                expert_engine.sopret.model_trained = True
                logging.info("✅ Pre-trained ML model loaded from Redis")
        except Exception as e:
            logging.warning(f"Could not load ML model: {e}")
