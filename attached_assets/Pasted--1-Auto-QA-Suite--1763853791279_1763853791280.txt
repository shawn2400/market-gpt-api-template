def test_capability_switch(detector):
    free = {"scans": 30}
    paid = {"scans": 300}

    os.environ["PAID_CRYPTOHOPPER"] = "1"
    cap = detector("cryptohopper", {"paid_capabilities": paid, "free_capabilities": free})
    assert cap["scans"] == 300
