# core/rl_trading_agent.py
import tensorflow as tf
import numpy as np

class TradingRLAgent:
    """
    סוכן מסחר עם Deep Q-Learning - לומד מהניסיון
    """
    def __init__(self, state_size=50, action_size=3):  # hold, buy, sell
        self.state_size = state_size
        self.action_size = action_size
        self.memory = []  # שמירת ניסיון קודם
        self.model = self._build_model()
    
    def _build_model(self):
        model = tf.keras.Sequential([
            tf.keras.layers.Dense(128, activation='relu', input_shape=(self.state_size,)),
            tf.keras.layers.Dropout(0.3),
            tf.keras.layers.Dense(64, activation='relu'),
            tf.keras.layers.Dropout(0.3),
            tf.keras.layers.Dense(self.action_size, activation='linear')
        ])
        model.compile(optimizer='adam', loss='mse')
        return model
    
    def remember(self, state, action, reward, next_state, done):
        """שומר את החוויה למידה עתידית"""
        self.memory.append((state, action, reward, next_state, done))
    
    def act(self, state):
        """מקבל החלטת מסחר based on state"""
        q_values = self.model.predict(state.reshape(1, -1), verbose=0)
        return np.argmax(q_values[0])  # מחזיר action
```

# core/deep_learning_predictor.py
import torch
import torch.nn as nn

class PricePredictor(nn.Module):
    """
    LSTM Neural Network לחיזוי מחירים
    """
    def __init__(self, input_size=10, hidden_size=50, num_layers=2):
        super(PricePredictor, self).__init__()
        self.hidden_size = hidden_size
        self.num_layers = num_layers
        
        self.lstm = nn.LSTM(input_size, hidden_size, num_layers, 
                           batch_first=True, dropout=0.2)
        self.fc = nn.Linear(hidden_size, 3)  # up, down, neutral
    
    def forward(self, x):
        h0 = torch.zeros(self.num_layers, x.size(0), self.hidden_size)
        c0 = torch.zeros(self.num_layers, x.size(0), self.hidden_size)
        
        out, _ = self.lstm(x, (h0, c0))
        out = self.fc(out[:, -1, :])  # רק ה-output האחרון
        return out

class AdvancedPredictor:
    def __init__(self):
        self.model = PricePredictor()
        self.optimizer = torch.optim.Adam(self.model.parameters(), lr=0.001)
        self.criterion = nn.CrossEntropyLoss()
    
    def train_on_market_data(self, sequences, targets):
        """מאמן את המודל על נתוני שוק"""
        self.model.train()
        outputs = self.model(sequences)
        loss = self.criterion(outputs, targets)
        
        self.optimizer.zero_grad()
        loss.backward()
        self.optimizer.step()
        
        return loss.item()
```

# core/feature_engineer.py
import ta  # technical analysis library
import pandas as pd

class AdvancedFeatureEngineer:
    """
    יצירת features מתקדמים ל-machine learning
    """
    def create_advanced_features(self, df):
        """יוצר features טכניים מתקדמים"""
        # Features בסיסיים
        df['returns'] = df['close'].pct_change()
        df['volatility'] = df['returns'].rolling(20).std()
        df['momentum'] = df['close'] / df['close'].shift(10) - 1
        
        # RSI variations
        df['rsi_6'] = ta.momentum.RSIIndicator(df['close'], window=6).rsi()
        df['rsi_14'] = ta.momentum.RSIIndicator(df['close'], window=14).rsi()
        df['rsi_21'] = ta.momentum.RSIIndicator(df['close'], window=21).rsi()
        
        # MACD features
        macd = ta.trend.MACD(df['close'])
        df['macd'] = macd.macd()
        df['macd_signal'] = macd.macd_signal()
        df['macd_histogram'] = macd.macd_diff()
        
        # Bollinger Bands features
        bollinger = ta.volatility.BollingerBands(df['close'])
        df['bb_upper'] = bollinger.bollinger_hband()
        df['bb_lower'] = bollinger.bollinger_lband()
        df['bb_position'] = (df['close'] - df['bb_lower']) / (df['bb_upper'] - df['bb_lower'])
        
        # Volume features
        df['volume_sma'] = df['volume'].rolling(20).mean()
        df['volume_ratio'] = df['volume'] / df['volume_sma']
        
        # Price position features
        df['high_20'] = df['high'].rolling(20).max()
        df['low_20'] = df['low'].rolling(20).min()
        df['price_position'] = (df['close'] - df['low_20']) / (df['high_20'] - df['low_20'])
        
        return df.dropna()
```

# core/ai_trading_engine.py
class AITradingEngine(AutonomousTradingEngine):
    """
    מערכת מסחר אוטונומית עם AI מתקדם
    """
    def __init__(self, exchange_client):
        super().__init__(exchange_client)
        self.rl_agent = TradingRLAgent()
        self.ensemble_ai = EnsembleTradingAI()
        self.sentiment_analyzer = MarketSentimentAnalyzer()
        self.feature_engineer = AdvancedFeatureEngineer()
    
    async def enhanced_analysis(self, market_data):
        """ניתוח משופר עם AI"""
        # 1. ניתוח טכני בסיסי
        technical_signals = await super().analyze_and_trade(market_data)
        
        # 2. ניתוח סנטימנט
        sentiment = await self.sentiment_analyzer.analyze_news_sentiment(
            market_data['symbol']
        )
        
        # 3. חיזוי עם ensemble AI
        features = self.feature_engineer.create_advanced_features(
            pd.DataFrame([market_data])
        )
        ai_prediction = self.ensemble_ai.predict_trend(features.values[0])
        
        # 4. Reinforcement Learning decision
        state = self._create_rl_state(market_data, sentiment, ai_prediction)
        rl_action = self.rl_agent.act(state)
        
        # שילוב כל ההחלטות
        final_signal = self._combine_signals(
            technical_signals,
            sentiment,
            ai_prediction,
            rl_action
        )
        
        return final_signal
    
    def _combine_signals(self, technical, sentiment, ai, rl):
        """משלב את כל האותות להחלטה סופית"""
        weights = {
            'technical': 0.3,
            'sentiment': 0.2, 
            'ai': 0.3,
            'rl': 0.2
        }
        
        # לוגיקה לשילוב האותות
        combined_confidence = (
            technical.get('confidence', 0) * weights['technical'] +
            sentiment.get('average_sentiment', 0.5) * weights['sentiment'] +
            ai['confidence'] * weights['ai'] +
            (rl / 2) * weights['rl']  # RL action normalization
        )
        
        return {
            'final_confidence': combined_confidence,
            'components': {
                'technical': technical,
                'sentiment': sentiment,
                'ai': ai,
                'rl_action': rl
            },
            'recommended_action': 'BUY' if combined_confidence > 0.6 else 'SELL' if combined_confidence < 0.4 else 'HOLD'
        }
```
