## 📋 קוד תיקון מלא למערכת הדינאמית שלך

```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
מערכת מסחר מתוקנת - ניהול סיכונים דינאמי
DYNAMIC_RISK_MANAGEMENT_SYSTEM_V4
"""

import pandas as pd
import numpy as np
import ta
from binance.client import Client
import time
import logging
from datetime import datetime, timedelta
import math

class DynamicRiskManagementSystem:
    def __init__(self, api_key, api_secret):
        self.client = Client(api_key, api_secret)
        self.equity = self.get_account_balance()
        
        # 🎯 הגדרות ניהול סיכונים דינאמיות
        self.risk_parameters = {
            'max_portfolio_risk': 0.02,  # 2% סיכון מקסימלי לפורטפוליו
            'max_trade_risk': 0.01,     # 1% סיכון מקסימלי לעסקה בודדת
            'max_leverage': 5,          # מינוף מקסימלי
            'volatility_adjusted_sl': True,
            'dynamic_position_sizing': True,
            'correlation_protection': True
        }
        
        # 📊 מאגר נתונים היסטוריים
        self.trade_history = []
        self.performance_metrics = {}
        self.volatility_regime = "NORMAL"
        
        self.setup_advanced_logging()
        self.initialize_risk_system()
    
    def setup_advanced_logging(self):
        """הגדר מערכת לוגינג מתקדמת"""
        logger = logging.getLogger()
        logger.setLevel(logging.INFO)
        
        formatter = logging.Formatter(
            '%(asctime)s | %(levelname)s | %(message)s',
            datefmt='%Y-%m-%d %H:%M:%S'
        )
        
        file_handler = logging.FileHandler('dynamic_risk_system.log', encoding='utf-8')
        file_handler.setFormatter(formatter)
        
        console_handler = logging.StreamHandler()
        console_handler.setFormatter(formatter)
        
        logger.addHandler(file_handler)
        logger.addHandler(console_handler)
        
        self.logger = logging.getLogger(__name__)
    
    def initialize_risk_system(self):
        """אתחול מערכת ניהול הסיכונים"""
        self.logger.info("🛡️ מערכת ניהול סיכונים דינאמית מאותחלת...")
        
        # עדכן פרמטרים based on ביצועים היסטוריים
        self.update_risk_parameters()
        
        # בדוק את מצב השוק הנוכחי
        self.analyze_market_conditions()
        
        self.logger.info("✅ מערכת ניהול סיכונים מוכנה!")
    
    def get_account_balance(self):
        """קבל יתרת חשבון עם טיפול בשגיאות"""
        try:
            account = self.client.get_account()
            usdt_balance = next(
                (float(x['free']) for x in account['balances'] if x['asset'] == 'USDT'), 
                0.0
            )
            self.logger.info(f"💰 יתרת חשבון: {usdt_balance:.2f} USDT")
            return usdt_balance
        except Exception as e:
            self.logger.error(f"❌ שגיאה בקבלת יתרה: {e}")
            return 1000.0  # ברירת מחדל
    
    def update_risk_parameters(self):
        """עדכן פרמטרי סיכון based on ביצועים היסטוריים"""
        if len(self.trade_history) < 5:
            return  # לא מספיק נתונים
        
        # חשב win rate אחרון
        recent_trades = self.trade_history[-10:]
        wins = len([t for t in recent_trades if t.get('pnl', 0) > 0])
        win_rate = wins / len(recent_trades) if recent_trades else 0.5
        
        # התאם סיכון based on win rate
        if win_rate > 0.7:
            # ביצועים מעולים - אפשר סיכון מעט גבוה יותר
            self.risk_parameters['max_trade_risk'] = 0.015
            self.risk_parameters['max_leverage'] = 5
        elif win_rate < 0.3:
            # ביצועים גרועים - הפחת סיכון
            self.risk_parameters['max_trade_risk'] = 0.005
            self.risk_parameters['max_leverage'] = 3
        else:
            # ביצועים סבירים - הגדרות רגילות
            self.risk_parameters['max_trade_risk'] = 0.01
            self.risk_parameters['max_leverage'] = 4
        
        self.logger.info(f"📊 עדכון סיכון: Win Rate {win_rate:.2%} -> סיכון: {self.risk_parameters['max_trade_risk']:.3f}")
    
    def analyze_market_conditions(self):
        """ניתוח תנאי שוק להתאמת סיכון"""
        try:
            # בדוק וולאטיליות שוק כללית
            symbols = ['BTCUSDT', 'ETHUSDT', 'BNBUSDT']
            volatility_scores = []
            
            for symbol in symbols:
                data = self.get_volatility_data(symbol)
                if data is not None:
                    volatility_scores.append(data)
            
            if volatility_scores:
                avg_volatility = np.mean(volatility_scores)
                
                if avg_volatility > 0.04:
                    self.volatility_regime = "HIGH_VOL"
                    self.risk_parameters['max_leverage'] = min(3, self.risk_parameters['max_leverage'])
                    self.risk_parameters['max_trade_risk'] *= 0.7  # הפחת סיכון
                elif avg_volatility < 0.015:
                    self.volatility_regime = "LOW_VOL"
                    self.risk_parameters['max_trade_risk'] *= 1.2  # הגדל סיכון במעט
                else:
                    self.volatility_regime = "NORMAL"
                
                self.logger.info(f"🌡️ מצב שוק: {self.volatility_regime} | וולאטיליות: {avg_volatility:.3f}")
                
        except Exception as e:
            self.logger.error(f"❌ שגיאה בניתוח שוק: {e}")
    
    def get_volatility_data(self, symbol, period=20):
        """קבל נתוני וולאטיליות עבור סימבול"""
        try:
            klines = self.client.get_klines(symbol=symbol, interval='1d', limit=period)
            closes = [float(k[4]) for k in klines]
            
            returns = np.diff(closes) / closes[:-1]
            volatility = np.std(returns) * math.sqrt(365)  # וולאטיליות שנתית
            
            return volatility
            
        except Exception as e:
            self.logger.warning(f"⚠️ שגיאה בקבלת וולאטיליות עבור {symbol}: {e}")
            return None
    
    def calculate_dynamic_position_size(self, symbol, entry_price, stop_loss_price):
        """חשב גודל פוזיציה דינאמי עם הגנות"""
        try:
            # 1. חישוב סיכון בסיסי
            risk_amount = self.equity * self.risk_parameters['max_trade_risk']
            
            # 2. חישוב מרחק stop loss
            price_diff_pct = abs(entry_price - stop_loss_price) / entry_price
            
            # 3. הגבלת סיכון מקסימלי לעסקה
            max_risk_pct = 0.05  # 5% מקסימום הפסד לעסקה בודדת
            if price_diff_pct > max_risk_pct:
                self.logger.warning(f"⚠️ Stop Loss רחוק מדי - מתאים מחדש")
                stop_loss_price = entry_price * (1 - max_risk_pct) if entry_price > stop_loss_price else entry_price * (1 + max_risk_pct)
                price_diff_pct = max_risk_pct
            
            # 4. חישוב גודל פוזיציה
            position_size = risk_amount / abs(entry_price - stop_loss_price)
            
            # 5. הגבלות based on מינוף ווולאטיליות
            max_position_value = self.equity * self.risk_parameters['max_leverage']
            current_position_value = position_size * entry_price
            
            if current_position_value > max_position_value:
                position_size = max_position_value / entry_price
                self.logger.info(f"📏 הגבלת פוזיציה based on מינוף מקסימלי")
            
            # 6. עיגול למספר הנכון של עשרוניות
            precision = self.get_symbol_precision(symbol)
            position_size = round(position_size, precision)
            
            # 7. וידוא כמות מינימלית
            min_quantity = self.get_min_quantity(symbol)
            if position_size < min_quantity:
                self.logger.warning(f"⚠️ כמות קטנה מדי - מבטל עסקה")
                return 0
            
            self.logger.info(f"📏 גודל פוזיציה: {position_size} {symbol.split('USDT')[0]}")
            
            return position_size
            
        except Exception as e:
            self.logger.error(f"❌ שגיאה בחישוב פוזיציה: {e}")
            return 0
    
    def get_symbol_precision(self, symbol):
        """קבל דיוק כמות עבור סימבול"""
        try:
            info = self.client.get_symbol_info(symbol)
            lot_size_filter = next(
                (filt for filt in info['filters'] if filt['filterType'] == 'LOT_SIZE'),
                None
            )
            if lot_size_filter:
                step_size = float(lot_size_filter['stepSize'])
                return abs(int(np.log10(step_size)))
        except:
            pass
        return 2  # ברירת מחדל
    
    def get_min_quantity(self, symbol):
        """קבל כמות מינימלית עבור סימבול"""
        try:
            info = self.client.get_symbol_info(symbol)
            lot_size_filter = next(
                (filt for filt in info['filters'] if filt['filterType'] == 'LOT_SIZE'),
                None
            )
            if lot_size_filter:
                return float(lot_size_filter['minQty'])
        except:
            pass
        return 0.001  # ברירת מחדל
    
    def calculate_dynamic_stop_loss_take_profit(self, symbol, entry_price, direction, volatility_data=None):
        """חישוב SL/TP דינאמי based on וולאטיליות"""
        try:
            if volatility_data is None:
                volatility_data = self.get_volatility_data(symbol, 10) or 0.02
            
            # בסיס based on וולאטיליות
            base_sl_distance = volatility_data * 1.5
            base_tp_distance = volatility_data * 2.5
            
            # התאם based on מצב שוק
            if self.volatility_regime == "HIGH_VOL":
                sl_distance = base_sl_distance * 0.8  # SL יותר הדוק
                tp_distance = base_tp_distance * 0.8  # TP יותר צנוע
            elif self.volatility_regime == "LOW_VOL":
                sl_distance = base_sl_distance * 1.2  # SL יותר מרווח
                tp_distance = base_tp_distance * 1.5  # TP יותר אגרסיבי
            else:
                sl_distance = base_sl_distance
                tp_distance = base_tp_distance
            
            # הגבלות מינימליות ומקסימליות
            sl_distance = max(0.005, min(0.08, sl_distance))  # 0.5%-8%
            tp_distance = max(0.015, min(0.15, tp_distance))  # 1.5%-15%
            
            if direction == "LONG":
                stop_loss = entry_price * (1 - sl_distance)
                take_profit = entry_price * (1 + tp_distance)
            else:  # SHORT
                stop_loss = entry_price * (1 + sl_distance)
                take_profit = entry_price * (1 - tp_distance)
            
            risk_reward = abs(take_profit - entry_price) / abs(entry_price - stop_loss)
            
            # ודא Risk/Reward סביר
            if risk_reward < 1.5:
                # התאם את ה-TP ל-R/R מינימלי
                min_rr = 1.5
                if direction == "LONG":
                    take_profit = entry_price + (min_rr * (entry_price - stop_loss))
                else:
                    take_profit = entry_price - (min_rr * (stop_loss - entry_price))
                
                risk_reward = min_rr
            
            return {
                'stop_loss': round(stop_loss, 6),
                'take_profit': round(take_profit, 6),
                'risk_reward_ratio': risk_reward,
                'sl_distance_percent': sl_distance * 100,
                'tp_distance_percent': tp_distance * 100
            }
            
        except Exception as e:
            self.logger.error(f"❌ שגיאה בחישוב SL/TP: {e}")
            # ערכי ברירת מחדל
            return {
                'stop_loss': round(entry_price * 0.98, 6),
                'take_profit': round(entry_price * 1.04, 6),
                'risk_reward_ratio': 2.0,
                'sl_distance_percent': 2.0,
                'tp_distance_percent': 4.0
            }
    
    def validate_trade_conditions(self, symbol, signal_confidence, position_size, sl_tp_data):
        """בדיקת תוקף עסקה לפני ביצוע"""
        validation_checks = []
        
        # 1. בדיקת ביטחון אות
        if signal_confidence < 0.65:
            validation_checks.append(("ביטחון אות נמוך", False))
        else:
            validation_checks.append(("ביטחון אות", True))
        
        # 2. בדיקת גודל פוזיציה
        if position_size <= 0:
            validation_checks.append(("גודל פוזיציה לא תקין", False))
        else:
            validation_checks.append(("גודל פוזיציה", True))
        
        # 3. בדיקת Risk/Reward
        if sl_tp_data['risk_reward_ratio'] < 1.3:
            validation_checks.append(("Risk/Reward נמוך", False))
        else:
            validation_checks.append(("Risk/Reward", True))
        
        # 4. בדיקת סיכון מקסימלי
        position_value = position_size * sl_tp_data.get('entry_price', 0)
        risk_exposure = position_value / self.equity
        
        if risk_exposure > 0.1:  # 10% מקסימום חשיפה לעסקה בודדת
            validation_checks.append(("חשיפת סיכון גבוהה", False))
        else:
            validation_checks.append(("חשיפת סיכון", True))
        
        # 5. בדיקת וולאטיליות
        if self.volatility_regime == "HIGH_VOL" and risk_exposure > 0.05:
            validation_checks.append(("וולאטיליות גבוהה - סיכון מוגבל", False))
        else:
            validation_checks.append(("תנאי וולאטיליות", True))
        
        # סיכום הבדיקות
        all_valid = all(check[1] for check in validation_checks)
        
        if not all_valid:
            failed_checks = [check[0] for check in validation_checks if not check[1]]
            self.logger.warning(f"⚠️ עסקה נדחתה: {', '.join(failed_checks)}")
        
        return all_valid, validation_checks
    
    def execute_safe_trade(self, symbol, action, entry_price, signal_confidence):
        """ביצוע עסקה בטוחה עם כל ההגנות"""
        try:
            # 1. חישוב SL/TP דינאמי
            sl_tp_data = self.calculate_dynamic_stop_loss_take_profit(symbol, entry_price, action)
            
            # 2. חישוב גודל פוזיציה
            position_size = self.calculate_dynamic_position_size(
                symbol, entry_price, sl_tp_data['stop_loss']
            )
            
            if position_size <= 0:
                return None
            
            # 3. בדיקת תוקף עסקה
            is_valid, checks = self.validate_trade_conditions(
                symbol, signal_confidence, position_size, sl_tp_data
            )
            
            if not is_valid:
                return None
            
            # 4. ביצוע העסקה
            if action == "BUY":
                order = self.client.order_market_buy(
                    symbol=symbol,
                    quantity=position_size
                )
            else:  # SELL
                order = self.client.order_market_sell(
                    symbol=symbol,
                    quantity=position_size
                )
            
            # 5. רישום SL/TP
            self.place_stop_loss_take_profit(symbol, position_size, sl_tp_data, action)
            
            # 6. שמירת נתוני עסקה
            trade_record = {
                'symbol': symbol,
                'action': action,
                'entry_price': entry_price,
                'position_size': position_size,
                'stop_loss': sl_tp_data['stop_loss'],
                'take_profit': sl_tp_data['take_profit'],
                'risk_reward': sl_tp_data['risk_reward_ratio'],
                'timestamp': datetime.now(),
                'signal_confidence': signal_confidence,
                'volatility_regime': self.volatility_regime
            }
            
            self.trade_history.append(trade_record)
            
            self.logger.info(f"""
✅ עסקה בוצעה בהצלחה עם הגנות:
   🎯 {symbol} | {action}
   💰 כמות: {position_size}
   ⬆️ Entry: {entry_price}
   ⬇️ SL: {sl_tp_data['stop_loss']} ({sl_tp_data['sl_distance_percent']:.1f}%)
   ⬆️ TP: {sl_tp_data['take_profit']} ({sl_tp_data['tp_distance_percent']:.1f}%)
   📊 R/R: {sl_tp_data['risk_reward_ratio']:.2f}
   🛡️ הגנות: {len([c for c in checks if c[1]])}/{len(checks)}
            """)
            
            return order
            
        except Exception as e:
            self.logger.error(f"❌ שגיאה בביצוע עסקה: {e}")
            return None
    
    def place_stop_loss_take_profit(self, symbol, quantity, sl_tp_data, action):
        """רישום SL/TP עם הגנות"""
        try:
            if action == "BUY":
                # STOP LOSS
                self.client.create_order(
                    symbol=symbol,
                    side=Client.SIDE_SELL,
                    type=Client.ORDER_TYPE_STOP_LOSS_LIMIT,
                    timeInForce='GTC',
                    quantity=quantity,
                    stopPrice=round(sl_tp_data['stop_loss'], 6),
                    price=round(sl_tp_data['stop_loss'], 6)
                )
                
                # TAKE PROFIT
                self.client.create_order(
                    symbol=symbol,
                    side=Client.SIDE_SELL,
                    type=Client.ORDER_TYPE_TAKE_PROFIT_LIMIT,
                    timeInForce='GTC',
                    quantity=quantity,
                    stopPrice=round(sl_tp_data['take_profit'], 6),
                    price=round(sl_tp_data['take_profit'], 6)
                )
            else:  # SELL
                # STOP LOSS
                self.client.create_order(
                    symbol=symbol,
                    side=Client.SIDE_BUY,
                    type=Client.ORDER_TYPE_STOP_LOSS_LIMIT,
                    timeInForce='GTC',
                    quantity=quantity,
                    stopPrice=round(sl_tp_data['stop_loss'], 6),
                    price=round(sl_tp_data['stop_loss'], 6)
                )
                
                # TAKE PROFIT
                self.client.create_order(
                    symbol=symbol,
                    side=Client.SIDE_BUY,
                    type=Client.ORDER_TYPE_TAKE_PROFIT_LIMIT,
                    timeInForce='GTC',
                    quantity=quantity,
                    stopPrice=round(sl_tp_data['take_profit'], 6),
                    price=round(sl_tp_data['take_profit'], 6)
                )
            
            self.logger.info(f"🛡️ SL/TP רשומים: SL={sl_tp_data['stop_loss']:.6f}, TP={sl_tp_data['take_profit']:.6f}")
            
        except Exception as e:
            self.logger.error(f"❌ שגיאה ברישום SL/TP: {e}")
    
    def run_safe_trading_cycle(self, trading_pairs):
        """הרצת מחזור מסחר בטוח"""
        self.logger.info("🔄 מתחיל מחזור מסחר בטוח...")
        
        # עדכן פרמטרי סיכון
        self.update_risk_parameters()
        self.analyze_market_conditions()
        
        for symbol in trading_pairs:
            try:
                # קבל אות מסחר (מהמערכת הקיימת שלך)
                signal = self.get_trading_signal(symbol)
                
                if signal and signal.get('confidence', 0) > 0.65:
                    # בצע עסקה בטוחה
                    result = self.execute_safe_trade(
                        symbol=symbol,
                        action=signal['action'],
                        entry_price=signal['entry_price'],
                        signal_confidence=signal['confidence']
                    )
                    
                    if result:
                        self.logger.info(f"🎯 מסחר בוצע: {symbol}")
                
                # המתנה בין סריקות
                time.sleep(1)
                
            except Exception as e:
                self.logger.error(f"❌ שגיאה בסריקת {symbol}: {e}")
                continue
    
    def get_trading_signal(self, symbol):
        """
        פונקציה זו תחליף את המערכת הקיימת שלך
        אתה תשלב כאן את הלוגיקה של יצירת האותות שלך
        """
        # 🎯 כאן אתה משלב את האלגוריתם הקיים שלך ליצירת אותות
        # לדוגמה:
        try:
            klines = self.client.get_klines(symbol=symbol, interval='15m', limit=50)
            df = pd.DataFrame(klines, columns=[
                'timestamp', 'open', 'high', 'low', 'close', 'volume',
                'close_time', 'quote_volume', 'trades', 'taker_buy', 'taker_quote', 'ignore'
            ])
            
            for col in ['open', 'high', 'low', 'close', 'volume']:
                df[col] = pd.to_numeric(df[col])
            
            current_price = df['close'].iloc[-1]
            
            # 🎯 החלף את זה באלגוריתם האמיתי שלך
            signal_confidence = 0.7  # דוגמה
            action = "BUY"  # דוגמה
            
            return {
                'symbol': symbol,
                'action': action,
                'entry_price': current_price,
                'confidence': signal_confidence
            }
            
        except Exception as e:
            self.logger.error(f"❌ שגיאה ביצירת אות: {e}")
            return None
    
    def get_performance_report(self):
        """דוח ביצועים מפורט"""
        if not self.trade_history:
            return "📊 אין נתוני מסחר עדיין"
        
        total_trades = len(self.trade_history)
        winning_trades = len([t for t in self.trade_history if t.get('pnl', 0) > 0])
        win_rate = winning_trades / total_trades if total_trades > 0 else 0
        
        total_pnl = sum(t.get('pnl', 0) for t in self.trade_history)
        avg_rr = np.mean([t.get('risk_reward', 0) for t in self.trade_history])
        
        report = f"""
📊 דוח ביצועים - מערכת ניהול סיכונים
══════════════════════════════════
🔢 סה"כ עסקאות: {total_trades}
✅ עסקאות רווחיות: {winning_trades}
📈 Win Rate: {win_rate:.2%}
💰 PNL כולל: {total_pnl:.2f} USDT
🎯 ממוצע R/R: {avg_rr:.2f}
```
