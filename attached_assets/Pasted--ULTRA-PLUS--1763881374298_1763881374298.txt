# ✅ **1) utils/ultra_plus_manager.py**

```python
import time
from datetime import datetime, timedelta
from collections import defaultdict

### CONFIG ###

RISK_FREEZE_THRESHOLD = 0.85
FREEZE_MIN = 20 * 60
FREEZE_MAX = 3 * 60 * 60

NEWS_BLOCK_BEFORE_MIN = 90
NEWS_BLOCK_AFTER_MIN = 30

FUNDING_BLOCK_THRESHOLD = 0.035  # 3.5%
FUNDING_EXTREME = 0.06           # 6%

SENTIMENT_NEGATIVE = -0.35
SENTIMENT_CRITICAL = -0.55

AUTO_PAUSE_LOSS_LIMIT = -150      # $/day
AUTO_PAUSE_HOURS = 12

ML_THRESHOLD_ENTER = 0.62
ML_THRESHOLD_EXIT = 0.18


class UltraPlusManager:
    def __init__(self):
        self.frozen = {}                     # symbol → {until, reason}
        self.news_events = []                # list of datetime UTC
        self.last_pause_until = 0
        self.symbol_heatmap = defaultdict(lambda: {"win": 0, "loss": 0})
        self.ml_model = None

    # =============================
    # ✅ NEWS WINDOW BLOCK
    # =============================
    def load_news_events(self, events: list):
        self.news_events = events

    def is_news_blocked(self) -> bool:
        now = datetime.utcnow()
        for event in self.news_events:
            before = event - timedelta(minutes=NEWS_BLOCK_BEFORE_MIN)
            after = event + timedelta(minutes=NEWS_BLOCK_AFTER_MIN)
            if before <= now <= after:
                return True
        return False

    # =============================
    # ✅ AUTO-FREEZE SYMBOL RISK
    # =============================
    def evaluate_symbol_risk(self, symbol: str, risk_score: float):
        if risk_score < RISK_FREEZE_THRESHOLD:
            return False

        duration = min(
            FREEZE_MIN + int(risk_score * 3600),
            FREEZE_MAX
        )

        self.frozen[symbol] = {
            "until": time.time() + duration,
            "reason": f"risk={risk_score:.2f}"
        }
        return True

    def is_symbol_frozen(self, symbol: str) -> bool:
        data = self.frozen.get(symbol)
        if not data:
            return False
        if time.time() >= data["until"]:
            del self.frozen[symbol]
            return False
        return True

    # =============================
    # ✅ FUNDING REGIME FILTER
    # =============================
    def funding_filter(self, funding_rate: float):
        if funding_rate >= FUNDING_EXTREME:
            return "block"
        if funding_rate >= FUNDING_BLOCK_THRESHOLD:
            return "reduce"
        return "normal"

    # =============================
    # ✅ AI SENTIMENT FILTER
    # =============================
    def sentiment_filter(self, sentiment: float):
        if sentiment <= SENTIMENT_CRITICAL:
            return "block"
        if sentiment <= SENTIMENT_NEGATIVE:
            return "caution"
        return "normal"

    # =============================
    # ✅ AUTO-PAUSE BASED ON PNL
    # =============================
    def auto_pause(self, daily_pnl: float):
        if daily_pnl <= AUTO_PAUSE_LOSS_LIMIT:
            self.last_pause_until = time.time() + (AUTO_PAUSE_HOURS * 3600)
            return True
        return False

    def is_paused(self):
        if time.time() < self.last_pause_until:
            return True
        return False

    # =============================
    # ✅ HEATMAP PERFORMANCE
    # =============================
    def update_heatmap(self, symbol: str, pnl: float):
        if pnl >= 0:
            self.symbol_heatmap[symbol]["win"] += 1
        else:
            self.symbol_heatmap[symbol]["loss"] += 1

    def get_heatmap(self):
        return self.symbol_heatmap

    # =============================
    # ✅ ML-PREDICTION (stub model)
    # =============================
    def load_ml_model(self, model):
        self.ml_model = model

    def ml_decision(self, features: dict):
        if not self.ml_model:
            return None

        prob = self.ml_model.predict_proba(features)

        if prob >= ML_THRESHOLD_ENTER:
            return "enter"
        if prob <= ML_THRESHOLD_EXIT:
            return "exit"
        return "hold"

    # =============================
    # ✅ STATUS EXPORT FOR DASHBOARD
    # =============================
    def status(self):
        return {
            "news_block": self.is_news_blocked(),
            "paused": self.is_paused(),
            "frozen": self.frozen,
            "heatmap": self.symbol_heatmap,
        }
```

```python
from utils.ultra_plus_manager import UltraPlusManager
```

ואז לפני פתיחת טרייד:

```python
# 1) GLOBAL NEWS BLOCK
if ultra.is_news_blocked():
    return {"status": "blocked_news"}

# 2) DAILY AUTO-PAUSE
if ultra.is_paused():
    return {"status": "auto_paused"}

# 3) SYMBOL RISK
if ultra.evaluate_symbol_risk(symbol, candidate.get("risk", 0)):
    return {"status": "symbol_frozen_new"}

if ultra.is_symbol_frozen(symbol):
    return {"status": "symbol_frozen_existing"}

# 4) FUNDING REGIME
funding_state = ultra.funding_filter(candidate.get("funding_rate", 0))
if funding_state == "block":
    return {"status": "blocked_funding"}
elif funding_state == "reduce":
    size *= 0.5

# 5) AI SENTIMENT
sent_state = ultra.sentiment_filter(candidate.get("sentiment", 0))
if sent_state == "block":
    return {"status": "blocked_sentiment"}
elif sent_state == "caution":
    size *= 0.7

# 6) ML DECISION
decision = ultra.ml_decision(candidate.get("features", {}))
if decision == "exit":
    return {"status": "ai_exit_signal"}
```

# ✅ **4) בדיקות pytest**

`tests/test_ultra.py`

```python
from utils.ultra_plus_manager import UltraPlusManager
from datetime import datetime, timedelta

def test_news_block():
    u = UltraPlusManager()
    event = datetime.utcnow() + timedelta(minutes=10)
    u.load_news_events([event])
    assert u.is_news_blocked() is True

def test_freeze():
    u = UltraPlusManager()
    assert u.evaluate_symbol_risk("BTCUSDT", 0.9) is True
    assert u.is_symbol_frozen("BTCUSDT") is True

def test_funding():
    u = UltraPlusManager()
    assert u.funding_filter(0.07) == "block"
    assert u.funding_filter(0.04) == "reduce"
    assert u.funding_filter(0.01) == "normal"
```
