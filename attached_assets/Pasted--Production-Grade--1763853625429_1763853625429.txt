# 🟩 external/plugin_manager.py

```python
# plugin_manager.py
import importlib
from external.plugin_registry import AVAILABLE_PLUGINS
from algo_core.capability_detector import detect_capabilities

class PluginManager:
    def __init__(self):
        self.plugins = {}

    def load_all(self):
        for name, cfg in AVAILABLE_PLUGINS.items():
            module_path, class_name = cfg["client"].rsplit(".", 1)
            module = importlib.import_module(module_path)
            cls = getattr(module, class_name)

            # Auto detect free/paid plan
            capabilities = detect_capabilities(name, cfg)

            self.plugins[name] = cls(capabilities)

    def get(self, name):
        return self.plugins.get(name)

    def get_by_type(self, ptype):
        return [p for p in self.plugins.values() if p.ptype == ptype]
```

# 🟩 algo_core/hybrid_router.py

```python
# hybrid_router.py
class HybridRouter:
    def __init__(self, plugins):
        self.plugins = plugins

    def get_scans(self):
        scans = []
        for p in self.plugins.get_by_type("scanner"):
            scans.append(p.get_market_scan())
        return scans

    def get_signals(self):
        signals = []
        for p in self.plugins.get_by_type("signals"):
            signals.extend(p.get_signals().get("signals", []))
        return signals

    def execute_order(self, order):
        # Delegation hierarchy:
        # 1) HyperTrader
        # 2) Binance (native)
        # 3) 3Commas SmartTrade
        for ptype in ["execution", "manager"]:
            for p in self.plugins.get_by_type(ptype):
                return p.execute(order) if hasattr(p, "execute") else p.manage_position(order["symbol"], order["sl"], order["tp"])
```

```python
# consensus_engine.py
class ConsensusEngine:
    def merge(self, scans, signals):
        fused = []

        # Combine Hopper+TradingView+WunderTrading
        for s in scans:
            for item in s.get("data", []):
                fused.append({
                    "symbol": item.get("symbol"),
                    "score": item.get("score", 0),
                    "source": s["source"]
                })

        for sig in signals:
            fused.append({"symbol": sig["symbol"], "score": 10, "source": "Signal"})

        # Sort → High score wins
        return sorted(fused, key=lambda x: x["score"], reverse=True)[:10]
```

```python
# self_optimizer.py
class SelfOptimizer:
    def evaluate(self, trade_results):
        score = {}
        for result in trade_results:
            bot = result["source"]
            if bot not in score:
                score[bot] = []
            score[bot].append(result["outcome"])

        for bot, lst in score.items():
            score[bot] = sum(lst) / max(len(lst), 1)

        return score
```

```python
class DynamicManager:
    def __init__(self, interval=4):
        self.interval = interval

    def update(self, position):
        # Example: dynamic adjustment (placeholder)
        position["sl"] *= 0.999
        position["tp"] *= 1.001
        return position

    def run(self, open_positions):
        while True:
            for pos in open_positions:
                self.update(pos)
```
