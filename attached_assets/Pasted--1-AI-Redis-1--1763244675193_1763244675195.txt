### **שלב 2 - AI קל משקל (קוד מוכן):**
```python
# core/lightweight_ai.py
from sklearn.ensemble import RandomForestClassifier, GradientBoostingClassifier
import pandas as pd
import numpy as np

class SmartTradingAI:
    """AI חכם אבל קל - מתאים ל-2GB"""
    
    def __init__(self):
        self.models = {
            'trend_predictor': RandomForestClassifier(n_estimators=50, max_depth=10),
            'momentum_analyzer': GradientBoostingClassifier(n_estimators=50)
        }
        self.memory_footprint = "250-350MB"
    
    def predict_signal(self, market_data):
        """חיזוי אותות עם מודלים קלים"""
        features = self.extract_light_features(market_data)
        
        # השתמש רק ב-10,000 דגימות לחסוך זיכרון
        if len(features) > 10000:
            features = features.sample(10000)
            
        prediction = self.models['trend_predictor'].predict_proba([features])
        return {
            'direction': 'long' if prediction[0][1] > 0.6 else 'short',
            'confidence': float(prediction[0][1]),
            'memory_used': 'light'
        }
```

### **שלב 3 - Redis חכם (קוד מוכן):**
```python
# utils/smart_redis_manager.py
class SmartRedisManager:
    """Redis שמתקן את עצמו - לא צורך worker קבוע"""
    
    async def health_check_and_fix(self):
        """בודק ומתקן רק כשצריך"""
        try:
            # בדיקה אם TOP 50 חסר או פגום
            top_50_count = await self.redis.scard('symbols:top_50')
            
            if top_50_count < 40:  # אם חסרים סמלים
                await self.refill_top_50_intelligently()
                return f"🔄 Refilled top_50 - was missing {50 - top_50_count} symbols"
            
            return "✅ Redis is healthy"
            
        except Exception as e:
            # recovery אוטומטי במקרה של שגיאה
            await self.emergency_redis_recovery()
            return f"🚨 Auto-recovered from: {e}"
```
