```python
class SmartTop50Scanner:
    def __init__(self):
        self.previous_top_100 = []  # זיכרון מהסריקה הקודמת
        self.candidate_pool_size = 120  # רק 120 מועמדים вместо 538!
        
    async def get_optimized_candidate_pool(self):
        """מחזיר רק 120 מועמדים פוטנציאליים במקום 538!"""
        
        # 1. קח את ה-TOP 100 מהסריקה הקודמת (80% מהמועמדים)
        previous_candidates = self.previous_top_100[:80] if self.previous_top_100 else []
        
        # 2. הוסף 40 מועמדים חדשים אקראיים לגיוון
        all_symbols = await self.binance.get_all_symbols()
        new_candidates = random.sample(
            [s for s in all_symbols if s not in previous_candidates], 
            min(40, len(all_symbols) - len(previous_candidates))
        )
        
        candidate_pool = previous_candidates + new_candidates
        
        # 3. שמור ל-next scan
        self.previous_top_100 = candidate_pool[:100]
        
        logger.info(f"🔍 Scanning {len(candidate_pool)} candidates instead of 538!")
        return candidate_pool
    
    async def calculate_optimized_top_50(self):
        """חישוב TOP 50 מסריקה אופטימלית"""
        
        # 1. קבל רק 120 מועמדים במקום 538!
        candidate_pool = await self.get_optimized_candidate_pool()
        
        # 2. חשב ציונים רק למועמדים האלה
        scored_symbols = []
        for symbol in candidate_pool:
            try:
                symbol_data = await self.get_symbol_data(symbol)
                score = self.scorer.calculate_symbol_score(symbol_data)
                
                if score > 0:
                    scored_symbols.append({
                        'symbol': symbol,
                        'score': score,
                        'volume': symbol_data['volume_24h']
                    })
                    
            except Exception as e:
                continue
        
        # 3. מיין וקח TOP 50
        scored_symbols.sort(key=lambda x: x['score'], reverse=True)
        top_50 = [s['symbol'] for s in scored_symbols[:50]]
        
        logger.info(f"✅ Optimized TOP 50 calculated from {len(candidate_pool)} candidates")
        return top_50
```

```python
class ProfitGuardTradingEngine:
    def __init__(self):
        self.profit_targets = {
            'quick_profit': 0.015,    # 1.5% - רווח מהיר
            'normal_profit': 0.025,   # 2.5% - רווח נורמלי  
            'stop_loss': -0.008,      # 0.8% - עצירת הפסד
            'breakeven': 0.002        # 0.2% - יציאה ללא רווח/הפסד
        }
        self.active_trades = {}
        
    async def monitor_active_trades(self):
        """ניטור רצוף של פוזיציות פתוחות"""
        while True:
            try:
                # 1. קבל את כל הפוזיציות הפתוחות
                open_positions = await self.binance.get_open_positions()
                
                for position in open_positions:
                    await self.evaluate_trade_exit(position)
                    
                # 2. המתנה 30 שניות בין סריקות
                await asyncio.sleep(30)
                
            except Exception as e:
                logger.error(f"❌ Trade monitoring failed: {e}")
                await asyncio.sleep(60)
    
    async def evaluate_trade_exit(self, position):
        """הערכת יציאה אוטומטית מפוזיציה"""
        symbol = position['symbol']
        unrealized_pnl = position['unrealizedPnl']
        entry_price = position['entryPrice']
        current_price = position['currentPrice']
        
        # חשב ROI נוכחי
        current_roi = (current_price - entry_price) / entry_price
        
        # החלטת יציאה אוטומטית
        exit_decision = await self.calculate_exit_decision(
            symbol, current_roi, unrealized_pnl, position
        )
        
        if exit_decision['should_exit']:
            await self.execute_auto_exit(position, exit_decision)
    
    async def calculate_exit_decision(self, symbol, current_roi, unrealized_pnl, position):
        """חישוב החלטת יציאה אוטומטית"""
        
        # 1. רווח מהיר - יציאה מיידית
        if current_roi >= self.profit_targets['quick_profit']:
            return {
                'should_exit': True,
                'reason': 'QUICK_PROFIT_ACHIEVED',
                'target': self.profit_targets['quick_profit'],
                'actual_roi': current_roi
            }
        
        # 2. עצירת הפסד - הגנה
        elif current_roi <= self.profit_targets['stop_loss']:
            return {
                'should_exit': True, 
                'reason': 'STOP_LOSS_TRIGGERED',
                'target': self.profit_targets['stop_loss'],
                'actual_roi': current_roi
            }
        
        # 3. רווח נורמלי - יציאה מתונה
        elif current_roi >= self.profit_targets['normal_profit']:
            return {
                'should_exit': True,
                'reason': 'NORMAL_PROFIT_TARGET',
                'target': self.profit_targets['normal_profit'], 
                'actual_roi': current_roi
            }
        
        # 4. יציאה ללא רווח/הפסד - לאחר זמן מסוים
        elif await self.should_break_even(symbol, position):
            return {
                'should_exit': True,
                'reason': 'BREAKEVEN_TIMEOUT',
                'target': self.profit_targets['breakeven'],
                'actual_roi': current_roi
            }
        
        return {'should_exit': False}
    
    async def should_break_even(self, symbol, position):
        """האם הגיע זמן ליציאה ללא רווח/הפסד?"""
        position_age = datetime.now() - position['opened_at']
        
        # אם הפוזיציה פתוחה יותר מ-4 שעות ובאפס
        if position_age.total_seconds() > 14400:  # 4 שעות
            current_roi = await self.get_current_roi(position)
            if abs(current_roi) <= self.profit_targets['breakeven']:
                return True
                
        return False
    
    async def execute_auto_exit(self, position, exit_decision):
        """ביצוע יציאה אוטומטית"""
        try:
            symbol = position['symbol']
            quantity = position['quantity']
            
            # סגור את הפוזיציה
            exit_result = await self.binance.close_position(
                symbol=symbol,
                quantity=quantity
            )
            
            # לוג את היציאה
            logger.info(
                f"💰 AUTO EXIT | {symbol} | "
                f"Reason: {exit_decision['reason']} | "
                f"ROI: {exit_decision['actual_roi']:.3%} | "
                f"P&L: {position['unrealizedPnl']:.2f} USDT"
            )
            
            # עדכן סטטיסטיקות
            await self.update_performance_stats(exit_decision, position)
            
        except Exception as e:
            logger.error(f"❌ Auto exit failed for {symbol}: {e}")
```
