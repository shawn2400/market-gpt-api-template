```python
# utils/quantum_top50_fixer.py
import asyncio
import logging
from typing import Dict, List
from datetime import datetime, timedelta

logger = logging.getLogger(__name__)

class QuantumTop50Fixer:
    """מתקן את Quantum TOP 50 Worker - הבעיה השורשית"""
    
    def __init__(self, db_client, binance_client):
        self.db = db_client
        self.binance = binance_client
        self.quantum_worker = None
        
    async def fix_quantum_worker(self):
        """תיקון מלא של Quantum Worker"""
        logger.info("🔧 מתקן Quantum TOP 50 Worker...")
        
        try:
            # 1. אתחול מחדש של Worker
            await self._reinitialize_worker()
            
            # 2. עדכון קריטריונים
            await self._update_selection_criteria()
            
            # 3. הפעלת סריקה מיידית
            await self._run_immediate_scan()
            
            # 4. וידוא עדכון GRID APPROVED
            await self._verify_grid_approved_update()
            
            logger.info("✅ Quantum TOP 50 Worker תוקן בהצלחה")
            return True
            
        except Exception as e:
            logger.error(f"❌ תיקון Quantum Worker נכשל: {e}")
            return False
    
    async def _reinitialize_worker(self):
        """אתחול מחדש של Worker"""
        logger.info("🔄 מאתחל מחדש Quantum Worker...")
        
        # עצירת Worker קיים אם רץ
        if self.quantum_worker and self.quantum_worker.is_running:
            await self.quantum_worker.stop()
            
        # יצירת Worker חדש עם הגדרות מתוקנות
        from workers.quantum_top50_worker import QuantumTop50Worker
        self.quantum_worker = QuantumTop50Worker(self.db, self.binance)
        
        # עדכון הגדרות
        await self._update_worker_settings()
    
    async def _update_worker_settings(self):
        """עדכון הגדרות Worker"""
        new_settings = {
            'scan_interval': 300,  # 5 דקות במקום 30
            'min_volume': 1000000,  # volume מינימלי
            'max_volatility': 0.15,  # volatility מקסימלי
            'min_liquidity': 50000,   # liquidity מינימלי
            'required_metrics': ['volume', 'liquidity', 'volatility', 'trend'],
            'auto_approve_threshold': 0.7  # 70% score מינימלי
        }
        
        await self.db.update_worker_settings('quantum_top50', new_settings)
        logger.info("✅ עדכנו הגדרות Quantum Worker")
    
    async def _update_selection_criteria(self):
        """עדכון קריטריוני בחירה"""
        criteria = {
            'volume_24h': {'min': 1000000, 'weight': 0.25},
            'liquidity': {'min': 50000, 'weight': 0.20},
            'volatility': {'max': 0.15, 'weight': 0.15},
            'price_stability': {'min_score': 0.6, 'weight': 0.15},
            'market_trend': {'weight': 0.25},
            'grid_suitability': {'min_score': 0.7, 'weight': 1.0}
        }
        
        await self.db.update_selection_criteria('grid_approved', criteria)
        logger.info("✅ עדכנו קריטריוני בחירה")
    
    async def _run_immediate_scan(self):
        """סריקה מיידית של סימבולים"""
        logger.info("🔍 מריץ סריקה מיידית...")
        
        # קבלת כל הסימבולים הזמינים
        all_symbols = await self.binance.fetch_markets()
        spot_symbols = [s for s in all_symbols if s['spot'] and s['active']]
        
        # סינון ראשוני
        filtered_symbols = await self._prefilter_symbols(spot_symbols)
        
        # ניתוח מתקדם
        analyzed_symbols = await self._analyze_symbols(filtered_symbols)
        
        # דירוג ובחירת TOP 50
        top_50 = await self._rank_and_select(analyzed_symbols, 50)
        
        # עדכון GRID APPROVED LIST
        await self._update_grid_approved(top_50)
        
        logger.info(f"✅ סריקה הושלמה - נמצאו {len(top_50)} סימבולים מתאימים")
    
    async def _prefilter_symbols(self, symbols: List) -> List:
        """סינון ראשוני של סימבולים"""
        filtered = []
        
        for symbol in symbols:
            try:
                # בדיקות בסיסיות
                if not await self._passes_basic_checks(symbol):
                    continue
                    
                # בדיקת volume
                if not await self._has_sufficient_volume(symbol['symbol']):
                    continue
                    
                # בדיקת liquidity
                if not await self._has_sufficient_liquidity(symbol['symbol']):
                    continue
                    
                filtered.append(symbol)
                
            except Exception as e:
                logger.debug(f"סימבול {symbol['symbol']} נכשל בסינון: {e}")
                continue
                
        return filtered
    
    async def _analyze_symbols(self, symbols: List) -> List[Dict]:
        """ניתוח מתקדם של סימבולים"""
        analyzed = []
        
        for symbol_info in symbols:
            try:
                symbol = symbol_info['symbol']
                
                # נתונים היסטוריים
                ohlcv = await self.binance.fetch_ohlcv(symbol, '1h', limit=24)
                if not ohlcv or len(ohlcv) < 24:
                    continue
                
                # חישוב מדדים
                metrics = await self._calculate_metrics(symbol, ohlcv)
                
                # ציון suitability ל-Grid
                grid_score = await self._calculate_grid_suitability(metrics)
                
                analyzed.append({
                    'symbol': symbol,
                    'metrics': metrics,
                    'grid_score': grid_score,
                    'analysis_time': datetime.now()
                })
                
            except Exception as e:
                logger.debug(f"ניתוח נכשל עבור {symbol_info['symbol']}: {e}")
                continue
                
        return analyzed
    
    async def _calculate_grid_suitability(self, metrics: Dict) -> float:
        """חישוב ציון התאמה ל-Grid Trading"""
        score = 0.0
        
        # volatility אופטימלי ל-Grid (בינוני)
        volatility = metrics.get('volatility', 0)
        if 0.02 <= volatility <= 0.08:
            score += 0.3
        elif 0.01 <= volatility <= 0.12:
            score += 0.2
        else:
            score += 0.1
            
        # volume יציב
        volume_consistency = metrics.get('volume_consistency', 0)
        score += volume_consistency * 0.3
        
        # trend יציב (לא חזק מדי)
        trend_strength = abs(metrics.get('trend_strength', 0))
        if 0.1 <= trend_strength <= 0.3:
            score += 0.2
        elif trend_strength <= 0.5:
            score += 0.1
            
        # price stability
        stability = metrics.get('price_stability', 0)
        score += stability * 0.2
        
        return min(score, 1.0)
```

```python
# utils/blacklist_cleaner.py
import logging
from datetime import datetime, timedelta

logger = logging.getLogger(__name__)

class BlacklistCleaner:
    """מנקה את TEMP BLACKLIST"""
    
    def __init__(self, db_client):
        self.db = db_client
        
    async def clear_temp_blacklist(self):
        """ניקוי מלא של TEMP BLACKLIST"""
        logger.info("🧹 מנקה TEMP BLACKLIST...")
        
        try:
            # 1. קבלת כל הרשומות ב-BLACKLIST
            blacklisted = await self.db.get_temp_blacklist()
            
            if not blacklisted:
                logger.info("✅ TEMP BLACKLIST כבר ריק")
                return True
                
            # 2. סינון רשומות ישנות (יותר מ-24 שעות)
            now = datetime.now()
            to_remove = []
            
            for item in blacklisted:
                blacklist_time = item.get('blacklisted_at', now)
                if isinstance(blacklist_time, str):
                    blacklist_time = datetime.fromisoformat(blacklist_time)
                    
                # אם עברו יותר מ-24 שעות - מסיר
                if now - blacklist_time > timedelta(hours=24):
                    to_remove.append(item['symbol'])
                else:
                    logger.info(f"🔒 שומר {item['symbol']} ב-BLACKLIST (טרי)")
            
            # 3. הסרת רשומות ישנות
            if to_remove:
                await self.db.remove_from_blacklist(to_remove)
                logger.info(f"✅ הסרנו {len(to_remove)} סימבולים מ-BLACKLIST")
                
            # 4. איפוס failure counts
            await self.reset_failure_counts()
            
            logger.info("✅ TEMP BLACKLIST נוקה בהצלחה")
            return True
            
        except Exception as e:
            logger.error(f"❌ ניקוי BLACKLIST נכשל: {e}")
            return False
    
    async def reset_failure_counts(self):
        """איפוס ספירת failures לכל הסימבולים"""
        try:
            # קבלת כל הסימבולים עם failure counts
            symbols_with_failures = await self.db.get_symbols_with_failures()
            
            # איפוס ל-0
            for symbol_data in symbols_with_failures:
                await self.db.update_failure_count(symbol_data['symbol'], 0)
                
            logger.info(f"✅ איפסנו failure counts עבור {len(symbols_with_failures)} סימבולים")
            
        except Exception as e:
            logger.error(f"❌ איפוס failure counts נכשל: {e}")
```

```python
# utils/grid_approved_rebuilder.py
import logging
from typing import List, Dict

logger = logging.getLogger(__name__)

class GridApprovedRebuilder:
    """בונה מחדש את GRID APPROVED LIST"""
    
    def __init__(self, db_client, binance_client):
        self.db = db_client
        self.binance = binance_client
        
    async def rebuild_grid_approved_list(self):
        """בנייה מחדש של GRID APPROVED LIST"""
        logger.info("🏗️ בונה מחדש GRID APPROVED LIST...")
        
        try:
            # 1. קבלת TOP 50 מסונן
            top_50_symbols = await self._get_filtered_top_50()
            
            # 2. וידוא שהם מתאימים ל-Grid
            grid_approved = await self._verify_grid_suitability(top_50_symbols)
            
            # 3. עדכון הדאטאבייס
            await self._update_grid_approved_db(grid_approved)
            
            # 4. וידוא עדכון
            await self._verify_update()
            
            logger.info(f"✅ GRID APPROVED LIST נבנתה מחדש עם {len(grid_approved)} סימבולים")
            return True
            
        except Exception as e:
            logger.error(f"❌ בניית GRID APPROVED LIST נכשלה: {e}")
            return False
    
    async def _get_filtered_top_50(self) -> List[Dict]:
        """קבלת TOP 50 מסונן"""
        # שימוש ב-Quantum Worker המתוקן
        quantum_fixer = QuantumTop50Fixer(self.db, self.binance)
        top_50 = await quantum_fixer.get_current_top_50()
        
        # סינון נוסף עבור Grid
        filtered = []
        for symbol_data in top_50:
            if await self._is_grid_suitable(symbol_data):
                filtered.append(symbol_data)
                
        return filtered[:50]  # הגבלה ל-50
    
    async def _is_grid_suitable(self, symbol_data: Dict) -> bool:
        """בדיקה אם סימבול מתאים ל-Grid Trading"""
        try:
            symbol = symbol_data['symbol']
            metrics = symbol_data.get('metrics', {})
            
            # קריטריונים מינימליים ל-Grid
            conditions = [
                metrics.get('grid_score', 0) >= 0.6,  # ציון התאמה מינימלי
                metrics.get('volatility', 0) >= 0.01,  # volatility מינימלי
                metrics.get('volatility', 0) <= 0.12,  # volatility מקסימלי
                metrics.get('volume_24h', 0) >= 500000,  # volume מינימלי
                metrics.get('liquidity', 0) >= 25000,    # liquidity מינימלי
                await self._has_reasonable_spread(symbol)  # spread סביר
            ]
            
            return all(conditions)
            
        except Exception as e:
            logger.debug(f"Grid suitability check failed for {symbol_data.get('symbol')}: {e}")
            return False
    
    async def _has_reasonable_spread(self, symbol: str) -> bool:
        """בדיקה אם spread סביר"""
        try:
            ticker = await self.binance.fetch_ticker(symbol)
            bid = ticker['bid']
            ask = ticker['ask']
            
            if bid == 0 or ask == 0:
                return False
                
            spread = (ask - bid) / bid
            return spread <= 0.002  # spread של 0.2% או פחות
            
        except Exception:
            return False
```

```python
# main_fix_script.py
import asyncio
import logging
from utils.quantum_top50_fixer import QuantumTop50Fixer
from utils.blacklist_cleaner import BlacklistCleaner
from utils.grid_approved_rebuilder import GridApprovedRebuilder

logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

async def execute_root_fix():
    """הפעלת התיקון השורשי המלא"""
    logger.info("🎯 מתחיל תיקון שורשי מלא...")
    
    try:
        # אתחול clients
        db_client = await get_db_client()  # החלף עם הקוד שלך
        binance_client = await get_binance_client()  # החלף עם הקוד שלך
        
        # 1. תיקון Quantum Worker
        quantum_fixer = QuantumTop50Fixer(db_client, binance_client)
        quantum_success = await quantum_fixer.fix_quantum_worker()
        
        if not quantum_success:
            logger.error("❌ נכשל בתיקון Quantum Worker - מפסיק")
            return False
        
        # 2. ניקוי BLACKLIST
        blacklist_cleaner = BlacklistCleaner(db_client)
        blacklist_success = await blacklist_cleaner.clear_temp_blacklist()
        
        if not blacklist_success:
            logger.error("❌ נכשל בניקוי BLACKLIST - ממשיך בכל זאת")
        
        # 3. בניית GRID APPROVED מחדש
        grid_rebuilder = GridApprovedRebuilder(db_client, binance_client)
        grid_success = await grid_rebuilder.rebuild_grid_approved_list()
        
        if not grid_success:
            logger.error("❌ נכשל בבניית GRID APPROVED")
            return False
        
        # 4. וידוא סופי
        await verify_fix_success(db_client)
        
        logger.info("🎉 תיקון שורשי הושלם בהצלחה!")
        logger.info("✅ Quantum TOP 50 Worker תוקן")
        logger.info("✅ TEMP BLACKLIST נוקה") 
        logger.info("✅ GRID APPROVED LIST נבנתה מחדש")
        logger.info("🚀 Grid Trading אמור לחזור לפעולה!")
        
        return True
        
    except Exception as e:
        logger.error(f"💥 תיקון שורשי נכשל: {e}")
        return False

async def verify_fix_success(db_client):
    """וידוא שהתיקון הצליח"""
    # בדיקה ש-GRID APPROVED לא ריקה
    grid_approved = await db_client.get_grid_approved_symbols()
    logger.info(f"📊 GRID APPROVED LIST מכילה {len(grid_approved)} סימבולים")
    
    # בדיקה ש-BLACKLIST נוקה
    blacklisted = await db_client.get_temp_blacklist()
    logger.info(f"🔒 TEMP BLACKLIST מכיל {len(blacklisted)} סימבולים")
    
    if len(grid_approved) > 10:  # לפחות 10 סימבולים
        logger.info("✅ התיקון הצליח - Grid Trading יכול לפעול")
    else:
        logger.warning("⚠️ GRID APPROVED קטן מדי - ייתכן ש仍需 adjustments")

if __name__ == "__main__":
    asyncio.run(execute_root_fix())
```
