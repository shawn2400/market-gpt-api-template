## ✅ **שלב 3 — מנוע SL/TP דינמי מלא (Copy-Paste מוכן)**

`/agent_core/sl_tp_engine.py`

```python
class SLTPDynamicEngine:
    def __init__(self):
        self.min_lock = 0.35    # R אחרי BE
        self.progress_trigger = 0.5  # חייב התקדמות
        self.multiplier = 1.2   # ATR trailing

    def update(self, position, price, atr):
        entry = position["entry"]
        sl = position["sl"]
        tp1 = position["tp1"]

        # BE+
        if not position.get("be_done") and price >= tp1:
            position["sl"] = entry * 1.0006
            position["be_done"] = True
            position["last_high"] = price
            return position

        # TRAILING רק אם יש התקדמות אמיתית
        if position.get("be_done") and price > position.get("last_high", entry):
            new_sl = price - (atr * self.multiplier)
            if new_sl > sl:
                position["sl"] = new_sl
                position["last_high"] = price

        return position
```

```python
def detect_market_mode(adx, rsi, atr, funding, sentiment):
    if adx > 30 and rsi > 58:
        return "TREND_UP"
    if adx > 30 and rsi < 42:
        return "TREND_DOWN"
    if atr > 4.2:
        return "HIGH_VOL"
    if abs(rsi - 50) < 3:
        return "CHOPPY"
    if sentiment == "REVERSAL" or funding > 0.03:
        return "REVERSAL_SETUP"
    return "NORMAL"
```

`/agent_core/kill_switch.py`

```python
class KillSwitch:
    active = False

    @classmethod
    def trigger(cls, reason):
        cls.active = True
        print(f"⚠️ KILL-SWITCH: {reason}")

    @classmethod
    def release(cls):
        cls.active = False
```
