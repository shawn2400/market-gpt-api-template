## 🚀 **שלב 2: Redis Auto-Fix פשוט ויעיל**

### **🔧 Worker אוטומטי קטן (5 דקות):**
```python
# workers/redis_health_worker.py
import asyncio
import redis

class RedisHealthWorker:
    def __init__(self):
        self.redis = redis.Redis(**REDIS_CONFIG)
        self.check_interval = 300  # 5 דקות
    
    async def run(self):
        """בודק ומתקן Redis כל 5 דקות"""
        while True:
            try:
                await self.check_and_fix_top_50()
                await asyncio.sleep(self.check_interval)
            except Exception as e:
                print(f"❌ Redis health worker error: {e}")
                await asyncio.sleep(60)  # המתן דקה ונסה שוב
    
    async def check_and_fix_top_50(self):
        """בודק אם TOP 50 קיים - אם לא, ממלא אותו"""
        try:
            # בדיקה אם הרשימה קיימת
            count = self.redis.scard('symbols:top_50')
            
            if count < 30:  # אם חסרים סמלים
                print(f"🔄 ממלא TOP 50 - היה רק {count} סמלים")
                await self.refill_top_50()
                print("✅ TOP 50 שוחזר")
            else:
                print(f"✅ TOP 50 תקין - {count} סמלים")
                
        except Exception as e:
            print(f"�️ שגיאת TOP 50: {e}")
            await self.emergency_refill()

    async def refill_top_50(self):
        """ממלא את TOP 50 עם הסמלים הטובים ביותר"""
        # כאן הקוד שלך למילוי TOP 50
        from utils.symbol_manager import SymbolManager
        symbol_mgr = SymbolManager()
        top_symbols = await symbol_mgr.get_top_50_symbols()
        
        # ניקוי והוספה ל-Redis
        self.redis.delete('symbols:top_50')
        for symbol in top_symbols:
            self.redis.sadd('symbols:top_50', symbol)
```
