    def initialize_system(self, total_balance: float):

    def calculate_adaptive_settings(self, balance: float) -> Dict:

        if balance >= 10000:

    def dynamic_risk_management(self, symbol: str, current_price: float) -> Dict:
        """ניהול סיכונים דינמי ואינטליגנטי"""
        
        # ניתוח תנודתיות
        volatility = self.calculate_volatility(symbol)
        market_regime = self.assess_market_regime()
        
        # חישוב מינוף דינמי
        leverage = self.calculate_smart_leverage(symbol, volatility, market_regime)
        
        # חישוב stop loss חכם
        stop_loss = self.calculate_intelligent_stop_loss(
            symbol, current_price, leverage, volatility, market_regime
        )
        
        # חישוב take profit אופטימלי
        take_profit = self.calculate_optimal_take_profit(
            current_price, stop_loss, volatility, market_regime
        )
        
        risk_metrics = {
            'leverage': leverage,
            'stop_loss': stop_loss,
            'take_profit': take_profit,
            'volatility_score': volatility,
            'market_regime': market_regime,
            'position_score': self.calculate_position_score(symbol, current_price)
        }
        
        return risk_metrics
    
    def calculate_smart_leverage(self, symbol: str, volatility: float, regime: str) -> float:
        """מינוף חכם based on תנודתיות, סוג נכס, ומצב שוק"""
        
        # בסיס מינוף based on תנודתיות
        if volatility > 0.20:
            base_leverage = 2.0  # תנודתיות גבוהה מאוד
        elif volatility > 0.15:
            base_leverage = 3.0  # תנודתיות גבוהה
        elif volatility > 0.10:
            base_leverage = 4.0  # תנודתיות בינונית
        elif volatility > 0.05:
            base_leverage = 5.0  # תנודתיות נמוכה
        else:
            base_leverage = 6.0  # תנודתיות נמוכה מאוד
            
        # התאמה לפי סוג נכס
        asset_multiplier = self.get_asset_risk_multiplier(symbol)
        
        # התאמה לפי מצב שוק
        regime_multiplier = 1.2 if regime == "bull" else 0.8
        
        raw_leverage = base_leverage * asset_multiplier * regime_multiplier
        
        # הגבלה לפי הגדרות המערכת
        max_allowed = self.adaptive_settings['max_leverage']
        return max(1.0, min(max_allowed, round(raw_leverage, 1)))
    
    def calculate_intelligent_stop_loss(self, symbol: str, entry_price: float, 
                                     leverage: float, volatility: float, regime: str) -> float:
        """מחשב stop loss אינטליגנטי based on multiple factors"""
        
        # בסיס based on תנודתיות
        if volatility > 0.20:
            base_sl = 0.05  # 5% for high volatility
        elif volatility > 0.15:
            base_sl = 0.04  # 4% for medium-high
        elif volatility > 0.10:
            base_sl = 0.03  # 3% for medium
        else:
            base_sl = 0.02  # 2% for low
            
        # התאמה למינוף - מינוף גבוה = stop tighter אבל לא יותר מדי
        leverage_adjustment = 1.0 / (leverage ** 0.35)
        dynamic_sl = base_sl * leverage_adjustment
        
        # התאמה למצב שוק
        if regime == "bear":
            dynamic_sl *= 1.3  # stop wider בשוק bearish
        elif regime == "high_volatility":
            dynamic_sl *= 1.2  # stop wider בשוק תנודתי
            
        # הגבלות בטיחות
        min_sl = 0.015  # 1.5% minimum
        max_sl = 0.10   # 10% maximum
        
        final_sl_percent = max(min_sl, min(max_sl, dynamic_sl))
        stop_loss_price = entry_price * (1 - final_sl_percent)
        
        print(f"🛡️  SL for {symbol}: {final_sl_percent*100:.1f}% ({stop_loss_price:.3f})")
        return stop_loss_price
    
    def calculate_optimal_take_profit(self, entry_price: float, stop_loss: float, 
                                    volatility: float, regime: str) -> Dict:
        """מחשב אסטרטגיית take profit אופטימלית"""
        
        risk_amount = entry_price - stop_loss
        base_rrr = 2.0  # base risk-reward ratio
        
        # התאמה based on תנודתיות
        if volatility > 0.15:
            rrr = base_rrr * 1.3  #更高的risk-reward לתנודתיות גבוהה
        elif volatility < 0.05:
            rrr = base_rrr * 0.8  #更低risk-reward לתנודתיות נמוכה
        else:
            rrr = base_rrr
            
        # התאמה based on מצב שוק
        if regime == "bull":
            rrr *= 1.2
        elif regime == "bear":
            rrr *= 0.8
            
        # חישוב מחירי TP
        tp1_price = entry_price + (risk_amount * rrr * 0.5)  # 50% ב-RRR בסיס
        tp2_price = entry_price + (risk_amount * rrr)        # 100% ב-RRR מלא
        tp3_price = entry_price + (risk_amount * rrr * 1.5)  # 150% ב-RRR
        
        tp_strategy = {
            'targets': [
                {'price': tp1_price, 'percent': 0.3},  # 30% מהפוזיציה
                {'price': tp2_price, 'percent': 0.4},  # 40% מהפוזיציה
                {'price': tp3_price, 'percent': 0.3}   # 30% מהפוזיציה
            ],
            'trailing_stop': {
                'activation': tp1_price,
                'trailing_percent': 0.03  # 3% trailing
            },
            'risk_reward_ratio': rrr
        }
        
        return tp_strategy
    
    def portfolio_optimization(self) -> List[str]:
        """מחזיר רשימת נכסים אופטימלית לפיזור"""
        
        # קטגוריות נכסים לפיזור
        asset_categories = {
            'large_caps': ['BTCUSDT', 'ETHUSDT', 'BNBUSDT'],
            'defi': ['AAVEUSDT', 'UNIUSDT', 'MKRUSDT'],
            'layer1': ['SOLUSDT', 'AVAXUSDT', 'DOTUSDT', 'ADAUSDT'],
            'gaming': ['SANDUSDT', 'MANAUSDT', 'AXSUSDT'],
            'meme': ['DOGEUSDT', 'SHIBUSDT', 'PEPEUSDT']
        }
        
        # בחירת נכסים based on מצב שוק וגודל חשבון
        selected_assets = []
        
        # בסיס - large caps
        selected_assets.extend(asset_categories['large_caps'][:1])
        
        # התאמה based on גודל חשבון
        if self.total_balance >= 5000:
            selected_assets.extend(asset_categories['defi'][:2])
            selected_assets.extend(asset_categories['layer1'][:2])
            selected_assets.extend(asset_categories['gaming'][:1])
        elif self.total_balance >= 1000:
            selected_assets.extend(asset_categories['defi'][:1])
            selected_assets.extend(asset_categories['layer1'][:2])
        else:
            selected_assets.extend(asset_categories['layer1'][:1])
            
        # הוספת נכסים based on strength
        strong_assets = self.identify_strong_assets()
        selected_assets.extend(strong_assets[:2])
        
        return list(set(selected_assets))[:self.adaptive_settings['max_positions']]
    
    def create_comprehensive_order(self, symbol: str, strategy: str = "auto") -> Optional[Dict]:
        """יוצר הזמנה מקיפה עם כל ניהול הסיכונים"""
        
        # בדיקות מקדימות
        if not self.pre_trade_checks(symbol):
            return None
            
        current_price = self.get_current_price(symbol)
        
        # ניהול סיכונים דינמי
        risk_management = self.dynamic_risk_management(symbol, current_price)
        
        # חישוב גודל פוזיציה
        position_size = self.total_balance * self.adaptive_settings['position_size_percent']
        leveraged_size = position_size * risk_management['leverage']
        quantity = leveraged_size / current_price
        
        # יצירת packages הזמנות
        order_package = {
            'main_order': {
                'symbol': symbol,
                'side': 'BUY',
                'type': 'LIMIT',
                'price': round(current_price * 0.995, 4),  # 0.5% below current
                'quantity': round(quantity, 4),
                'leverage': risk_management['leverage'],
                'timeInForce': 'GTC'
            },
            'stop_loss': {
                'type': 'STOP_MARKET',
                'price': round(risk_management['stop_loss'], 4),
                'reduce_only': True
            },
            'take_profit': self.create_tp_orders(symbol, risk_management['take_profit'], quantity),
            'risk_metrics': {
                'position_value': round(leveraged_size, 2),
                'base_value': round(position_size, 2),
                'risk_reward_ratio': risk_management['take_profit']['risk_reward_ratio'],
                'volatility': risk_management['volatility_score'],
                'position_score': risk_management['position_score']
            },
            'monitoring': {
                'check_interval': 300,  # 5 minutes
                'adjustment_threshold': 0.02  # 2% change triggers adjustment
            }
        }
        
        print(f"✅ הזמנה מקיפה נוצרה עבור {symbol}")
        print(f"   Lev: {risk_management['leverage']}x, Size: ${leveraged_size:.2f}")
        print(f"   SL: {risk_management['stop_loss']:.3f}")
        print(f"   R/R: {risk_management['take_profit']['risk_reward_ratio']:.1f}")
        
        return order_package
    
    def create_tp_orders(self, symbol: str, tp_strategy: Dict, quantity: float) -> List[Dict]:
        """יוצר multiple TP orders"""
        tp_orders = []
        
        for target in tp_strategy['targets']:
            tp_orders.append({
                'symbol': symbol,
                'side': 'SELL',
                'type': 'LIMIT',
                'price': round(target['price'], 4),
                'quantity': round(quantity * target['percent'], 4),
                'reduce_only': True,
                'timeInForce': 'GTC'
            })
            
        # trailing stop
        tp_orders.append({
            'symbol': symbol,
            'side': 'SELL',
            'type': 'TRAILING_STOP_MARKET',
            'activationPrice': round(tp_strategy['trailing_stop']['activation'], 4),
            'callbackRate': tp_strategy['trailing_stop']['trailing_percent'] * 100,
            'quantity': round(quantity * 0.3, 4),  # 30% for trailing
            'reduce_only': True
        })
        
        return tp_orders
    
    def pre_trade_checks(self, symbol: str) -> bool:
        """בדיקות מקדימות לפני כל trade"""
        
        # בדיקת מגבלת פוזיציות
        if len(self.portfolio) >= self.adaptive_settings['max_positions']:
            print(f"❌ מגבלה: {len(self.portfolio)}/{self.adaptive_settings['max_positions']} פוזיציות")
            return False
            
        # בדיקת חשיפה לנכס
        if symbol in self.portfolio:
            print(f"❌ כבר יש פוזיציה פתוחה ב-{symbol}")
            return False
            
        # בדיקת correlation עם נכסים קיימים
        if self.check_high_correlation(symbol):
            print(f"❌ {symbol} בעל correlation גבוה עם נכסים קיימים")
            return False
            
        # בדיקת מגבלת סיכון יומית
        if self.daily_risk_limit_reached():
            print("❌ הגעת למגבלת הסיכון היומית")
            return False
            
        return True
    
    def real_time_monitoring(self):
        """ניטור real-time והתאמות אוטומטיות"""
        while True:
            try:
                # עדכון נתונים
                self.update_portfolio_values()

                # בדיקת התאמות

                # עדכון הגדרות if needed
                self.adaptive_rebalancing()
                
                # דיווח סטטוס
                self.print_system_status()
                
                time.sleep(60)  # בדיקה כל דקה
                
            except Exception as e:
                print(f"❌ שגיאה בניטור: {e}")
                time.sleep(300)  # המתנה 5 דקות אם יש שגיאה
    
    def auto_adjust_positions(self):
        """התאמה אוטומטית של פוזיציות based on תנאי שוק"""
        for symbol, position in self.portfolio.items():
            current_price = self.get_current_price(symbol)
            
            # התאמת stop loss אם הרווחים גדולים
            if self.should_adjust_stop_loss(position, current_price):
                self.adjust_trailing_stop(symbol, position, current_price)
                
            # סגירה אוטומטית אם indicators השתנו
            if self.should_close_position(symbol, position, current_price):
                self.close_position(symbol, "Signal change")
    
    def adaptive_rebalancing(self):
        """איזון מחדש אוטומטי של הפורטפוליו"""
        total_value = self.calculate_portfolio_value()
        
        # אם יש שינוי גדול בערך - עדכן הגדרות
        if abs(total_value - self.total_balance) / self.total_balance > 0.1:
            print("🔄 שינוי משמעותי בערך - מבצע adaptive rebalancing")
            self.initialize_system(total_value)
    
    def should_adjust_stop_loss(self, position: Dict, current_price: float) -> bool:
        """בודק אם צריך להתאים stop loss"""
        unrealized_pnl = (current_price - position['entry_price']) / position['entry_price']
        return unrealized_pnl > 0.08  # אם ברווח של 8%+
    
    def adjust_trailing_stop(self, symbol: str, position: Dict, current_price: float):
        """מתאים stop loss ל-break even או trailing"""
        new_sl = current_price * 0.98  # 2% below current
        if new_sl > position['stop_loss']:
            self.update_stop_loss(symbol, new_sl)
            print(f"📈 Adjusted SL for {symbol} to {new_sl:.3f}")
