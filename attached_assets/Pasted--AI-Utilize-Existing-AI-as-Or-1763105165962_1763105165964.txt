```python
async def coordinate_modules(self, ai_decisions):
    """ה-AI מתאם בין כל חלקי המערכת"""
    
    # AI מחליט על אסטרטגיית Orders
    if ai_decisions['order_strategy'] == 'AGGRESSIVE_ENTRY':
        await self.system_modules['order_management'].set_aggressive_mode()
        await self.system_modules['risk_control'].adjust_risk_limits(1.3)
    
    elif ai_decisions['order_strategy'] == 'DEFENSIVE_EXIT':
        await self.system_modules['order_management'].set_defensive_mode()
        await self.system_modules['risk_control'].adjust_risk_limits(0.7)
    
    # AI מתאם בין Scanner ל-Trading
    scanner_signals = await self.system_modules['market_analysis'].get_scanner_results()
    trading_decisions = await self.ai_brain.convert_signals_to_trades(scanner_signals)
    
    # AI שולח פקודות לביצוע
    await self.execute_ai_commands(trading_decisions)
```

```python
class AIStateManager:
    def __init__(self):
        self.system_state = {}
        self.performance_metrics = {}
        self.learning_data = []
    
    async def ai_controlled_state_management(self):
        """ה-AI מנהל את כל ה-state של המערכת"""
        
        # AI עוקב אחרי כל ה-orders
        orders_state = await self.track_all_orders()
        self.system_state['orders'] = orders_state
        
        # AI עוקב אחרי performance
        performance = await self.calculate_performance_metrics()
        self.performance_metrics = performance
        
        # AI לומד ומשפר את המערכת
        await self.ai_learning_loop()
    
    async def track_all_orders(self):
        """AI עוקב אחרי כל ה-orders במערכת"""
        orders = await self.get_all_active_orders()
        
        ai_analysis = {}
        for order in orders:
            # AI מנתח כל order בנפרד
            order_health = await self.ai_brain.analyze_order_health(order)
            ai_analysis[order['id']] = {
                'health_score': order_health['score'],
                'recommended_action': order_health['action'],
                'risk_level': order_health['risk']
            }
        
        return ai_analysis
```

### **🔄 AI Auto-Recovery:**

```python
class AIAutoRecovery:
    async def ai_driven_error_recovery(self):
        """AI מתקן שגיאות אוטומטית"""
        while True:
            try:
                # AI סורק בעיות
                system_issues = await self.detect_system_issues()
                
                for issue in system_issues:
                    # AI מחליט איך לתקן
                    recovery_plan = await self.ai_brain.create_recovery_plan(issue)
                    
                    # AI מבצע את התיקון
                    await self.execute_recovery(recovery_plan)
                
                await asyncio.sleep(15)  # סריקה כל 15 שניות
                
            except Exception as e:
                logger.error(f"❌ AI Recovery failed: {e}")
                await asyncio.sleep(60)
    
    async def detect_system_issues(self):
        """AI מזהה בעיות במערכת"""
        issues = []
        
        # AI בודק orders תקועים
        stuck_orders = await self.find_stuck_orders()
        if stuck_orders:
            issues.append({'type': 'STUCK_ORDERS', 'data': stuck_orders})
        
        # AI בודק performance degradation
        performance_issues = await self.find_performance_issues()
        if performance_issues:
            issues.append({'type': 'PERFORMANCE', 'data': performance_issues})
        
        # AI בודק connection problems
        connection_issues = await self.find_connection_issues()
        if connection_issues:
            issues.append({'type': 'CONNECTION', 'data': connection_issues})
        
        return issues
```

```python
class AIDecisionEngine:

        """מנוע החלטות AI - המוח של המערכת"""
        
        # AI אוסף נתונים מכל המקורות
        market_data = await self.get_market_data()
        system_health = await self.get_system_health()
        performance_data = await self.get_performance_data()
        risk_metrics = await self.get_risk_metrics()
        
        # AI מקבל החלטות
        decisions = await self.ai_brain.process_decisions({
            'market': market_data,
            'system': system_health, 
            'performance': performance_data,
            'risk': risk_metrics
        })
        
        return {
            'trading_strategy': decisions['trading'],
            'risk_adjustments': decisions['risk'],
            'system_optimizations': decisions['system'],
            'recovery_actions': decisions['recovery']
        }
```
