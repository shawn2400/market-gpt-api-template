# 🚀 קובץ יחיד מלא — `ultimate-bootstrap.sh`

**העתק-הדבק — זה כולל EVERYTHING:**

```bash
#!/bin/bash
set -e

echo "✅ STARTING FULL AUTO-PILOT INSTALL (AlgoGPT ULTRA+)"

### --- 0) SYSTEM PREP ---
apt-get update -y
apt-get install -y curl wget git ufw jq ca-certificates

### --- 1) SECURITY & FIREWALL ---
ufw allow ssh
ufw allow 10000    # AlgoGPT API
ufw allow 8080     # Dashboard
ufw allow 3000     # Grafana
ufw allow 9090     # Prometheus
ufw allow 11434    # OLLaMA
ufw --force enable

### --- 2) INSTALL DOCKER + COMPOSE ---
if ! command -v docker >/dev/null; then
  curl -fsSL https://get.docker.com | sh
  systemctl enable docker
fi
curl -L "https://github.com/docker/compose/releases/download/2.27.0/docker-compose-linux-x86_64" \
  -o /usr/local/bin/docker-compose
chmod +x /usr/local/bin/docker-compose

### --- 3) DEPLOY DIRECTORY ---
mkdir -p /opt/algogpt
cd /opt/algogpt

### --- 4) CREATE docker-compose.yml ---
cat > docker-compose.yml << 'EOF'
version: "3.9"

services:
  algogpt:
    image: algogpt/core:latest
    restart: always
    environment:
      AUTO_RUN: "1"
      EXECUTE_TRADES: "0"            # SAFE by default
      MIN_QUALITY_SCORE: "8.5"
      AUTO_RECOVERY: "on"
      AUTO_EXPAND: "on"
      AUTO_SCALE: "on"
      SELF_HEALING: "on"
      CIRCUIT_BREAKER: "on"
      KILL_SWITCH: "on"
      HA_CLUSTER: "standby"
      MULTI_ACCOUNT: "ready"
      PROFIT_SHARE: "enabled"
      AUTO_HEDGE: "on"
      SENTIMENT_AI: "on"
      NEWS_FILTER: "on"
      FUNDING_FILTER: "on"
      TZ: "Asia/Jerusalem"
    ports:
      - "10000:10000"
    volumes:
      - ./data:/app/data
      - ./backups:/app/backups

  ollama:
    image: ollama/ollama:latest
    restart: always
    ports:
      - "11434:11434"
    volumes:
      - ./ollama:/root/.ollama

  dashboard:
    image: algogpt/dashboard:latest
    restart: always
    environment:
      API_URL: "http://algogpt:10000"
      OLLAMA_URL: "http://ollama:11434"
      PROMETHEUS_URL: "http://prometheus:9090"
    ports:
      - "8080:8080"

  grafana:
    image: grafana/grafana:latest
    restart: always
    ports:
      - "3000:3000"
    volumes:
      - ./grafana:/var/lib/grafana

  prometheus:
    image: prom/prometheus:latest
    restart: always
    ports:
      - "9090:9090"
    volumes:
      - ./prometheus.yml:/etc/prometheus/prometheus.yml

volumes:
  data:
EOF

### --- 5) CREATE PROMETHEUS CONFIG ---
cat > prometheus.yml << 'EOF'
global:
  scrape_interval: 5s

scrape_configs:
  - job_name: 'algogpt'
    static_configs:
      - targets: ['algogpt:10000']
EOF

### --- 6) START STACK ---
docker-compose pull
docker-compose up -d

### --- 7) LOAD OLLaMA MODEL ---
sleep 4
curl -s -X POST http://localhost:11434/api/pull -d '{"name":"llama3"}'

### --- 8) ENABLE AUTO-BACKUPS ---
mkdir -p /opt/algogpt/backups
cat > /etc/cron.daily/algogpt-backup << 'EOF'
#!/bin/bash
tar -czf /opt/algogpt/backups/backup-$(date +%F).tar.gz /opt/algogpt/data
EOF
chmod +x /etc/cron.daily/algogpt-backup

echo ""
echo "✅ INSTALL COMPLETE"
echo "🌍 API:         http://<SERVER-IP>:10000/readyz"
echo "📊 Dashboard:   http://<SERVER-IP>:8080"
echo "📈 Grafana:     http://<SERVER-IP>:3000"
echo "🧠 OLLaMA:      http://<SERVER-IP>:11434"
echo "🔥 FULL AUTO-PILOT READY"
```
