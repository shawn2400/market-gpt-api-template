## ✅ `/routes/auto_pilot.py`

```python
from fastapi import APIRouter
from agent_core.state import SYSTEM_STATE

router = APIRouter()

@router.post("/enable")
def enable_auto():
    SYSTEM_STATE["auto_pilot"] = True
    return {"status": "AUTO_PILOT_ON"}

@router.post("/disable")
def disable_auto():
    SYSTEM_STATE["auto_pilot"] = False
    return {"status": "AUTO_PILOT_OFF"}

@router.get("/status")
def status():
    return SYSTEM_STATE
```

## ✅ `/routes/accounts.py`

```python
from fastapi import APIRouter

router = APIRouter()

ACCOUNTS = {}

@router.post("/add")
def add_account(user_id: str, api_key: str, secret: str):
    ACCOUNTS[user_id] = {"key": api_key, "secret": secret}
    return {"status": "ACCOUNT_ADDED"}

@router.post("/suspend")
def suspend(user_id: str):
    ACCOUNTS[user_id]["suspended"] = True
    return {"status": "SUSPENDED"}
```
