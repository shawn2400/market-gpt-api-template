class AdvancedRiskManager:
    def __init__(self):
        self.min_hold_time = 60  # 60 שניות minimum
        self.atr_multiplier = 2.0  # 2x ATR
        
    def calculate_dynamic_sl(self, symbol, entry_price, atr):
        """חשב SL דינמי based on ATR"""
        sl_distance = atr * self.atr_multiplier
        
        # התאם multiplier לפי volatility
        if self.get_volatility(symbol) > 0.08:
            sl_distance = atr * 2.5  # SL רחב יותר לתנודתי
        else:
            sl_distance = atr * 1.5  # SL צמוד ליציב
            
        return entry_price - sl_distance  # עבור long position
    
    def should_activate_sl(self, position):
        """בדוק אם לאפשר SL - הגנת זמן"""
        position_age = time.time() - position.entry_time
        return position_age >= self.min_hold_time
```

def smart_sl_placement(symbol, entry_price, direction):
    """שים SL מעבר ל-support/resistance zones"""
    recent_lows = get_recent_lows(symbol, periods=20)
    recent_highs = get_recent_highs(symbol, periods=20)
    
    if direction == "LONG":
        # תמצא support level הכי קרוב
        support_level = find_nearest_support(entry_price, recent_lows)
        sl_price = support_level * 0.995  # 0.5% מתחת ל-support
    else:  # SHORT
        resistance_level = find_nearest_resistance(entry_price, recent_highs)
        sl_price = resistance_level * 1.005  # 0.5% מעל resistance
        
    return sl_price
```
