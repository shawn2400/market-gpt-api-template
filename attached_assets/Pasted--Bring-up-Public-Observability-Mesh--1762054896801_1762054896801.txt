python - <<'PY'
from __future__ import annotations
from pathlib import Path
p=Path("utils/mesh_bus.py")
if not p.exists():
  p.parent.mkdir(parents=True, exist_ok=True)
  p.write_text(r'''# -*- coding: utf-8 -*-
from __future__ import annotations
import os, json, time, hmac, hashlib
from typing import Dict, Any

AI_MESH_SECRET = os.getenv("AI_MESH_SECRET","")
NODE_ID = os.getenv("NODE_ID","core-api")

_STATE: Dict[str, Any] = {
  "node": {"id": NODE_ID, "ts": int(time.time())},
  "peers": {},  # url -> meta
  "ok": True,
  "last_ping_ts": 0,
}

def _sig(msg: str) -> str:
    if not AI_MESH_SECRET: return ""
    return hmac.new(AI_MESH_SECRET.encode(), msg.encode(), hashlib.sha256).hexdigest()

def snapshot() -> Dict[str,Any]:
    # פינג אמיתי בלולאה נוכל להוסיף בהמשך; כרגע משמר מצב בזיכרון
    return {"node": _STATE["node"], "peers": dict(_STATE["peers"]), "ok": _STATE["ok"], "last_ping_ts": _STATE["last_ping_ts"]}

def ensure_peer(peer: Dict[str,Any]) -> None:
    url=(peer.get("url") or "").rstrip("/")
    if not url: return
    rec=_STATE["peers"].get(url,{})
    rec.update({
      "id": peer.get("id"),
      "role": peer.get("role"),
      "scopes": peer.get("scopes") or [],
      "weight": int(peer.get("weight",1)),
      "enabled": bool(peer.get("enabled", True)),
      "ok": True,
      "last_ts": int(time.time()),
      "latency_ms": None,
      "version": None
    })
    _STATE["peers"][url]=rec
''', encoding="utf-8")
  print("✅ created utils/mesh_bus.py")
else:
  print("ℹ️ utils/mesh_bus.py exists")
PY

from __future__ import annotations
from pathlib import Path
rp=Path("routes/public_endpoints.py")
if not rp.exists():
  rp.parent.mkdir(parents=True, exist_ok=True)
  rp.write_text(r'''from __future__ import annotations
import os, time, hmac, hashlib
from typing import Any, Dict
from fastapi import APIRouter, HTTPException, Request, Query, Header
from contextlib import suppress

router = APIRouter(tags=["Public"])

# ---- /status/public ----
@router.get("/status/public")
def status_public():
    app = {"name": os.getenv("APP_TITLE","AlgoGPT Service"),
           "version": os.getenv("APP_VERSION",""),
           "ts": int(time.time())}
    sup = {"ok": False, "interval_sec": None}
    with suppress(Exception):
        from workers import ai_supervisor as w  # type: ignore
        sup["ok"]=True; sup["interval_sec"]=getattr(w,"INTERVAL_SEC",None)
    mesh_ok=True
    with suppress(Exception):
        from utils import mesh_bus as mb  # type: ignore
        mesh_ok = bool(mb.snapshot().get("ok", True))
    return {"ok": True, "app": app, "supervisors": sup, "mesh_ok": mesh_ok, "ts": int(time.time())}

# ---- /mesh/public | /mesh/register | /mesh/peers ----
def _sig(msg: str) -> str:
    sec=os.getenv("AI_MESH_SECRET","")
    return hmac.new(sec.encode(), msg.encode(), hashlib.sha256).hexdigest() if sec else ""

@router.get("/mesh/public")
def mesh_public() -> Dict[str,Any]:
    try:
        from utils.mesh_bus import snapshot  # type: ignore
        s = snapshot()
    except Exception:
        s = {"node":{"id": os.getenv("NODE_ID","core-api")}, "peers":{}, "ok": True}
    app = {"version": os.getenv("APP_VERSION","")}
    # סינון ציבורי
    pub={}
    for url, meta in (s.get("peers") or {}).items():
        pub[url]={k: meta.get(k) for k in ("id","role","scopes","weight","enabled","ok","latency_ms","last_ts","version")}
    return {"ok": True, "node": s.get("node"), "peers": pub, "app": app, "ts": int(time.time())}

@router.post("/mesh/register")
async def mesh_register(req: Request) -> Dict[str,Any]:
    body = await req.json()
    need = ["id","url","role","scopes","ts","sig"]
    if any(k not in body for k in need):
        raise HTTPException(400, "bad_payload")
    expect = _sig(f"{body['id']}:{body['ts']}:{body['url']}")
    if not expect or expect != body["sig"]:
        raise HTTPException(403, "bad_sig")
    try:
        from utils.mesh_bus import ensure_peer  # type: ignore
        ensure_peer({
            "id": body["id"],
            "url": body["url"].rstrip("/"),
            "role": body["role"],
            "scopes": body.get("scopes", []),
            "weight": int(body.get("weight",1)),
            "enabled": bool(body.get("enabled", True))
        })
        return {"ok": True, "registered": body["id"]}
    except Exception as e:
        raise HTTPException(500, f"mesh_error: {e}")

@router.get("/mesh/peers")
def mesh_peers():
    try:
        from utils.mesh_bus import snapshot  # type: ignore
        s=snapshot()
        return {"ok": True, "peers": list((s.get("peers") or {}).keys())}
    except Exception:
        return {"ok": True, "peers": []}

# ---- Placeholders בטוחים לקריאה-בלבד (מנוקים מסודות) ----
@router.get("/git/public")
def git_public():
    with suppress(Exception):
        from utils.gh_status import get_status  # type: ignore
        import anyio
        return anyio.run(get_status, False)
    return {"ok": False, "reason": "gh_status_not_available"}

@router.post("/git/refresh")
def git_refresh(authorization: str = Header(default="")):
    token=os.getenv("API_BEARER_TOKEN_RO","").strip()
    if token and authorization != f"Bearer {token}":
        raise HTTPException(403, "bad_bearer")
    with suppress(Exception):
        from utils.gh_status import get_status  # type: ignore
        import anyio
        return anyio.run(get_status, True)
    return {"ok": False, "reason": "gh_status_not_available"}

@router.get("/render/public")
def render_public():
    base = os.getenv("RENDER_EXTERNAL_URL") or os.getenv("PUBLIC_HOST") or "http://localhost:8000"
    out = {"ok": True, "external_url": base, "ts": int(time.time())}
    return out

@router.get("/trades/public")
def trades_public(limit: int = Query(50, ge=1, le=200)):
    return {"ok": True, "items": []}

@router.get("/pnl/public")
def pnl_public():
    return {"ok": True, "pnl": {}}

@router.get("/logs/tail/public")
def logs_tail_public(name: str = Query("main"), lines: int = Query(200, ge=1, le=2000)):
    # לא קוראים קבצים כרגע — רק placeholder בטוח
    return {"ok": True, "name": name, "lines": 0, "content": []}
''', encoding="utf-8")
  print("✅ created routes/public_endpoints.py")
else:
  print("ℹ️ routes/public_endpoints.py exists")
