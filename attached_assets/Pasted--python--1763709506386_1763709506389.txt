    def update_market_analysis(self):
        """ניתוח שוק מתקדם עם multiple indicators"""
        try:
            # בדוק מספר זוגות לווידוא מצב שוק כללי

            volatility_scores = []

                data = self.get_multi_timeframe_data(symbol)
                volatility_scores.append(self.calculate_volatility_score(data))
                trend_scores.append(self.calculate_trend_strength(data))
            
            avg_volatility = np.mean(volatility_scores)
            avg_trend = np.mean(trend_scores)
            
            # קבע מצב שוק
            if avg_volatility > 0.04:

            elif avg_volatility < 0.012:
                self.market_regime = "LOW_VOL" 
                self.adaptive_risk = 0.015
                self.volatility_regime = "LOW"
            else:
                self.market_regime = "NORMAL"
                self.adaptive_risk = 0.011
                self.volatility_regime = "MEDIUM"
            
            self.logger.info(f"📊 ניתוח שוק: {self.market_regime} | סיכון: {self.adaptive_risk:.3f} | טרנד: {avg_trend:.2f}")
            
        except Exception as e:
            self.logger.error(f"❌ שגיאה בניתוח שוק: {e}")
    
    def get_multi_timeframe_data(self, symbol: str) -> Dict:
        """קבל נתונים מ-multiple timeframes"""
        timeframes = ['5m', '15m', '1h', '4h']
        data = {}
        
        for tf in timeframes:
            try:
                klines = self.client.get_klines(
                    symbol=symbol, 
                    interval=tf, 
                    limit=100
                )
                
                df = pd.DataFrame(klines, columns=[
                    'timestamp', 'open', 'high', 'low', 'close', 'volume',
                    'close_time', 'quote_volume', 'trades', 'taker_buy', 'taker_quote', 'ignore'
                ])
                
                for col in ['open', 'high', 'low', 'close', 'volume']:
                    df[col] = pd.to_numeric(df[col])
                
                # הוסף אינדיקטורים
                df['ema_20'] = ta.trend.EMAIndicator(df['close'], window=20).ema_indicator()
                df['ema_50'] = ta.trend.EMAIndicator(df['close'], window=50).ema_indicator()
                df['rsi'] = ta.momentum.RSIIndicator(df['close'], window=14).rsi()
                df['atr'] = ta.volatility.AverageTrueRange(df['high'], df['low'], df['close']).average_true_range()
                df['volume_ma'] = df['volume'].rolling(20).mean()
                
                data[tf] = df.iloc[-1]
                
            except Exception as e:
                self.logger.warning(f"⚠️  שגיאה ב-timeframe {tf} עבור {symbol}: {e}")
                continue
        
        return data
    
    def calculate_volatility_score(self, data: Dict) -> float:
        """חשב ציון וולאטיליות משוקלל"""
        volatility_scores = []
        
        for tf, tf_data in data.items():
            if tf_data is not None:
                atr_percent = tf_data['atr'] / tf_data['close']
                volatility_scores.append(atr_percent)
        
        return np.mean(volatility_scores) if volatility_scores else 0.02
    
    def calculate_trend_strength(self, data: Dict) -> float:
        """חשב חוזק טרנד משוקלל"""
        trend_scores = []
        
        for tf, tf_data in data.items():
            if tf_data is not None:
                # ציון based on EMA alignment ומרחק
                if tf_data['close'] > tf_data['ema_20'] > tf_data['ema_50']:
                    score = 1.0
                elif tf_data['close'] < tf_data['ema_20'] < tf_data['ema_50']:
                    score = -1.0
                else:
                    score = 0.0
                
                trend_scores.append(score)
        
        return np.mean(trend_scores) if trend_scores else 0.0
    
    def calculate_dynamic_position_size(self, symbol: str, entry_price: float) -> float:
        """חשב גודל פוזיציה דינאמי based on וולאטיליות וסיכון"""
        try:
            data = self.get_multi_timeframe_data(symbol)
            current_volatility = self.calculate_volatility_score(data)
            
            # התאם סיכון based on וולאטיליות
            if current_volatility > 0.035:
                risk_multiplier = 0.6  # הפחת סיכון בוולאטיליות גבוהה
            elif current_volatility < 0.015:
                risk_multiplier = 1.4  # הגדל סיכון בוולאטיליות נמוכה
            else:
                risk_multiplier = 1.0
            
            risk_amount = self.equity * self.adaptive_risk * risk_multiplier
            position_size = risk_amount / entry_price
            
            # הגבל למקסימום המותר
            max_size = self.equity * self.max_position_size_percent / entry_price
            
            final_size = min(position_size, max_size)
            final_size = round(final_size, self.get_symbol_precision(symbol))
            
            self.logger.info(f"📏 גודל פוזיציה עבור {symbol}: {final_size} | סיכון: {risk_multiplier:.2f}x")
            
            return final_size
            
        except Exception as e:
            self.logger.error(f"❌ שגיאה בחישוב פוזיציה: {e}")
            # ברירת מחדל שמרנית
            return (self.equity * 0.005) / entry_price
    
    def get_symbol_precision(self, symbol: str) -> int:
        """קבל דיוק כמות עבור סימבול"""
        try:
            info = self.client.get_symbol_info(symbol)
            step_size = next(
                float(filt['stepSize']) 
                for filt in info['filters'] 
                if filt['filterType'] == 'LOT_SIZE'
            )
            return abs(int(np.log10(step_size)))
        except:
            return 2  # ברירת מחדל
    
    def generate_intelligent_signal(self, symbol: str) -> Optional[Dict]:
        """הפק אות מסחר אינטליגנטי עם ניתוח רב-ממדי"""
        if self.daily_trades_count >= self.max_daily_trades:
            self.logger.info("🎯 הגעת למקסימום מסחר יומי")
            return None
        
        try:
            # קבל נתונים מפורטים
            data = self.get_multi_timeframe_data(symbol)
            current_price = self.get_current_price(symbol)
            
            if not data:
                return None
            
            # ניתוח טכני מתקדם
            momentum_analysis = self.momentum_analysis(data)
            volume_analysis = self.volume_analysis(data) 
            trend_analysis = self.trend_analysis(data)
            pattern_analysis = self.pattern_analysis(data)
            
            # ציון ביטחון משוקלל
            confidence_score = (
                momentum_analysis['score'] * 0.35 +
                volume_analysis['score'] * 0.25 +
                trend_analysis['score'] * 0.25 +
                pattern_analysis['score'] * 0.15
            )
            
            # בדוק אם עובר סף מינימום
            if confidence_score < self.min_confidence:
                return None
            
            # חישוב SL/TP דינאמי
            sl_tp_levels = self.calculate_adaptive_sl_tp(symbol, current_price, data)
            
            # בדוק Risk/Reward
            if sl_tp_levels['risk_reward_ratio'] < 1.8:
                self.logger.info(f"⚠️  R/R נמוך עבור {symbol}: {sl_tp_levels['risk_reward_ratio']:.2f}")
                return None
            
            # חישוב גודל פוזיציה
            position_size = self.calculate_dynamic_position_size(symbol, current_price)
            
            signal = {
                'symbol': symbol,
                'action': 'BUY',
                'entry_price': current_price,
                'stop_loss': sl_tp_levels['stop_loss'],
                'take_profit': sl_tp_levels['take_profit'],
                'position_size': position_size,
                'timestamp': datetime.now(),
                'confidence': confidence_score,
                'risk_reward': sl_tp_levels['risk_reward_ratio'],
                'market_regime': self.market_regime,
                'analysis': {
                    'momentum': momentum_analysis,
                    'volume': volume_analysis,
                    'trend': trend_analysis,
                    'pattern': pattern_analysis
                }
            }
            
            self.logger.info(f"🎯 אות איכותי: {symbol} | ביטחון: {confidence_score:.2f} | R/R: {sl_tp_levels['risk_reward_ratio']:.2f}")
            
            return signal
            
        except Exception as e:
            self.logger.error(f"❌ שגיאה בהפקת אות עבור {symbol}: {e}")
            return None
    
    def momentum_analysis(self, data: Dict) -> Dict:
        """ניתוח מומנטום מתקדם"""
        scores = []
        details = []
        
        for tf, tf_data in data.items():
            if tf_data is not None:
                # RSI analysis
                rsi_score = 1.0 if 40 <= tf_data['rsi'] <= 65 else 0.3
                
                # Price vs EMA momentum
                ema_score = 1.0 if tf_data['close'] > tf_data['ema_20'] else 0.4
                
                # Combined momentum score for this timeframe
                tf_momentum = (rsi_score + ema_score) / 2
                scores.append(tf_momentum)
                
                details.append({
                    'timeframe': tf,
                    'rsi': tf_data['rsi'],
                    'price_vs_ema20': tf_data['close'] - tf_data['ema_20'],
                    'score': tf_momentum
                })
        
        avg_score = np.mean(scores) if scores else 0.0
        
        return {
            'score': avg_score,
            'details': details
        }
    
    def volume_analysis(self, data: Dict) -> Dict:
        """ניתוח ווליום מתקדם"""
        scores = []
        details = []
        
        for tf, tf_data in data.items():
            if tf_data is not None:
                volume_ratio = tf_data['volume'] / tf_data['volume_ma']
                
                if volume_ratio > 2.0:
                    volume_score = 1.0
                elif volume_ratio > 1.5:
                    volume_score = 0.8
                elif volume_ratio > 1.2:
                    volume_score = 0.6
                else:
                    volume_score = 0.3
                
                scores.append(volume_score)
                details.append({
                    'timeframe': tf,
                    'volume_ratio': volume_ratio,
                    'score': volume_score
                })
        
        avg_score = np.mean(scores) if scores else 0.0
        
        return {
            'score': avg_score,
            'details': details
        }
    
    def trend_analysis(self, data: Dict) -> Dict:
        """ניתוח טרנד רב-ממדי"""
        scores = []
        details = []
        
        for tf, tf_data in data.items():
            if tf_data is not None:
                # Trend alignment
                if tf_data['close'] > tf_data['ema_20'] > tf_data['ema_50']:
                    trend_score = 1.0
                elif tf_data['close'] < tf_data['ema_20'] < tf_data['ema_50']:
                    trend_score = 0.0
                else:
                    trend_score = 0.5
                
                scores.append(trend_score)
                details.append({
                    'timeframe': tf,
                    'trend_alignment': trend_score,
                    'ema20_vs_ema50': tf_data['ema_20'] - tf_data['ema_50']
                })
        
        # שקלל יותר timeframes גבוהים
        weights = {'5m': 0.1, '15m': 0.2, '1h': 0.3, '4h': 0.4}
        weighted_score = 0.0
        
        for detail in details:
            tf = detail['timeframe']
            if tf in weights:
                weighted_score += detail['trend_alignment'] * weights[tf]
        
        return {
            'score': weighted_score,
            'details': details
        }
    
    def pattern_analysis(self, data: Dict) -> Dict:
        """ניתוח תבניות מסחר בסיסי"""
        # כאן ניתן להוסיף ניתוח תבניות מתקדם
        # כרגע מחזיר ציון based on יציבות
        volatility_scores = []
        
        for tf, tf_data in data.items():
            if tf_data is not None:
                atr_percent = tf_data['atr'] / tf_data['close']
                # ציון גבוה יותר לוולאטיליות נמוכה (יציבות)
                volatility_score = 1.0 - min(atr_percent * 10, 0.8)
                volatility_scores.append(volatility_score)
        
        avg_score = np.mean(volatility_scores) if volatility_scores else 0.5
        
        return {
            'score': avg_score,
            'details': [{'volatility_score': avg_score}]
        }
    
    def calculate_adaptive_sl_tp(self, symbol: str, entry_price: float, data: Dict) -> Dict:
        """חשב SL/TP אדפטיבי based on תנאי שוק"""
        try:
            # השתמש ב-15m data לחישוב SL/TP
            tf_data = data.get('15m')
            if tf_data is None:
                tf_data = next(iter(data.values()))
            
            atr = tf_data['atr']
            current_volatility = atr / entry_price
            
            # חישוב SL based on ATR ווולאטיליות
            if current_volatility > 0.03:
                sl_atr_multiplier = 1.5  # SL tighter בוולאטיליות גבוהה
                tp_atr_multiplier = 2.5   # TP יותר צנוע
            elif current_volatility < 0.01:
                sl_atr_multiplier = 2.0   # SL יותר מרווח בוולאטיליות נמוכה
                tp_atr_multiplier = 4.0   # TP יותר אגרסיבי
            else:
                sl_atr_multiplier = 1.8
                tp_atr_multiplier = 3.0
            
            stop_loss = entry_price - (atr * sl_atr_multiplier)
            take_profit = entry_price + (atr * tp_atr_multiplier)
            
            # הגבלות הגיוניות
            max_sl_percent = 0.06  # 6% מקסימום stop loss
            min_tp_percent = 0.025 # 2.5% מינימום take profit
            
            stop_loss = max(stop_loss, entry_price * (1 - max_sl_percent))
            take_profit = max(take_profit, entry_price * (1 + min_tp_percent))
            
            risk_reward = (take_profit - entry_price) / (entry_price - stop_loss)
            
            return {
                'stop_loss': round(stop_loss, 6),
                'take_profit': round(take_profit, 6),
                'risk_reward_ratio': risk_reward,
                'atr_multipliers': {'sl': sl_atr_multiplier, 'tp': tp_atr_multiplier}
            }
            
        except Exception as e:
            self.logger.error(f"❒ שגיאה בחישוב SL/TP: {e}")
            # ערכי ברירת מחדל
            return {
                'stop_loss': round(entry_price * 0.97, 6),
                'take_profit': round(entry_price * 1.045, 6),
                'risk_reward_ratio': 2.0,
                'atr_multipliers': {'sl': 2.0, 'tp': 3.0}
            }
    
    def get_current_price(self, symbol: str) -> float:
        """קבל מחיר נוכחי"""
        try:
            ticker = self.client.get_symbol_ticker(symbol=symbol)
            return float(ticker['price'])
        except Exception as e:
            self.logger.error(f"❌ שגיאה בקבלת מחיר עבור {symbol}: {e}")
            return 0.0
    
    def execute_trade_with_management(self, signal: Dict) -> bool:
        """בצע מסחר עם ניהול פוזיציה מלא"""
        try:
            symbol = signal['symbol']
            quantity = signal['position_size']
            
            # בדוק אם יש מספיק יתרה
            required_amount = quantity * signal['entry_price']
            if required_amount > self.equity * 0.9:
                self.logger.warning(f"⚠️  אין מספיק יתרה עבור {symbol}")
                return False
            
            # בצע קניה
            buy_order = self.client.order_market_buy(
                symbol=symbol,
                quantity=quantity
            )
            
            # רשום STOP LOSS
            sl_order = self.client.create_order(
                symbol=symbol,
                side=Client.SIDE_SELL,
                type=Client.ORDER_TYPE_STOP_LOSS_LIMIT,
                timeInForce='GTC',
                quantity=quantity,
                stopPrice=round(signal['stop_loss'], 6),
                price=round(signal['stop_loss'], 6)
            )
            
            # רשום TAKE PROFIT
            tp_order = self.client.create_order(
                symbol=symbol,
                side=Client.SIDE_SELL,
                type=Client.ORDER_TYPE_TAKE_PROFIT_LIMIT,
                timeInForce='GTC',
                quantity=quantity,
                stopPrice=round(signal['take_profit'], 6),
                price=round(signal['take_profit'], 6)
            )
            
            # שמור מידע על הפוזיציה
            self.active_positions[symbol] = {
                'buy_order': buy_order,
                'sl_order': sl_order,
                'tp_order': tp_order,
                'entry_price': signal['entry_price'],
                'quantity': quantity,
                'stop_loss': signal['stop_loss'],
                'take_profit': signal['take_profit'],
                'timestamp': datetime.now()
            }
            
            self.daily_trades_count += 1
            self.total_trades += 1
            
            self.logger.info(f"""
✅ מסחר בוצע בהצלחה:
   🎯 {symbol}
   💰 כמות: {quantity}
   ⬆️ Entry: {signal['entry_price']}
   ⬇️ SL: {signal['stop_loss']}
   ⬆️ TP: {signal['take_profit']}
   📊 R/R: {signal['risk_reward']:.2f}
   🎯 ביטחון: {signal['confidence']:.2f}
            """)
            
            return True
            
        except BinanceAPIException as e:
            self.logger.error(f"❌ שגיאת Binance: {e.code} - {e.message}")
            return False
        except Exception as e:
            self.logger.error(f"❌ שגיאה כללית בביצוע מסחר: {e}")
            return False
    
    def start_position_monitoring(self):
        """התחל ניטור פוזיציות אוטומטי"""
        def monitor_positions():
            while True:
                try:
                    self.monitor_and_manage_positions()
                    time.sleep(30)  # בדוק כל 30 שניות
                except Exception as e:
                    self.logger.error(f"❌ שגיאה בניטור פוזיציות: {e}")
                    time.sleep(60)
        
        self.position_monitor_thread = threading.Thread(target=monitor_positions, daemon=True)
        self.position_monitor_thread.start()
        self.logger.info("🔍 ניטור פוזיציות הופעל")
    
    def monitor_and_manage_positions(self):
        """ניטור וניהול פוזיציות פעילות"""
        try:
            for symbol, position in list(self.active_positions.items()):
                current_price = self.get_current_price(symbol)
                
                if current_price == 0:
                    continue
                
                entry_price = position['entry_price']
                current_profit_pct = (current_price - entry_price) / entry_price * 100
                
                # Trail Stop Logic - הזז SL אם ברווח טוב
                if current_profit_pct > 2.0:  # אם ברווח של 2%+
                    new_sl = current_price * 0.995  # SL של 0.5% מתחת למחיר נוכחי
                    
                    if new_sl > position['stop_loss']:
                        # עדכן stop loss
                        try:
                            # בטל SL ישן
                            self.client.cancel_order(
                                symbol=symbol,
                                orderId=position['sl_order']['orderId']
                            )
                            
                            # צור SL חדש
                            new_sl_order = self.client.create_order(
                                symbol=symbol,
                                side=Client.SIDE_SELL,
                                type=Client.ORDER_TYPE_STOP_LOSS_LIMIT,
                                timeInForce='GTC',
                                quantity=position['quantity'],
                                stopPrice=round(new_sl, 6),
                                price=round(new_sl, 6)
                            )
                            
                            # עדכן פוזיציה
                            position['sl_order'] = new_sl_order
                            position['stop_loss'] = new_sl
                            
                            self.logger.info(f"🔄 Trail SL: {symbol} -> {new_sl:.6f} (+{current_profit_pct:.1f}%)")
                            
                        except Exception as e:
                            self.logger.warning(f"⚠️  לא ניתן לעדכן SL עבור {symbol}: {e}")
                
                # בדוק אם הפוזיציה נסגרה
                open_orders = self.client.get_open_orders(symbol=symbol)
                sl_order_id = position['sl_order']['orderId']
                tp_order_id = position['tp_order']['orderId']
                
                # אם אין orders פתוחים - הפוזיציה נסגרה
                if not any(order['orderId'] in [sl_order_id, tp_order_id] for order in open_orders):
                    # הפוזיציה נסגרה - עדכן סטטיסטיקות
                    closed_pnl = self.calculate_position_pnl(position, current_price)
                    
                    if closed_pnl > 0:
                        self.wins += 1
                        self.logger.info(f"🎉 {symbol} נסגר ברווח: {closed_pnl:.2f} USDT")
                    else:
                        self.losses += 1
                        self.logger.info(f"💸 {symbol} נסגר בהפסד: {closed_pnl:.2f} USDT")
                    
                    # הסר מהפוזיציות הפעילות
                    del self.active_positions[symbol]
                    
                    # עדכן equity
                    self.equity = self.get_account_balance()
        
        except Exception as e:
            self.logger.error(f"❌ שגיאה בניטור פוזיציות: {e}")
    
    def calculate_position_pnl(self, position: Dict, exit_price: float) -> float:
        """חשב PNL עבור פוזיציה"""
        quantity = position['quantity']
        entry_price = position['entry_price']
        
        pnl = (exit_price - entry_price) * quantity
        return pnl
    
    def run_trading_cycle(self):
        """הרץ מחזור מסחר מלא"""
        self.logger.info("🔄 מתחיל מחזור מסחר...")
        
        # עדכן ניתוח שוק
        self.update_market_analysis()
        
        # רשימת זוגות מועדפים
        trading_pairs = [
            'BTCUSDT', 'ETHUSDT', 'BNBUSDT', 
            'XRPUSDT', 'ADAUSDT', 'DOTUSDT',
            'LINKUSDT', 'LTCUSDT', 'MATICUSDT'
        ]
        
        # סרוק זוגות לאותות מסחר
        for symbol in trading_pairs:
            try:
                signal = self.generate_intelligent_signal(symbol)
                
                if signal and signal['confidence'] >= self.min_confidence:
                    if self.execute_trade_with_management(signal):
                        self.logger.info(f"🎯 מסחר חדש: {symbol} עם ביטחון {signal['confidence']:.2f}")
                
                # המתנה בין סריקות
                time.sleep(1)
                
            except Exception as e:
                self.logger.error(f"❌ שגיאה בסריקת {symbol}: {e}")
                continue
    
    def reset_daily_counter(self):
        """אפס מונה מסחר יומי"""
        current_time = datetime.now()
        if current_time.date() > self.last_trade_reset.date():
            self.daily_trades_count = 0
            self.last_trade_reset = current_time
            
            # שמור סטטיסטיקות יומיות
            daily_stats = {
                'date': current_time.date(),
                'total_trades': self.total_trades,
                'wins': self.wins,
                'losses': self.losses,
                'win_rate': self.wins / self.total_trades if self.total_trades > 0 else 0
            }
            self.performance_history.append(daily_stats)
            
            self.logger.info(f"🔄 מונה מסחר יומי אופס | WIN RATE: {daily_stats['win_rate']:.2%}")
    
    def get_performance_summary(self) -> Dict:
        """קבל סיכום ביצועים"""
        win_rate = self.wins / self.total_trades if self.total_trades > 0 else 0
        
        return {
            'total_trades': self.total_trades,
            'wins': self.wins,
            'losses': self.losses,
            'win_rate': win_rate,
            'daily_trades': self.daily_trades_count,
            'equity': self.equity,
            'market_regime': self.market_regime,
            'active_positions': len(self.active_positions)
        }
