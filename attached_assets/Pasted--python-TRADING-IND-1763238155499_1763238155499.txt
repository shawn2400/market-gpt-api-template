    def calculate_bearish_combos(self, price_data):

        combos = {

            "combo_downtrend_momentum_resistance": {

                "conditions": [

            "combo_breakdown_confirmation": {

    def _calc_trend_momentum_support(self, data):
        """חישוב COMBO 1: מגמה + מומנטום + תמיכה"""
        score = 0
        max_score = 4
        
        # EMA trend check
        if data.get('ema_20', 0) > data.get('ema_50', 0):
            score += 1
        
        # RSI momentum check
        rsi = data.get('rsi', 50)
        if 35 <= rsi <= 65:
            score += 1
        
        # Support level check
        if data.get('near_support', False):
            score += 1
        
        # Volume confirmation
        if data.get('volume_above_avg', False):
            score += 1
        
        confidence = score / max_score
        return {
            "score": score,
            "max_score": max_score,
            "confidence": confidence,
            "grade": "A" if confidence >= 0.75 else "B" if confidence >= 0.5 else "C"
        }
    
    def _calc_breakout_confirmation(self, data):
        """חישוב COMBO 2: פריצת התנגדות"""
        score = 0
        max_score = 4
        
        if data.get('breakout_above_resistance', False):
            score += 1
        
        if data.get('volume_breakout', False):
            score += 1
        
        if data.get('rsi', 50) < 70:
            score += 1
        
        if data.get('ema_20', 0) > data.get('ema_50', 0):
            score += 1
        
        confidence = score / max_score
        return {
            "score": score,
            "max_score": max_score,
            "confidence": confidence,
            "grade": "A" if confidence >= 0.75 else "B" if confidence >= 0.5 else "C"
        }
    
    def _calc_pullback_trend_rsi(self, data):
        """חישוב COMBO 3: פולבק למגמה עם RSI"""
        score = 0
        max_score = 4
        
        if data.get('ema_20', 0) > data.get('ema_50', 0):
            score += 1
        
        if data.get('pullback_to_ema', False):
            score += 1
        
        if 30 <= data.get('rsi', 50) <= 40:
            score += 1
        
        if data.get('volume_confirmation', False):
            score += 1
        
        confidence = score / max_score
        return {
            "score": score,
            "max_score": max_score,
            "confidence": confidence,
            "grade": "A" if confidence >= 0.75 else "B" if confidence >= 0.5 else "C"
        }
    
    def _calc_downtrend_momentum_resistance(self, data):
        """חישוב COMBO 1 SHORT: מגמת ירידה + מומנטום + התנגדות"""
        score = 0
        max_score = 4
        
        if data.get('ema_20', 0) < data.get('ema_50', 0):
            score += 1
        
        rsi = data.get('rsi', 50)
        if 35 <= rsi <= 65:
            score += 1
        
        if data.get('near_resistance', False):
            score += 1
        
        if data.get('volume_above_avg', False):
            score += 1
        
        confidence = score / max_score
        return {
            "score": score,
            "max_score": max_score,
            "confidence": confidence,
            "grade": "A" if confidence >= 0.75 else "B" if confidence >= 0.5 else "C"
        }
    
    def analyze_current_setup(self, market_data):
        """מנתח את המצב הנוכחי ומחזיר את הקומבו המומלץ"""
        
        bullish_combos = self.calculate_bullish_combos(market_data)
        bearish_combos = self.calculate_bearish_combos(market_data)
        
        results = {
            "bullish_setups": [],
            "bearish_setups": [],
            "recommendation": None
        }
        
        # בדיקת כל הקומבוים הלונגיים
        for combo_id, combo_info in bullish_combos.items():
            calculation_func = combo_info["calculation"]
            result = calculation_func(market_data)
            
            if result["confidence"] >= 0.5:  # רק קומבוים עם confidence מעל 50%
                results["bullish_setups"].append({
                    "name": combo_info["name"],
                    "confidence": result["confidence"],
                    "grade": result["grade"],
                    "conditions": combo_info["conditions"]
                })
        
        # בדיקת כל הקומבוים השורטיים
        for combo_id, combo_info in bearish_combos.items():
            calculation_func = combo_info["calculation"]
            result = calculation_func(market_data)
            
            if result["confidence"] >= 0.5:
                results["bearish_setups"].append({
                    "name": combo_info["name"],
                    "confidence": result["confidence"],
                    "grade": result["grade"],
                    "conditions": combo_info["conditions"]
                })
        
        # מציאת ההמלצה הטובה ביותר
        all_setups = results["bullish_setups"] + results["bearish_setups"]
        if all_setups:
            best_setup = max(all_setups, key=lambda x: x["confidence"])
            results["recommendation"] = best_setup
        
        return results
