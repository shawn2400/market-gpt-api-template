import requests

class OllamaClient:
    def __init__(self, host="http://localhost:11434"):
        self.host = host

    def generate(self, prompt):
        r = requests.post(f"{self.host}/api/generate",
                          json={"model": "llama3.1", "prompt": prompt},
                          timeout=20)
        return r.json().get("response", "")

--------------------------------------------------------------------
app/ai/gpt_fallback.py
--------------------------------------------------------------------
import os
import openai

class GPTFallback:
    def __init__(self):
        openai.api_key = os.getenv("OPENAI_API_KEY")

    def generate(self, prompt):
        try:
            return openai.chat.completions.create(
                model="gpt-4.1-mini",
                messages=[{"role":"user","content":prompt}]
            ).choices[0].message.content
        except:
            return "AI unavailable"

--------------------------------------------------------------------
app/ai/ai_router.py
--------------------------------------------------------------------
from .ollama_client import OllamaClient
from .gpt_fallback import GPTFallback

class AIRouter:
    def __init__(self):
        self.local = OllamaClient()
        self.remote = GPTFallback()

    def ask(self, prompt):
        try:
            return self.local.generate(prompt)
        except:
            return self.remote.generate(prompt)

import requests
import os

ALGO_URL = os.getenv("ALGO_API_URL")
ALGO_TOKEN = os.getenv("ALGO_API_TOKEN")

def get_trade_signal():
    r = requests.get(
        f"{ALGO_URL}/autopilot/next",
        headers={"Authorization": f"Bearer {ALGO_TOKEN}"},
        timeout=8
    )
    return r.json()

def execute_decision(control, signal):
    if signal["action"] == "open":
        control.exec(f"binance_open {signal['symbol']}")
    elif signal["action"] == "close":
        control.exec(f"binance_close {signal['symbol']}")

from app.autopilot import get_trade_signal, execute_decision

@app.on_event("startup")
async def autopilot_loop():
    import asyncio
    while True:
        signal = get_trade_signal()
        if signal["status"] == "ready":
            execute_decision(exec_manager, signal)
        await asyncio.sleep(20)
