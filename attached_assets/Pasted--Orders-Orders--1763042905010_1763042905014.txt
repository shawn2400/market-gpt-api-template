class SystemOrderOptimizer:
    async def optimize_all_symbols(self):
        symbols = await self.get_all_traded_symbols()
        
        for symbol in symbols:
            # 1. קבל את כל ה-orders הנוכחיים
            current_orders = await self.get_orders(symbol)
            
            # 2. הפעל את פרוטוקול האופטימיזציה
            optimized_orders = await self.consolidate_orders(current_orders)
            
            # 3. בצע את השינויים
            await self.execute_optimization(symbol, optimized_orders)
            
            # 4. דווח על השינויים
            await self.report_optimization(symbol, current_orders, optimized_orders)
```

def consolidate_orders(current_orders):
    """מאחד orders דומים וממטב את הספר"""
    
    # קבוצ orders לפי סוג
    tp_orders = [o for o in current_orders if o.type == 'TAKE_PROFIT']
    sl_orders = [o for o in current_orders if o.type == 'STOP_LOSS']
    
    # אופטימיזציית TP - מקסימום 3 רמות
    if len(tp_orders) > 3:
        optimized_tp = self.optimize_tp_levels(tp_orders)
    else:
        optimized_tp = self.space_tp_levels(tp_orders)
    
    # החזר הספר הממוטב
    return optimized_tp + sl_orders
```
