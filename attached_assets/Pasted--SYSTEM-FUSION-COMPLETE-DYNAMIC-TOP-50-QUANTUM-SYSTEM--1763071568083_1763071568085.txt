```python
class StrictTop50Scorer:
    def __init__(self):
        self.min_volume = 20_000_000  # $20M - סף מינימלי
        self.min_liquidity = 100_000   # $100k - עומק order book
        
    def calculate_symbol_score(self, symbol_data):
        """ציון 0-100 - רק הטובים ביותר עוברים"""
        
        # 1. בדיקות סף - נכשל? ציון 0!
        if not self.passes_minimum_criteria(symbol_data):
            return 0
            
        scores = {
            'volume_score': self.calculate_volume_score(symbol_data['volume_24h']),
            'liquidity_score': self.calculate_liquidity_score(symbol_data['liquidity']),
            'performance_score': self.calculate_performance_score(symbol_data['history']),
            'volatility_score': self.calculate_volatility_score(symbol_data['atr']),
            'trend_score': self.calculate_trend_score(symbol_data['technical'])
        }
        
        # ממוצע משוקלל
        total_score = sum(score * weight for score, weight in zip(
            scores.values(), 
            [0.25, 0.20, 0.20, 0.15, 0.20]
        ))
        
        return round(total_score, 2)
    
    def passes_minimum_criteria(self, symbol_data):
        """סף מינימלי - זבל נופל כאן!"""
        return (symbol_data['volume_24h'] >= self.min_volume and 
                symbol_data['liquidity'] >= self.min_liquidity and
                symbol_data.get('win_rate', 0) >= 0.4)  # לפחות 40% win rate
```

```python
class Top50Calculator:
    def __init__(self):
        self.redis = redis_manager
        self.binance = binance_client
        self.scorer = StrictTop50Scorer()
    
    async def calculate_top_50(self):
        """חישוב TOP 50 - Zero Compromise!"""
        logger.info("🔄 Calculating new TOP 50 rankings...")
        
        # 1. קבל את כל הסמלים הזמינים
        all_symbols = await self.binance.get_all_symbols()
        
        # 2. חשב ציון לכל סמל שעובר סף מינימלי
        scored_symbols = []
        for symbol in all_symbols:
            try:
                symbol_data = await self.get_symbol_data(symbol)
                score = self.scorer.calculate_symbol_score(symbol_data)
                
                if score > 0:  # רק אם עבר את הסף המינימלי
                    scored_symbols.append({
                        'symbol': symbol,
                        'score': score,
                        'volume': symbol_data['volume_24h'],
                        'liquidity': symbol_data['liquidity']
                    })
                    
            except Exception as e:
                logger.warning(f"⚠️ Failed to score {symbol}: {e}")
                continue
        
        # 3. מיין וקח רק את ה-TOP 50 המוחלט
        scored_symbols.sort(key=lambda x: x['score'], reverse=True)
        top_50 = [s['symbol'] for s in scored_symbols[:50]]
        
        # 4. שמור והפץ
        await self.save_top_50(top_50)
        logger.info(f"✅ TOP 50 Updated: {len(top_50)} symbols")
        
        return top_50
    
    async def save_top_50(self, top_50):
        """שמירת TOP 50 ב-Redis + DB"""
        # Redis למהירות
        await self.redis.setex('current_top_50', 7200, json.dumps(top_50))
```

```python
            }
        
        # 2. הגנות נוספות

            self.check_grid_trading(symbol, order_type),
            self.check_leverage_limits(symbol),
            self.check_time_based_protections(symbol)
        ]
        
        for protection in protections:
            if not protection['approved']:
                return protection
                
        return {
            'approved': True,
            'reason': '✅ TOP 50 Approved Symbol',
            'allowed_leverage': await self.calculate_max_leverage(symbol)
        }

        """האם הסמל נמצא ב-TOP 50 הנוכחי?"""
        top_50 = await self.get_current_top_50()
        return symbol in top_50
    
    def check_grid_trading(self, symbol, order_type):
        """GRID trading - רק ל-TOP 10!"""
        if order_type == "GRID":
            top_10 = self.get_current_top_10()
            if symbol not in top_10:
                return {
                    'approved': False,
                    'reason': f'🚫 GRID trading blocked for {symbol} - TOP 10 only',
                    'blocked': True
                }
        return {'approved': True}
```
