#### B. Router חדש — `routes/benchmarks_tv.py`

```python
# -*- coding: utf-8 -*-
from __future__ import annotations
import os, json, time, logging, hashlib
from typing import Optional, Any, Dict
from datetime import datetime, timezone
from fastapi import APIRouter, Request, Header, HTTPException
from pydantic import BaseModel, Field, ValidationError, condecimal
from utils.benchmarks_ingest import save_tv_alert_as_benchmark
from utils.rate_limit import sliding_window_ok  # אם אין, נטפל מטה ב-fallback
from utils.safe_headers import to_str  # אם אין, נטפל מטה ב-fallback

router = APIRouter(prefix="/benchmarks", tags=["Benchmarks"])

TV_WEBHOOK_TOKEN = os.getenv("TV_WEBHOOK_TOKEN", "")
TV_ALERT_RATE_MAX_PER_MIN = int(os.getenv("TV_ALERT_RATE_MAX_PER_MIN", "300"))  # רחב כי זה Any alert()
ALLOW_ANON_TV = os.getenv("ALLOW_ANON_TV", "0") == "1"

# Fallbacks אם אין כלים פנימיים
def _sliding_window_ok_fallback(key: str, limit: int, window_sec: int = 60) -> bool:
    # זיכרון מקומי — פרוסס יחיד; בפרודקשן עדיף Redis. לא נופל אם utils.rate_limit חסר.
    try:
        _store = _sliding_window_ok_fallback.__dict__.setdefault("_store", {})
        now = time.time()
        bucket = _store.setdefault(key, [])
        # נקה ישנים
        bucket[:] = [t for t in bucket if now - t < window_sec]
        if len(bucket) >= limit:
            return False
        bucket.append(now)
        return True
    except Exception:
        return True

def _to_str_fallback(v: Any) -> str:
    try:
        if v is None: return ""
        if isinstance(v, (bytes, bytearray)): return v.decode("utf-8", "ignore")
        return str(v)
    except Exception:
        return ""

if "sliding_window_ok" not in globals():
    sliding_window_ok = _sliding_window_ok_fallback  # type: ignore
if "to_str" not in globals():
    to_str = _to_str_fallback  # type: ignore

class TvAlertPayload(BaseModel):
    sym: str = Field(..., min_length=3, max_length=32)
    tf: str = Field(..., min_length=1, max_length=16)
    sig: str = Field(..., pattern="^(Breakout|Retest|Sweep)$")
    price: condecimal(gt=0)  # Decimal לשמירה על דיוק
    ema21: condecimal(gt=0)
    ema50: condecimal(gt=0)
    adx: condecimal(gt=0)
    atr: condecimal(gt=0)
    tp1_x: condecimal(gt=0)
    tp2_x: condecimal(gt=0)
    sl_x: condecimal(gt=0)
    ts: int  # unix seconds (TV)

def _auth_guard(x_tv_token: Optional[str]) -> None:
    if ALLOW_ANON_TV:
        return
    token = to_str(x_tv_token)
    if not token or not TV_WEBHOOK_TOKEN or token != TV_WEBHOOK_TOKEN:
        raise HTTPException(status_code=401, detail="unauthorized")

@router.post("/tv-alert")
async def tv_alert_ingest(
    request: Request,
    x_tv_token: Optional[str] = Header(None, convert_underscores=False),
    x_forwarded_for: Optional[str] = Header(None, convert_underscores=False),
):
    _auth_guard(x_tv_token)
    ip = to_str(x_forwarded_for) or request.client.host if request.client else "unknown"
    key = f"tv:{ip}"
    if not sliding_window_ok(key, TV_ALERT_RATE_MAX_PER_MIN, 60):
        raise HTTPException(status_code=429, detail="rate_limited")

    body = await request.body()
    text = body.decode("utf-8", "ignore") if isinstance(body, (bytes, bytearray)) else str(body)
    try:
        data = json.loads(text)
    except Exception:
        raise HTTPException(status_code=400, detail="invalid_json")

    try:
        payload = TvAlertPayload(**data)
    except ValidationError as e:
        raise HTTPException(status_code=422, detail=f"schema_error:{e.errors()}")

    # שמירה כ-Benchmark (ינרמל ויזריק ל-DB)
    try:
        saved = await save_tv_alert_as_benchmark(payload.model_dump())
    except Exception as ex:
        logging.exception("tv_alert_ingest save failed")
        raise HTTPException(status_code=500, detail=f"save_failed:{ex}")

    return {
        "ok": True,
        "source": "tradingview",
        "id": saved.get("id"),
        "ts_server": datetime.now(timezone.utc).isoformat(),
    }
```

```python
from __future__ import annotations
import os, json, asyncio
from datetime import datetime, timezone, timedelta
from typing import Dict, Any
import decimal
import pathlib
import psycopg2, psycopg2.extras  # אם אתם כבר עם asyncpg/SQLAlchemy, אפשר להתאים

DB_DSN = os.getenv("DATABASE_URL") or os.getenv("POSTGRES_DSN") or "postgresql://postgres:postgres@127.0.0.1:5432/algogpt"

# משקולות לציון קומפוזיט
W = {
    "sharpe": 0.25,
    "profit_factor": 0.20,
    "win_rate": 0.20,
    "max_drawdown": 0.15,   # נורמליזציה עם סימן מינוס

def _to_float(x: Any) -> float:
    if isinstance(x, decimal.Decimal): return float(x)
    try: return float(x)
    except Exception: return 0.0

def _calc_composite(example: Dict[str, float]) -> float:
    # TV Alert בודד לא כולל סטטיסטיקות מלאות — נרשום composite מינימלי מהקשר:
    # כאן נשמור 0 ונאפשר אנליטיקה בהמשך (TV משמש “ויזואלי/טריגר”).
    return 0.0

async def save_tv_alert_as_benchmark(tv: Dict[str, Any]) -> Dict[str, Any]:
    """
    tv: dict מטען מאומת מ-TvAlertPayload
    שמירה:
      - JSON לקובץ: benchmarks/tradingview/YYYYMMDD/<sym>-<ts>.json
      - שורת DB לטבלת benchmark_results (source='tradingview') בתאריך הנר
    """
    # שמירה כקובץ (ארכיון)
    day = datetime.now(timezone.utc).strftime("%Y%m%d")
    d = pathlib.Path("benchmarks") / "tradingview" / day
    d.mkdir(parents=True, exist_ok=True)
    fname = d / f"{tv['sym']}-{tv['ts']}.json"
    with open(fname, "w", encoding="utf-8") as f:
        json.dump(tv, f, ensure_ascii=False)

    # הערכים ל-DB (מינימלי; TV = “event” ולא סשן מלא)
    now = datetime.now(timezone.utc)
    period_start = datetime.fromtimestamp(int(tv["ts"]), timezone.utc)
    period_end   = period_start + timedelta(minutes=1)

    row = dict(
        source="tradingview",
        period_start=period_start.isoformat(),
        period_end=period_end.isoformat(),
        total_trades=1,
        win_rate=None,
        profit_factor=None,
        sharpe_ratio=None,
        max_drawdown=None,
        total_pnl_usd=None,
        avg_trade_duration_min=None,
        composite_score=_calc_composite({}),
    )

    sql = """
    CREATE TABLE IF NOT EXISTS benchmark_results (
      id SERIAL PRIMARY KEY,
      source VARCHAR(20),
      period_start TIMESTAMPTZ,
      period_end   TIMESTAMPTZ,
      total_trades INT,
      win_rate DOUBLE PRECISION,
      profit_factor DOUBLE PRECISION,
      sharpe_ratio DOUBLE PRECISION,
      max_drawdown DOUBLE PRECISION,
      total_pnl_usd DOUBLE PRECISION,
      avg_trade_duration_min INT,
      composite_score DOUBLE PRECISION,
      created_at TIMESTAMPTZ DEFAULT NOW()
    );
    CREATE INDEX IF NOT EXISTS idx_bench_source_period ON benchmark_results(source, period_start);
    INSERT INTO benchmark_results
    (source, period_start, period_end, total_trades, win_rate, profit_factor, sharpe_ratio, max_drawdown, total_pnl_usd, avg_trade_duration_min, composite_score)
    VALUES (%(source)s, %(period_start)s, %(period_end)s, %(total_trades)s, %(win_rate)s, %(profit_factor)s, %(sharpe_ratio)s, %(max_drawdown)s, %(total_pnl_usd)s, %(avg_trade_duration_min)s, %(composite_score)s)
    RETURNING id;
    """
    # חיבור זמני סינכרוני (פשוט). ואם יש PgBouncer על 6432, הדס"ן יפנה אליו.
    conn = psycopg2.connect(DB_DSN)
    try:
        with conn:
            with conn.cursor(cursor_factory=psycopg2.extras.DictCursor) as cur:
                cur.execute(sql, row)
                new_id = cur.fetchone()[0]
    finally:
        conn.close()

    row["id"] = new_id
    return row
```

```python
# utils/rate_limit.py
import time
from typing import Dict
_store: Dict[str, list] = {}
def sliding_window_ok(key: str, limit: int, window_sec: int = 60) -> bool:
    now = time.time()
    bucket = _store.setdefault(key, [])
    # ניקוי
    while bucket and now - bucket[0] >= window_sec:
        bucket.pop(0)
    if len(bucket) >= limit:
        return False
    bucket.append(now)
    return True
```

### 4) הוראות ידניות ל-TradingView (חד-פעמי לשנה)

1. העתק את תוכן `tradingview/algogpt_bench_v1.pine` לאינדיקטור חדש ב-TV ושמור.
2. פתח Alert חדש:

   * **Condition:** האינדיקטור שלך
   * **Trigger:** **Any alert() function call**
   * **Webhook URL:** `https://<PUBLIC_HOST>/benchmarks/tv-alert`
   * **HTTP Header (Optional):** `X-TV-Token: rnd_tv_ChangeMe123456`
   * **Message:** לא צריך — האינדיקטור שולח `alert(msg)`
   * **Expiration:** שנה קדימה
3. שמור. אין צורך לגעת שוב — האלרט “חי לנצח” ומזרים JSON דינמי.

### 5) הערות פרודקשן

* אם מותקן PgBouncer, הפנה את `POSTGRES_DSN` ל-6432.
* אם יש Anti-Replay/Signature אצלכם, אפשר להוסיף גם חתימה ל-TV: ה-TV לא תומך ב־HMAC מובנה, לכן נשארים עם טוקן כותרת ורייט-לימיט סביר.
* קצב `Any alert()` עלול להיות גבוה — שמנו Rate-Limit רחב וארכיון קבצים יומי.
* ה-DB שומר “אירוע” TV כאינדיקציה ויזואלית; חיבור מלא ל-Composite נעשה במודול האנליזה הקיים.
