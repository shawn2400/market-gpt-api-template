# -*- coding: utf-8 -*-
from __future__ import annotations
import os, time, json, urllib.request

NEON_API_KEY   = os.getenv("NEON_API_KEY", "")
NEON_PROJECT_ID= os.getenv("NEON_PROJECT_ID", "")
NEON_ENDPOINT_ID=os.getenv("NEON_ENDPOINT_ID", "")

def resume_endpoint(timeout_sec: int = 30) -> tuple[bool,str]:
    if not (NEON_API_KEY and NEON_PROJECT_ID and NEON_ENDPOINT_ID):
        return False, "NEON vars missing (skip)"
    url = f"{NEON_BASE}/projects/{NEON_PROJECT_ID}/endpoints/{NEON_ENDPOINT_ID}/resume"

    try:
        with urllib.request.urlopen(req, timeout=timeout_sec) as r:
            _ = r.read()
        return True, "Neon resume requested"
    except Exception as e:
        return False, f"Neon resume failed: {e!r}"

# -*- coding: utf-8 -*-
from __future__ import annotations
from decimal import Decimal, ROUND_DOWN
from typing import List

def clamp_price(v: float, min_tick: float) -> float:
    return max(v, float(min_tick))

def quantize(v: float, step: float) -> float:
    return float(Decimal(str(v)).quantize(Decimal(str(step)), rounding=ROUND_DOWN))

def tp_from_entry(side: str, entry: float, percents: List[float], tick: float = 0.0001) -> List[float]:
    out: List[float] = []
    for p in percents:
        k = float(p) / 100.0
        if side.upper() == "LONG":
            price = entry * (1.0 + k)
        else:
            price = entry * (1.0 - k)
        price = clamp_price(price, tick)
        out.append(quantize(price, tick))
    return out

# -*- coding: utf-8 -*-
from __future__ import annotations
import os, time, threading
from collections import deque
from typing import Deque, Tuple, List

DIGEST_ENABLE = os.getenv("TELEGRAM_DIGEST_ENABLED", "1") == "1"
DIGEST_WINDOW_SEC = int(os.getenv("TELEGRAM_DIGEST_INTERVAL_SEC", "900"))

class DigestBus:
    def __init__(self) -> None:
        self.q: Deque[Tuple[str,str]] = deque()
        self.lock = threading.Lock()
        self.last_ts = 0.0

    def push(self, level: str, msg: str) -> None:
        if not DIGEST_ENABLE: return
        with self.lock:
            self.q.append((level, msg))

    def pop_all(self) -> List[Tuple[str,str]]:
        with self.lock:
            items = list(self.q)
            self.q.clear()
            return items

BUS = DigestBus()

def enqueue(level: str, msg: str) -> None:
    BUS.push(level, msg)

def should_flush() -> bool:
    if not DIGEST_ENABLE: return False
    now = time.time()
    if now - BUS.last_ts >= DIGEST_WINDOW_SEC:
        BUS.last_ts = now
        return True
    return False

def format_digest(items: List[Tuple[str,str]]) -> str:
    from collections import defaultdict
    g = defaultdict(list)
    for lvl, m in items:
        g[lvl].append(m)
    lines = ["📬 *Digest*"]
    for k, arr in g.items():
        lines.append(f"\n*{k}* ({len(arr)}):")
        lines += [f"• {s}" for s in arr[:20]]
        if len(arr) > 20:
            lines.append(f"… ועוד {len(arr)-20}")
    return "\n".join(lines)

# -*- coding: utf-8 -*-
from __future__ import annotations
import os, sys, logging

REQUIRED_SOFT = ["OPENAI_API_KEY","GITHUB_REPO"]
REQUIRED_HARD = []  # הוסף כאן אם תרצה לעצור ריצה אם חסר

def validate_env(strict: bool=False) -> None:
    miss_soft = [k for k in REQUIRED_SOFT if not os.getenv(k)]
    miss_hard = [k for k in REQUIRED_HARD if not os.getenv(k)]
    if miss_soft:
        logging.warning("Missing (soft) ENV: %s", ",".join(miss_soft))
    if miss_hard:
        logging.error("Missing (hard) ENV: %s", ",".join(miss_hard))
        if strict:
            sys.exit(2)

### מה זה נותן לך בפועל (בקצר):

* **Neon Auto-Resume**: מודול Python + סקריפט (ריצה בתחילת ה-app + ידנית לפי הצורך).
* **Binance Hedge ENFORCE**: הפעלה חד-פעמית של Hedge Mode אם הדגל דלוק, ובניית פרמטרים בטוחה שמונעת `-4061` ו-`-1106`.
* **TP Ladder בטוח**: לעולם לא מחירים שליליים, כיווניות נכונה, קוונטיזציה לפי tick.
* **Telegram Digest**: איגוד התראות לכל 15 דק׳ (ניתן לשינוי ב-ENV), בלי להציף.
* **ENV Validation**: אזהרות על חסרים + אפשרות להקשיח בהמשך; `.env.example` מעודכן.
* **psycopg/SQLAlchemy/requests/tenacity**: תלויות מותקנות ב־requirements.
* **Hooks ל-main.py**: `validate_env()` + `resume_endpoint()` לפני חיבור DB; ו-`ensure_hedge_mode()` בתחילת הריצה כשדגל פעיל.
