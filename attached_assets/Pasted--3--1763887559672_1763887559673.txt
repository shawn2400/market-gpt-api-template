## ✅ 1) `routes/dashboard.py` — ממשק שליטה מלא

```python
from fastapi import APIRouter
from utils.state import SYSTEM_STATE
from utils.account_manager import AccountManager
from utils.control_flags import ControlFlags

router = APIRouter(prefix="/dashboard", tags=["Dashboard"])

@router.get("/live-status")
def live_status():
    return {
        "mode": SYSTEM_STATE.mode,
        "auto_pilot": SYSTEM_STATE.auto_pilot,
        "accounts_connected": AccountManager.count(),
        "hedge_active": SYSTEM_STATE.hedge_active,
        "circuit_breaker": SYSTEM_STATE.cb_active,
        "kill_switch": SYSTEM_STATE.kill_switch,
        "last_action": SYSTEM_STATE.last_action
    }

@router.post("/toggle-auto")
def toggle_auto():
    SYSTEM_STATE.auto_pilot = not SYSTEM_STATE.auto_pilot
    return {"auto_pilot": SYSTEM_STATE.auto_pilot}

@router.post("/freeze")
def freeze_all():
    ControlFlags.freeze_all()
    return {"status": "ALL TRADING PAUSED"}

@router.post("/resume")
def resume_all():
    ControlFlags.resume_all()
    return {"status": "SYSTEM RESUMED"}
```

from utils.account_manager import AccountManager
from utils.pnl_tracker import get_user_pnl
from utils.payments import send_payment_request

def run_weekly_billing():
    users = AccountManager.get_all()

    for user in users:
        pnl = get_user_pnl(user.id)
        if pnl <= 0:
            continue

        fee = pnl * 0.18
        send_payment_request(user.wallet, fee)

    return "BILLING COMPLETE"

## ✅ 7) `utils/account_manager.py` — עד 7 משתמשים

```python
class AccountManager:
    accounts = {}

    @classmethod
    def add(cls, user_id, api_key, api_secret):
        if len(cls.accounts) >= 7:
            raise Exception("MAX_USERS_REACHED")
        cls.accounts[user_id] = {"key": api_key, "secret": api_secret}

    @classmethod
    def remove(cls, user_id):
        cls.accounts.pop(user_id, None)

    @classmethod
    def count(cls):
        return len(cls.accounts)

    @classmethod
    def get_all(cls):
        return cls.accounts.values()
```

def auto_hedge(exposure, vol, sentiment):
    if exposure > 2.0 and vol > 4:
        return {"action": "HEDGE", "ratio": 0.5}

    if sentiment == "BEARISH" and exposure > 1.2:
        return {"action": "REDUCE", "ratio": 0.3}

    return {"action": "NONE"}
