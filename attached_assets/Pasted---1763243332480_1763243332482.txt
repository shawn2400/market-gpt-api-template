```python
# utils/blacklist_manager.py

class BlacklistManager:
    def clear_blacklist(self):
        """מנקה את ה-blacklist ומתחיל מחדש"""
        try:
            r = redis.Redis(**REDIS_CONFIG)
            
            # מחיקת כל הרשומות ב-blacklist
            blacklist_keys = r.keys('blacklist:*')
            if blacklist_keys:
                r.delete(*blacklist_keys)
                print(f"✅ נוקו {len(blacklist_keys)} סימנים מ-blacklist")
            
            # איפוס סטטיסטיקות כישלון
            failure_keys = r.keys('failure_count:*')
            if failure_keys:
                r.delete(*failure_keys)
                print(f"✅ אופסו {len(failure_keys)} סטטיסטיקות כישלון")
            
            return True
            
        except Exception as e:
            print(f"❌ שגיאה בניקוי blacklist: {e}")
            return False

    def analyze_failure_reasons(self, symbol):
        """מנתח את סיבות הכישלון עבור סימן ספציפי"""
        failure_data = {
            'recent_trades': self.get_recent_trades(symbol),
            'market_conditions': self.get_market_conditions(symbol),
            'signal_quality': self.analyze_signal_quality(symbol),
            'execution_issues': self.check_execution_issues(symbol)
        }
        return failure_data
```

```python
class ImprovedTradingStrategy:
    def __init__(self):
        self.failure_threshold = 3  # במקום 5
        self.cooldown_period = 3600  # 1 hour cooldown
    
    def enhanced_signal_validation(self, signal):
        """אימות משו�בר לאותות מסחר"""
        validation_checks = {
            'volume_confirmation': signal['volume'] > signal['volume_ma_20'] * 1.2,
            'trend_alignment': self.check_trend_alignment(signal),
            'volatility_check': signal['atr_percentage'] < 0.05,  # 5% מקסימום
            'timeframe_confirmation': self.multi_timeframe_confirmation(signal),
            'news_sentiment': self.check_news_sentiment(signal['symbol'])
        }
        
        passed_checks = sum(validation_checks.values())
        return passed_checks >= 3  # לפחות 3 checks חיוביים
    
    def dynamic_position_sizing(self, symbol, failure_count):
        """התאמה דינמית של גודל פוזיציה based on היסטוריה"""
        base_size = 0.02  # 2% מהון
        
        if failure_count > 0:
            reduction_factor = 1 / (failure_count + 1)
            adjusted_size = base_size * reduction_factor
            return max(adjusted_size, 0.005)  # מינימום 0.5%
        
        return base_size
```

```python
# analytics/failure_analyzer.py

class FailureAnalyzer:
    def generate_failure_report(self, symbols):
        """מייצר דוח מפורט על כישלונות"""
        report = {}
        
        for symbol in symbols:
            symbol_data = {
                'failure_count': self.get_failure_count(symbol),
                'common_exit_reasons': self.get_exit_reasons(symbol),
                'avg_trade_duration': self.get_avg_duration(symbol),
                'market_conditions_during_failures': self.get_market_conditions(symbol),
                'recommendations': self.generate_recommendations(symbol)
            }
            report[symbol] = symbol_data
        
        return report
    
    def get_exit_reasons(self, symbol):
        """מחזיר את הסיבות הנפוצות ליציאה"""
        trades = self.get_recent_trades(symbol, count=10)
        exit_reasons = {}
        
        for trade in trades:
            reason = trade.get('exit_reason')
            if reason:
                exit_reasons[reason] = exit_reasons.get(reason, 0) + 1
        
        return exit_reasons
    
    def generate_recommendations(self, symbol):
        """מייצר המלצות based on דפוסי כישלון"""
        failure_pattern = self.identify_failure_pattern(symbol)
        
        recommendations = {
            'stop_loss_adjustment': "התאם SL ל-2x ATR",
            'entry_timing': "הוסף confirmation candle",
            'position_size': "הקטן גודל פוזיציה ב-50%",
            'timeframe': "עבור ל-timeframe גבוה יותר"
        }
        
        return recommendations.get(failure_pattern, "בדוק את הגדרות האסטרטגיה")
```

## 🛡️ **מערכת הגנה משופרת - קוד מלא:**

```python
# core/enhanced_protection.py

class EnhancedProtectionSystem:
    def __init__(self):
        self.redis = redis.Redis(**REDIS_CONFIG)
        self.failure_threshold = 3
        self.cooldown_hours = 4
    
    def should_allow_trade(self, symbol, signal_confidence):
        """קובע אם לאפשר מסחר based on היסטוריה"""
        failure_count = self.redis.get(f"failure_count:{symbol}") or 0
        last_failure = self.redis.get(f"last_failure:{symbol}")
        
        # בדיקת cooldown period
        if last_failure:
            time_since_failure = time.time() - float(last_failure)
            if time_since_failure < self.cooldown_hours * 3600:
                return False, "בתקופ� cooldown"
        
        # בדיקת failure count
        if int(failure_count) >= self.failure_threshold:
            return False, "יותר מדי כשלונות"
        
        # בדיקת confidence מספיק גבוה
        if signal_confidence < 0.7:
            return False, "confidence נמוך מדי"
        
        return True, "מאושר למסחר"
    
    def record_failure(self, symbol, reason):
        """מתועד כשלון ומעדכן סטטיסטיקות"""
        # עדכון failure count
        current_count = self.redis.incr(f"failure_count:{symbol}")
        self.redis.set(f"last_failure:{symbol}", time.time())
        
        # שמירת סיבת הכשלון
        self.redis.rpush(f"failure_reasons:{symbol}", reason)
        
        # אם עבר threshold - הוסף ל-blacklist
        if current_count >= self.failure_threshold:
            self.redis.sadd("blacklist:symbols", symbol)
            self.redis.setex(f"blacklist:{symbol}", 86400, "auto_blacklisted")  # 24 hours
    
    def record_success(self, symbol):
        """מתועד הצלחה - מאפס סטטיסטיקות כישלון"""
        self.redis.delete(f"failure_count:{symbol}")
        self.redis.delete(f"last_failure:{symbol}")
        self.redis.srem("blacklist:symbols", symbol)
```
