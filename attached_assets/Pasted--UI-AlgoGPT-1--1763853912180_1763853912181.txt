# ⚙️ **7) קוד מלא — דשבורד עם FastAPI + HTMX**

## 📁 dashboard.py

```python
from fastapi import APIRouter, Request
from fastapi.responses import HTMLResponse
from external.plugin_manager import PluginManager
from algo_core.self_optimizer import SelfOptimizer

router = APIRouter()

plugin_manager = PluginManager()
plugin_manager.load_all()
optimizer = SelfOptimizer()

@router.get("/dashboard", response_class=HTMLResponse)
async def dashboard(request: Request):
    bots = plugin_manager.plugins
    return HTML_TEMPLATE.format(
        bots_html="".join(render_bot(name, client) for name, client in bots.items())
    )

def render_bot(name, client):
    return f"""
    <div class="bot-card">
        <h3>{name.upper()}</h3>
        <p>Type: {client.ptype}</p>
        <p>Capabilities: {client.cap}</p>
        <button hx-post="/control/{name}/on">ON</button>
        <button hx-post="/control/{name}/off">OFF</button>
        <button hx-post="/control/{name}/auto">AUTO</button>
    </div>
    """

HTML_TEMPLATE = """
<html>
<head>
<style>
.bot-card {
  background:#0d1117;
  border:1px solid #30363d;
  padding:15px;
  margin:10px;
  color:white;
  border-radius:8px;
}
button {
  margin-right:10px;
}
body { background:#010409; color:white; }
</style>
</head>
<body>
<h1>AlgoGPT — External Brain Control</h1>
<div>{bots_html}</div>
</body>
</html>
"""
```
