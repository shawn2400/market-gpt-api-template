# routes/live_dashboard.py

from fastapi import APIRouter, WebSocket
from utils.performance_heatmap import heat
from core.mode_manager import mode
from utils.freeze_manager import freeze
from utils.news_guard import news
from utils.kill_switch import kill
from utils.auto_optimizer import optimizer

router = APIRouter()

@router.websocket("/ws/live")
async def live_ws(ws: WebSocket):
    await ws.accept()
    while True:
        await ws.send_json({
            "mode": mode.mode,
            "auto_pause": news.is_paused(),
            "kill_switch": kill.is_active(),
            "optimizer_state": optimizer.current_state(),
            "freeze_list": freeze.list_active(),
            "heatmap": heat.summary(),
        })
        await asyncio.sleep(3)

# ✅ חלק 2 — AUTO-OPTIMIZE (לומד לבד)

### 📌 `utils/auto_optimizer.py`

```python
# utils/auto_optimizer.py

import numpy as np

class AutoOptimizer:
    def __init__(self):
        self.history = []
        self.params = {
            "atr_mult": 1.2,
            "take_profit_r": 2.0,
            "max_trades_day": 6
        }

    def update(self, pnl):
        self.history.append(pnl)

        if len(self.history) < 15:
            return

        win_rate = sum(p > 0 for p in self.history[-15:]) / 15
        avg_r = np.mean([p for p in self.history[-15:]])

        if win_rate > 0.7 and avg_r > 0:
            self.params["atr_mult"] = min(self.params["atr_mult"] + 0.1, 2.0)
            self.params["take_profit_r"] += 0.2

        if win_rate < 0.45:
            self.params["atr_mult"] = max(self.params["atr_mult"] - 0.1, 0.6)
            self.params["max_trades_day"] = max(self.params["max_trades_day"] - 1, 2)

    def current_state(self):
        return self.params

optimizer = AutoOptimizer()
```

# utils/kill_switch.py

import time

class KillSwitch:
    def __init__(self):
        self.active = False
        self.reason = None
        self.triggered_at = None

    def activate(self, reason):
        self.active = True
        self.reason = reason
        self.triggered_at = time.time()

    def reset(self):
        self.active = False
        self.reason = None
        self.triggered_at = None

    def is_active(self):
        return self.active

kill = KillSwitch()
```

# utils/circuit_breaker.py

import time

class CircuitBreaker:
    def __init__(self):
        self.locked = False
        self.unlock_time = None

    def trigger(self, minutes, reason):
        self.locked = True
        self.unlock_time = time.time() + (minutes * 60)

    def is_locked(self):
        if self.locked and time.time() >= self.unlock_time:
            self.locked = False
        return self.locked

breaker = CircuitBreaker()
```

import { useEffect, useState } from "react";

export default function App() {
  const [data, setData] = useState({});

  useEffect(() => {
    const ws = new WebSocket("wss://YOUR_DOMAIN/ws/live");

    ws.onmessage = (msg) => {
      setData(JSON.parse(msg.data));
    };

    return () => ws.close();
  }, []);

  return (
    <div style={{ fontFamily: "sans-serif", padding: 20 }}>
      <h1>AlgoGPT Live Dashboard</h1>

}
