```python
# 📁 ב-position_monitor.py - הוספת trailing logic

class EnhancedPositionMonitor:
    def __init__(self):
        self.trailing_positions = {}  # {symbol: {peak_price, trailing_active, etc}}
    
    async def check_and_update_trailing_tp(self):
        """בודק ועדכן trailing TP כל 30 שניות"""
        for position in self.get_active_positions():
            if self.should_activate_trailing(position):
                self.activate_trailing(position)
            
            if self.should_close_by_trailing(position):
                await self.execute_trailing_close(position)
    
    def should_activate_trailing(self, position):
        """האם להפעיל trailing TP?"""
        current_pnl = position['unrealizedProfit'] / position['positionAmt']
        return current_pnl >= 0.25  # 25% רווח - הפעל trailing
    
    def activate_trailing(self, position):
        """הפעל trailing TP לפוזיציה"""
        symbol = position['symbol']
        self.trailing_positions[symbol] = {
            'peak_price': position['markPrice'],
            'activation_time': datetime.now(),
            'trailing_distance': 0.15,  # 15% מתחת לשיא
            'original_entry': position['entryPrice']
        }
        logger.info(f"🎯 Trailing TP activated for {symbol}")
    
    def should_close_by_trailing(self, position):
        """בדוק אם צריך לסגור לפי trailing"""
        symbol = position['symbol']
        if symbol not in self.trailing_positions:
            return False
        
        trailing_data = self.trailing_positions[symbol]
        current_price = position['markPrice']
        peak_price = trailing_data['peak_price']
        
        # עדכן שיא חדש אם יש
        if current_price > peak_price:
            trailing_data['peak_price'] = current_price
            return False
        
        # בדוק אם ירד מספיק מהשיא
        drawdown = (peak_price - current_price) / peak_price
        return drawdown >= trailing_data['trailing_distance']
```

```python
class InsuranceMonitor:
    def __init__(self):
        self.protection_rules = {
            'drawdown': {'threshold': -0.05, 'active': True},
            'margin_ratio': {'threshold': 0.10, 'active': True},
            'circuit_breaker': {'threshold': -0.08, 'active': True}
        }
    
    async def run_insurance_checks(self):
        """רוץ כל 60 שניות - לא צריך real-time כמו trailing"""
        while True:
            try:
                await self.check_drawdown_protection()
                await self.check_margin_ratio() 
                await self.check_circuit_breaker()
                await asyncio.sleep(60)  # כל דקה
            except Exception as e:
                logger.error(f"Insurance check failed: {e}")
```
