## ✅ קובץ יחיד — **`ha-global-bootstrap.sh`**

**העתק-הדבק — מוכן להרצה:**

```bash
#!/bin/bash
set -e

echo "🌍 STARTING GLOBAL HA AUTO-PILOT DEPLOY"

### --- REQUIRED ENV ---
PRIMARY_IP="<PRIMARY-SERVER-IP>"
SECONDARY_IP="<SECONDARY-SERVER-IP>"
DNS_API_KEY="<CLOUDFLARE_API_KEY>"
DNS_ZONE="<YOUR_DOMAIN>"
DNS_RECORD="api.${DNS_ZONE}"

### --- INSTALL KEEPALIVED (FAILOVER) ---
apt-get update -y
apt-get install -y keepalived haproxy curl jq

### --- CREATE KEEPALIVED CONFIG ---
cat > /etc/keepalived/keepalived.conf << EOF
vrrp_instance VI_1 {
    state MASTER
    interface eth0
    virtual_router_id 51
    priority 150
    advert_int 1
    authentication {
        auth_type PASS
        auth_pass ALGOGPTHA
    }
    virtual_ipaddress {
        ${PRIMARY_IP}
    }
    notify_fault "/usr/local/bin/ha-failover.sh fault"
    notify_master "/usr/local/bin/ha-failover.sh master"
}
EOF

### --- CREATE FAILOVER SCRIPT ---
cat > /usr/local/bin/ha-failover.sh << EOF
#!/bin/bash
EVENT=\$1
if [ "\$EVENT" = "fault" ]; then
  curl -s -X PATCH \
    "https://api.cloudflare.com/client/v4/zones/${DNS_ZONE}/dns_records" \
    -H "Authorization: Bearer ${DNS_API_KEY}" \
    -H "Content-Type: application/json" \
    --data '{"content":"${SECONDARY_IP}"}'
elif [ "\$EVENT" = "master" ]; then
  curl -s -X PATCH \
    "https://api.cloudflare.com/client/v4/zones/${DNS_ZONE}/dns_records" \
    -H "Authorization: Bearer ${DNS_API_KEY}" \
    -H "Content-Type: application/json" \
    --data '{"content":"${PRIMARY_IP}"}'
fi
EOF

chmod +x /usr/local/bin/ha-failover.sh

systemctl enable keepalived
systemctl restart keepalived

### --- AUTO-HEALING SERVICE ---
cat > /etc/systemd/system/algogpt-heal.service << EOF
[Unit]
Description=AlgoGPT Self-Healing
After=docker.service

[Service]
ExecStart=/usr/local/bin/algogpt-heal.sh
Restart=always

[Install]
WantedBy=multi-user.target
EOF

### --- SELF-REPAIR LOGIC ---
cat > /usr/local/bin/algogpt-heal.sh << EOF
#!/bin/bash
while true; do
  if ! curl -fs http://localhost:10000/readyz >/dev/null; then
    docker restart algogpt || true
  fi
  sleep 5
done
EOF

chmod +x /usr/local/bin/algogpt-heal.sh
systemctl enable algogpt-heal
systemctl start algogpt-heal

echo "✅ GLOBAL HA READY"
echo "🌍 AUTO-FAILOVER ENABLED"
echo "🚀 ZERO-DOWNTIME ACTIVE"
```
