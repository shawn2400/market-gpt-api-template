s=open(p,"r",encoding="utf-8").read()
def strip_env(cmd):
    parts=cmd.strip().split()
    out=[]; started=False
    for t in parts:
        if not started and "=" in t and not t.startswith(("python","gunicorn","uvicorn","bash","sh","cd")):
            continue
        started=True; out.append(t)
    return " ".join(out)
def repl(m): 
    before,val,after=m.group(1),m.group(2),m.group(3)
    return f'{before}{strip_env(val)}{after}'

from __future__ import annotations
def clamp_to_step(value: float, step: float) -> float:
    if not step: return value
    return (int(value/step))*step
def round_to_tick(price: float, tick: float) -> float:
    if not tick: return price
    return (int(price/tick))*tick
EOF

import logging
from typing import List, Any
from utils.precision_utils import round_to_tick, clamp_to_step
logger = logging.getLogger(__name__)
async def set_tp_ladder(symbol:str, side:str, tp_prices:List[float], qty:float, client:Any,
                        price_tick:float=0.0, qty_step:float=0.0):
    try:
        for o in await client.futures_get_open_orders(symbol=symbol):
            if o.get("type") in ("TAKE_PROFIT","TAKE_PROFIT_MARKET") and o.get("reduceOnly"):
                await client.futures_cancel_order(symbol=symbol, orderId=o["orderId"])
    except Exception as e:
        logger.debug(f"cancel existing TPs failed: {e}")
    weights = [0.5, 0.3, 0.2, 0.0][:len(tp_prices)]
    if len(tp_prices)==4: weights=[0.4,0.3,0.2,0.1]
    wsum = sum(weights) or 1.0
    weights=[w/wsum for w in weights]
    for i,(tp,w) in enumerate(zip(tp_prices, weights), start=1):
        price = round_to_tick(tp, price_tick or 0.0)
        q = clamp_to_step(qty*w, qty_step or 0.0) if qty_step else qty*w
        try:
            await client.futures_create_order(
                symbol=symbol,
                side="SELL" if side=="LONG" else "BUY",
                type="TAKE_PROFIT_LIMIT",
                price=price, stopPrice=price,
                quantity=q, timeInForce="GTC", reduceOnly=True
            )
        except Exception as e:
            logger.error(f"TP{i} place failed {symbol}: {e}")
EOF

from typing import Literal
RegimeType = Literal["TRENDING","CHOPPY","VOLATILE","SIDEWAYS"]
BASE = {
  "TRENDING": {"sl_mult":2.0, "tps":[2.5,4.0,6.0,8.0]},
  "CHOPPY":   {"sl_mult":0.8, "tps":[1.5,2.5]},
  "VOLATILE": {"sl_mult":3.0, "tps":[4.0,7.0]},
  "SIDEWAYS": {"sl_mult":1.2, "tps":[2.0,3.2]},
}
def pick_params(regime:RegimeType, quality:float):
    p = dict(BASE.get(regime, BASE["TRENDING"]))
    if quality >= 8.5:
        p["sl_mult"] *= 1.10
        p["tps"] = [x*1.20 for x in p["tps"]]
    elif quality <= 7.5:
        p["sl_mult"] *= 0.90
        p["tps"] = [x*0.85 for x in p["tps"]]
    return p
EOF

async def _mini_dynamic_manage(position, client, exch_filters:dict|None=None):

    symbol = position["symbol"]
