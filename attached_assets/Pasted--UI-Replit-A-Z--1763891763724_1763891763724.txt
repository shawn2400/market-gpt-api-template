# ✅ **PART 5 — Terminal חי (כמו Replit)**

📌 `src/components/Terminal.js`

```javascript
import { useEffect, useRef } from "react";
import { Terminal as XTerm } from "xterm";
import { FitAddon } from "xterm-addon-fit";
import { api } from "../api/client";

export default function Terminal() {
  const termRef = useRef(null);

  useEffect(() => {
    const term = new XTerm({ theme: { background: "#0B0F17" } });
    const fit = new FitAddon();
    term.loadAddon(fit);
    term.open(termRef.current);
    fit.fit();

    term.write("✅ Connected to Algo-Agent\n");

    term.onData(async (cmd) => {
      if (cmd === "\r") {
        term.write("\n");
        const res = await api.post("/exec", { command: "" });
        term.write(res.data.output + "\n");
      } else {
        term.write(cmd);
      }
    });
  }, []);

  return <div ref={termRef} style={{ height: "50%" }} />;
}
```
