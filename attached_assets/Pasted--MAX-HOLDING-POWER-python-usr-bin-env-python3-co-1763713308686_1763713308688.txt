#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
מערכת ניהול פוזיציות דינאמית - MAX HOLDING POWER
DYNAMIC_POSITION_MANAGER_V12

import pandas as pd
import numpy as np
import ta
from binance.client import Client
import time
import logging
from datetime import datetime, timedelta
from enum import Enum
import threading
from dataclasses import dataclass
from typing import Dict, List, Optional

class PositionState(Enum):
    EARLY_PROFIT = "early_profit"
    STRONG_TREND = "strong_trend"
    CONSOLIDATION = "consolidation"
    MATURE_PROFIT = "mature_profit"
    EXTREME_PROFIT = "extreme_profit"

@dataclass
class DynamicPosition:
    symbol: str
    entry_price: float
    quantity: float
    direction: str
    position_id: str
    timestamp: datetime
    
    # ניהול דינאמי
    current_stop_loss: float
    current_take_profit: float
    original_take_profit: float
    position_state: PositionState = PositionState.EARLY_PROFIT
    
    # מדדים דינאמיים
    momentum_strength: float = 0.0
    trend_quality: float = 0.0
    volatility_score: float = 0.0
    volume_confirmation: float = 0.0
    
    # סטטיסטיקות ביצועים
    unrealized_pnl: float = 0.0
    pnl_percentage: float = 0.0
    time_in_trade: float = 0.0
    max_favorable_move: float = 0.0

class MaxHoldingManager:
    def __init__(self, api_key: str, api_secret: str):
        self.client = Client(api_key, api_secret)
        self.active_positions: Dict[str, DynamicPosition] = {}
        self.performance_metrics = {}
        
        # הגדרות דינאמיות
        self.holding_config = {
            'min_hold_time': 300,  # 5 דקות מינימום
            'max_hold_time': 86400,  # 24 שעות מקסימום
            'profit_thresholds': [0.02, 0.05, 0.10, 0.20],  # 2%, 5%, 10%, 20%
            'momentum_requirements': [0.6, 0.7, 0.8, 0.9],
            'volume_requirements': [1.5, 2.0, 2.5, 3.0]
        }
        
        self.setup_advanced_logging()
        self.start_dynamic_monitoring()
    
    def setup_advanced_logging(self):
        """לוגינג מתקדם"""
        logging.basicConfig(
            level=logging.INFO,
            format='%(asctime)s | %(levelname)s | %(message)s',
            handlers=[
                logging.FileHandler('max_holding.log', encoding='utf-8'),
                logging.StreamHandler()
            ]
        )
        self.logger = logging.getLogger(__name__)
    
    def start_dynamic_monitoring(self):
        """התחל ניטור דינאמי"""
        def monitor_positions():
            while True:
                try:
                    self.analyze_all_positions()
                    time.sleep(10)  # בדיקה כל 10 שניות
                except Exception as e:
                    self.logger.error(f"❌ שגיאה בניטור: {e}")
                    time.sleep(30)
        
        threading.Thread(target=monitor_positions, daemon=True).start()
        self.logger.info("🔍 ניטור דינאמי הופעל")
    
    def analyze_all_positions(self):
        """ניתוח כל הפוזיציות הפעילות"""
        for position_id, position in list(self.active_positions.items()):
            try:
                # עדכן נתונים נוכחיים
                self.update_position_data(position)
                
                # חשב מדדים דינאמיים
                self.calculate_dynamic_metrics(position)
                
                # קבע state חדש
                new_state = self.determine_position_state(position)
                
                # בצע התאמות if needed
                if new_state != position.position_state:
                    self.handle_state_transition(position, new_state)
                
                # החלט whether to hold or adjust
                self.make_holding_decision(position)
                
            except Exception as e:
                self.logger.error(f"❌ שגיאה בניתוח פוזיציה {position_id}: {e}")
    
    def update_position_data(self, position: DynamicPosition):
        """עדכן נתוני פוזיציה נוכחיים"""
        try:
            current_price = self.get_current_price(position.symbol)
            
            # חשב PNL
            if position.direction == "LONG":
                position.unrealized_pnl = (current_price - position.entry_price) * position.quantity
                position.pnl_percentage = (current_price - position.entry_price) / position.entry_price
            else:
                position.unrealized_pnl = (position.entry_price - current_price) * position.quantity
                position.pnl_percentage = (position.entry_price - current_price) / position.entry_price
            
            # עדכן time in trade
            position.time_in_trade = (datetime.now() - position.timestamp).total_seconds()
            
            # עדכן max favorable move
            if position.pnl_percentage > position.max_favorable_move:
                position.max_favorable_move = position.pnl_percentage
            
        except Exception as e:
            self.logger.error(f"❌ שגיאה בעדכון נתונים: {e}")
    
    def calculate_dynamic_metrics(self, position: DynamicPosition):
        """חשב מדדים דינאמיים להחלטת החזקה"""
        try:
            # קבל נתונים טכניים
            data = self.get_technical_data(position.symbol, '15m', 50)
            
            # 1. חוזק מומנטום
            position.momentum_strength = self.calculate_momentum_strength(data)
            
            # 2. איכות טרנד
            position.trend_quality = self.calculate_trend_quality(data)
            
            # 3. מדד וולאטיליות
            position.volatility_score = self.calculate_volatility_score(data)
            
            # 4. אישור ווליום
            position.volume_confirmation = self.calculate_volume_confirmation(data)
            
        except Exception as e:
            self.logger.error(f"❌ שגיאה בחישוב מדדים: {e}")
    
    def calculate_momentum_strength(self, data: pd.DataFrame) -> float:
        """חישוב חוזק מומנטום"""
        try:
            rsi = ta.momentum.RSIIndicator(data['close'], window=14).rsi().iloc[-1]
            macd = ta.trend.MACD(data['close']).macd().iloc[-1]
            macd_signal = ta.trend.MACD(data['close']).macd_signal().iloc[-1]
            
            # נורמליזציה של מדדים
            rsi_score = 1 - abs(rsi - 50) / 50  # קרוב יותר ל-50 = טוב יותר
            macd_score = 1.0 if macd > macd_signal else 0.3
            
            momentum_score = (rsi_score * 0.4 + macd_score * 0.6)
            return max(0.0, min(1.0, momentum_score))
            
        except:
            return 0.5
    
    def calculate_trend_quality(self, data: pd.DataFrame) -> float:
        """חישוב איכות טרנד"""
        try:
            # EMA alignment
            ema_20 = ta.trend.EMAIndicator(data['close'], window=20).ema_indicator().iloc[-1]
            ema_50 = ta.trend.EMAIndicator(data['close'], window=50).ema_indicator().iloc[-1]
            ema_100 = ta.trend.EMAIndicator(data['close'], window=100).ema_indicator().iloc[-1]
            
            current_price = data['close'].iloc[-1]
            
            # ציון alignement
            alignment_score = 0.0
            if current_price > ema_20 > ema_50 > ema_100:
                alignment_score = 1.0
            elif current_price < ema_20 < ema_50 < ema_100:
                alignment_score = 1.0
            elif (current_price > ema_20 > ema_50) or (current_price < ema_20 < ema_50):
                alignment_score = 0.7
            else:
                alignment_score = 0.3
            
            # ADX for trend strength
            adx = ta.trend.ADXIndicator(data['high'], data['low'], data['close']).adx().iloc[-1]
            adx_score = min(1.0, adx / 50)  # נורמליזציה ל-50
            
            trend_quality = (alignment_score * 0.6 + adx_score * 0.4)
            return max(0.0, min(1.0, trend_quality))
            
        except:
            return 0.5
    
    def calculate_volatility_score(self, data: pd.DataFrame) -> float:
        """חישוב מדד וולאטיליות אופטימלי"""
        try:
            atr = ta.volatility.AverageTrueRange(data['high'], data['low'], data['close']).average_true_range().iloc[-1]
            current_price = data['close'].iloc[-1]
            atr_percent = atr / current_price
            
            # וולאטיליות אופטימלית בין 1% ל-3%
            if 0.01 <= atr_percent <= 0.03:
                volatility_score = 1.0
            elif 0.005 <= atr_percent <= 0.05:
                volatility_score = 0.7
            else:
                volatility_score = 0.3
            
            return volatility_score
            
        except:
            return 0.5
    
    def calculate_volume_confirmation(self, data: pd.DataFrame) -> float:
        """חישוב אישור ווליום"""
        try:
            current_volume = data['volume'].iloc[-1]
            volume_ma = data['volume'].rolling(20).mean().iloc[-1]
            volume_ratio = current_volume / volume_ma
            
            if volume_ratio >= 2.0:
                volume_score = 1.0
            elif volume_ratio >= 1.5:
                volume_score = 0.8
            elif volume_ratio >= 1.2:
                volume_score = 0.6
            else:
                volume_score = 0.3
            
            return volume_score
            
        except:
            return 0.5
    
    def determine_position_state(self, position: DynamicPosition) -> PositionState:
        """קבע את state הנוכחי של הפוזיציה"""
        
        # בדיקת תנאים לכל state
        if position.pnl_percentage >= 0.20 and self.is_extreme_conditions(position):
            return PositionState.EXTREME_PROFIT
        
        elif position.pnl_percentage >= 0.10 and self.is_strong_conditions(position):
            return PositionState.MATURE_PROFIT
        
        elif position.pnl_percentage >= 0.05 and self.is_good_conditions(position):
            return PositionState.STRONG_TREND
        
        elif position.pnl_percentage >= 0.02 and self.is_acceptable_conditions(position):
            return PositionState.CONSOLIDATION
        
        else:
            return PositionState.EARLY_PROFIT
    
    def is_extreme_conditions(self, position: DynamicPosition) -> bool:
        """בדיקה אם התנאים מצדיקים החזקה קיצונית"""
        return (position.momentum_strength >= 0.9 and 
                position.trend_quality >= 0.9 and 
                position.volume_confirmation >= 2.5 and
                position.time_in_trade >= 3600)  # שעה לפחות
    
    def is_strong_conditions(self, position: DynamicPosition) -> bool:
        """בדיקה אם התנאים חזקים"""
        return (position.momentum_strength >= 0.8 and 
                position.trend_quality >= 0.8 and 
                position.volume_confirmation >= 2.0)
    
    def is_good_conditions(self, position: DynamicPosition) -> bool:
        """בדיקה אם התנאים טובים"""
        return (position.momentum_strength >= 0.7 and 
                position.trend_quality >= 0.7 and 
                position.volume_confirmation >= 1.5)
    
    def is_acceptable_conditions(self, position: DynamicPosition) -> bool:
        """בדיקה אם התנאים מקובלים"""
        return (position.momentum_strength >= 0.6 and 
                position.trend_quality >= 0.6 and 
                position.volume_confirmation >= 1.2)
    
    def handle_state_transition(self, position: DynamicPosition, new_state: PositionState):
        """טיפול במעבר between states"""
        old_state = position.position_state
        position.position_state = new_state
        
        self.logger.info(f"🔄 מעבר state: {position.symbol} {old_state} -> {new_state}")
        
        # בצע התאמות based on state חדש
        self.adjust_position_management(position)
    
    def adjust_position_management(self, position: DynamicPosition):
        """התאם את ניהול הפוזיציה based on state"""
        try:
            current_price = self.get_current_price(position.symbol)
            
            if position.position_state == PositionState.EXTREME_PROFIT:
                # במצב קיצוני - הרחק מאוד את ה-SL ואפשר רווחים נוספים
                new_sl = current_price * 0.90  # רק 10% stop loss
                new_tp = current_price * 1.50  # 50% take profit
                
            elif position.position_state == PositionState.MATURE_PROFIT:
                # רווח בוגר - trailing stop aggressive
                new_sl = current_price * 0.95  # 5% stop loss
                new_tp = current_price * 1.30  # 30% take profit
                
            elif position.position_state == PositionState.STRONG_TREND:
                # טרנד חזק - trailing stop standard
                new_sl = current_price * 0.97  # 3% stop loss
                new_tp = current_price * 1.20  # 20% take profit
                
            elif position.position_state == PositionState.CONSOLIDATION:
                # איחוד - הדק stop loss
                new_sl = current_price * 0.98  # 2% stop loss
                new_tp = current_price * 1.10  # 10% take profit
                
            else:  # EARLY_PROFIT
                # רווח מוקדם - הגדרות שמרניות
                new_sl = current_price * 0.99  # 1% stop loss
                new_tp = position.original_take_profit
            
            # עדכן את ההגנות
            if position.direction == "LONG":
                if new_sl > position.current_stop_loss:
                    self.update_stop_loss(position, new_sl)
                if new_tp > position.current_take_profit:
                    self.update_take_profit(position, new_tp)
            else:  # SHORT
                if new_sl < position.current_stop_loss:
                    self.update_stop_loss(position, new_sl)
                if new_tp < position.current_take_profit:
                    self.update_take_profit(position, new_tp)
                    
        except Exception as e:
            self.logger.error(f"❌ שגיאה בהתאמת ניהול: {e}")
    
    def make_holding_decision(self, position: DynamicPosition):
        """החלט whether to hold, add, or reduce"""
        try:
            hold_confidence = self.calculate_hold_confidence(position)
            
            if hold_confidence >= 0.8 and position.position_state in [PositionState.STRONG_TREND, PositionState.MATURE_PROFIT]:
                # תנאים מעולים - שקול להוסיף לפוזיציה
                if self.should_add_to_position(position):
                    self.consider_adding_to_position(position)
                    
            elif hold_confidence >= 0.6:
                # תנאים טובים - החזק חזק
                self.logger.info(f"💎 מחזיק חזק: {position.symbol} | ביטחון: {hold_confidence:.2f}")
                
            elif hold_confidence <= 0.3:
                # תנאים גרועים - שקול יציאה
                if self.should_early_exit(position):
                    self.consider_early_exit(position)
                    
        except Exception as e:
            self.logger.error(f"❌ שגיאה בהחלטת החזקה: {e}")
    
    def calculate_hold_confidence(self, position: DynamicPosition) -> float:
        """חשב ביטחון בהחזקת הפוזיציה"""
        scores = []
        
        # 1. מדד מומנטום (30%)
        scores.append(position.momentum_strength * 0.3)
        
        # 2. איכות טרנד (25%)
        scores.append(position.trend_quality * 0.25)
        
        # 3. אישור ווליום (20%)
        scores.append(position.volume_confirmation * 0.2)
        
        # 4. מדד וולאטיליות (15%)
        scores.append(position.volatility_score * 0.15)
        
        # 5. זמן במסחר (10%) - יותר זמן = יותר ביטחון (עד נקודה מסוימת)
        time_score = min(1.0, position.time_in_trade / 7200)  # 2 שעות מקסימום
        scores.append(time_score * 0.1)
        
        return sum(scores)
    
    def should_add_to_position(self, position: DynamicPosition) -> bool:
        """האם להוסיף לפוזיציה?"""
        return (position.pnl_percentage >= 0.05 and 
                position.momentum_strength >= 0.8 and
                position.volume_confirmation >= 2.0 and
                position.time_in_trade >= 900)  # 15 דקות לפחות
    
    def consider_adding_to_position(self, position: DynamicPosition):
        """שקול הוספה לפוזיציה"""
        try:
            current_price = self.get_current_price(position.symbol)
            
            # חשב גודל תוספת (50% מהפוזיציה המקורית)
            add_quantity = position.quantity * 0.5
            
            # בצע קניה נוספת
            if position.direction == "LONG":
                order = self.client.order_market_buy(
                    symbol=position.symbol,
                    quantity=add_quantity
                )
            else:
                order = self.client.order_market_sell(
                    symbol=position.symbol,
                    quantity=add_quantity
                )
            
            # עדכן את הפוזיציה
            position.quantity += add_quantity
            position.entry_price = ((position.entry_price * (position.quantity - add_quantity)) + 
                                  (current_price * add_quantity)) / position.quantity
            
            self.logger.info(f"➕ הוספה לפוזיציה: {position.symbol} | כמות: {add_quantity}")
            
            # עדכן הגנות
            self.adjust_position_management(position)
            
        except Exception as e:
            self.logger.error(f"❌ שגיאה בהוספת פוזיציה: {e}")
    
    def should_early_exit(self, position: DynamicPosition) -> bool:
        """האם לצאת מוקדם מהפוזיציה?"""
        # צא מוקדם רק אם:
        # 1. המומנטום התהפך
        # 2. הווליום לא תומך
        # 3. יש איתות חזק נגד הכיוון
        return (position.momentum_strength <= 0.3 and 
                position.volume_confirmation <= 0.8 and
                position.pnl_percentage < 0.02)  # לא בשיא רווח
    
    def consider_early_exit(self, position: DynamicPosition):
        """שקול יציאה מוקדמת"""
        try:
            self.logger.warning(f"⚠️ שקול יציאה מוקדמת: {position.symbol}")
            
            # כאן ניתן להוסיף לוגיקה ליציאה מוקדמת
            # כרגע רק מתריע
            
        except Exception as e:
            self.logger.error(f"❌ שגיאה בבחינת יציאה מוקדמת: {e}")
    
    def update_stop_loss(self, position: DynamicPosition, new_sl: float):
        """עדכן stop loss"""
        try:
            # בטל stop loss ישן
            open_orders = self.client.get_open_orders(symbol=position.symbol)
            for order in open_orders:
                if order['type'] in ['STOP_LOSS_LIMIT', 'STOP_LOSS']:
                    self.client.cancel_order(symbol=position.symbol, orderId=order['orderId'])
            
            # צור stop loss חדש
            if position.direction == "LONG":
                self.client.create_order(
                    symbol=position.symbol,
                    side=Client.SIDE_SELL,
                    type=Client.ORDER_TYPE_STOP_LOSS_LIMIT,
                    timeInForce='GTC',
                    quantity=position.quantity,
                    stopPrice=round(new_sl, 6),
                    price=round(new_sl, 6)
                )
            else:
                self.client.create_order(
                    symbol=position.symbol,
                    side=Client.SIDE_BUY,
                    type=Client.ORDER_TYPE_STOP_LOSS_LIMIT,
                    timeInForce='GTC',
                    quantity=position.quantity,
                    stopPrice=round(new_sl, 6),
                    price=round(new_sl, 6)
                )
            
            position.current_stop_loss = new_sl
            self.logger.info(f"🔄 Stop Loss עודכן: {position.symbol} -> {new_sl:.6f}")
            
        except Exception as e:
            self.logger.error(f"❌ שגיאה בעדכון Stop Loss: {e}")
    
    def update_take_profit(self, position: DynamicPosition, new_tp: float):
        """עדכן take profit"""
        try:
            # בטל take profit ישן
            open_orders = self.client.get_open_orders(symbol=position.symbol)
            for order in open_orders:
                if order['type'] in ['TAKE_PROFIT_LIMIT', 'TAKE_PROFIT']:
                    self.client.cancel_order(symbol=position.symbol, orderId=order['orderId'])
            
            # צור take profit חדש
            if position.direction == "LONG":
                self.client.create_order(
                    symbol=position.symbol,
                    side=Client.SIDE_SELL,
                    type=Client.ORDER_TYPE_TAKE_PROFIT_LIMIT,
                    timeInForce='GTC',
                    quantity=position.quantity,
                    stopPrice=round(new_tp, 6),
                    price=round(new_tp, 6)
                )
            else:
                self.client.create_order(
                    symbol=position.symbol,
                    side=Client.SIDE_BUY,
                    type=Client.ORDER_TYPE_TAKE_PROFIT_LIMIT,
                    timeInForce='GTC',
                    quantity=position.quantity,
                    stopPrice=round(new_tp, 6),
                    price=round(new_tp, 6)
                )
            
            position.current_take_profit = new_tp
            self.logger.info(f"🎯 Take Profit עודכן: {position.symbol} -> {new_tp:.6f}")
            
        except Exception as e:
            self.logger.error(f"❌ שגיאה בעדכון Take Profit: {e}")
    
    # Utility functions
    def get_current_price(self, symbol: str) -> float:
        """קבל מחיר נוכחי"""
        try:
            ticker = self.client.get_symbol_ticker(symbol=symbol)
            return float(ticker['price'])
        except:
            return 0.0
    
    def get_technical_data(self, symbol: str, interval: str, limit: int) -> pd.DataFrame:
        """קבל נתונים טכניים"""
        try:
            klines = self.client.get_klines(symbol=symbol, interval=interval, limit=limit)
            df = pd.DataFrame(klines, columns=[
                'timestamp', 'open', 'high', 'low', 'close', 'volume',
                'close_time', 'quote_volume', 'trades', 'taker_buy', 'taker_quote', 'ignore'
            ])
            
            for col in ['open', 'high', 'low', 'close', 'volume']:
                df[col] = pd.to_numeric(df[col])
            
            return df
            
        except Exception as e:
            self.logger.error(f"❌ שגיאה בקבלת נתונים: {e}")
            return pd.DataFrame()
    
    def add_position(self, symbol: str, entry_price: float, quantity: float, direction: str):
        """הוסף פוזיציה חדשה לניהול"""
        position_id = f"POS_{datetime.now().strftime('%Y%m%d_%H%M%S')}"
        
        # חשב הגנות התחלתיות
        if direction == "LONG":
            stop_loss = entry_price * 0.98
            take_profit = entry_price * 1.04
        else:
            stop_loss = entry_price * 1.02
            take_profit = entry_price * 0.96
        
        position = DynamicPosition(
            symbol=symbol,
            entry_price=entry_price,
            quantity=quantity,
            direction=direction,
            position_id=position_id,
            timestamp=datetime.now(),
            current_stop_loss=stop_loss,
            current_take_profit=take_profit,
            original_take_profit=take_profit
        )
        
        self.active_positions[position_id] = position
        
        # רשום הגנות התחלתיות
        self.place_initial_protections(position)
        
        self.logger.info(f"🎯 פוזיציה נוספה לניהול: {symbol} | ID: {position_id}")
        
        return position_id
    
    def place_initial_protections(self, position: DynamicPosition):
        """רשום הגנות התחלתיות"""
        try:
            if position.direction == "LONG":
                # Stop Loss
                self.client.create_order(
                    symbol=position.symbol,
                    side=Client.SIDE_SELL,
                    type=Client.ORDER_TYPE_STOP_LOSS_LIMIT,
                    timeInForce='GTC',
                    quantity=position.quantity,
                    stopPrice=round(position.current_stop_loss, 6),
                    price=round(position.current_stop_loss, 6)
                )
                
                # Take Profit
                self.client.create_order(
                    symbol=position.symbol,
                    side=Client.SIDE_SELL,
                    type=Client.ORDER_TYPE_TAKE_PROFIT_LIMIT,
                    timeInForce='GTC',
                    quantity=position.quantity,
                    stopPrice=round(position.current_take_profit, 6),
                    price=round(position.current_take_profit, 6)
                )
            
            self.logger.info(f"🛡️ הגנות ראשוניות רשומות עבור {position.position_id}")
            
        except Exception as e:
            self.logger.error(f"❌ שגיאה ברישום הגנות: {e}")
