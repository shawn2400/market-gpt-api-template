```python
# core/lightweight_ai.py
import pandas as pd
import numpy as np
from sklearn.ensemble import RandomForestClassifier, GradientBoostingClassifier
from sklearn.preprocessing import StandardScaler
import logging
import joblib
from datetime import datetime
import redis

class LightweightTradingAI:
    """
    AI קל משקל למסחר - עובד עם רק 200-300MB זיכרון
    """
    
    def __init__(self, redis_client=None):
        self.redis = redis_client
        self.models = {}
        self.scaler = StandardScaler()
        self.is_trained = False
        self.model_size = "250MB"
        self.logger = logging.getLogger(__name__)
        
        # אתחול מודלים קלים
        self._initialize_models()
    
    def _initialize_models(self):
        """מאתחל מודלים קלים - לא טוענים הרבה זיכרון"""
        try:
            # מודל חיזוי מגמה - מאוד קל
            self.models['trend'] = RandomForestClassifier(
                n_estimators=30,      # פחות trees - פחות זיכרון
                max_depth=8,          # פחות deep - פחות זיכרון
                min_samples_split=20, # יותר samples - פחות overfitting
                random_state=42,
                n_jobs=1              # רק core אחד - פחות זיכרון
            )
            
            # מודל סיווג אותות - קל
            self.models['signal'] = GradientBoostingClassifier(
                n_estimators=50,
                max_depth=5,
                learning_rate=0.1,
                random_state=42
            )
            
            self.logger.info("✅ AI קל משקל אותחל - מוכן לאימון")
            
        except Exception as e:
            self.logger.error(f"❌ שגיאה באתחול AI: {e}")
    
    def create_lightweight_features(self, market_data):
        """
        יוצר תכונות קלות - לא דורש הרבה זיכרון
        """
        features = {}
        
        # תכונות מחיר בסיסיות
        features['price'] = market_data.get('close', 0)
        features['volume'] = market_data.get('volume', 0)
        features['rsi'] = market_data.get('rsi', 50)
        features['macd'] = market_data.get('macd', 0)
        features['atr'] = market_data.get('atr', 0)
        
        # תכונות מגמה פשוטות
        if all(k in market_data for k in ['ema_20', 'ema_50']):
            features['trend_strength'] = market_data['ema_20'] - market_data['ema_50']
            features['above_ema'] = 1 if market_data['close'] > market_data['ema_20'] else 0
        
        # תכונות ווליום
        if 'volume_ma_20' in market_data:
            features['volume_ratio'] = market_data['volume'] / market_data['volume_ma_20']
        else:
            features['volume_ratio'] = 1.0
        
        # תכונות תנודתיות
        if 'bb_upper' in market_data and 'bb_lower' in market_data:
            bb_width = market_data['bb_upper'] - market_data['bb_lower']
            features['bb_position'] = (market_data['close'] - market_data['bb_lower']) / bb_width if bb_width > 0 else 0.5
        
        return pd.DataFrame([features])
    
    def train_with_memory_limit(self, X, y, max_samples=5000):
        """
        מאמן עם הגבלת זיכרון - לא יותר מ-5000 דגימות
        """
        try:
            # הגבלת דגימות לחסוך זיכרון
            if len(X) > max_samples:
                X = X.sample(max_samples, random_state=42)
                y = y.loc[X.index]
            
            # קנה מידה לתכונות
            X_scaled = self.scaler.fit_transform(X)
            
            # אימון המודלים
            self.models['trend'].fit(X_scaled, y)
            self.models['signal'].fit(X_scaled, (y > 0.5).astype(int))
            
            self.is_trained = True
            
            # שמירה ל-Redis (אופציונלי)
            self._save_models_to_redis()
            
            self.logger.info(f"✅ AI אומן עם {len(X)} דגימות - זיכרון: {self.model_size}")
            return True
            
        except Exception as e:
            self.logger.error(f"❌ שגיאה באימון AI: {e}")
            return False
    
    def predict_signal(self, market_data):
        """
        חיזוי אות מסחר - קל ומהיר
        """
        if not self.is_trained:
            return self._default_prediction()
        
        try:
            features = self.create_lightweight_features(market_data)
            features_scaled = self.scaler.transform(features)
            
            # חיזוי ממודל מגמה
            trend_pred = self.models['trend'].predict_proba(features_scaled)[0]
            
            # חיזוי ממודל אותות
            signal_pred = self.models['signal'].predict_proba(features_scaled)[0]
            
            # שילוב החיזויים
            combined_confidence = (trend_pred[1] + signal_pred[1]) / 2
            
            # קבלת החלטה
            if combined_confidence > 0.65:
                direction = 'long'
                action = 'BUY'
            elif combined_confidence < 0.35:
                direction = 'short' 
                action = 'SELL'
            else:
                direction = 'hold'
                action = 'HOLD'
            
            return {
                'direction': direction,
                'action': action,
                'confidence': float(combined_confidence),
                'trend_confidence': float(trend_pred[1]),
                'signal_confidence': float(signal_pred[1]),
                'ai_used': True,
                'memory_footprint': self.model_size
            }
            
        except Exception as e:
            self.logger.error(f"❌ שגיאה בחיזוי AI: {e}")
            return self._default_prediction()
    
    def _default_prediction(self):
        """חיזוי ברירת מחדל כאשר AI לא מאומן"""
        return {
            'direction': 'hold',
            'action': 'HOLD', 
            'confidence': 0.5,
            'trend_confidence': 0.5,
            'signal_confidence': 0.5,
            'ai_used': False,
            'memory_footprint': '0MB'
        }
    
    def _save_models_to_redis(self):
        """שומר מודלים ל-Redis (אופציונלי)"""
        try:
            if not self.redis:
                return
                
            # שמירת Scaler
            scaler_data = joblib.dumps(self.scaler)
            self.redis.set('ai:scaler', scaler_data)
            
            self.logger.info("✅ מודלי AI נשמרו ל-Redis")
        except Exception as e:
            self.logger.warning(f"⚠️ לא ניתן לשמור מודלים ל-Redis: {e}")
    
    def load_models_from_redis(self):
        """טוען מודלים מ-Redis (אופציונלי)"""
        try:
            if not self.redis:
                return False
                
            scaler_data = self.redis.get('ai:scaler')
            if scaler_data:
                self.scaler = joblib.loads(scaler_data)
                self.is_trained = True
                self.logger.info("✅ מודלי AI נטענו מ-Redis")
                return True
        except Exception as e:
            self.logger.warning(f"⚠️ לא ניתן לטעון מודלים מ-Redis: {e}")
        
        return False

# יצירת instance גלובלי
trading_ai = LightweightTradingAI()
```

```python
# core/ai_manager.py
import asyncio
import logging
from core.lightweight_ai import trading_ai

class AIManager:
    """
    מנהל את חיי ה-AI במערכת
    """
    
    def __init__(self):
        self.logger = logging.getLogger(__name__)
        self.ai_active = False
        
    async def initialize_ai_system(self):
        """
        מאתחל את מערכת ה-AI
        """
        try:
            self.logger.info("🚀 מאתחל מערכת AI קלה...")
            
            # נסה לטעון מודלים קיימים
            models_loaded = trading_ai.load_models_from_redis()
            
            if not models_loaded:
                self.logger.info("🤖 אין מודלים קיימים - AI יופעל ללא אימון ראשוני")
                # כאן ניתן להוסיף אימון ראשוני אם יש נתונים היסטוריים
            else:
                self.logger.info("✅ AI אותחל עם מודלים קיימים")
            
            self.ai_active = True
            return True
            
        except Exception as e:
            self.logger.error(f"❌ שגיאה באתחול AI: {e}")
            self.ai_active = False
            return False
    
    async def train_ai_with_data(self, historical_data):
        """
        מאמן את ה-AI עם נתונים היסטוריים
        """
        if not self.ai_active:
            await self.initialize_ai_system()
        
        try:
            # כאן הלוגיקה להכנת נתוני אימון
            X = historical_data['features']
            y = historical_data['labels']
            
            success = trading_ai.train_with_memory_limit(X, y)
            
            if success:
                self.logger.info("✅ AI אומן בהצלחה עם נתונים היסטוריים")
            else:
                self.logger.warning("⚠️ AI לא אומן - ישמש בחיזוי ברירת מחדל")
                
            return success
            
        except Exception as e:
            self.logger.error(f"❌ שגיאה באימון AI: {e}")
            return False

# יצירת מנהל AI גלובלי
ai_manager = AIManager()
```

```python
# workers/ai_enhanced_scanner.py
import asyncio
import logging
from datetime import datetime
from core.lightweight_ai import trading_ai

class AIEnhancedScanner:
    """
    סורק משובע עם AI קל משקל
    """
    
    def __init__(self):
        self.logger = logging.getLogger(__name__)
        self.ai_enabled = True
        self.min_ai_confidence = 0.6
        
    async def scan_with_ai_enhancement(self, symbol, market_data):
        """
        סורק סמל עם שיפור AI
        """
        try:
            # ניתוח בסיסי
            basic_signal = await self._basic_analysis(symbol, market_data)
            
            # אם AI פעיל וה-signal בסיסי חזק מספיק
            if (self.ai_enabled and 
                basic_signal.get('confidence', 0) > 0.4):
                
                # קבלת חיזוי AI
                ai_prediction = trading_ai.predict_signal(market_data)
                
                if ai_prediction['ai_used']:
                    # שילוב התוצאות
                    enhanced_signal = self._combine_signals(basic_signal, ai_prediction)
                    return enhanced_signal
            
            # אם AI לא פעיל או confidence נמוך - חזור לבסיסי
            return basic_signal
            
        except Exception as e:
            self.logger.error(f"❌ שגיאה בסריקה עם AI: {e}")
            return await self._basic_analysis(symbol, market_data)
    
    def _combine_signals(self, basic_signal, ai_signal):
        """משלב אות בסיסי עם חיזוי AI"""
        # משקלל את ה-confidence
        basic_conf = basic_signal.get('confidence', 0.5)
        ai_conf = ai_signal.get('confidence', 0.5)
        
        # weighting - 70% AI, 30% בסיסי אם AI confidence גבוה
        if ai_conf > self.min_ai_confidence:
            combined_confidence = (ai_conf * 0.7) + (basic_conf * 0.3)
        else:
            combined_confidence = (ai_conf * 0.3) + (basic_conf * 0.7)
        
        # החלטת כיוון סופית
        if ai_conf > self.min_ai_confidence:
            final_direction = ai_signal['direction']
        else:
            final_direction = basic_signal.get('direction', 'hold')
        
        return {
            'symbol': basic_signal['symbol'],
            'direction': final_direction,
            'confidence': combined_confidence,
            'basic_confidence': basic_conf,
            'ai_confidence': ai_conf,
            'ai_analysis': ai_signal,
            'enhanced': True,
            'timestamp': datetime.now().isoformat()
        }
    
    async def _basic_analysis(self, symbol, market_data):
        """
        ניתוח בסיסי ללא AI (הקוד הקיים שלך)
        """
        # כאן הקוד הקיים של הניתוח הטכני
        # מחליף ל-logic האמיתי שלך
        
        # דוגמה בסיסית
        rsi = market_data.get('rsi', 50)
        if rsi < 30:
            direction = 'long'
            confidence = 0.7
        elif rsi > 70:
            direction = 'short'
            confidence = 0.7
        else:
            direction = 'hold'
            confidence = 0.5
        
        return {
            'symbol': symbol,
            'direction': direction,
            'confidence': confidence,
            'reason': 'technical_analysis'
        }
```
