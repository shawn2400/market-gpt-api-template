echo "🔁 Setting Render as sole trade executor..."
set -euo pipefail

# === 1️⃣ נטרול ביצוע טריידים מ-Replit ===
export DRY_RUN=true
export EXECUTE_TRADES=false

# === 2️⃣ הפעלת מצב EXECUTOR מלא ב-Render ===
export RENDER_TRADE_MASTER=1
export RENDER_EXECUTE_TRADES=true
export RENDER_ALLOW_FULL_ACCESS=true

# === 3️⃣ שמירת קובץ קונפיג להוראות AlgoGPT ===
mkdir -p config
cat > config/trade_handoff.json <<'JSON'
{
  "mode": "handoff",
  "render_executor_only": true,
  "replit_interface_only": true,
  "telegram_interface_only": true,
  "render_has_full_binance_access": true,
  "execute_on_render": true,
  "dry_run_on_replit": true,
  "allow_cross_env_commands": true
}
JSON

# === 4️⃣ בדיקת סביבת Render ===
echo "🔍 Checking environment..."
if [ -n "${RENDER:-}" ] || [ -n "${RENDER_EXTERNAL_URL:-}" ]; then
  echo "✅ Detected Render environment"
  echo "🌐 External URL: ${RENDER_EXTERNAL_URL:-unknown}"
else
  echo "⚠️ Not running inside Render! (running in Replit or local)"
  echo "❌ Live trades disabled here."
fi

# === 5️⃣ בדיקת משתני סביבה חיוניים ב-Render ===
echo "🧩 Checking required environment variables..."
missing=()
for var in BINANCE_API_KEY BINANCE_API_SECRET EXECUTE_TRADES RENDER_TRADE_MASTER DRY_RUN; do
  if [ -z "${!var:-}" ]; then
    missing+=("$var")
  fi
done

if [ ${#missing[@]} -gt 0 ]; then
  echo "❌ Missing variables in Render environment:"
  printf '  - %s\n' "${missing[@]}"
  echo "➡️ Go to Render → Dashboard → Environment → add missing keys manually."
  exit 1
else
  echo "✅ All critical environment variables found."
fi

# === 6️⃣ בדיקת גישה ל-Binance ===
echo "🔍 Verifying Binance connectivity..."
TS=$(date +%s%3N)
QS="timestamp=$TS"
SIG=$(echo -n "$QS" | openssl dgst -sha256 -hmac "$BINANCE_API_SECRET" -hex | sed 's/^.* //')
RSP=$(curl -s -w "\nHTTP %{http_code}\n" -H "X-MBX-APIKEY: $BINANCE_API_KEY" \
  "https://fapi.binance.com/fapi/v2/account?$QS&signature=$SIG")

CODE=$(echo "$RSP" | tail -n1 | awk '{print $2}')
BODY=$(echo "$RSP" | head -n1)

if [[ "$CODE" == "200" ]]; then
  echo "✅ Binance Futures live connection OK"
else
  echo "❌ Binance connection failed ($CODE)"
  echo "➡️ Check that API permissions include:"
  echo "   ☑️ Enable Reading"
  echo "   ☑️ Enable Futures"
  echo "   ☑️ Unrestricted (no IP restriction)"
fi

# === 7️⃣ הדפסה סופית ===
echo "-------------------------------------------"
echo "✅ Render is now the sole live trade executor."
echo "🔒 Replit/Telegram act as control interfaces only."
echo "-------------------------------------------"
