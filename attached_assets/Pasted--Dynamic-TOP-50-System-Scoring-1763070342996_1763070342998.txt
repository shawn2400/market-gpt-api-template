### **📊 Scoring Algorithm - קריטריונים לדירוג:**

```python
class Top50Scorer:
    def calculate_score(self, symbol_data):
        weights = {
            'volume_score': 0.25,      # 24h volume ($20M+)
            'liquidity_score': 0.20,   # Order book depth  
            'performance_score': 0.20, # Win rate + historical ROI
            'volatility_score': 0.15,  # Healthy ATR (1-4%)
            'trend_score': 0.10,       # Technical strength
            'stability_score': 0.10    # Low drawdown
        }
        
        total_score = 0
        for factor, weight in weights.items():
            score = getattr(self, f'calculate_{factor}')(symbol_data)
            total_score += score * weight
            
        return total_score
```

### **🔄 Auto-Ranking Worker (רץ כל שעה):**

```python
class Top50Calculator:
    def __init__(self):
        self.redis = redis_manager
        self.binance = binance_client
    
    def calculate_top_50(self):
        # 1. קבל את כל הסמלים הזמינים
        all_symbols = self.binance.get_all_symbols()
        
        # 2. חשב ציון לכל סמל
        scored_symbols = []
        for symbol in all_symbols:
            if self.passes_minimum_criteria(symbol):
                score = self.scorer.calculate_score(symbol)
                scored_symbols.append((symbol, score))
        
        # 3. מיין וקח רק את ה-TOP 50
        scored_symbols.sort(key=lambda x: x[1], reverse=True)
        top_50 = [symbol for symbol, score in scored_symbols[:50]]
        
        # 4. שמור ב-Redis + Database
        self.redis.set('top_50_symbols', json.dumps(top_50), ex=3700)  # 1 hour + buffer
        self.save_to_database(top_50)
        
        return top_50
    
    def passes_minimum_criteria(self, symbol):
        """סף מינימלי - זבל נופל כאן"""
        min_volume = 20_000_000  # $20M
        min_liquidity = 100_000  # $100k order book
        
        volume_24h = self.binance.get_24h_volume(symbol)
        liquidity = self.binance.get_liquidity_depth(symbol)
        
        return volume_24h >= min_volume and liquidity >= min_liquidity
```

### **🚫 Strict Gatekeeper Integration:**

```python
class Top50Gatekeeper:
    def __init__(self):
        self.redis = redis_manager
    
    def is_approved(self, symbol):
        """האם הסמל נמצא ב-TOP 50 הנוכחי?"""
        top_50 = self.get_current_top_50()
        return symbol in top_50
    
    def validate_trade(self, symbol, order_type, amount):
        """וולידציה לפני כל trade"""
        if not self.is_approved(symbol):
            return {
                "approved": False,
                "reason": f"Symbol {symbol} not in TOP 50",
                "blocked": True
            }
        
        # הגנות נוספות
        if order_type == "GRID" and not self.allow_grid_trading(symbol):
            return {
                "approved": False, 
                "reason": "GRID trading blocked for non-TOP symbols",
                "blocked": True
            }
            
        return {"approved": True, "reason": "TOP 50 approved symbol"}
    
    def get_current_top_50(self):
        """קבל את רשימת ה-TOP 50 העדכנית"""
        cached = self.redis.get('top_50_symbols')
        if cached:
            return json.loads(cached)
        else:
            # fallback - חשב מחדש
            return self.calculator.calculate_top_50()
```

```python
class Top50Scanner:
    def __init__(self):
        self.gatekeeper = Top50Gatekeeper()
    
    def scan_opportunities(self):
        """סרוק רק את ה-TOP 50 להזדמנויות"""
        top_50 = self.gatekeeper.get_current_top_50()
        opportunities = []
        
        for symbol in top_50:
            if self.has_trading_opportunity(symbol):
                opportunities.append({
                    'symbol': symbol,
                    'signal': self.get_signal(symbol),
                    'confidence': self.get_confidence(symbol),
                    'allowed_leverage': self.calculate_leverage(symbol)
                })
        
        return opportunities
```
