```python
# config/redis_config.py
import redis

REDIS_CONFIG = {
    'host': 'redis-10327.c269.eu-west-1-3.ec2.cloud.redislabs.com',
    'port': 10327,
    'password': 'הסיסמה-האמיתית-שתראה-כשתלחץ-על-העין',  # ⬅️ כאן!
    'ssl': True,  # Redis Cloud דורש SSL
    'ssl_cert_reqs': None,
    'decode_responses': True,
    'socket_connect_timeout': 10,
    'retry_on_timeout': True
}

def get_redis_connection():
    """
    מחזיר חיבור Redis למערכת המסחר האוטונומית
    """
    try:
        client = redis.Redis(**REDIS_CONFIG)
        client.ping()  # בדיקת חיבור
        print("✅ Redis Cloud connected - ready for trading!")
        return client
    except Exception as e:
        print(f"❌ Redis connection failed: {e}")
        return None

# בדיקה מיידית
if __name__ == "__main__":
    r = get_redis_connection()
    if r:
        # בדיקת מערכת מסחר
        r.set('trading:status', 'operational')
        r.set('trading:bot:active', 'true')
        print("✅ Trading system Redis test passed!")
```
