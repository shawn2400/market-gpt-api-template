def strip_env(val:str)->str:
    parts = val.strip().split()
    out, started = [], False
    for token in parts:
        if not started and "=" in token and not token.startswith(("python","gunicorn","uvicorn","cd","bash","sh")):

        started = True
        out.append(token)
    return " ".join(out)

from __future__ import annotations
import os, logging
from typing import Literal, Dict, List
from dataclasses import dataclass

logger = logging.getLogger(__name__)

RegimeType = Literal["TRENDING", "CHOPPY", "VOLATILE", "SIDEWAYS"]
PositionSide = Literal["LONG", "SHORT"]

@dataclass
class RegimeParams:
    sl_atr_mult: float
    tps_atr_mults: List[float]
    trail_mode: str  # "aggr" | "mid" | "mild" | "off"
    use_grid: bool = False

class DynamicRegimeManager:
    _REGIME_PARAMS: Dict[RegimeType, RegimeParams] = {
        "TRENDING": RegimeParams(2.0, [2.5, 4.0, 6.0, 8.0], "aggr"),
        "CHOPPY":   RegimeParams(0.8, [1.5, 2.5], "off", use_grid=True),
        "VOLATILE": RegimeParams(3.0, [4.0, 7.0], "mid"),
        "SIDEWAYS": RegimeParams(1.2, [2.0, 3.2], "mild"),
    }

    @staticmethod
    def get_current_regime() -> RegimeType:
        forced = os.getenv("FORCE_REGIME", "").upper().strip()
        if forced in {"TRENDING","CHOPPY","VOLATILE","SIDEWAYS"}:
            logger.info(f"Using forced regime: {forced}")
            return forced  # type: ignore[return-value]
        # TODO: חבר ל-detectors האמיתיים שלך (ADX/ATR/EMA/BTC trend)
        return "TRENDING"

    @staticmethod
    def calculate_dynamic_targets(
        regime: RegimeType, quality_score: float, atr_value: float,
        swing_price: float, side: PositionSide, current_price: float
    ) -> dict:
        base = DynamicRegimeManager._REGIME_PARAMS.get(regime, DynamicRegimeManager._REGIME_PARAMS["TRENDING"])
        adj = DynamicRegimeManager._apply_quality_adjustments(quality_score, base)
        if side == "LONG":
            sl = max(0.0, swing_price - adj.sl_atr_mult * atr_value)
            tps = [current_price + m * atr_value for m in adj.tps_atr_mults]
        else:
            sl = swing_price + adj.sl_atr_mult * atr_value
            tps = [current_price - m * atr_value for m in adj.tps_atr_mults]
        return {"sl_price": sl, "tp_prices": tps, "trail_mode": adj.trail_mode, "use_grid": adj.use_grid,
                "regime": regime, "quality_score": quality_score}

    @staticmethod
    def _apply_quality_adjustments(qs: float, base: RegimeParams) -> RegimeParams:
        adj = RegimeParams(base.sl_atr_mult, base.tps_atr_mults.copy(), base.trail_mode, base.use_grid)
        if qs >= 8.5:
            adj.sl_atr_mult *= 1.10
            adj.tps_atr_mults = [m*1.20 for m in adj.tps_atr_mults]
        elif qs <= 7.5:
            adj.sl_atr_mult *= 0.90
            adj.tps_atr_mults = [m*0.85 for m in adj.tps_atr_mults]
        return adj

import logging
from typing import Dict, Any

logger = logging.getLogger(__name__)

class ZeroGapSLManager:
    """
    AtoMic SL replace: place new -> verify -> cancel old (no gap window).
    client צריך לתמוך במתודות futures_create_order / futures_get_order / futures_get_open_orders / futures_cancel_order
    """
    @staticmethod
    async def safe_replace_sl(symbol: str, position_side: str, current_price: float,
                              atr_value: float, desired_sl: float, client: Any) -> Dict:
        try:
            new = await ZeroGapSLManager._place_new_sl_order(symbol, position_side, desired_sl, client)
            if not new.get("success"):
                return {"ok": False, "error": f"New SL placement failed: {new.get('error')}"}
            new_id = new["order_id"]
            ok = await ZeroGapSLManager._verify_order_active(symbol, new_id, client)
            if not ok:
                with logger: await ZeroGapSLManager._cancel_order(symbol, new_id, client)  # type: ignore
                return {"ok": False, "error": "New SL order not verified"}
            cancelled = await ZeroGapSLManager._cancel_old_sl_orders(symbol, position_side, client)
            return {"ok": True, "new_order_id": new_id, "old_orders_cancelled": cancelled.get("cancelled_count", 0),
                    "desired_sl": desired_sl}
        except Exception as e:
            logger.error(f"Zero-Gap SL replacement failed for {symbol}: {e}")
            return {"ok": False, "error": str(e)}

    @staticmethod
    async def _place_new_sl_order(symbol: str, side: str, sl_price: float, client: Any) -> Dict:
        try:
            order = await client.futures_create_order(
                symbol=symbol,
                side="SELL" if side == "LONG" else "BUY",
                type="STOP_MARKET",
                stopPrice=sl_price,
                reduceOnly=True,
                timeInForce="GTC"
            )
            return {"success": True, "order_id": order["orderId"]}
        except Exception as e:
            logger.error(f"Failed to place new SL for {symbol}: {e}")
            return {"success": False, "error": str(e)}

    @staticmethod
    async def _verify_order_active(symbol: str, order_id: int, client: Any, max_retries: int = 3) -> bool:
        for _ in range(max_retries):
            try:
                info = await client.futures_get_order(symbol=symbol, orderId=order_id)
                if info and info.get("status") in ("NEW","PARTIALLY_FILLED"):
                    return True
            except Exception as e:
                logger.warning(f"Order verify failed: {e}")
        return False

    @staticmethod
    async def _cancel_old_sl_orders(symbol: str, position_side: str, client: Any) -> Dict:
        try:
            open_orders = await client.futures_get_open_orders(symbol=symbol)
            c = 0
            for o in open_orders:
                if o.get("type") in ("STOP_MARKET","STOP") and o.get("reduceOnly") and o.get("positionSide")==position_side:
                    await client.futures_cancel_order(symbol=symbol, orderId=o["orderId"])
                    c += 1
            return {"cancelled_count": c}
        except Exception as e:
            logger.error(f"Failed cancel old SL: {e}")
            return {"cancelled_count": 0, "error": str(e)}

    @staticmethod
    async def _cancel_order(symbol: str, order_id: int, client: Any):
        try:
            await client.futures_cancel_order(symbol=symbol, orderId=order_id)
        except Exception as e:
            logger.warning(f"cancel new SL failed silently: {e}")
EOF
