# ===========================================
# 🔧 AlgoGPT Unified AI Setup – Phase A
# ===========================================
# מטרה: לחבר את כל סוכני ה-AI (Replit, DeepSeek, AI X, GPT-5)
# לשרת הנוכחי שלך (Render/Replit) בצורה חכמה, יציבה ואוטומטית.
# ===========================================

echo "🚀 Starting Phase A – Unified AI Setup"

# --- הגדרות כלליות ---
export BASE_URL="https://algogpt-docker.onrender.com"
export APP_MODE="production"
export AGENT_ROLE="maintenance"
export AGENT_MODE="autofix"
export AIX_MODE="supervisor"

# --- מפתחות חיבור (ממולאים מראש ב-Replit Secrets או Render Secrets) ---
export GITHUB_TOKEN="$GITHUB_TOKEN"
export XAI_API_KEY="$XAI_API_KEY"
export AIX_SUPERVISOR_TOKEN="$AIX_SUPERVISOR_TOKEN"
export DEEPSEEK_API_KEY="$DEEPSEEK_API_KEY"
export OPENAI_API_KEY="$OPENAI_API_KEY"
export API_BEARER_TOKEN="$API_BEARER_TOKEN"
export BINANCE_API_KEY="$BINANCE_API_KEY"
export BINANCE_API_SECRET="$BINANCE_API_SECRET"

# --- התקנות לינוקס הכרחיות ---
sudo apt-get update -y && sudo apt-get install -y \
  curl jq procps coreutils net-tools iproute2

echo "✅ Base packages installed successfully."

# --- חיבור GitHub (Auto Deploy + Self-Fix) ---
if [ -n "$GITHUB_TOKEN" ]; then
  git config --global user.name "AlgoGPT-Auto"
  git config --global user.email "algogpt@system.local"
  git remote set-url origin "https://$GITHUB_TOKEN@github.com/shawn2400/algogpt.git"
  echo "✅ GitHub Auto-Deploy Connected"
else
  echo "⚠️ GitHub Token Missing – Skipping."
fi

# --- בדיקת Replit Agent ---
echo "🧩 Checking Replit Maintenance Agent..."
export REPLIT_AGENT_STATUS=$(ps aux | grep -v grep | grep replit | wc -l)
if [ "$REPLIT_AGENT_STATUS" -gt 0 ]; then
  echo "✅ Replit Agent Detected and Running."
else
  echo "⚠️ Replit Agent not active – please enable it."
fi

# --- בדיקת חיבור ל-AI X (Supervisor) ---
echo "🧠 Checking AI X (Grok) Connection..."
curl -s https://api.x.ai/v1/chat/completions \
  -H "Content-Type: application/json" \
  -H "Authorization: Bearer $AIX_SUPERVISOR_TOKEN" \
  -d '{"messages":[{"role":"user","content":"Say: ✅ Connected to Grok API successfully"}],"model":"grok-4-fast-non-reasoning"}' \
  | grep -q "✅ Connected" && echo "✅ AI X Connected" || echo "⚠️ AI X Connection Failed"

# --- בדיקת DeepSeek Optimizer ---
echo "🧮 Checking DeepSeek API..."
curl -s https://api.deepseek.com/v1/chat/completions \
  -H "Authorization: Bearer $DEEPSEEK_API_KEY" \
  -H "Content-Type: application/json" \
  -d '{"model":"deepseek-coder-v2","messages":[{"role":"user","content":"ping"}]}' \
  | grep -q "ping" && echo "✅ DeepSeek Connected" || echo "⚠️ DeepSeek API not responding"

# --- בדיקת GPT-5 ---
echo "🤖 Checking GPT-5 (Orchestrator)..."
curl -s https://api.openai.com/v1/chat/completions \
  -H "Content-Type: application/json" \
  -H "Authorization: Bearer $OPENAI_API_KEY" \
  -d '{"model":"gpt-5-turbo","messages":[{"role":"user","content":"AlgoGPT system test"}]}' \
  | grep -q "AlgoGPT" && echo "✅ GPT-5 Connected" || echo "⚠️ GPT-5 Connection Failed"

# --- תקינות מערכת ---
echo "📊 Running System Sanity Check..."
curl -s "$BASE_URL/readyz" && echo
curl -s "$BASE_URL/version" && echo
curl -s -H "Authorization: Bearer $API_BEARER_TOKEN" "$BASE_URL/metrics" | head -20

# --- שמירה ודוחות ---
echo "🧾 Logging System Status..."
{
  echo "=== AlgoGPT System Status ==="
  date
  uname -a
  free -h || echo "no free"
  uptime || echo "no uptime"
  df -h /app | tail -1
  echo "--- AI Connections ---"
  echo "AI X: $( [ -n "$AIX_SUPERVISOR_TOKEN" ] && echo OK || echo MISSING )"
  echo "DeepSeek: $( [ -n "$DEEPSEEK_API_KEY" ] && echo OK || echo MISSING )"
  echo "OpenAI: $( [ -n "$OPENAI_API_KEY" ] && echo OK || echo MISSING )"
  echo "Replit Agent: $( [ "$REPLIT_AGENT_STATUS" -gt 0 ] && echo RUNNING || echo INACTIVE )"
} > /app/system_state.log

echo "✅ Phase A Completed Successfully"
echo "🧠 All agents connected and ready for orchestration."
