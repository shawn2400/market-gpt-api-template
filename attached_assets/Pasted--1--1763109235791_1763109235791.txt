```python
# utils/synergistic_manager.py
class SynergisticRiskManager:
    def __init__(self):
        self.sl_tp_communication = True
        self.update_interval = 30  # שניות
        
    async def continuous_sl_tp_dialogue(self, position_data):
        """SL ו-TP מתקשרים כל 30 שניות"""
        while position_data['is_open']:
            # חישוב מצב נוכחי
            profit_pct = position_data['unrealized_pnl_pct']
            market_condition = await self.get_market_condition()
            
            # בחירת מצב סינרגטי
            if profit_pct <= 5:
                plan = self.attack_mode_synergy(position_data)
            elif 5 < profit_pct <= 15:
                plan = self.harvest_mode_synergy(position_data)
            else:
                plan = self.trail_mode_synergy(position_data)
            
            # יישום התוכנית
            await self.execute_synergistic_plan(position_data, plan)
            await asyncio.sleep(30)
```

### **שלב 3: הגדרות הטריילינג הסינרגטי**
```python
def calculate_synergistic_trailing(self, position_data):
    """טריילינג מתואם SL-TP"""
    current_profit = position_data['unrealized_pnl_pct']
    atr = self.get_current_atr(position_data['symbol'])
    
    if current_profit <= 3:
        return {'type': 'breakeven_guard', 'trigger_at': 3.0}
    elif current_profit <= 8:
        return {'type': 'soft_trail', 'distance': atr * 1.2}
    else:
        return {'type': 'aggressive_trail', 'distance': atr * 0.8}
```

4. **✅ אינטגרציה מלאה** עם כל ההגנות הקיימות:
