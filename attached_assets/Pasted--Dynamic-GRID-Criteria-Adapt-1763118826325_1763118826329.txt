```python
def calculate_optimal_grids(atr_percent, base_grids=25):
    """מתאים כמות GRIDs לתנודתיות"""
    if atr_percent < 2:   # שקט מדי
        return max(15, base_grids * 0.7)  # פחות GRIDs
    elif atr_percent > 8: # תנודתי מדי  
        return min(35, base_grids * 1.4)  # יותר GRIDs
    else:
        return base_grids  # 25 GRIDs default
```

### 📊 **יישום מיידי במערכת:**

```python
class DynamicGridAdapter:
    def __init__(self):
        self.base_criteria = {
            'min_volume': 40_000_000,
            'min_liquidity': 100_000, 
            'max_atr': 5.0,
            'max_spread': 15
        }
    
    def adapt_criteria(self, market_regime):
        multipliers = self.get_regime_multipliers(market_regime)
        adapted = {}
        
        for key, value in self.base_criteria.items():
            adapted_value = value * multipliers[key]
            # הגנות מינימום/מקסימום
            adapted[key] = self.apply_safety_limits(key, adapted_value)
            
        return adapted
    
    def get_regime_multipliers(self, regime):
        return {
            'CHOPPY': {'min_volume': 0.7, 'min_liquidity': 0.7, 'max_atr': 1.3, 'max_spread': 1.3},
            'TRENDING': {'min_volume': 1.2, 'min_liquidity': 1.2, 'max_atr': 0.8, 'max_spread': 0.8},
            'VOLATILE': {'min_volume': 1.0, 'min_liquidity': 1.0, 'max_atr': 1.1, 'max_spread': 1.0}
        }[regime]
```
