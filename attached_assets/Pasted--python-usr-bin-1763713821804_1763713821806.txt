    def calculate_momentum(self, symbol: str, tech_data: dict) -> float:
        """חישוב מומנטום"""
        try:
            rsi = tech_data.get('rsi', 50)
            macd = tech_data.get('macd', 0)
            macd_signal = tech_data.get('macd_signal', 0)
            
            # נורמליזציה של RSI (קרוב ל-50 = ניטרלי)
            rsi_score = 1 - abs(rsi - 50) / 50
            
            # MACD momentum
            macd_score = 1.0 if macd > macd_signal else 0.3
            
            momentum = (rsi_score * 0.4 + macd_score * 0.6)
            return max(0.0, min(1.0, momentum))
            
        except:
            return 0.5
    
    def calculate_trend_strength(self, symbol: str, tech_data: dict) -> float:
        """חישוב חוזק טרנד"""
        try:
            ema_20 = tech_data.get('ema_20', 0)
            ema_50 = tech_data.get('ema_50', 0)
            current_price = tech_data.get('current_price', 0)
            
            if current_price > ema_20 > ema_50:
                trend_score = 1.0
            elif current_price < ema_20 < ema_50:
                trend_score = 1.0
            elif current_price > ema_20:
                trend_score = 0.7
            else:
                trend_score = 0.3
            
            return trend_score
            
        except:
            return 0.5
    
    def calculate_volume_confirmation(self, symbol: str, tech_data: dict) -> float:

                return 0.8
            elif volume_trend > 1.2:
                return 0.6

        except:

    def calculate_volatility(self, symbol: str, tech_data: dict) -> float:
        """חישוב וולאטיליות"""
        try:
            atr = tech_data.get('atr', 0)
            current_price = tech_data.get('current_price', 1)
            atr_percent = atr / current_price
            
            # וולאטיליות אופטימלית בין 1% ל-3%
            if 0.01 <= atr_percent <= 0.03:
                return 1.0
            elif 0.005 <= atr_percent <= 0.05:
                return 0.7
            else:

        except:
            return 0.5
    
    def assess_position_risk(self, position: dict, tech_data: dict) -> dict:
        """הערכת סיכון פוזיציה"""
        try:
            pnl_percent = position['pnl_percent']
            liq_price = position['liq_price']
            current_price = position['mark_price']
            leverage = position['leverage']
            
            # סיכון נזילות
            if liq_price > 0:
                distance_to_liq = abs(current_price - liq_price) / current_price
                if distance_to_liq < 0.05:  # 5% מנזילות
                    liq_risk = "HIGH"
                elif distance_to_liq < 0.10:
                    liq_risk = "MEDIUM"
                else:
                    liq_risk = "LOW"
            else:
                liq_risk = "LOW"
            
            # סיכון מינוף
            if leverage >= 5:
                leverage_risk = "HIGH"
            elif leverage >= 3:
                leverage_risk = "MEDIUM"
            else:
                leverage_risk = "LOW"
            
            # סיכון כללי
            if liq_risk == "HIGH" or leverage_risk == "HIGH":
                overall_risk = "HIGH"
            elif liq_risk == "MEDIUM" or leverage_risk == "MEDIUM":
                overall_risk = "MEDIUM"
            else:
                overall_risk = "LOW"
            
            return {
                'liquidation_risk': liq_risk,
                'leverage_risk': leverage_risk,
                'overall_risk': overall_risk,
                'distance_to_liquidation': distance_to_liq if liq_price > 0 else 1.0
            }
            
        except Exception as e:
            self.logger.error(f"❌ שגיאה בהערכת סיכון: {e}")
            return {'overall_risk': 'UNKNOWN'}
    
    def generate_action_recommendation(self, symbol: str, position: dict, 
                                    momentum: float, trend_strength: float, 
                                    volume_confirmation: float, volatility: float) -> dict:
        """המלצת פעולה based on ניתוח"""
        
        pnl_percent = position['pnl_percent']
        current_price = position['mark_price']
        entry_price = position['entry_price']
        
        # ציון החזקה
        hold_score = (momentum * 0.3 + trend_strength * 0.3 + 
                     volume_confirmation * 0.2 + volatility * 0.2)
        
        if hold_score >= 0.7 and pnl_percent > 0:
            # תנאים מעולים - החזק והגדל רווחים
            action = "HOLD_STRONG"
            new_sl = entry_price * 0.98  # 2% stop loss
            new_tp = current_price * 1.15  # 15% take profit
            
        elif hold_score >= 0.5 and pnl_percent >= -2:
            # תנאים טובים - החזק עם הגנות
            action = "HOLD"
            new_sl = entry_price * 0.99  # 1% stop loss
            new_tp = current_price * 1.08  # 8% take profit
            
        elif pnl_percent <= -5 or hold_score <= 0.3:
            # תנאים גרועים - שקול יציאה
            action = "CONSIDER_EXIT"
            new_sl = current_price * 0.995  # 0.5% stop loss
            new_tp = current_price * 1.02  # 2% take profit
            
        else:
            # תנאים ניטרליים - החזק עם זהירות
            action = "HOLD_CAREFUL"
            new_sl = entry_price * 0.995  # 0.5% stop loss
            new_tp = current_price * 1.05  # 5% take profit
        
        return {
            'action': action,
            'hold_score': hold_score,
            'recommended_sl': new_sl,
            'recommended_tp': new_tp,
            'confidence': hold_score
        }
    
    # Utility functions for technical analysis
    def calculate_rsi(self, df, period=14):
        """חישוב RSI"""
        delta = df['close'].diff()
        gain = (delta.where(delta > 0, 0)).rolling(window=period).mean()
        loss = (-delta.where(delta < 0, 0)).rolling(window=period).mean()
        rs = gain / loss
        rsi = 100 - (100 / (1 + rs))
        return rsi.iloc[-1]
    
    def calculate_macd(self, df):
        """חישוב MACD"""
        exp1 = df['close'].ewm(span=12).mean()
        exp2 = df['close'].ewm(span=26).mean()
        macd = exp1 - exp2
        signal = macd.ewm(span=9).mean()
        return macd.iloc[-1], signal.iloc[-1]
    
    def calculate_ema(self, df, period):
        """חישוב EMA"""
        return df['close'].ewm(span=period).mean().iloc[-1]
    
    def calculate_atr(self, df, period=14):
        """חישוב ATR"""
        high_low = df['high'] - df['low']
        high_close = np.abs(df['high'] - df['close'].shift())
        low_close = np.abs(df['low'] - df['close'].shift())
        ranges = pd.concat([high_low, high_close, low_close], axis=1)
        true_range = np.max(ranges, axis=1)
        atr = true_range.rolling(period).mean()
        return atr.iloc[-1]
    
    def calculate_volume_trend(self, df):
        """חישוב מגמת ווליום"""
        current_volume = df['volume'].iloc[-1]
        avg_volume = df['volume'].rolling(20).mean().iloc[-1]
        return current_volume / avg_volume

            """)

            if recommendation['action'] in ['HOLD_STRONG', 'HOLD', 'HOLD_CAREFUL']:

        except Exception as e:

    def cancel_existing_orders(self, symbol: str):
        """ביטול כל ה-orders הקיימים עבור סימבול"""
        try:
            orders = self.client.get_open_orders(symbol=symbol)
            for order in orders:
                self.client.cancel_order(symbol=symbol, orderId=order['orderId'])
                self.logger.info(f"🗑️ ביטול order: {symbol} - {order['type']}")
            
        except Exception as e:
            self.logger.error(f"❌ שגיאה בביטול orders: {e}")
    
    def create_new_protections(self, symbol: str, analysis: dict, recommendation: dict):
        """יצירת הגנות חדשות"""
        try:
            position = analysis
            rec = recommendation
            
            # Stop Loss
            self.client.create_order(
                symbol=symbol,
                side='SELL' if position['entry_price'] > 0 else 'BUY',
                type='STOP_LOSS_LIMIT',
                timeInForce='GTC',
                quantity=abs(position['size']),
                stopPrice=round(rec['recommended_sl'], 6),
                price=round(rec['recommended_sl'], 6)
            )
            
            # Take Profit
            self.client.create_order(
                symbol=symbol,
                side='SELL' if position['entry_price'] > 0 else 'BUY',
                type='TAKE_PROFIT_LIMIT',
                timeInForce='GTC',
                quantity=abs(position['size']),
                stopPrice=round(rec['recommended_tp'], 6),
                price=round(rec['recommended_tp'], 6)
            )
            
            self.logger.info(f"🛡️ הגנות חדשות ל-{symbol}: SL={rec['recommended_sl']:.6f}, TP={rec['recommended_tp']:.6f}")
            
        except Exception as e:
            self.logger.error(f"❌ שגיאה ביצירת הגנות: {e}")
    
    def consider_early_exit(self, symbol: str, analysis: dict):

        except Exception as e:
            self.logger.error(f"❌ שגיאה בבחינת יציאה: {e}")
    
    def generate_comprehensive_report(self):

        report = {
            'timestamp': datetime.now(),
            'positions_analyzed': len(analysis),

            'summary': self.generate_summary(analysis),

        }
        
        return report
    
    def generate_summary(self, analysis: dict) -> dict:
        """יצירת סיכום כללי"""
        total_positions = len(analysis)
        profitable = len([a for a in analysis.values() if a['pnl_percent'] > 0])
        avg_hold_score = np.mean([a['action_recommendation']['hold_score'] for a in analysis.values()])
        
        return {
            'total_positions': total_positions,
            'profitable_positions': profitable,
            'profitability_rate': profitable / total_positions if total_positions > 0 else 0,
            'average_hold_score': avg_hold_score,
            'high_risk_positions': len([a for a in analysis.values() if a['risk_assessment']['overall_risk'] == 'HIGH'])
        }
    
    def generate_overall_recommendations(self, analysis: dict) -> list:
        """יצירת המלצות כלליות"""
        recommendations = []
        
        for symbol, data in analysis.items():
            rec = data['action_recommendation']
            risk = data['risk_assessment']

                'risk_level': risk['overall_risk'],
                'details': f"PNL: {data['pnl_percent']:.2f}% | Hold Score: {rec['hold_score']:.2f}"
            })
        
        return recommendations
