**🎯 מערכת שילובים דינמית ואוטומטית מלאה**

אפתח מערכת שיודעת ליצור שילובים דינמיים אוטומטית based on נתונים היסטוריים וביצועים. הנה הארכיטקטורה המלאה:

## 🚀 **Dynamic Combo Generator System**

```python
# utils/dynamic_combo_engine.py

import numpy as np
import pandas as pd
from typing import Dict, List, Any, Optional
from dataclasses import dataclass
from enum import Enum
import logging
from datetime import datetime, timedelta

class ComboType(Enum):
    TREND_FOLLOWING = "trend_following"
    MEAN_REVERSION = "mean_reversion" 
    BREAKOUT = "breakout"
    MOMENTUM = "momentum"
    HYBRID = "hybrid"

@dataclass
class IndicatorConfig:
    name: str
    weight: float
    params: Dict[str, Any]
    condition: str  # 'above', 'below', 'cross', 'divergence'

@dataclass
class DynamicCombo:
    id: str
    name: str
    combo_type: ComboType
    indicators: List[IndicatorConfig]
    conditions: List[str]
    performance: Dict[str, float]
    active: bool = True

class DynamicComboGenerator:
    """
    🧠 מערכת ליצירה אוטומטית ודינמית של שילובי אינדיקטורים
    - לומדת מההיסטוריה אילו שילובים עובדים
    - יוצרת קומבויים חדשים אוטומטית
    - מתאימה את המשקולות based on ביצועים
    """
    
    def __init__(self, data_source=None):
        self.data_source = data_source
        self.available_indicators = self._load_available_indicators()
        self.active_combos = {}
        self.combo_performance_history = {}
        self.learning_rate = 0.1
        
        # מאגר אינדיקטורים בסיסי
        self.indicator_pool = {
            'trend': ['EMA_9', 'EMA_20', 'EMA_50', 'ADX', 'MACD', 'VWAP'],
            'momentum': ['RSI', 'Stochastic', 'Williams_R', 'CCI', 'ROC'],
            'volatility': ['ATR', 'Bollinger_Bands', 'Keltner_Channels', 'Standard_Dev'],
            'volume': ['Volume', 'OBV', 'Volume_Profile', 'MFI'],
            'support_resistance': ['Pivot_Points', 'Fibonacci', 'Dynamic_SR']
        }
        
        self._initialize_default_combos()
    
    def _load_available_indicators(self) -> Dict:
        """טוען את כל האינדיקטורים הזמינים מהמערכת"""
        # כאן תתחבר למערכת האינדיקטורים הקיימת שלך
        return {
            'EMA_9': {'type': 'trend', 'default_params': {'period': 9}},
            'EMA_20': {'type': 'trend', 'default_params': {'period': 20}},
            'EMA_50': {'type': 'trend', 'default_params': {'period': 50}},
            'RSI': {'type': 'momentum', 'default_params': {'period': 14}},
            'MACD': {'type': 'trend', 'default_params': {'fast': 12, 'slow': 26, 'signal': 9}},
            'ADX': {'type': 'trend', 'default_params': {'period': 14}},
            'ATR': {'type': 'volatility', 'default_params': {'period': 14}},
            'Bollinger_Bands': {'type': 'volatility', 'default_params': {'period': 20, 'std': 2}},
            'VWAP': {'type': 'volume', 'default_params': {}},
            'Volume': {'type': 'volume', 'default_params': {}},
            # ... הוסף את כל האינדיקטורים שלך
        }
    
    def _initialize_default_combos(self):
        """מאתחל קומבויים בסיסיים"""
        
        # קומבו 1: מגמה + מומנטום
        trend_momentum = DynamicCombo(
            id="TREND_MOM_001",
            name="מגמה ומומנטום דינמי",
            combo_type=ComboType.TREND_FOLLOWING,
            indicators=[
                IndicatorConfig("EMA_20", 0.3, {'period': 20}, 'above'),
                IndicatorConfig("EMA_50", 0.3, {'period': 50}, 'above'), 
                IndicatorConfig("RSI", 0.2, {'period': 14}, 'range_30_70'),
                IndicatorConfig("MACD", 0.2, {}, 'positive')
            ],
            conditions=[
                "EMA_20 > EMA_50",
                "RSI BETWEEN 30 AND 70", 
                "MACD histogram > 0",
                "Volume > SMA(Volume, 20)"
            ],
            performance={'win_rate': 0.0, 'profit_factor': 0.0, 'sharpe': 0.0}
        )
        
        self.active_combos[trend_momentum.id] = trend_momentum
        
        # קומבו 2: פריצה עם נפח
        breakout_combo = DynamicCombo(
            id="BREAKOUT_002", 
            name="פריצה עם אישורי נפח",
            combo_type=ComboType.BREAKOUT,
            indicators=[
                IndicatorConfig("Bollinger_Bands", 0.4, {'period': 20, 'std': 2}, 'break_upper'),
                IndicatorConfig("Volume", 0.3, {}, 'above_avg'),
                IndicatorConfig("RSI", 0.2, {'period': 14}, 'below_70'),
                IndicatorConfig("ATR", 0.1, {'period': 14}, 'high_volatility')
            ],
            conditions=[
                "Price > Bollinger_Upper",
                "Volume > 1.5 * SMA(Volume, 20)",
                "RSI < 70",
                "ATR > SMA(ATR, 14)"
            ],
            performance={'win_rate': 0.0, 'profit_factor': 0.0, 'sharpe': 0.0}
        )
        
        self.active_combos[breakout_combo.id] = breakout_combo
    
    def generate_dynamic_combo(self, market_regime: str, success_patterns: List) -> DynamicCombo:
        """
        🎯 יוצר קומבו חדש dynamically based on מצב שוק ודפוסי הצלחה
        """
        combo_id = f"DYN_{len(self.active_combos) + 1:03d}"
        
        # בחירת אינדיקטורים based on market regime
        selected_indicators = self._select_indicators_for_regime(market_regime)
        
        # יצירת תנאים based on success patterns
        conditions = self._generate_conditions(selected_indicators, success_patterns)
        
        # חישוב משקולות אוטומטי
        weights = self._calculate_weights(selected_indicators, market_regime)
        
        new_combo = DynamicCombo(
            id=combo_id,
            name=f"קומבו דינמי {combo_id} - {market_regime}",
            combo_type=self._detect_combo_type(selected_indicators),
            indicators=selected_indicators,
            conditions=conditions,
            performance={'win_rate': 0.5, 'profit_factor': 1.0, 'sharpe': 0.0}
        )
        
        self.active_combos[combo_id] = new_combo
        logging.info(f"🎯 נוצר קומבו חדש: {new_combo.name}")
        
        return new_combo
    
    def _select_indicators_for_regime(self, regime: str) -> List[IndicatorConfig]:
        """בוחר אינדיקטורים מתאימים based on סוג השוק"""
        
        regime_indicators = {
            'trending': ['EMA_20', 'EMA_50', 'ADX', 'MACD', 'ATR'],
            'ranging': ['Bollinger_Bands', 'RSI', 'Stochastic', 'Williams_R'],
            'volatile': ['ATR', 'Bollinger_Bands', 'Standard_Dev', 'Volume'],
            'breakout': ['Bollinger_Bands', 'Volume', 'VWAP', 'RSI']
        }
        
        selected_names = regime_indicators.get(regime, ['EMA_20', 'RSI', 'Volume'])
        selected_indicators = []
        
        for indicator_name in selected_names[:4]:  # לוקח עד 4 אינדיקטורים
            if indicator_name in self.available_indicators:
                config = IndicatorConfig(
                    name=indicator_name,
                    weight=0.25,  # weight אחיד בהתחלה
                    params=self.available_indicators[indicator_name]['default_params'],
                    condition=self._get_default_condition(indicator_name)
                )
                selected_indicators.append(config)
        
        return selected_indicators
    
    def _get_default_condition(self, indicator_name: str) -> str:
        """מחזיר condition default based on סוג האינדיקטור"""
        conditions_map = {
            'EMA_20': 'above',
            'EMA_50': 'above', 
            'RSI': 'range_30_70',
            'MACD': 'positive',
            'Volume': 'above_avg',
            'Bollinger_Bands': 'break_upper',
            'ADX': 'above_25'
        }
        return conditions_map.get(indicator_name, 'above')
    
    def _generate_conditions(self, indicators: List[IndicatorConfig], success_patterns: List) -> List[str]:
        """מייצר תנאים based on אינדיקטורים ודפוסי הצלחה"""
        conditions = []
        
        for indicator in indicators:
            if indicator.name.startswith('EMA'):
                conditions.append(f"{indicator.name} > EMA_50")
            elif indicator.name == 'RSI':
                conditions.append("RSI BETWEEN 30 AND 70")
            elif indicator.name == 'Volume':
                conditions.append("Volume > SMA(Volume, 20)")
            elif indicator.name == 'Bollinger_Bands':
                conditions.append("Price > Bollinger_Upper OR Price < Bollinger_Lower")
            elif indicator.name == 'MACD':
                conditions.append("MACD histogram > 0")
        
        return conditions
    
    def evaluate_combo_performance(self, combo_id: str, trade_results: List[Dict]):
        """מעריך ומעדכן ביצועי קומבו"""
        if combo_id not in self.active_combos:
            return
        
        wins = len([t for t in trade_results if t['pnl'] > 0])
        total = len(trade_results)
        win_rate = wins / total if total > 0 else 0
        
        total_profit = sum([t['pnl'] for t in trade_results if t['pnl'] > 0])
        total_loss = abs(sum([t['pnl'] for t in trade_results if t['pnl'] < 0]))
        profit_factor = total_profit / total_loss if total_loss > 0 else 1.0
        
        # עדכון הביצועים
        self.active_combos[combo_id].performance.update({
            'win_rate': win_rate,
            'profit_factor': profit_factor,
            'total_trades': total
        })
        
        # התאמת משקולות based on ביצועים
        self._adjust_combo_weights(combo_id, win_rate, profit_factor)
    
    def _adjust_combo_weights(self, combo_id: str, win_rate: float, profit_factor: float):
        """מתאים את משקולות האינדיקטורים based on ביצועים"""
        combo = self.active_combos[combo_id]
        
        performance_score = (win_rate + min(profit_factor, 3)) / 2
        
        if performance_score > 0.6:  # ביצועים טובים - הגדל משקולות
            for indicator in combo.indicators:
                indicator.weight *= (1 + self.learning_rate)
        elif performance_score < 0.4:  # ביצועים גרועים - שנה קומבו
            self._evolve_or_remove_combo(combo_id)
        
        # נורמליזציה של משקולות
        total_weight = sum(ind.weight for ind in combo.indicators)
        for indicator in combo.indicators:
            indicator.weight /= total_weight
    
    def _evolve_or_remove_combo(self, combo_id: str):
        """מפתח קומבו חדש או מסיר קומבו כושל"""
        combo = self.active_combos[combo_id]
        
        if combo.performance['total_trades'] > 20 and combo.performance['win_rate'] < 0.4:
            # הסר קומבו כושל
            del self.active_combos[combo_id]
            logging.info(f"🗑️ הקומבו {combo_id} הוסר due to poor performance")
        else:
            # evolve - החלף אינדיקטור אחד
            self._evolve_combo(combo_id)
    
    def _evolve_combo(self, combo_id: str):
        """מפתח קומבו קיים על ידי החלפת אינדיקטור"""
        combo = self.active_combos[combo_id]
        
        # מצא את האינדיקטור עם הביצועים הגרועים ביותר
        worst_indicator = min(combo.indicators, key=lambda x: x.weight)
        
        # החלף באינדיקטור חדש מאותה קטגוריה
        indicator_type = self.available_indicators[worst_indicator.name]['type']
        alternative_indicators = [
            name for name, info in self.available_indicators.items() 
            if info['type'] == indicator_type and name not in [i.name for i in combo.indicators]
        ]
        
        if alternative_indicators:
            new_indicator_name = np.random.choice(alternative_indicators)
            new_indicator = IndicatorConfig(
                name=new_indicator_name,
                weight=worst_indicator.weight,
                params=self.available_indicators[new_indicator_name]['default_params'],
                condition=self._get_default_condition(new_indicator_name)
            )
            
            # החלף את האינדיקטור
            combo.indicators = [
                new_indicator if ind.name == worst_indicator.name else ind 
                for ind in combo.indicators
            ]
            
            logging.info(f"🔄 הקומבו {combo_id} התפתח: {worst_indicator.name} -> {new_indicator_name}")
    
    def get_best_combos(self, count: int = 5) -> List[DynamicCombo]:
        """מחזיר את הקומבויים עם הביצועים הטובים ביותר"""
        active_combos = [c for c in self.active_combos.values() if c.active]
        
        # דירוג based on win rate ו-profit factor
        scored_combos = []
        for combo in active_combos:
            score = (
                combo.performance['win_rate'] * 0.6 +
                min(combo.performance['profit_factor'], 3) * 0.4
            )
            scored_combos.append((score, combo))
        
        # מיון מהגבוה לנמוך
        scored_combos.sort(key=lambda x: x[0], reverse=True)
        
        return [combo for score, combo in scored_combos[:count]]
    
    def analyze_market_and_generate_combos(self, market_data: Dict) -> List[DynamicCombo]:
        """מנתח את השוק ומייצר קומבויים מתאימים"""
        # זיהוי סוג השוק
        market_regime = self._detect_market_regime(market_data)
        
        # בדיקה אם צריך קומבויים חדשים
        existing_combos_for_regime = [
            c for c in self.active_combos.values() 
            if self._combo_matches_regime(c, market_regime)
        ]
        
        if len(existing_combos_for_regime) < 2:
            # צור קומבויים חדשים עבור regime זה
            success_patterns = self._get_success_patterns(market_regime)
            new_combo = self.generate_dynamic_combo(market_regime, success_patterns)
            return [new_combo] + existing_combos_for_regime
        
        return existing_combos_for_regime
    
    def _detect_market_regime(self, market_data: Dict) -> str:
        """מזהה את סוג השוק הנוכחי"""
        # כאן ישולב הלוגיקה מזיהוי סוג השוק שכבר פיתחנו
        if market_data.get('adx', 0) > 25:
            if market_data.get('ema_20', 0) > market_data.get('ema_50', 0):
                return 'trending_up'
            else:
                return 'trending_down'
        elif market_data.get('atr_percentage', 0) > 0.03:
            return 'volatile'
        else:
            return 'ranging'
    
    def _combo_matches_regime(self, combo: DynamicCombo, regime: str) -> bool:
        """בודק אם קומבו מתאים ל-regime הנוכחי"""
        regime_map = {
            'trending_up': ComboType.TREND_FOLLOWING,
            'trending_down': ComboType.TREND_FOLLOWING, 
            'ranging': ComboType.MEAN_REVERSION,
            'volatile': ComboType.BREAKOUT
        }
        
        expected_type = regime_map.get(regime, ComboType.HYBRID)
        return combo.combo_type == expected_type
    
    def _get_success_patterns(self, regime: str) -> List:
        """מחזיר דפוסי הצלחה היסטוריים ל-regime נתון"""
        # כאן תתחבר למאגר הנתונים ההיסטורי
        # עבור עכשיו - מחזיר patterns ברירת מחדל
        return [f"pattern_{regime}_1", f"pattern_{regime}_2"]
    
    def _detect_combo_type(self, indicators: List[IndicatorConfig]) -> ComboType:
        """מזהה את סוג הקומבו based on האינדיקטורים"""
        indicator_types = [self.available_indicators[ind.name]['type'] for ind in indicators]
        
        if 'trend' in indicator_types and 'momentum' in indicator_types:
            return ComboType.TREND_FOLLOWING
        elif 'volatility' in indicator_types and 'volume' in indicator_types:
            return ComboType.BREAKOUT
        else:
            return ComboType.HYBRID

class ComboPerformanceTracker:
    """עוקב אחר ביצועי קומבויים לאורך זמן"""
    
    def __init__(self):
        self.performance_db = {}
        self.combo_evolution_history = {}
    
    def record_trade(self, combo_id: str, trade_data: Dict):
        """מתועד עסקה עבור קומבו ספציפי"""
        if combo_id not in self.performance_db:
            self.performance_db[combo_id] = []
        
        self.performance_db[combo_id].append({
            'timestamp': datetime.now(),
            'symbol': trade_data['symbol'],
            'direction': trade_data['direction'],
            'pnl': trade_data['pnl'],
            'duration': trade_data['duration'],
            'market_regime': trade_data.get('market_regime', 'unknown')
        })
    
    def generate_combo_analytics(self, combo_id: str) -> Dict:
        """מייצר אנליטיקה מפורטת עבור קומבו"""
        if combo_id not in self.performance_db:
            return {}
        
        trades = self.performance_db[combo_id]
        
        if not trades:
            return {}
        
        wins = [t for t in trades if t['pnl'] > 0]
        losses = [t for t in trades if t['pnl'] < 0]
        
        analytics = {
            'total_trades': len(trades),
            'win_rate': len(wins) / len(trades),
            'total_profit': sum(t['pnl'] for t in trades),
            'average_win': np.mean([t['pnl'] for t in wins]) if wins else 0,
            'average_loss': np.mean([t['pnl'] for t in losses]) if losses else 0,
            'profit_factor': abs(sum(t['pnl'] for t in wins) / sum(t['pnl'] for t in losses)) if losses else float('inf'),
            'best_trade': max(trades, key=lambda x: x['pnl']) if trades else None,
            'worst_trade': min(trades, key=lambda x: x['pnl']) if trades else None,
        }
        
        return analytics

class AdvancedTradingSystem:
    """מערכת מסחר מתקדמת עם קומבויים דינמיים"""
    
    def __init__(self):
        self.combo_generator = DynamicComboGenerator()
        self.performance_tracker = ComboPerformanceTracker()
        self.active_combos = {}
    
    def analyze_and_trade(self, market_data: Dict):
        """מנתח את השוק ומייצר אותות מסחר"""
        
        # שלב 1: קבלת קומבויים מומלצים עבור השוק הנוכחי
        recommended_combos = self.combo_generator.analyze_market_and_generate_combos(market_data)
        
        # שלב 2: הערכת כל קומבו
        trade_signals = []
        for combo in recommended_combos:
            signal = self._evaluate_combo_signal(combo, market_data)
            if signal['confidence'] > 0.6:  # threshold מינימלי
                trade_signals.append(signal)
        
        # שלב 3: בחירת האות הטוב ביותר
        if trade_signals:
            best_signal = max(trade_signals, key=lambda x: x['confidence'])
            return self._execute_trade(best_signal)
        
        return None
    
    def _evaluate_combo_signal(self, combo: DynamicCombo, market_data: Dict) -> Dict:
        """מעריך קומבו ומחזיר אות מסחר"""
        total_score = 0
        max_score = len(combo.indicators)
        conditions_met = []
        
        for indicator in combo.indicators:
            indicator_value = market_data.get(indicator.name)
            if indicator_value is not None:
                # כאן תתווסף הלוגיקה לבדיקת כל condition
                score = self._check_indicator_condition(indicator, indicator_value, market_data)
                total_score += score * indicator.weight
        
        confidence = total_score / max_score
        
        return {
            'combo_id': combo.id,
            'combo_name': combo.name,
            'direction': self._determine_direction(combo, market_data),
            'confidence': confidence,
            'conditions_met': conditions_met,
            'recommended_size': self._calculate_position_size(confidence, combo.performance['win_rate'])
        }
    
    def _check_indicator_condition(self, indicator: IndicatorConfig, value: float, market_data: Dict) -> float:
        """בודק condition של אינדיקטור ומחזיר score"""
        # הלוגיקה המלאה תתווסף כאן
        return 0.8  # placeholder
    
    def _determine_direction(self, combo: DynamicCombo, market_data: Dict) -> str:
        """קובע את כיוון העסקה based on הקומבו"""
        if combo.combo_type in [ComboType.TREND_FOLLOWING, ComboType.BREAKOUT]:
            return 'long' if market_data.get('ema_20', 0) > market_data.get('ema_50', 0) else 'short'
        else:
            return 'long'  # default
    
    def _calculate_position_size(self, confidence: float, win_rate: float) -> float:
        """מחשב גודל פוזיציה based on confidence ו-win rate"""
        base_size = 0.02  # 2% מהון
        adjusted_size = base_size * confidence * win_rate
        return min(adjusted_size, 0.05)  # max 5% מהון

        print(f"   גודל מומלץ: {trade_signal['recommended_size']:.2%}")
    else:
        print("❌ אין אות מסחר מתאים")
    
    # הצגת הקומבויים הפעילים
    print(f"\n📊 קומבויים פעילים: {len(trading_system.combo_generator.active_combos)}")
    for combo_id, combo in list(trading_system.combo_generator.active_combos.items())[:3]:
        print(f"   • {combo.name} (Win Rate: {combo.performance['win_rate']:.1%})")

if __name__ == "__main__":
    test_dynamic_system()
```
