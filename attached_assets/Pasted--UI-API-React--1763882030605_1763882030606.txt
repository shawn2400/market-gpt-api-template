# ✅ חלק 1 — REST API שליטה מלאה (FastAPI)

### 📌 `routes/control_panel.py`

```python
# routes/control_panel.py

from fastapi import APIRouter
from utils.kill_switch import kill
from utils.circuit_breaker import breaker
from utils.mode_manager import mode
from utils.freeze_manager import freeze
from utils.news_guard import news

router = APIRouter(prefix="/control")

# 🟢 Pause trading
@router.post("/pause")
def pause_trading():
    kill.activate("Manual admin pause")
    return {"status": "paused"}

# 🟢 Resume trading
@router.post("/resume")
def resume_trading():
    kill.reset()
    return {"status": "resumed"}

# 🔄 Flip direction
@router.post("/flip")
def flip_mode():
    mode.flip()
    return {"status": "flipped", "new_mode": mode.mode}

# ❄ Freeze symbol
@router.post("/freeze/{symbol}")
def freeze_symbol(symbol: str):
    freeze.add(symbol.upper())
    return {"status": "frozen", "symbol": symbol}

# ❄ Unfreeze symbol
@router.delete("/freeze/{symbol}")
def unfreeze_symbol(symbol: str):
    freeze.remove(symbol.upper())
    return {"status": "unfrozen", "symbol": symbol}

# 🚨 Manual circuit-breaker
@router.post("/lock/{minutes}")
def lock(minutes: int):
    breaker.trigger(minutes, "Manual admin lock")
    return {"status": "locked", "minutes": minutes}

# 📡 Toggle news auto-pause
@router.post("/news/auto/{state}")
def toggle_news(state: str):
    news.enable(state == "on")
    return {"news_auto": state}
```

✅ אין ריסטארט
✅ אין רגרסיות
✅ 100% admin-only

# ✅ חלק 3 — UI React מלא עם כפתורים

### 📌 `src/App.jsx`

```jsx
import { useEffect, useState } from "react";

export default function App() {
  const [data, setData] = useState({});

  const api = (path, method = "POST") =>
    fetch(`/control/${path}`, { method });

  useEffect(() => {
    const ws = new WebSocket("wss://YOUR_DOMAIN/ws/live");
    ws.onmessage = (msg) => setData(JSON.parse(msg.data));
    return () => ws.close();
  }, []);

  return (
    <div style={{ padding: 20, fontFamily: "sans-serif" }}>
      <h1>AlgoGPT Control Center</h1>

      <section>
        <h2>System Status</h2>
        <p>Mode: {data.mode}</p>
        <p>Kill-Switch: {data.kill ? "ACTIVE" : "OK"}</p>
        <p>Circuit-Breaker: {data.breaker ? "LOCKED" : "OPEN"}</p>
        <p>News Pause: {data.news_pause ? "YES" : "NO"}</p>
      </section>

      <section>
        <h2>Actions</h2>
        <button onClick={() => api("pause")}>Pause</button>
        <button onClick={() => api("resume")}>Resume</button>
        <button onClick={() => api("flip")}>Flip</button>
        <button onClick={() => api("lock/120")}>Lock 2h</button>
        <button onClick={() => api("news/auto/on")}>Enable News Auto</button>
        <button onClick={() => api("news/auto/off")}>Disable News Auto</button>
      </section>

      <section>
        <h2>Optimizer</h2>
        <pre>{JSON.stringify(data.optimizer, null, 2)}</pre>
      </section>

      <section>
        <h2>Heatmap</h2>
        <pre>{JSON.stringify(data.heatmap, null, 2)}</pre>
      </section>
    </div>
  );
}
```
