```python
        }
        
        self.decision_history = []
        self.performance_tracking = {}
    
    def council_vote(self, trade_opportunity):
        """הצבעת מועצה עם weights"""
        
        votes = {}
        total_weight = 0
        
        # כל מומחה מצביע לפי התמחותו
        for role, member in self.council_members.items():
            if self.should_activate_member(member, trade_opportunity):
                vote = member['voting_function'](trade_opportunity)
                weight = member['weight']
                votes[role] = {'vote': vote, 'weight': weight}
                total_weight += weight
        
        # חישוב החלטה סופית
        if total_weight > 0:
            final_decision = self.calculate_weighted_decision(votes)
            
            # רק אם ההחלטה חיובית - משתמשים ב-tokens
            if final_decision['should_trade']:
                self.activate_council_tokens(votes)
            
            return final_decision
        else:
            return {'should_trade': False, 'reason': 'No qualified voters'}
```

### **רק טריידים איכותיים משתמשים ב-tokens:**
```python
class TradeQualificationSystem:
    def __init__(self):
        self.minimum_confidence = 0.75  # 75% confidence minimum
        self.risk_reward_threshold = 1.5  # 1:1.5 minimum
        self.volume_confirmation = True
        self.trend_alignment = True
    
    def qualify_trade(self, signal):
        """בודק אם הטרייד qualified להשקעת tokens"""
        
        qualification_score = 0
        
        # 1. ניתוח טכני
        if signal['technical_score'] > 0.7:
            qualification_score += 0.3
        
        # 2. ניתוח fundamental  
        if signal['fundamental_score'] > 0.6:
            qualification_score += 0.2
        
        # 3. volume confirmation
        if signal['volume_spike'] and signal['volume'] > signal['avg_volume'] * 1.5:
            qualification_score += 0.2
        
        # 4. trend alignment
        if signal['trend_alignment']:
            qualification_score += 0.2
        
        # 5. risk/reward ratio
        if signal['risk_reward_ratio'] >= self.risk_reward_threshold:
            qualification_score += 0.1
        
        return qualification_score >= self.minimum_confidence
```
