```python
# hybrid_timeout_manager.py
class HybridTimeoutManager:
    """
    🎯 ניהול טיימאאוט היברידי - דינמי + חכם
    """
    
    def __init__(self):
        self.base_timeout = 60
        self.performance_data = {}
        
    async def calculate_dynamic_timeout(self, order_data: Dict) -> float:
        """חישוב טיימאאוט דינמי לפי תנאים"""
        
        symbol = order_data['symbol']
        order_type = order_data.get('type', 'LIMIT')
        
        # Base calculation
        timeout = self.base_timeout
        
        # Market volatility adjustment
        volatility = await self.get_symbol_volatility(symbol)
        if volatility > 0.04:    # High volatility
            timeout *= 0.6       # 60% - more aggressive
        elif volatility < 0.015: # Low volatility  
            timeout *= 1.4       # 140% - more patience
        
        # Order type adjustment
        type_multipliers = {
            'ENTRY_LIMIT': 1.2,
            'TP_LIMIT': 0.8, 
            'STOP_LIMIT': 0.7,
            'GRID_ORDER': 1.5,
            'MARKET': 0.3
        }
        timeout *= type_multipliers.get(order_type, 1.0)
        
        # Time of day adjustment
        hour = time.localtime().tm_hour
        if 14 <= hour <= 22:  # Active hours
            timeout *= 0.8
        else:  # Quiet hours
            timeout *= 1.3
        
        return max(30, min(timeout, 180))  # Limit 30-180 seconds
    
    async def check_order_timeouts(self):
        """בדיקת טיימאאוטים עם intelligence"""
        current_time = time.time()
        
        for order_id, order_state in self.order_states.items():
            if order_state['status'] != 'ACTIVE':
                continue
                
            order_age = current_time - order_state['created_at']
            timeout_threshold = order_state.get('timeout_at', current_time + 60)
            
            # Pre-timeout check (80% of timeout)
            if order_age >= timeout_threshold * 0.8:
                await self.pre_timeout_analysis(order_id, order_state)
            
            # Actual timeout
            if current_time >= timeout_threshold:
                await self.handle_order_timeout(order_id, order_state)
    
    async def pre_timeout_analysis(self, order_id: str, order_state: Dict):
        """ניתוח pro-active לפני טיימאאוט"""
        symbol = order_state['symbol']
        current_conditions = await self.get_market_conditions(symbol)
        
        # Check if order is near fill
        if await self.is_near_fill_price(symbol, order_state.get('price', 0)):
            # Extend timeout for orders near fill
            new_timeout = order_state['timeout_at'] + 30  # +30 seconds
            order_state['timeout_at'] = new_timeout
            logger.info(f"⏱️ Extended timeout for {order_id} - near fill")
    
    async def handle_order_timeout(self, order_id: str, order_state: Dict):
        """טיפול חכם ב-Order timeout"""
        symbol = order_state['symbol']
        
        # Get current market analysis
        market_analysis = await self.analyze_current_market(symbol)
        
        # Make intelligent decision
        decision = await self.make_escalation_decision(order_state, market_analysis)
        
        # Execute decision
        if decision['action'] == 'CONVERT_TO_MARKET':
            await self.convert_to_market_order(order_id, order_state)
        elif decision['action'] == 'ADJUST_AND_RETRY':
            await self.adjust_and_retry_order(order_id, order_state)
        elif decision['action'] == 'CANCEL':
            await self.cancel_order(order_id, order_state)
        
        # Update state
        order_state['status'] = 'TIMEOUT'
        order_state['events'].append({
            'timestamp': time.time(),
            'event': 'TIMEOUT',
            'action': decision['action'],
            'reason': decision['reason']
        })
```

```python
class SmartCleanupManager:
    """
    🧹 ניהול ניקוי אוטומטי ואינטליגנטי
    """
    
    def __init__(self):
        self.cleanup_interval = 300  # 5 minutes
        self.orphaned_check_interval = 120  # 2 minutes
        
    async def start_cleanup_engine(self):
        """הפעלת מנוע הניקוי"""
        while True:
            try:
                cleanup_tasks = [
                    self.cleanup_orphaned_orders(),
                    self.cleanup_old_order_states(),
                    self.cleanup_performance_data(),
                    self.cleanup_connection_pools()
                ]
                
                # Run all cleanup tasks in parallel
                await asyncio.gather(*cleanup_tasks)
                
                await asyncio.sleep(self.cleanup_interval)
                
            except Exception as e:
                logger.error(f"❌ Cleanup engine failed: {e}")
                await asyncio.sleep(60)
    
    async def cleanup_orphaned_orders(self):
        """ניקוי Orphaned Orders"""
        try:
            # Get all active orders from exchange
            exchange_orders = await self.binance.get_open_orders()
            exchange_order_ids = {order['id'] for order in exchange_orders}
            
            # Find orphaned orders (in our system but not in exchange)
            orphaned_count = 0
            for order_id, order_state in list(self.order_states.items()):
                if (order_state['status'] == 'ACTIVE' and 
                    order_id not in exchange_order_ids):
                    
                    # Mark as orphaned
                    order_state['status'] = 'ORPHANED'
                    order_state['events'].append({
                        'timestamp': time.time(),
                        'event': 'ORPHANED_DETECTED'
                    })
                    orphaned_count += 1
            
            if orphaned_count > 0:
                logger.warning(f"🧹 Cleaned up {orphaned_count} orphaned orders")
                
        except Exception as e:
            logger.error(f"❌ Orphaned cleanup failed: {e}")
    
    async def cleanup_old_order_states(self):
        """ניקוי state ישן"""
        current_time = time.time()
        max_age = 3600  # 1 hour
        
        cleaned_count = 0
        for order_id, order_state in list(self.order_states.items()):
            order_age = current_time - order_state['created_at']
            
            # Remove old completed orders
            if (order_state['status'] in ['FILLED', 'CANCELLED', 'TIMEOUT'] and 
                order_age > max_age):
                
                del self.order_states[order_id]
                cleaned_count += 1
        
        if cleaned_count > 0:
            logger.info(f"🧹 Cleaned {cleaned_count} old order states")
```

```python
class ResilientErrorHandler:
    """
    🛡️ מנוע טיפול בשגיאות עם התאוששות אוטומטית
    """
    
    def __init__(self):
        self.error_counts = {}
        self.circuit_breakers = {}
        self.recovery_strategies = {}
        
    async def handle_error(self, error_type: str, context: Dict):
        """טיפול מרכזי בכל השגיאות"""
        error_id = f"{error_type}_{context.get('symbol', 'global')}"
        
        # Track error frequency
        self.error_counts[error_id] = self.error_counts.get(error_id, 0) + 1
        
        # Check if circuit breaker should trigger
        if await self.should_trigger_circuit_breaker(error_id):
            await self.activate_circuit_breaker(error_id, context)
            return
        
        # Execute recovery strategy
        recovery_success = await self.execute_recovery_strategy(error_type, context)
        
        if not recovery_success:
            await self.escalate_error(error_type, context)
    
    async def execute_recovery_strategy(self, error_type: str, context: Dict) -> bool:
        """ביצוע אסטרטגיית התאוששות"""
        strategies = {
            'ORDER_TIMEOUT': self.recover_order_timeout,
            'NETWORK_ERROR': self.recover_network_error,
            'INSUFFICIENT_BALANCE': self.recover_balance_error,
            'API_LIMIT': self.recover_api_limit,
            'MARKET_CLOSED': self.recover_market_closed
        }
        
        strategy = strategies.get(error_type)
        if strategy:
            return await strategy(context)
        
        return False
    
    async def recover_order_timeout(self, context: Dict) -> bool:
        """התאוששות מ-Order Timeout"""
        try:
            order_id = context.get('order_id')
            symbol = context.get('symbol')
            
            # 1. Cancel the stuck order
            await self.binance.cancel_order(symbol, order_id)
            
            # 2. Decide retry strategy
            market_conditions = await self.get_market_conditions(symbol)
            
            if market_conditions['volatility'] < 0.02:
                # Low volatility - retry with limit
                await self.place_limit_order(context)
            else:
                # High volatility - use market order
                await self.place_market_order(context)
            
            logger.info(f"🛡️ Recovered from order timeout for {symbol}")
            return True
            
        except Exception as e:
            logger.error(f"❌ Order timeout recovery failed: {e}")
            return False
```

```python
# unified_state_tracker.py
class UnifiedStateTracker:
    """
    📊 מעקב state מאוחד - Single Source of Truth
    """
    
    def __init__(self):
        self.system_state = {
            'orders': {},
            'positions': {},
            'performance': {},
            'health': {},
            'connections': {}
        }
        self.state_history = []
        self.last_sync = time.time()
        
    async def track_system_state(self):
        """מעקב רצוף אחרי state המערכת"""
        while True:
            try:
                # Capture comprehensive system state
                state_snapshot = {
                    'timestamp': time.time(),
                    'orders': await self.capture_order_states(),
                    'positions': await self.capture_position_states(),
                    'performance': await self.capture_performance_metrics(),
                    'health': await self.capture_health_metrics(),
                    'connections': await self.capture_connection_states()
                }
                
                # Store in history (keep last 1000 snapshots)
                self.state_history.append(state_snapshot)
                if len(self.state_history) > 1000:
                    self.state_history.pop(0)
                
                # Update current state
                self.system_state = state_snapshot
                
                # Sync to Redis every 30 seconds
                if time.time() - self.last_sync > 30:
                    await self.sync_to_redis()
                    self.last_sync = time.time()
                
                await asyncio.sleep(10)  # Every 10 seconds
                
            except Exception as e:
                logger.error(f"❌ State tracking failed: {e}")
                await asyncio.sleep(30)
    
    async def capture_order_states(self) -> Dict:
        """לכידת state של כל Orders"""
        order_states = {}
        
        for order_id, order_data in self.order_states.items():
            order_states[order_id] = {
                'symbol': order_data['symbol'],
                'type': order_data['type'],
                'status': order_data['status'],
                'age': time.time() - order_data['created_at'],
                'event_count': len(order_data['events']),
                'last_event': order_data['events'][-1] if order_data['events'] else None
            }
        
        return order_states
```
