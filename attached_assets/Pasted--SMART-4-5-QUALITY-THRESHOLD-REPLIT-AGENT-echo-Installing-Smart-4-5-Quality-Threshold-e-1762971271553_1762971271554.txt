def validate_trade(symbol, quality_score, indicators, volume_ratio, trend_alignment, rr_ratio):
    """Validate trade using smart 4.5 threshold"""
    
    # Base quality check
    if quality_score < QUALITY_CONFIG['MIN_QUALITY_THRESHOLD']:
        return False, "Quality score too low"
    
    # Smart filters check
    filters = QUALITY_CONFIG['SMART_FILTERS']
    
    # Confluence check
    if filters['confluence']['required']:
        aligned_indicators = sum(1 for indicator in indicators if indicator['aligned'])
        if aligned_indicators < filters['confluence']['min_indicators']:
            return False, "Insufficient indicator confluence"
    
    # Volume check
    if filters['volume']['required']:
        if volume_ratio < filters['volume']['min_ratio']:
            return False, "Volume too low"
    
    # Trend check  
    if filters['trend']['required'] and not trend_alignment:
        return False, "Against trend"
    
    # Risk/Reward check
    if filters['risk_reward']['required'] and rr_ratio < filters['risk_reward']['min_ratio']:
        return False, "Risk/reward too low"
    
    return True, "Trade validated"
