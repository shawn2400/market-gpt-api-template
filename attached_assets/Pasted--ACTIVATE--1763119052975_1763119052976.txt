```python
# commands/system_orchestrator.py
import asyncio
import logging
from utils.central_orchestrator import CentralOrderOrchestrator
from utils.smart_blacklist_manager import SmartBlacklistManager
from utils.dynamic_quantum_manager import DynamicQuantumManager
from utils.adaptive_grid_manager import AdaptiveGridManager

logger = logging.getLogger(__name__)

class SystemOrchestrator:
    """
    🧠 המנהל האוטומטי השלם - מטפל בכל ההזמנות באופן אוטומטי
    """
    
    def __init__(self, db_client, binance_client):
        self.db = db_client
        self.binance = binance_client
        self.is_running = False
        
        # כל המנהלים האוטומטיים
        self.central_orchestrator = CentralOrderOrchestrator(binance_client)
        self.blacklist_manager = SmartBlacklistManager(db_client)
        self.quantum_manager = DynamicQuantumManager(db_client, binance_client)
        self.grid_manager = AdaptiveGridManager(db_client, binance_client)
        
    async def activate_full_auto_mode(self):
        """הפעלת מצב אוטומטי מלא"""
        logger.info("🚀 ACTIVATING FULL AUTO TRADING SYSTEM")
        
        try:
            # 1. ניקוי והכנת מערכת
            await self._prepare_system()
            
            # 2. הפעלת כל המנהלים
            await self._start_all_managers()
            
            # 3. וידוא פעולה
            await self._verify_system_health()
            
            self.is_running = True
            logger.info("✅ FULL AUTO SYSTEM ACTIVATED - הכל עובד אוטומטית!")
            
            return {
                'status': 'success',
                'message': 'מערכת אוטומטית הופעלה - כל ההזמנות מנוהלות אוטומטית',
                'managers_active': [
                    'Central Orchestrator',
                    'Smart Blacklist Manager', 
                    'Dynamic Quantum Manager',
                    'Adaptive Grid Manager'
                ]
            }
            
        except Exception as e:
            logger.error(f"❌ Failed to activate auto system: {e}")
            return {'status': 'error', 'message': str(e)}
    
    async def _prepare_system(self):
        """הכנת המערכת לפעולה אוטומטית"""
        logger.info("🔧 Preparing system for auto operation...")
        
        # 1. ניקוי blacklist
        await self.blacklist_manager.smart_cleanup()
        
        # 2. עדכון quantum top 50
        await self.quantum_manager.dynamic_scan()
        
        # 3. בניית grid approved
        await self.grid_manager.adaptive_rebuild()
        
        # 4. איפוס errors
        await self.db.reset_error_counts()
        
        logger.info("✅ System preparation completed")
    
    async def _start_all_managers(self):
        """הפעלת כל המנהלים האוטומטיים"""
        
        # 1. Central Orchestrator - המוח הראשי
        await self.central_orchestrator.start_orchestration()
        
        # 2. Smart Timeout Management
        await self.central_orchestrator.timeout_manager.start_monitoring()
        
        # 3. Auto Cleanup Engine
        await self.central_orchestrator.cleanup_manager.start_cleanup_engine()
        
        # 4. Error Recovery System
        await self.central_orchestrator.error_handler.start_recovery_engine()
        
        logger.info("✅ All auto managers started")
    
    async def _verify_system_health(self):
        """וידוא שהמערכת פועלת כראוי"""
        health_checks = [
            self._check_order_monitoring(),
            self._check_timeout_management(),
            self._check_cleanup_system(),
            self._check_recovery_system()
        ]
        
        results = await asyncio.gather(*health_checks, return_exceptions=True)
        
        healthy_checks = sum(1 for r in results if not isinstance(r, Exception))
        logger.info(f"✅ System health: {healthy_checks}/{len(health_checks)} checks passed")
    
    async def smart_order_management(self):
        """ניהול הזמנות חכם - הפונקציה הראשית"""
        while self.is_running:
            try:
                # 1. בדיקת הזמנות תקועות
                await self._manage_stuck_orders()
                
                # 2. אופטימיזציית Grid
                await self._optimize_grid_orders()
                
                # 3. ניהול סיכונים
                await self._manage_risks()
                
                # 4. עדכון אסטרטגיות
                await self._update_strategies()
                
                await asyncio.sleep(60)  # בדיקה כל דקה
                
            except Exception as e:
                logger.error(f"❌ Smart order management failed: {e}")
                await asyncio.sleep(30)
```

```python
# utils/auto_order_manager.py
import asyncio
import time
import logging
from typing import Dict, List
from datetime import datetime, timedelta

logger = logging.getLogger(__name__)

class AutoOrderManager:
    """
    🤖 מנהל הזמנות אוטומטי - מטפל בכל סוגי ההזמנות
    """
    
    def __init__(self, binance_client, db_client):
        self.binance = binance_client
        self.db = db_client
        self.order_cache = {}
        self.performance_metrics = {}
        
    async def manage_all_orders(self):
        """ניהול אוטומטי של כל ההזמנות"""
        logger.info("🤖 Starting auto order management for ALL orders")
        
        while True:
            try:
                # 1. קבלת כל ההזמנות הפעילות
                all_orders = await self._get_all_active_orders()
                
                # 2. ניתוח וקטגוריזציה
                categorized_orders = await self._categorize_orders(all_orders)
                
                # 3. ניהול לפי קטגוריה
                management_tasks = [
                    self._manage_grid_orders(categorized_orders['grid_orders']),
                    self._manage_entry_orders(categorized_orders['entry_orders']),
                    self._manage_tp_orders(categorized_orders['tp_orders']),
                    self._manage_stop_orders(categorized_orders['stop_orders'])
                ]
                
                await asyncio.gather(*management_tasks)
                
                # 4. עדכון metrics
                await self._update_performance_metrics(all_orders)
                
                await asyncio.sleep(30)  # בדיקה כל 30 שניות
                
            except Exception as e:
                logger.error(f"❌ Auto order management failed: {e}")
                await asyncio.sleep(60)
    
    async def _get_all_active_orders(self) -> List[Dict]:
        """קבלת כל ההזמנות הפעילות מכל הסוגים"""
        try:
            # הזמנות פתוחות מהאקסצ'יינג'
            exchange_orders = await self.binance.fetch_open_orders()
            
            # הזמנות ממערכת הניהול שלנו
            system_orders = await self.db.get_system_orders()
            
            # שילוב והסרת duplicates
            all_orders = self._merge_orders(exchange_orders, system_orders)
            
            logger.debug(f"📊 Found {len(all_orders)} active orders")
            return all_orders
            
        except Exception as e:
            logger.error(f"❌ Failed to get active orders: {e}")
            return []
    
    async def _categorize_orders(self, orders: List[Dict]) -> Dict:
        """קטגוריזציה חכמה של ההזמנות"""
        categorized = {
            'grid_orders': [],
            'entry_orders': [], 
            'tp_orders': [],
            'stop_orders': [],
            'unknown_orders': []
        }
        
        for order in orders:
            order_type = await self._determine_order_type(order)
            categorized[order_type].append(order)
            
        return categorized
    
    async def _determine_order_type(self, order: Dict) -> str:
        """זיהוי אוטומטי של סוג ההזמנה"""
        symbol = order['symbol']
        order_side = order['side']  # 'buy' or 'sell'
        order_price = float(order['price'])
        
        # ניתוח הקונטקסט של ההזמנה
        current_price = await self._get_current_price(symbol)
        
        # Grid Order - קניות במחירים יורדים או מכירות בעולות
        if await self._is_grid_pattern(order, current_price):
            return 'grid_orders'
            
        # Entry Order - קניה מעל/מתחת למחיר נוכחי
        elif await self._is_entry_order(order, current_price):
            return 'entry_orders'
            
        # TP Order - מכירה מעל מחיר נוכחי
        elif order_side == 'sell' and order_price > current_price * 1.01:
            return 'tp_orders'
            
        # Stop Order - מכירה מתחת למחיר נוכחי
        elif order_side == 'sell' and order_price < current_price * 0.99:
            return 'stop_orders'
            
        else:
            return 'unknown_orders'
    
    async def _manage_grid_orders(self, grid_orders: List[Dict]):
        """ניהול אוטומטי של Grid Orders"""
        for order in grid_orders:
            try:
                # בדיקת זמן החיים של ההזמנה
                order_age = await self._get_order_age(order)
                
                if order_age > 3600:  # יותר משעה
                    # החלטה חכמה - לבטל או להתאים
                    decision = await self._make_grid_order_decision(order)
                    
                    if decision['action'] == 'CANCEL':
                        await self._cancel_order_safely(order)
                    elif decision['action'] == 'ADJUST':
                        await self._adjust_grid_order(order, decision['new_price'])
                        
            except Exception as e:
                logger.error(f"❌ Grid order management failed for {order['id']}: {e}")
    
    async def _manage_entry_orders(self, entry_orders: List[Dict]):
        """ניהול אוטומטי של Entry Orders"""
        for order in entry_orders:
            try:
                # בדיקת התאמה לתנאי שוק נוכחיים
                market_conditions = await self._analyze_market_conditions(order['symbol'])
                
                if not await self._is_entry_still_valid(order, market_conditions):
                    # ביטול והחלפה באסטרטגיה טובה יותר
                    await self._replace_invalid_entry(order, market_conditions)
                    
            except Exception as e:
                logger.error(f"❌ Entry order management failed for {order['id']}: {e}")
    
    async def _make_grid_order_decision(self, order: Dict) -> Dict:
        """קבלת החלטה חכמה עבור Grid Order"""
        symbol = order['symbol']
        current_price = await self._get_current_price(symbol)
        order_price = float(order['price'])
        
        # חישוב מרוחק מהמחיר הנוכחי
        price_distance = abs(current_price - order_price) / current_price
        
        if price_distance > 0.05:  # יותר מ-5% מהמחיר הנוכחי
            return {'action': 'CANCEL', 'reason': 'Too far from current price'}
        elif price_distance < 0.005:  # פחות מ-0.5% - קרוב מדי
            return {'action': 'ADJUST', 'new_price': current_price * 0.99}
        else:
            return {'action': 'KEEP', 'reason': 'Optimal distance'}
    
    async def _is_entry_still_valid(self, order: Dict, market_conditions: Dict) -> bool:
        """בדיקה אם Entry Order עדיין תקף"""
        symbol = order['symbol']
        order_price = float(order['price'])
        current_price = await self._get_current_price(symbol)
        
        conditions = [
            market_conditions['trend'] in ['BULLISH', 'SIDEWAYS'],  # מגמה מתאימה
            abs(order_price - current_price) / current_price < 0.03,  # לא רחוק מדי
            market_conditions['volatility'] < 0.08,  # volatility סביר
            await self._has_sufficient_volume(symbol)  # volume מספיק
        ]
        
        return all(conditions)
```
