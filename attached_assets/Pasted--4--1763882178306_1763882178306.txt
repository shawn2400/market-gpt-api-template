# ✅ 1) AUTO-HEDGING — בזמן אירוע קיצוני

### 📌 מודול חדש: `utils/auto_hedger.py`

```python
from utils.binance_client import futures_market_order
from utils.news_filter import is_extreme_news
from utils.volatility import atr_spike

class AutoHedger:
    def __init__(self):
        self.active = False
        self.min_atr_spike = 2.5
        self.hedge_size = 0.35  # 35% מהחשיפה

    def check_and_hedge(self, symbol, position, price, atr):
        if self.active:
            return

        if not is_extreme_news() and atr_spike(atr, threshold=self.min_atr_spike):
            self.open_hedge(symbol, position)
            self.active = True

    def open_hedge(self, symbol, position):
        side = "SELL" if position.side == "LONG" else "BUY"
        qty = position.size * self.hedge_size

        futures_market_order(
            symbol=symbol,
            side=side,
            quantity=qty,
            reduce_only=False
        )

        return True

    def release(self, symbol, position):
        if not self.active:
            return

        futures_market_order(
            symbol=symbol,
            side=position.side,
            quantity=position.size * self.hedge_size,
            reduce_only=True
        )

        self.active = False
```

### ✅ מה זה עושה בפועל

✔ מזהה חדות־סיכון (ATR + חדשות)
✔ פותח hedge נגדי אוטומטית
✔ משחרר רק כשהסיכון ירד
✔ לא מפריע ל-SL/TP
✔ 0 סיכוי ל-שאריות פוזיציה

# ✅ 2) MULTI-ACCOUNT ORCHESTRATION

### 📌 קובץ חדש: `utils/account_router.py`

```python
import json

class AccountRouter:
    def __init__(self):
        with open("accounts/accounts_config.json") as f:
            self.accounts = json.load(f)

    def get_client(self, user_id):
        cfg = self.accounts[user_id]
        return BinanceClient(
            api_key=cfg["api_key"],
            api_secret=cfg["api_secret"],
            mode=cfg["type"]  # spot/futures
        )

    def broadcast(self, fn, *args, **kwargs):
        results = {}
        for user_id in self.accounts:
            client = self.get_client(user_id)
            results[user_id] = fn(client, *args, **kwargs)
        return results

router = AccountRouter()
```

### ✅ איך זה עובד

✔ כל משתמש עם API שלו
✔ אתה שולט מי פעיל ומי מושעה
✔ אפשר להחליף משתמש בלחיצה
✔ תמיכה ב-Spot + Futures

# ✅ 3) PROFIT-SHARE BILLING אוטומטי

### 📌 קובץ חדש: `utils/profit_share.py`

```python
from utils.pnl_tracker import get_weekly_pnl
from utils.telegram_notifier import send_invoice
from utils.account_router import router

PERCENT = 0.18

def generate_billing():
    invoices = []

    for user_id in router.accounts:
        pnl = get_weekly_pnl(user_id)
        if pnl <= 0:
            continue

        amount = pnl * PERCENT

        invoice = {
            "user_id": user_id,
            "amount": round(amount, 2),
            "wallet": router.accounts[user_id]["usdt_wallet"],
            "pnl": pnl
        }

        send_invoice(
            user_id=user_id,
            msg=f"""
🔔 Weekly Billing Notice

Profit: ${pnl}
Share: {PERCENT*100}%
Amount Due: ${amount}

Pay USDT → {invoice['wallet']}
Deadline: 24h
"""
        )

        invoices.append(invoice)

    return invoices
```

### ✅ כולל חוקים

✔ לא משלם תוך 24h → suspension
✔ אין התעסקות ידנית
✔ המערכת מחשבת לבד
✔ עובד לכל משתמש בנפרד

# ✅ 4) AUTO-RECOVERY אחרי קריסה

### 📌 קובץ חדש: `utils/auto_recovery.py`

```python
import time
from utils.health import check_db, check_binance, check_ws
from utils.system_state import system_state
from utils.trade_manager import resync_positions

RETRY_INTERVAL = 10
MAX_RETRIES = 12

def auto_recovery():
    failures = 0

    while failures < MAX_RETRIES:
        if all([check_db(), check_binance(), check_ws()]):
            system_state.kill = False
            resync_positions()
            return True

        failures += 1
        time.sleep(RETRY_INTERVAL)

    system_state.kill = True
    return False
```

### ✅ מה זה עושה

✔ מזהה נפילה לבד
✔ מנסה לתקן עד 12 פעמים
✔ אם הצליח → חוזר לפעולה
✔ אם לא → נכנס ל-Kill-Switch
✔ לא פותח טריידים בלי סינכרון
