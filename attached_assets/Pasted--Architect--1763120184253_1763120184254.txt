```python
# scripts/immediate_blacklist_flush.py
import asyncio
import redis
import logging
from datetime import datetime

logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

async def immediate_blacklist_flush():
    """ניקוי מיידי של TEMP BLACKLIST - הרצה עכשיו!"""
    
    try:
        # התחברות ל-Redis
        r = redis.Redis(host='localhost', port=6379, db=0, decode_responses=True)
        
        # 1. קבלת כל הסימבולים ב-TEMP BLACKLIST
        temp_blacklist_key = "blacklist:temp"
        blacklisted_symbols = r.smembers(temp_blacklist_key)
        
        logger.info(f"🧹 Starting immediate flush - {len(blacklisted_symbols)} symbols in blacklist")
        
        # 2. ניקוי TEMP BLACKLIST
        if blacklisted_symbols:
            r.delete(temp_blacklist_key)
            logger.info(f"✅ Cleared TEMP BLACKLIST: {len(blacklisted_symbols)} symbols")
        
        # 3. איפוס failure counts
        failure_pattern = "failures:count:*"
        failure_keys = r.keys(failure_pattern)
        
        for key in failure_keys:
            r.delete(key)
            logger.debug(f"🗑️ Cleared failure count: {key}")
            
        logger.info(f"✅ Cleared {len(failure_keys)} failure counts")
        
        # 4. איפוס failure history
        history_pattern = "failures:history:*"
        history_keys = r.keys(history_pattern)
        
        for key in history_keys:
            r.delete(key)
            
        logger.info(f"✅ Cleared {len(history_keys)} failure histories")
        
        # 5. יצירת דוח
        report = {
            'timestamp': datetime.now().isoformat(),
            'cleaned_blacklist_count': len(blacklisted_symbols),
            'cleaned_failure_counts': len(failure_keys),
            'cleaned_histories': len(history_keys),
            'status': 'SUCCESS'
        }
        
        # שמירת דוח הניקוי
        r.set("cleanup:immediate_flush_report", str(report))
        
        logger.info("🎉 IMMEDIATE BLACKLIST FLUSH COMPLETED!")
        logger.info(f"📊 Report: {report}")
        
        return report
        
    except Exception as e:
        logger.error(f"❌ Immediate flush failed: {e}")
        return {'status': 'ERROR', 'error': str(e)}

if __name__ == "__main__":
    asyncio.run(immediate_blacklist_flush())
```

## **2. 🧠 BlacklistForgivenessManager**

```python
# utils/blacklist_forgiveness_manager.py
import asyncio
import redis
import logging
import json
from datetime import datetime, timedelta
from typing import Dict, List, Set

logger = logging.getLogger(__name__)

class BlacklistForgivenessManager:
    """
    🧠 מנהל סליחה אוטומטי חכם - Blacklist Forgiveness
    """
    
    def __init__(self, redis_client):
        self.redis = redis_client
        self.forgiveness_interval = 300  # 5 דקות
        
    async def start_forgiveness_engine(self):
        """הפעלת מנוע הסליחה האוטומטי"""
        logger.info("🔄 Starting Blacklist Forgiveness Engine")
        
        while True:
            try:
                await self.run_forgiveness_cycle()
                await asyncio.sleep(self.forgiveness_interval)
                
            except Exception as e:
                logger.error(f"❌ Forgiveness engine failed: {e}")
                await asyncio.sleep(60)
    
    async def run_forgiveness_cycle(self):
        """מחזור סליחה אחד - בודק TOP 50 ומעניק סליחה"""
        try:
            # 1. קבלת TOP 50 עדכני
            top_50 = await self.get_valid_top_50()
            if not top_50:
                logger.warning("⚠️ No valid TOP 50 found - skipping cycle")
                return
                
            # 2. קבלת TEMP BLACKLIST
            blacklisted = await self.get_temp_blacklist()
            
            # 3. מציאת סימבולים ב-TOP 50 שחסמים
            top_50_blacklisted = top_50.intersection(blacklisted)
            
            if not top_50_blacklisted:
                logger.info("✅ No TOP 50 symbols in blacklist - nothing to forgive")
                return
                
            logger.info(f"🎯 Found {len(top_50_blacklisted)} TOP 50 symbols in blacklist")
            
            # 4. עיבוד סליחה לכל סימבול
            forgiveness_results = []
            for symbol in top_50_blacklisted:
                result = await self.process_symbol_forgiveness(symbol)
                forgiveness_results.append(result)
            
            # 5. דוח סופי
            await self.generate_forgiveness_report(forgiveness_results)
            
        except Exception as e:
            logger.error(f"❌ Forgiveness cycle failed: {e}")
    
    async def get_valid_top_50(self) -> Set[str]:
        """קבלת TOP 50 תקף (פחות מ-10 דקות)"""
        try:
            # קבלת TOP 50 מ-Redis
            top_50_data = self.redis.get("market:top_50_symbols")
            if not top_50_data:
                return set()
                
            data = json.loads(top_50_data)
            
            # וידוא עדכניות
            timestamp_str = data.get('timestamp')
            if not timestamp_str:
                return set()
                
            timestamp = datetime.fromisoformat(timestamp_str)
            if datetime.now() - timestamp > timedelta(minutes=10):
                logger.warning("⚠️ TOP 50 data is older than 10 minutes")
                return set()
            
            symbols = set(data.get('symbols', []))
            logger.info(f"📊 Valid TOP 50 contains {len(symbols)} symbols")
            return symbols
            
        except Exception as e:
            logger.error(f"❌ Failed to get valid TOP 50: {e}")
            return set()
    
    async def get_temp_blacklist(self) -> Set[str]:
        """קבלת TEMP BLACKLIST"""
        try:
            blacklist_key = "blacklist:temp"
            symbols = self.redis.smembers(blacklist_key)
            logger.info(f"🔒 TEMP BLACKLIST contains {len(symbols)} symbols")
            return symbols
        except Exception as e:
            logger.error(f"❌ Failed to get TEMP BLACKLIST: {e}")
            return set()
    
    async def process_symbol_forgiveness(self, symbol: str) -> Dict:
        """עיבוד סליחה עבור סימבול בודד"""
        try:
            # קבלת failure count
            failure_count = await self.get_failure_count(symbol)
            
            # החלטת סליחה
            if failure_count < 6:
                # סליחה מלאה - הסרה מ-BLACKLIST + הפחתת failure count
                await self.remove_from_blacklist(symbol)
                new_count = await self.reduce_failure_count(symbol, reduction_percent=50)
                
                logger.info(f"✅ Full forgiveness for {symbol}: failures {failure_count} → {new_count}")
                
                return {
                    'symbol': symbol,
                    'action': 'FULL_FORGIVENESS',
                    'old_failures': failure_count,
                    'new_failures': new_count,
                    'removed_from_blacklist': True
                }
                
            else:
                # Cool-off של שעה - שומר ב-BLACKLIST אבל מאפס timer
                await self.reset_blacklist_timer(symbol)
                
                logger.info(f"⏳ Cool-off for {symbol}: {failure_count} failures - keeping in blacklist for 1h")
                
                return {
                    'symbol': symbol,
                    'action': 'COOL_OFF',
                    'failures': failure_count,
                    'cool_off_duration': '1h',
                    'removed_from_blacklist': False
                }
                
        except Exception as e:
            logger.error(f"❌ Forgiveness processing failed for {symbol}: {e}")
            return {
                'symbol': symbol,
                'action': 'ERROR',
                'error': str(e)
            }
    
    async def get_failure_count(self, symbol: str) -> int:
        """קבלת ספירת failures לסימבול"""
        try:
            count_key = f"failures:count:{symbol}"
            count = self.redis.get(count_key)
            return int(count) if count else 0
        except:
            return 0
    
    async def remove_from_blacklist(self, symbol: str):
        """הסרת סימבול מ-TEMP BLACKLIST"""
        try:
            blacklist_key = "blacklist:temp"
            self.redis.srem(blacklist_key, symbol)
            
            # עדכון זמן הסרה
            removal_key = f"blacklist:removed:{symbol}"
            self.redis.set(removal_key, datetime.now().isoformat(), ex=86400)  # 24h
        except Exception as e:
            logger.error(f"❌ Failed to remove {symbol} from blacklist: {e}")
    
    async def reduce_failure_count(self, symbol: str, reduction_percent: int = 50) -> int:
        """הפחתת failure count באחוזים"""
        try:
            count_key = f"failures:count:{symbol}"
            current_count = await self.get_failure_count(symbol)
            
            if current_count > 0:
                # הפחתה ב-50% - עיגול למעלה
                new_count = max(1, (current_count * reduction_percent) // 100)
                self.redis.set(count_key, new_count)
                return new_count
            return 0
        except Exception as e:
            logger.error(f"❌ Failed to reduce failure count for {symbol}: {e}")
            return current_count
    
    async def reset_blacklist_timer(self, symbol: str):
        """איפוס טיימר BLACKLIST (cool-off)"""
        try:
            # עדכון זמן כניסה ל-BLACKLIST לעכשיו
            entry_key = f"blacklist:entry_time:{symbol}"
            self.redis.set(entry_key, datetime.now().isoformat(), ex=3600)  # 1h
        except Exception as e:
            logger.error(f"❌ Failed to reset blacklist timer for {symbol}: {e}")
    
    async def generate_forgiveness_report(self, results: List[Dict]):
        """יצירת דוח סליחה"""
        try:
            full_forgiveness = [r for r in results if r.get('action') == 'FULL_FORGIVENESS']
            cool_off = [r for r in results if r.get('action') == 'COOL_OFF']
            errors = [r for r in results if r.get('action') == 'ERROR']
            
            report = {
                'timestamp': datetime.now().isoformat(),
                'total_processed': len(results),
                'full_forgiveness_count': len(full_forgiveness),
                'cool_off_count': len(cool_off),
                'error_count': len(errors),
                'details': {
                    'full_forgiveness': full_forgiveness,
                    'cool_off': cool_off,
                    'errors': errors
                }
            }
            
            # שמירת דוח
            self.redis.set("forgiveness:last_report", json.dumps(report), ex=3600)
            
            logger.info(f"📊 Forgiveness Report: {len(full_forgiveness)} forgiven, {len(cool_off)} cool-off, {len(errors)} errors")
            
        except Exception as e:
            logger.error(f"❌ Failed to generate forgiveness report: {e}")
```

## **3. 🎯 DynamicGridCriteriaAdapter**

```python
# utils/dynamic_grid_criteria.py
import asyncio
import redis
import logging
import json
from datetime import datetime, timedelta
from typing import Dict, Optional
from enum import Enum

logger = logging.getLogger(__name__)

class MarketRegime(Enum):
    CHOPPY = "choppy"
    TRENDING = "trending" 
    VOLATILE = "volatile"
    UNKNOWN = "unknown"

class DynamicGridCriteriaAdapter:
    """
    🎯 מתאם קריטריוני GRID דינמיים - מתאים לפי מצב שוק
    """
    
    def __init__(self, redis_client):
        self.redis = redis_client
        self.last_regime_change = None
        self.current_regime = MarketRegime.UNKNOWN
        self.debounce_period = 900  # 15 דקות
        
        # קריטריוני בסיס
        self.base_criteria = {
            'min_volume': 20000000,      # 20M
            'min_liquidity': 75000,      # 75k
            'max_atr': 0.07,             # 7%
            'max_spread': 0.0025,        # 25bps
        }
        
        # multipliers לפי regime
        self.regime_multipliers = {
            MarketRegime.CHOPPY: {
                'volume_multiplier': 0.7,
                'liquidity_multiplier': 0.7, 
                'atr_multiplier': 1.3
            },
            MarketRegime.TRENDING: {
                'volume_multiplier': 1.2,
                'liquidity_multiplier': 1.2,
                'atr_multiplier': 0.8
            },
            MarketRegime.VOLATILE: {
                'volume_multiplier': 1.0,
                'liquidity_multiplier': 1.0,
                'atr_multiplier': 1.1
            }
        }
    
    async def get_dynamic_criteria(self) -> Dict:
        """קבלת קריטריונים דינמיים לפי מצב שוק נוכחי"""
        try:
            # 1. קבלת מצב שוק נוכחי
            current_regime = await self.get_current_market_regime()
            
            # 2. וידוא debounce period
            if not await self.is_regime_stable(current_regime):
                logger.info("⏳ Regime not stable yet - using previous criteria")
                return await self.get_last_criteria()
            
            # 3. חישוב קריטריונים דינמיים
            dynamic_criteria = await self.calculate_dynamic_criteria(current_regime)
            
            # 4. אכיפת safety guardrails
            enforced_criteria = self.enforce_safety_guardrails(dynamic_criteria)
            
            # 5. שמירת קריטריונים נוכחיים
            await self.save_current_criteria(enforced_criteria, current_regime)
            
            logger.info(f"🎯 Dynamic criteria for {current_regime.value}: {enforced_criteria}")
            return enforced_criteria
            
        except Exception as e:
            logger.error(f"❌ Failed to get dynamic criteria: {e}")
            return await self.get_fallback_criteria()
    
    async def get_current_market_regime(self) -> MarketRegime:
        """קבלת מצב שוק נוכחי מ-MarketIntelligence"""
        try:
            # קבלת נתונים מ-MarketIntelligence
            regime_data = self.redis.get("market:intelligence:regime")
            if not regime_data:
                return MarketRegime.UNKNOWN
                
            data = json.loads(regime_data)
            regime_str = data.get('current_regime', 'unknown').lower()
            
            # מיפוי ל-enum
            regime_map = {
                'choppy': MarketRegime.CHOPPY,
                'sideways': MarketRegime.CHOPPY,
                'trending': MarketRegime.TRENDING,
                'bullish': MarketRegime.TRENDING,
                'bearish': MarketRegime.TRENDING,
                'volatile': MarketRegime.VOLATILE
            }
            
            new_regime = regime_map.get(regime_str, MarketRegime.UNKNOWN)
            
            # בדיקת שינוי regime
            if new_regime != self.current_regime:
                if self.last_regime_change and datetime.now() - self.last_regime_change < timedelta(seconds=self.debounce_period):
                    logger.info(f"⏳ Regime change debounced: {self.current_regime.value} → {new_regime.value}")
                    return self.current_regime
                else:
                    logger.info(f"🔄 Regime changed: {self.current_regime.value} → {new_regime.value}")
                    self.current_regime = new_regime
                    self.last_regime_change = datetime.now()
            
            return self.current_regime
            
        except Exception as e:
            logger.error(f"❌ Failed to get market regime: {e}")
            return MarketRegime.UNKNOWN
    
    async def is_regime_stable(self, regime: MarketRegime) -> bool:
        """בדיקה אם regime יציב (debounce)"""
        if not self.last_regime_change:
            return True
            
        time_since_change = datetime.now() - self.last_regime_change
        return time_since_change.total_seconds() >= self.debounce_period
    
    async def calculate_dynamic_criteria(self, regime: MarketRegime) -> Dict:
        """חישוב קריטריונים דינמיים לפי regime"""
        multipliers = self.regime_multipliers.get(regime, self.regime_multipliers[MarketRegime.VOLATILE])
        
        dynamic_criteria = {
            'regime': regime.value,
            'min_volume': self.base_criteria['min_volume'] * multipliers['volume_multiplier'],
            'min_liquidity': self.base_criteria['min_liquidity'] * multipliers['liquidity_multiplier'],
            'max_atr': self.base_criteria['max_atr'] * multipliers['atr_multiplier'],
            'max_spread': self.base_criteria['max_spread'],  # spread לא משתנה
            'calculated_at': datetime.now().isoformat(),
            'multipliers_used': multipliers
        }
        
        return dynamic_criteria
    
    def enforce_safety_guardrails(self, criteria: Dict) -> Dict:
        """אכיפת safety guardrails - ערכי מינימום/מקסימום מוחלטים"""
        enforced = criteria.copy()
        
        # אכיפת floors
        enforced['min_volume'] = max(criteria['min_volume'], 20000000)    # 20M floor
        enforced['min_liquidity'] = max(criteria['min_liquidity'], 75000) # 75k floor
        
        # אכיפת ceilings  
        enforced['max_atr'] = min(criteria['max_atr'], 0.07)              # 7% ceiling
        enforced['max_spread'] = min(criteria['max_spread'], 0.0025)      # 25bps ceiling
        
        return enforced
    
    async def save_current_criteria(self, criteria: Dict, regime: MarketRegime):
        """שמירת קריטריונים נוכחיים ל-Redis"""
        try:
            criteria_data = {
                'criteria': criteria,
                'regime': regime.value,
                'saved_at': datetime.now().isoformat()
            }
            
            self.redis.set("grid:dynamic_criteria:current", json.dumps(criteria_data))
            self.redis.set("grid:dynamic_criteria:history", json.dumps(criteria_data))
            
        except Exception as e:
            logger.error(f"❌ Failed to save current criteria: {e}")
    
    async def get_last_criteria(self) -> Dict:
        """קבלת הקריטריונים האחרונים"""
        try:
            criteria_data = self.redis.get("grid:dynamic_criteria:current")
            if criteria_data:
                data = json.loads(criteria_data)
                return data['criteria']
        except:
            pass
        
        return await self.get_fallback_criteria()
    
    async def get_fallback_criteria(self) -> Dict:
        """קבלת קריטריוני fallback"""
        return {
            'regime': 'fallback',
            'min_volume': self.base_criteria['min_volume'],
            'min_liquidity': self.base_criteria['min_liquidity'],
            'max_atr': self.base_criteria['max_atr'],
            'max_spread': self.base_criteria['max_spread'],
            'is_fallback': True
        }
```

```python
# utils/dynamic_grid_approver.py (עדכון)
import logging
from dynamic_grid_criteria import DynamicGridCriteriaAdapter

logger = logging.getLogger(__name__)

class DynamicGridApprover:
    """מאשר GRID דינמי - משודרג עם קריטריונים דינמיים"""
    
    def __init__(self, redis_client, binance_client):
        self.redis = redis_client
        self.binance = binance_client
        self.criteria_adapter = DynamicGridCriteriaAdapter(redis_client)
    
    async def approve_symbol_for_grid(self, symbol: str) -> bool:
        """אישור סימבול ל-GRID עם קריטריונים דינמיים"""
        try:
            # 1. קבלת קריטריונים דינמיים
            criteria = await self.criteria_adapter.get_dynamic_criteria()
            
            # 2. בדיקת קריטריונים
            checks = await self.run_dynamic_checks(symbol, criteria)
            
            # 3. החלטה סופית
            approved = all(checks.values())
            
            if approved:
                logger.info(f"✅ {symbol} approved for GRID (regime: {criteria['regime']})")
            else:
                failed_checks = [k for k, v in checks.items() if not v]
                logger.info(f"❌ {symbol} rejected for GRID - failed: {failed_checks}")
            
            # 4. שמירת תוצאה ל-audit
            await self.save_approval_result(symbol, approved, checks, criteria)
            
            return approved
            
        except Exception as e:
            logger.error(f"❌ Grid approval failed for {symbol}: {e}")
            return False
    
    async def run_dynamic_checks(self, symbol: str, criteria: Dict) -> Dict:
        """הרצת בדיקות עם קריטריונים דינמיים"""
        checks = {}
        
        try:
            # Volume check
            volume = await self.get_24h_volume(symbol)
            checks['volume'] = volume >= criteria['min_volume']
            
            # Liquidity check
            liquidity = await self.get_liquidity(symbol)
            checks['liquidity'] = liquidity >= criteria['min_liquidity']
            
            # ATR check
            atr = await self.get_atr(symbol)
            checks['atr'] = atr <= criteria['max_atr']
            
            # Spread check
            spread = await self.get_spread(symbol)
            checks['spread'] = spread <= criteria['max_spread']
            
            # Logging עם ערכים
            logger.debug(f"🔍 {symbol} checks - Volume: {volume:.0f}/{criteria['min_volume']:.0f}, "
                        f"Liquidity: {liquidity:.0f}/{criteria['min_liquidity']:.0f}, "
                        f"ATR: {atr:.3f}/{criteria['max_atr']:.3f}, "
                        f"Spread: {spread:.4f}/{criteria['max_spread']:.4f}")
                        
        except Exception as e:
            logger.error(f"❌ Dynamic checks failed for {symbol}: {e}")
            checks['error'] = False
            
        return checks
```
