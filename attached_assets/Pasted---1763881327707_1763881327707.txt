# ✅ **שלב 1 — קובץ AUTO-FREEZE לפי סיכון סימול**

`utils/risk_freeze.py`

```python
import time

FREEZE_MIN = 20 * 60     # 20 minutes
FREEZE_MAX = 3 * 60 * 60 # 3 hours

RISK_THRESHOLD = 0.85  # מעל זה → הקפאה

class AutoFreezeManager:
    def __init__(self):
        self.frozen = {}  # symbol: {until, reason}

    def is_frozen(self, symbol: str) -> bool:
        data = self.frozen.get(symbol)
        if not data:
            return False

        if time.time() >= data["until"]:
            del self.frozen[symbol]
            return False

        return True

    def evaluate_symbol(self, symbol, risk_score):
        """
        risk_score = 0 → רגוע
        risk_score = 1 → סכנה מוחלטת
        """

        if risk_score < RISK_THRESHOLD:
            return False

        duration = min(
            FREEZE_MIN + (risk_score * 60 * 60),
            FREEZE_MAX
        )

        self.frozen[symbol] = {
            "until": time.time() + duration,
            "reason": f"risk={risk_score:.2f}"
        }

        return True

    def get_status(self):
        return self.frozen
```

✅ דינמי
✅ לא קופא מערכת — רק סימולים
✅ משתחרר אוטומטית
✅ אין לעולם soft-lock

# ✅ **שלב 2 — פילטר חלונות זמן (CPI / FOMC וכו')**

`utils/news_window.py`

```python
import time
from datetime import datetime, timedelta

BLOCK_BEFORE_MIN = 90   # דקות לפני האירוע
BLOCK_AFTER_MIN = 30    # דקות אחרי האירוע

class NewsWindowManager:
    def __init__(self):
        self.events = []  # list of timestamps (UTC)
        self.active = False

    def load_events(self, events: list):
        """
        events = timestamps בצורת UTC datetime
        דוגמה: from API או קובץ
        """
        self.events = events

    def is_blocked(self):
        now = datetime.utcnow()

        for event in self.events:
            before = event - timedelta(minutes=BLOCK_BEFORE_MIN)
            after = event + timedelta(minutes=BLOCK_AFTER_MIN)

            if before <= now <= after:
                self.active = True
                return True

        self.active = False
        return False

    def status(self):
        return {
            "active": self.active,
        }
```

✅ אין טריידים חדשים בזמן חלון
✅ ניהול פוזיציות ממשיך
✅ אוטומטי לחלוטין
✅ לא דורש פעולה ידנית
