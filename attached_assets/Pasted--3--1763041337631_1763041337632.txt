    async def protect_position(self, position):
        # שכבה 1: המתן 60 שניות לפני SL
        await asyncio.sleep(self.min_hold_time)
        
        # שכבה 2: חשב SL דינמי

        dynamic_sl = self.calculate_dynamic_sl(position, atr)
        
        # שכבה 3: הגדר הגנת 2%
        hard_stop = position.entry_price * (1 - self.max_loss)
        
        # השתמש ב-2 ההגנות -取其 קריטי יותר
        final_sl = min(dynamic_sl, hard_stop)  # SL הכי צמוד
        
        await self.set_stop_loss(position.symbol, final_sl)
