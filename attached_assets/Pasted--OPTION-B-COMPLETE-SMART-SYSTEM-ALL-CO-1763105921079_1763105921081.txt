        """הפעלת האורכיסטרטור הראשי"""

        # Start all core processes

        ]

        logger.info("✅ All orchestration tasks started")
    
    async def continuous_order_monitoring(self):
        """ניטור רצוף של כל Orders - לב המערכת"""
        while True:
            try:
                current_time = time.time()
                
                # 1. Track all active orders

                # 3. Proactive cleanup
                await self.cleanup_manager.cleanup_orphaned_orders()

            except Exception as e:
                logger.error(f"❌ Order monitoring failed: {e}")
                await self.error_handler.handle_error('MONITORING_FAILURE', {'error': str(e)})
                await asyncio.sleep(10)
    
    async def track_order_lifecycle(self, order: Dict):
        """מעקב מלא אחרי מחזור חיי Order"""
        order_id = order['orderId']
        symbol = order['symbol']
        
        if order_id not in self.order_states:
            # Initialize new order state
            self.order_states[order_id] = {
                'order_id': order_id,
                'symbol': symbol,
                'type': order['type'],
                'side': order['side'],
                'price': float(order['price']),
                'quantity': float(order['origQty']),
                'created_at': time.time(),
                'status': 'NEW',
                'events': [{
                    'timestamp': time.time(),
                    'event': 'CREATED',
                    'price': float(order['price']),
                    'quantity': float(order['origQty'])
                }],
                'last_checked': time.time(),
                'timeout_at': time.time() + await self.timeout_manager.calculate_dynamic_timeout(order)
            }
        else:
            # Update existing order
            self.order_states[order_id]['last_checked'] = time.time()
            self.order_states[order_id]['status'] = order['status']
            
            # Add heartbeat event
            self.order_states[order_id]['events'].append({
                'timestamp': time.time(),
                'event': 'HEARTBEAT',
                'status': order['status'],
                'filled': float(order.get('executedQty', 0))
            })
    
    async def proactive_health_checks(self):
        """בדיקות בריאות pro-active"""
        while True:
            try:
                health_checks = [
                    self.check_system_resources(),
                    self.check_market_connections(),
                    self.check_api_limits(),
                    self.check_performance_degradation()
                ]
                
                results = await asyncio.gather(*health_checks, return_exceptions=True)
                
                # Update system health
                healthy_checks = sum(1 for r in results if not isinstance(r, Exception) and r)
                system_health = 'HEALTHY' if healthy_checks >= 3 else 'DEGRADED'
                
                self.system_state['health'] = system_health
                self.system_state['last_check'] = time.time()
                
                if system_health == 'DEGRADED':
                    logger.warning(f"⚠️ System health degraded - {healthy_checks}/4 checks passed")
                    await self.recovery_queue.put({
                        'type': 'SYSTEM_HEALTH',
                        'health': system_health,
                        'failed_checks': results
                    })
                
                await asyncio.sleep(30)  # Every 30 seconds
                
            except Exception as e:
                logger.error(f"❌ Health checks failed: {e}")
                await asyncio.sleep(60)
    
    async def auto_recovery_engine(self):
        """מנוע התאוששות אוטומטי"""
        while True:
            try:
                # Process recovery items with timeout
                try:
                    recovery_item = await asyncio.wait_for(
                        self.recovery_queue.get(), 
                        timeout=10.0
                    )
                    await self.execute_recovery(recovery_item)
                    self.performance_metrics['recoveries_executed'] += 1
                except asyncio.TimeoutError:
                    pass  # No recovery items
                
                # Auto-cleanup every 2 minutes
                if time.time() - getattr(self, 'last_cleanup', 0) > 120:
                    await self.cleanup_manager.cleanup_old_order_states()
                    self.last_cleanup = time.time()
                
                await asyncio.sleep(5)  # Fast recovery loop
                
            except Exception as e:
                logger.error(f"❌ Auto-recovery failed: {e}")
                await asyncio.sleep(30)
    
    async def execute_recovery(self, recovery_item: Dict):
        """ביצוע התאוששות"""
        try:
            recovery_type = recovery_item['type']
            
            if recovery_type == 'ORDER_TIMEOUT':
                await self.timeout_manager.handle_order_timeout(
                    recovery_item['order_id'], 
                    recovery_item['order_state']
                )
            elif recovery_type == 'SYSTEM_HEALTH':
                await self.error_handler.recover_system_health(recovery_item)
            elif recovery_type == 'NETWORK_ERROR':
                await self.error_handler.recover_network_error(recovery_item)
            elif recovery_type == 'API_LIMIT':
                await self.error_handler.recover_api_limit(recovery_item)
                
            logger.info(f"🛡️ Recovery executed for {recovery_type}")
            
        except Exception as e:
            logger.error(f"❌ Recovery execution failed: {e}")
            await self.error_handler.handle_error('RECOVERY_FAILED', {
                'recovery_type': recovery_item.get('type'),
                'error': str(e)
            })
    
    async def performance_optimizer(self):
        """אופטימיזציית ביצועים אוטומטית"""
        while True:
            try:
                # Analyze performance data
                performance_data = await self.analyze_performance()
                
                # Optimize timeouts based on performance
                if performance_data['avg_fill_time'] < 10:
                    # Fast fills - reduce timeouts
                    self.timeout_manager.base_timeout = max(30, self.timeout_manager.base_timeout * 0.9)
                elif performance_data['avg_fill_time'] > 45:
                    # Slow fills - increase timeouts  
                    self.timeout_manager.base_timeout = min(120, self.timeout_manager.base_timeout * 1.1)
                
                # Optimize cleanup intervals

                    self.cleanup_manager.cleanup_interval = max(60, self.cleanup_manager.cleanup_interval * 0.8)
                
                await asyncio.sleep(300)  # Every 5 minutes

    async def state_synchronizer(self):
