## ✅ קובץ חדש: `utils/performance_heatmap.py`

```python
import numpy as np
from collections import defaultdict

class PerformanceHeatmap:
    def __init__(self):
        self.map = defaultdict(lambda: {"wins": 0, "losses": 0})

    def update(self, mode, outcome):
        if outcome > 0:
            self.map[mode]["wins"] += 1
        else:
            self.map[mode]["losses"] += 1

    def get_score(self, mode):
        w = self.map[mode]["wins"]
        l = self.map[mode]["losses"]
        total = w + l
        if total == 0:
            return 1.0  # neutral
        return round(w / total, 2)

    def summary(self):
        return {
            mode: {
                "win_rate": self.get_score(mode),
                "trades": self.map[mode]["wins"] + self.map[mode]["losses"]
            }
            for mode in self.map
        }
```

## ✅ קובץ חדש: `utils/atr_controller.py`

```python
class ATRController:
    def __init__(self):
        self.multiplier = 1.2

    def adjust(self, mode, heatmap_score, agent_override=None):
        # If agent controls it
        if agent_override is not None:
            self.multiplier = agent_override
            return self.multiplier

        # System logic
        if mode == "TREND" and heatmap_score >= 0.7:
            self.multiplier = min(2.0, self.multiplier + 0.1)

        elif mode == "CHOP" and heatmap_score <= 0.4:
            self.multiplier = max(0.8, self.multiplier - 0.1)

        return self.multiplier
```

## ✅ קובץ חדש: `utils/auto_pause.py`

```python
class AutoPause:
    def __init__(self):
        self.active = False

    def evaluate(self, news):
        if news["impact"] == "critical" or news.get("score", 0) <= -0.7:
            self.active = True
        return self.active

    def reset(self):
        self.active = False
```
