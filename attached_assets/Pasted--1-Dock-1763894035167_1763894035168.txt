# ✅ **1) Dockerfile**

```dockerfile
FROM python:3.11-slim

ENV PYTHONUNBUFFERED=1
WORKDIR /app

# System deps
RUN apt-get update && apt-get install -y \
    git curl unzip wget nano jq build-essential \
    && rm -rf /var/lib/apt/lists/*

# Install code-server
RUN curl -fsSL https://code-server.dev/install.sh | sh

# Install Ollama
RUN curl -fsSL https://ollama.ai/install.sh | sh

COPY requirements.txt .
RUN pip install --no-cache-dir -r requirements.txt

COPY . .

EXPOSE 8000 8080 11434

CMD ["bash", "./start.sh"]
```

# ✅ **2) docker-compose.yml**

```yaml
version: "3.9"

services:
  algogpt-dev:
    build: .
    container_name: algogpt-dev
    restart: always
    ports:
      - "8000:8000"     # FastAPI
      - "8080:8080"     # code-server
      - "11434:11434"   # Ollama
    env_file:
      - config.env
    volumes:
      - ./workspace:/app/workspace
      - ./backups:/app/backups
    healthcheck:
      test: ["CMD", "curl", "-f", "http://localhost:8000/readyz"]
      interval: 30s
      timeout: 5s
      retries: 5
```

# ✅ **3) install.sh (הרצה בפקודה אחת)**

```bash
#!/bin/bash
set -e

echo "🚀 Installing AlgoGPT Private Dev Platform..."

# Create folders
mkdir -p workspace backups

# Pull Ollama models
echo "📥 Downloading AI models..."
ollama pull llama3
ollama pull codellama

# Start system
docker-compose up -d --build

echo "✅ DONE!"
echo "🌐 Open IDE at: https://YOUR-SERVER-IP:8080"
echo "🧠 Ollama running on port 11434"
echo "📊 FastAPI dashboard on port 8000"
```

Make executable:

```bash
chmod +x install.sh
```

Run:

```bash
./install.sh
```

# ✅ **4) start.sh**

```bash
#!/bin/bash
set -e

# Start ollama
ollama serve &

# Start code-server
code-server --bind-addr 0.0.0.0:8080 --auth password &

# Start backend
uvicorn main:app --host 0.0.0.0 --port 8000
```
