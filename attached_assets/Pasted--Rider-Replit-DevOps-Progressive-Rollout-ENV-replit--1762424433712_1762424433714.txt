**(1) `utils/precision.py` — עטיפות דיוק+מסננים**

```python
# utils/precision.py
from __future__ import annotations
from decimal import Decimal, ROUND_DOWN

def _to_dec(x): 
    return Decimal(str(x))

def quant_to_step(value: float, step: float) -> float:
    v = _to_dec(value); s = _to_dec(step)
    if s <= 0: return float(v)
    return float((v // s) * s)

def clamp_tick(price: float, tick: float) -> float:
    return quant_to_step(price, tick)

def clamp_step(qty: float, step: float) -> float:
    return quant_to_step(qty, step)

def ensure_min_notional(price: float, qty: float, min_notional: float) -> float:
    notional = _to_dec(price) * _to_dec(qty)
    if notional >= _to_dec(min_notional):
        return qty
    need_qty = (_to_dec(min_notional) / _to_dec(price)).quantize(_to_dec("0.00000001"), rounding=ROUND_DOWN)
    return float(need_qty)

def sanitize_order(price: float, qty: float, tick: float, step: float, min_notional: float) -> tuple[float,float]:
    p = clamp_tick(price, tick)
    q = clamp_step(qty, step)
    q = ensure_min_notional(p, q, min_notional)
    q = clamp_step(q, step)
    return p, q
```

**(2) `utils/regime_detector_v2.py` — זיהוי משופר + confidence**

```python
# utils/regime_detector_v2.py
from __future__ import annotations
from dataclasses import dataclass
import time, math

@dataclass
class RegimeResult:
    regime: str           # TRENDING/CHOPPY/VOLATILE/SIDEWAYS
    confidence: float     # 0..1
    ts: float

def detect_regime_v2(atr_pct: float, adx: float, macd_slope: float) -> RegimeResult:
    # היגיון פשוט ויציב (אפשר לשפר בהמשך):
    # atr_pct = ATR/Price * 100
    # adx: 0..50+
    # macd_slope: חיובי/שלילי/שטוח
    score_trend  = max(0.0, (adx-18)/22) * (1.0 if abs(macd_slope) > 0.0 else 0.7)
    score_chop   = max(0.0, (15-abs(macd_slope))/15) * (1.0 if adx<20 else 0.6)
    score_vol    = min(1.0, atr_pct/2.5)
    score_side   = max(0.0, 1.0 - (score_trend+score_vol))

    # נרמל:
    raw = [("TRENDING", score_trend), ("CHOPPY", score_chop), ("VOLATILE", score_vol), ("SIDEWAYS", score_side)]
    s = sum(v for _,v in raw) or 1.0
    probs = [(k, v/s) for k,v in raw]
    regime, conf = max(probs, key=lambda kv: kv[1])
    return RegimeResult(regime=regime, confidence=float(conf), ts=time.time())
```

**(3) `utils/adaptive_mixer.py` — מיקסר פשוט (שלב 1)**

```python
# utils/adaptive_mixer.py
from __future__ import annotations
from dataclasses import dataclass

@dataclass
class MixParams:
    sl_atr_mult: float
    tp_rr_mult: float
    trail_enable: bool

def basic_mix(regime: str, conf: float, pnl_state: str|None = None) -> MixParams:
    # בסיס בטוח; בהמשך נרחיב:
    base = {
        "TRENDING": (2.0, 2.2, True),
        "CHOPPY":   (1.2, 1.4, False),
        "VOLATILE": (2.5, 1.8, True),
        "SIDEWAYS": (1.0, 1.2, False),
    }.get(regime, (1.6, 1.6, False))
    sl, rr, te = base

    # confidence משפיע עד ±20%
    factor = 0.8 + 0.4*conf
    sl *= factor
    rr *= factor

    # מצב PnL זהיר (אם קיים): אם Drawdown → הגדל SL והקטן RR מעט
    if pnl_state == "drawdown":
        sl *= 1.1; rr *= 0.9

    return MixParams(sl_atr_mult=sl, tp_rr_mult=rr, trail_enable=te)
```

```python
import hashlib, time

def make_ik(symbol: str, side: str, action: str, seed: str|None=None) -> str:
    raw = f"{symbol}|{side}|{action}|{seed or ''}|{int(time.time()//5)}"
    return hashlib.sha256(raw.encode()).hexdigest()[:32]
```

```python
# utils/circuit_breaker.py
import time

class CircuitBreaker:
    def __init__(self, fail_threshold:int=5, cooldown_sec:int=900):
        self.fail_threshold = fail_threshold
        self.cooldown_sec   = cooldown_sec
        self._fails = 0
        self._opened_ts = 0.0

    def record_success(self):
        self._fails = 0

    def record_fail(self):
        self._fails += 1
        if self._fails >= self.fail_threshold:
            self._opened_ts = time.time()

    def is_open(self)->bool:
        if self._opened_ts==0: return False
        if time.time()-self._opened_ts >= self.cooldown_sec:
            self._opened_ts = 0; self._fails = 0
            return False
        return True
```

```python
        log.info("[DYN_ENFORCE_OK] %s", sample)
    else:
        log.error("[DYN_ENFORCE_FAIL] %s err=%s", sample, err)
```

```python
        place_or_replace_stop_loss(symbol, side, context["position_qty"], sl_price, ik)  # עטוף קיים
```
