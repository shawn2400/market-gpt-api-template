**A. סכמת DB — `migrations/xxxx_expenses.sql`** (הפעל אוטו-מיגרציה אם קיימת; אחרת תריץ ב־psql)

```sql
CREATE TABLE IF NOT EXISTS partners (
  id SERIAL PRIMARY KEY,
  name VARCHAR(64) NOT NULL UNIQUE,
  weight NUMERIC(6,4) NOT NULL DEFAULT 1.0,  -- משקל חלוקה
  active BOOLEAN NOT NULL DEFAULT TRUE,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS expense_categories (
  id SERIAL PRIMARY KEY,
  code VARCHAR(32) NOT NULL UNIQUE,          -- e.g. "infra","ai","ops","fees"
  name VARCHAR(64) NOT NULL,
  monthly_cap_usd NUMERIC(12,2) NOT NULL DEFAULT 0, -- 0 = ללא תקרה
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS expenses (
  id SERIAL PRIMARY KEY,
  ts TIMESTAMPTZ NOT NULL,                   -- תאריך הוצאה
  vendor VARCHAR(96) NOT NULL,
  description TEXT,
  amount_usd NUMERIC(14,2) NOT NULL CHECK (amount_usd >= 0),
  category_code VARCHAR(32) REFERENCES expense_categories(code),
  payer_partner_id INT REFERENCES partners(id),     -- מי שילם בפועל (אם רלוונטי)
  attachment_url TEXT,                        -- קישור קבלה/חשבונית (אופציונלי)
  status VARCHAR(16) NOT NULL DEFAULT 'PENDING', -- PENDING/APPROVED/REJECTED
  created_by VARCHAR(64),                     -- user id / telegram
  created_at TIMESTAMPTZ DEFAULT NOW(),
  UNIQUE(ts, vendor, amount_usd, coalesce(description,'')) -- מניעת כפילויות בסיסית
);

CREATE TABLE IF NOT EXISTS expense_allocations (
  id SERIAL PRIMARY KEY,
  expense_id INT NOT NULL REFERENCES expenses(id) ON DELETE CASCADE,
  partner_id INT NOT NULL REFERENCES partners(id),
  share_pct NUMERIC(6,4) NOT NULL,           -- אחוז חלוקה (0..1)
  share_usd NUMERIC(14,2) NOT NULL           -- סכום לשותף
);

CREATE TABLE IF NOT EXISTS expense_rules_audit (
  id SERIAL PRIMARY KEY,
  ts TIMESTAMPTZ DEFAULT NOW(),
  rule VARCHAR(64) NOT NULL,
  outcome VARCHAR(32) NOT NULL,
  expense_id INT,
  details JSONB
);
```

**C. שירות לוגיקה — `utils/expenses_manager.py`**

```python
# -*- coding: utf-8 -*-
from __future__ import annotations
import os, math, statistics
from typing import Dict, List, Tuple, Optional
from decimal import Decimal
import psycopg2, psycopg2.extras

DSN = os.getenv("POSTGRES_DSN") or os.getenv("DATABASE_URL") or "postgresql://postgres:postgres@127.0.0.1:5432/algogpt"

SOFT_PCT = float(os.getenv("EXPENSES_SOFT_CAP_ALERT_PCT", "0.8"))
HARD_BLOCK = os.getenv("EXPENSES_HARD_CAP_BLOCK","on")=="on"
ANOMALY_PCT = float(os.getenv("EXPENSES_ANOMALY_PCT","0.35"))
DEFAULT_SPLIT = os.getenv("EXPENSES_DEFAULT_SPLIT","weights")  # weights/equal

def _q(sql:str, params:dict|tuple=()):
    conn = psycopg2.connect(DSN); conn.autocommit=True
    try:
        with conn.cursor(cursor_factory=psycopg2.extras.RealDictCursor) as cur:
            cur.execute(sql, params); 
            if cur.description: return cur.fetchall()
            return []
    finally:
        conn.close()

def get_partners() -> List[dict]:
    return _q("SELECT * FROM partners WHERE active=TRUE ORDER BY id")

def get_category(code:str) -> Optional[dict]:
    r=_q("SELECT * FROM expense_categories WHERE code=%s", (code,)); return r[0] if r else None

def current_month_usage(code:str) -> Decimal:
    r=_q("""
        SELECT COALESCE(SUM(amount_usd),0)::numeric AS s
        FROM expenses 
        WHERE category_code=%s AND date_trunc('month', ts)=date_trunc('month', now()) AND status IN ('APPROVED','PENDING')
    """,(code,))
    return r[0]["s"] if r else Decimal(0)

def split_amount(amount: Decimal, partners: List[dict]) -> List[Tuple[int, Decimal, float]]:
    if not partners: return []
    if DEFAULT_SPLIT=="equal":
        pct = 1.0/len(partners)
        return [(p["id"], Decimal(amount)*Decimal(pct), pct) for p in partners]
    total_w = sum(float(p["weight"]) for p in partners)
    return [(p["id"], Decimal(amount)*Decimal(float(p["weight"])/total_w), float(p["weight"])/total_w) for p in partners]

def anomaly_check(code:str, amount: Decimal) -> Tuple[bool, float]:
    rows = _q("""
      SELECT amount_usd FROM expenses 
      WHERE category_code=%s AND date_trunc('month', ts)>=date_trunc('month', now()) - interval '6 months'
        AND status IN ('APPROVED','PENDING') 
      ORDER BY ts DESC LIMIT 500
    """,(code,))
    if len(rows) < 10: return (False, 0.0)
    vals = [float(r["amount_usd"]) for r in rows]
    med = statistics.median(vals)
    if med<=0: return (False, 0.0)
    delta = (float(amount)-med)/med
    return (delta>ANOMALY_PCT, delta)

def can_approve(category_code:str, add_amount: Decimal) -> Tuple[bool, str, float]:
    cat = get_category(category_code)
    if not cat: return (False, "unknown_category", 0.0)
    cap = float(cat["monthly_cap_usd"])
    if cap<=0: return (True,"no_cap", 0.0)
    used = float(current_month_usage(category_code))
    new = used + float(add_amount)
    ratio = new/cap
    if ratio >= 1.0 and HARD_BLOCK: return (False,"hard_cap_exceeded", ratio)
    return (True,"ok_soft" if ratio>=SOFT_PCT else "ok", ratio)

def persist_expense(ts, vendor, description, amount, category_code, payer_partner_id, created_by, attachment_url=None) -> int:
    rows = _q("""INSERT INTO expenses (ts,vendor,description,amount_usd,category_code,payer_partner_id,created_by,attachment_url,status)
                 VALUES (%s,%s,%s,%s,%s,%s,%s,%s,'PENDING') RETURNING id""",
              (ts, vendor, description, amount, category_code, payer_partner_id, created_by, attachment_url))
    return rows[0]["id"]

def persist_allocations(expense_id:int, allocs:List[Tuple[int, Decimal, float]]) -> None:
    for pid,amt,pct in allocs:
        _q("""INSERT INTO expense_allocations (expense_id,partner_id,share_pct,share_usd) 
              VALUES (%s,%s,%s,%s)""",(expense_id, pid, pct, amt))

def audit(rule:str, outcome:str, expense_id:int|None, details:dict) -> None:
    _q("""INSERT INTO expense_rules_audit (rule,outcome,expense_id,details) VALUES (%s,%s,%s,%s)""",
       (rule, outcome, expense_id, psycopg2.extras.Json(details)))

def approve(expense_id:int, approve:bool, reason:str=""):
    st = "APPROVED" if approve else "REJECTED"
    _q("UPDATE expenses SET status=%s WHERE id=%s",(st,expense_id))
    audit("manual_approve", st.lower(), expense_id, {"reason":reason})
```

**D. REST API — `routes/expenses.py`** (מאובטח Bearer; שמור על הסגנון אצלך)

```python
# -*- coding: utf-8 -*-
from __future__ import annotations
import os, datetime as dt
from decimal import Decimal
from typing import Optional, List
from fastapi import APIRouter, Depends, HTTPException
from pydantic import BaseModel, Field
from utils.expenses_manager import (persist_expense, persist_allocations, split_amount,
    can_approve, anomaly_check, get_partners, audit, approve)

def bearer_ok(): 
    # השתמש בפתרון ה-Bearer/HMAC הקיים אצלך
    return True

router = APIRouter(prefix="/expenses", tags=["Expenses"], dependencies=[Depends(bearer_ok)])

class ExpenseIn(BaseModel):
    ts: dt.datetime
    vendor: str = Field(min_length=2, max_length=96)
    description: Optional[str] = ""
    amount_usd: Decimal = Field(gt=0)
    category_code: str
    payer_partner_id: Optional[int] = None
    attachment_url: Optional[str] = None
    created_by: Optional[str] = "system"

class ExpenseOut(BaseModel):
    id: int
    status: str
    cap_state: str
    cap_ratio: float
    anomaly: bool
    anomaly_delta: float

@router.post("/add", response_model=ExpenseOut)
def add_expense(e: ExpenseIn):
    ok, cap_state, ratio = can_approve(e.category_code, e.amount_usd)
    anomaly, delta = anomaly_check(e.category_code, e.amount_usd)
    exp_id = persist_expense(e.ts, e.vendor, e.description, e.amount_usd, e.category_code, e.payer_partner_id, e.created_by, e.attachment_url)
    allocs = split_amount(e.amount_usd, get_partners())
    persist_allocations(exp_id, allocs)
    audit("cap_check", cap_state, exp_id, {"ratio":ratio})
    audit("anomaly_check", "alert" if anomaly else "ok", exp_id, {"delta":delta})
    if not ok:
        # מסמן כ-PENDING; נדרש אישור ידני
        return ExpenseOut(id=exp_id, status="PENDING", cap_state=cap_state, cap_ratio=ratio, anomaly=anomaly, anomaly_delta=delta)
    # אפשר אוטו-אישור אם רוצים: approve(exp_id, True, "auto-cap-ok")
    return ExpenseOut(id=exp_id, status="PENDING", cap_state=cap_state, cap_ratio=ratio, anomaly=anomaly, anomaly_delta=delta)

class ApproveIn(BaseModel):
    approve: bool
    reason: Optional[str] = ""

@router.post("/{expense_id}/approve")
def approve_expense(expense_id:int, body:ApproveIn):
    approve(expense_id, body.approve, body.reason or "")
    return {"ok": True, "id": expense_id, "status": "APPROVED" if body.approve else "REJECTED"}

@router.get("/summary/month")
def month_summary():
    # סיכומי חודש (תקציב/ניצול/חריגים) – לטבלאות ודוחות
    from utils.expenses_reports import month_snapshot
    return month_snapshot()
```

# -*- coding: utf-8 -*-
from __future__ import annotations
import os
from datetime import datetime, timezone
import psycopg2, psycopg2.extras

DSN = os.getenv("POSTGRES_DSN") or os.getenv("DATABASE_URL")

def _q(sql:str, params=()):
    conn = psycopg2.connect(DSN); conn.autocommit=True
    try:
        with conn.cursor(cursor_factory=psycopg2.extras.RealDictCursor) as cur:
            cur.execute(sql, params)
            return cur.fetchall()
    finally: conn.close()

def month_snapshot():
    rows = _q("""
      WITH e AS (
        SELECT category_code, SUM(amount_usd) AS total
        FROM expenses
        WHERE date_trunc('month', ts)=date_trunc('month', now()) 
          AND status IN ('APPROVED','PENDING')
        GROUP BY 1
      )
      SELECT c.code, c.name, c.monthly_cap_usd, COALESCE(e.total,0) AS used,
             CASE WHEN c.monthly_cap_usd>0 THEN (COALESCE(e.total,0)/c.monthly_cap_usd) ELSE NULL END AS ratio
      FROM expense_categories c
      LEFT JOIN e ON e.category_code=c.code
      ORDER BY c.code
    """)
    return {"month": datetime.now(timezone.utc).strftime("%Y-%m"),
            "categories": rows}

import os, datetime as dt

        send_message("\n".join(["📊 Soft Cap Alerts"]+msgs))

### 3) כללי ניהול/בקרה (חכם ולא מפוזר)

1. **תקרות חודשיות לפי קטגוריה** (DB)
2. **SOFT CAP Alerts** ב-80% ניצול (בר־שינוי)
3. **HARD CAP Block**: עצירת אישור אוטומטי מעל תקרה
4. **חלוקה בין שותפים**: `weights` (ברירת מחדל) או `equal`
5. **זיהוי חריגות**: 35% מעל חציון 6 חודשים ⇒ סימון/Audit
6. **Anti-Duplicate**: יוניק בסיסי (ts,vendor,amount,desc)
7. **אישורים**: API/טלגרם (Approve/Reject עם סיבה)
8. **דוחות**: סיכומי חודש + CSV/PDF דרך תשתית ה-Export הקיימת
9. **אבטחה**: Bearer/HMAC קיימים; Rate-Limit ברמת Gateway שלך
10. **4GB-Safe**: שאילתות קלות; בלי OCR/ML מקומי; הכל סינכרוני קצר
