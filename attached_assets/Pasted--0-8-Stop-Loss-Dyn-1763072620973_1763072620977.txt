class DynamicStopLoss:
    def __init__(self):
        self.base_stop_loss = -0.008  # -0.8% בסיסי
        self.volatility_multipliers = {
            'LOW_VOLATILITY': 1.2,    # -0.96%
            'MEDIUM_VOLATILITY': 1.0, # -0.8% 
            'HIGH_VOLATILITY': 0.6,   # -0.48%
            'EXTREME_VOLATILITY': 0.4 # -0.32%
        }
    
    async def calculate_dynamic_stop_loss(self, symbol, position_data):
        """חישוב Stop Loss דינמי לפי תנודתיות"""
        
        # 1. בדיקת תנודתיות הסמל
        volatility_level = await self.get_symbol_volatility(symbol)
        
        # 2. התאמת Stop Loss לפי תנודתיות
        multiplier = self.volatility_multipliers[volatility_level]
        dynamic_stop = self.base_stop_loss * multiplier
        
        # 3. הגבלות נוספות
        final_stop = await self.apply_additional_rules(symbol, dynamic_stop, position_data)
        
        return final_stop
    
    async def get_symbol_volatility(self, symbol):
        """זיהוי רמת התנודתיות של הסמל"""
        atr = await self.get_average_true_range(symbol)
        
        if atr > 0.06: return 'EXTREME_VOLATILITY'
        elif atr > 0.04: return 'HIGH_VOLATILITY'
        elif atr > 0.02: return 'MEDIUM_VOLATILITY'
        else: return 'LOW_VOLATILITY'
```

class SmartStopLossEngine:
    def __init__(self):
        self.protection_rules = {
            'max_daily_loss': -0.02,  # -2% מקסימום הפסד יומי
            'circuit_breaker': -0.05, # -5% הפסד כולל - עוצר הכל
            'time_based_stops': True,
            'volume_spike_protection': True
        }
    
    async def calculate_smart_stop_loss(self, symbol, entry_price, position_size):
        """Stop Loss חכם עם multiple layers"""
        
        stops = []
        
        # 1. Stop Loss בסיסי לפי תנודתיות
        base_stop = await self.dynamic_stop_loss.calculate_dynamic_stop_loss(symbol, {
            'entry_price': entry_price,
            'size': position_size
        })
        stops.append(('VOLATILITY_BASED', base_stop))
        
        # 2. Stop Loss לפי Support Levels
        support_stop = await self.get_support_based_stop(symbol, entry_price)
        stops.append(('SUPPORT_BASED', support_stop))
        
        # 3. Stop Loss לפי Time Decay
        time_stop = await self.get_time_based_stop(symbol, entry_price)
        stops.append(('TIME_BASED', time_stop))
        
        # 4. בחירת ה-Stop Loss ההגנתי ביותר (הכי קרוב)
        best_stop = max(stops, key=lambda x: x[1])  # highest stop loss (least negative)
        
        return {
            'stop_loss_price': entry_price * (1 + best_stop[1]),
            'stop_loss_percent': best_stop[1],
            'stop_type': best_stop[0],
            'all_calculated_stops': stops
        }
    
    async def get_support_based_stop(self, symbol, entry_price):
        """Stop Loss לפי רמות Support"""
        support_levels = await self.get_support_levels(symbol)
        nearest_support = self.find_nearest_support(support_levels, entry_price)
        
        if nearest_support:
            stop_percent = (nearest_support - entry_price) / entry_price
            # הגבל ל-max -3%
            return max(stop_percent, -0.03)
        
        return -0.015  # default -1.5% אם אין Support
        
    async def get_time_based_stop(self, symbol, entry_price):
        """Stop Loss שהולך ונהיה הדוק יותר עם הזמן"""
        position_age = await self.get_position_age(symbol)
        
        if position_age < 900:  # 15 דקות ראשונות
            return -0.012  # -1.2% - מרווח נשימה
        elif position_age < 3600:  # שעה ראשונה
            return -0.008  # -0.8%
        else:  # אחרי שעה
            return -0.005  # -0.5% - הדוק יותר
```

class AdvancedStopStrategies:
    async def trailing_stop_loss(self, symbol, current_price, highest_price):
        """Trailing Stop Loss אוטומטי"""
        trail_percent = 0.006  # 0.6% trailing
        
        if highest_price > current_price:
            trail_stop = highest_price * (1 - trail_percent)
            return max(trail_stop, current_price * 0.995)  # min -0.5%
        
        return current_price * 0.992  # -0.8% default
    
    async def volume_spike_stop(self, symbol, current_price):
        """Stop Loss בהתבסס על Volume Spikes"""
        volume_ratio = await self.get_volume_ratio(symbol)
        
        if volume_ratio > 3.0:  # volume spike של x3
            # בשוק תנודתי - Stop Loss יותר הדוק
            return -0.004  # -0.4%
        
        return -0.008  # -0.8% רגיל
    
    async def correlated_stop_loss(self, symbol, market_correlation):
        """Stop Loss בהתבסס על Correlation לשוק"""
        if market_correlation > 0.8:
            # סמל מאוד מקושר - Stop Loss יותר הדוק
            return -0.006  # -0.6%
        else:
            # סמל עצמאי - יותר מרווח
            return -0.010  # -1.0%

                'stop_type': position['stop_type'],
                'risk_level': self.assess_risk_level(position)
            }
            stop_analysis.append(analysis)
        
        return {
            'active_stops': stop_analysis,
            'total_risk_exposure': await self.calculate_total_risk(),
            'closest_stops': self.get_closest_stops(stop_analysis),
            'daily_stop_triggers': await self.get_daily_stop_stats()
        }
    
    def calculate_stop_distance(self, position):
        """חישוב מרחק מ-Stop Loss"""
        current = position['current_price']
        stop = position['stop_loss_price']
        return ((current - stop) / current) * 100
