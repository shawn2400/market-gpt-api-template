# ✅ 1) קובץ חדש: `utils/market_heatmap.py`

```python
import time

class MarketHeatmap:
    def __init__(self):
        self.data = {}  # {symbol: {regime: performance}}
        self.last_update = 0

    def update(self, symbol: str, regime: str, pnl: float, win: bool):
        if symbol not in self.data:
            self.data[symbol] = {}

        if regime not in self.data[symbol]:
            self.data[symbol][regime] = {
                "trades": 0,
                "wins": 0,
                "pnl": 0.0
            }

        bucket = self.data[symbol][regime]
        bucket["trades"] += 1
        bucket["pnl"] += pnl
        if win:
            bucket["wins"] += 1

        self.last_update = time.time()

    def score(self, symbol: str, regime: str):
        bucket = self.data.get(symbol, {}).get(regime)
        if not bucket or bucket["trades"] < 3:
            return None  # not enough data

        win_rate = bucket["wins"] / bucket["trades"]
        avg_pnl = bucket["pnl"] / bucket["trades"]

        return (win_rate * 0.7) + (avg_pnl * 0.3)

    def best_regime(self, symbol: str):
        regimes = self.data.get(symbol, {})
        if not regimes:
            return None

        scored = {
            r: self.score(symbol, r)
            for r in regimes
            if self.score(symbol, r) is not None
        }

        if not scored:
            return None

        return max(scored, key=scored.get)
```

# ✅ 4) מדד Heat-Index חכם

ב־`utils/ultra_plus_manager.py`:

```python
def compute_heat_index(self, volatility, funding, sentiment):
    score = 100

    if volatility > 0.05:
        score -= 25

    if funding > 0.01:
        score -= 20

    if sentiment == "fear":
        score -= 30

    if sentiment == "extreme_fear":
        score -= 50

    self.current_heat_index = max(0, score)
    return self.current_heat_index
```

🟢 70–100 = סחר רגיל
🟡 40–69 = זהירות
🔴 0–39 = pause auto

# ✅ 5) ML-Prediction (קל, לא כבד)

קובץ חדש: `utils/ml_predictor.py`

```python
import numpy as np

class MicroPredictor:
    def predict(self, prices):
        # prices = last N closes
        arr = np.array(prices)
        slope = arr[-1] - arr[-5]
        vol = np.std(arr[-10:])
        momentum = slope / (vol + 1e-6)

        if momentum > 0.8:
            return "up"
        if momentum < -0.8:
            return "down"
        return "flat"
```
